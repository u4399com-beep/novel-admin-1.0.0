   141→export function Category({

   144→  page = 1,

   151→  return (

   153→      <Crumb items={[{ label: '首页', view: { name: 'home' } }, { label: '小说书库' }]} navigate={navigate} />
   154→      <h1 className="border-l-4 border-[#2563eb] pl-4 text-[24px] leading-8 font-semibold text-[#1e293b]">
   155→        {catName ?? '小说书库'}
   156→        {cats.data ? (
   157→          <span className="ml-3 text-[13px] font-normal text-[#94a3b8]">
   158→            {cats.data.find((c) => c.id === categoryId)?.novelCount ?? list.data?.total ?? ''}
   159→          </span>
   160→        ) : null}
   161→      </h1>
   162→
   163→      {/* 筛选条 */}

   165→        <button
   166→          onClick={() => navigate({ name: 'category', page: 1 })}
   167→          className={
   168→            categoryId == null
   169→              ? 'cursor-pointer rounded-full bg-[#2563eb] px-4 py-1.5 text-[13px] text-white transition-colors duration-200'
   170→              : 'cursor-pointer rounded-full bg-[#f0f4fb] px-4 py-1.5 text-[13px] text-[#1e293b] transition-colors duration-200 hover:bg-[#e8f1ff] hover:text-[#2563eb]'

   172→        >

   181→                ? 'cursor-pointer rounded-full bg-[#2563eb] px-4 py-1.5 text-[13px] text-white transition-colors duration-200'

   188→      </div>

   421→        sort="clicks"
   422→        navigate={navigate}
   423→      />
   424→    </div>
   425→  )
   426→}

   429→
   430→export function Toc({ navigate, novelId }: ViewProps & { novelId: number }) {
   431→  const { data: novel } = useNovel(novelId)
   432→  const { data: chapters, isPending, isError, refetch } = useChapters(novelId)
   433→
   434→  return (
   435→    <div className="mx-auto max-w-[1180px] space-y-5 px-4 py-6">
   436→      <Crumb
   437→        items={[
   438→          { label: '首页', view: { name: 'home' } },
   439→          { label: novel?.title ?? '…', view: { name: 'book', novelId } },
   440→          { label: '章节目录' },
   441→        ]}
   442→        navigate={navigate}
   443→      />
   444→      {/* 最新章节（全书倒数 12 章，新→旧） */}
   445→      {chapters && chapters.length > 0 && (
   446→        <section className="rounded-[10px] border border-[#e6edf7] bg-white p-6 shadow-[0_1px_3px_rgba(37,99,235,0.06)]">
   447→          <SectionTitle
   448→            title="最新章节"
   449→            small="最近更新 12 章 · 新→旧"
   450→            action={
   451→              <button
   452→                onClick={() => navigate({ name: 'book', novelId })}
   453→                className="shrink-0 cursor-pointer rounded-[10px] border-[1.5px] border-[#dbe4f0] bg-white px-5 py-2 text-[13px] text-[#1e293b] transition-colors duration-200 hover:border-[#2563eb] hover:text-[#2563eb]"
   454→              >
   455→                返回书页
   456→              </button>
   457→            }
   458→          />
   459→          <ChapterPills chapters={[...chapters].slice(-12).reverse()} navigate={navigate} />
   460→        </section>
   461→      )}
   462→
   463→      <section className="rounded-[10px] border border-[#e6edf7] bg-white p-6 shadow-[0_1px_3px_rgba(37,99,235,0.06)]">
   464→        <SectionTitle
   465→          title={novel ? `${novel.title} · 章节目录` : '章节目录'}
   466→          small={chapters ? `共 ${chapters.length} 章` : undefined}
   467→          action={
   468→            <button
   469→              onClick={() => navigate({ name: 'book', novelId })}
   470→              className="shrink-0 cursor-pointer rounded-[10px] border-[1.5px] border-[#dbe4f0] bg-white px-5 py-2 text-[13px] text-[#1e293b] transition-colors duration-200 hover:border-[#2563eb] hover:text-[#2563eb]"
   471→            >
   472→              返回书页
   473→            </button>
   474→          }
   475→        />
   476→        {isPending ? (

   796→function SearchPanel({ query, navigate }: { query: string; navigate: Nav }) {
