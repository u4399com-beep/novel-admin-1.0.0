   234→
   235→  if (home.isError || !home.data) {
   236→    return (
   237→      <Container className="mt-4">

   239→      </Container>
   240→    )
   241→  }
   242→  if (updates.isError) {
   243→    return (
   244→      <Container className="mt-4">
   245→        <GErrorBox onRetry={() => updates.refetch()} />
   246→      </Container>
   247→    )
   248→  }
   249→
   250→  const d = home.data
   251→  /* 源站首页「热门小说推荐」为 2×2 四张封面简介卡（ggd66 实测），而非 6 张 */
   252→  const hotItems = d.hot.slice(0, 4)
   253→  const clickRank = d.rankings.clicks.slice(0, 12)
   254→  const updateRows = (updates.data?.list ?? []).slice(0, 30)
   255→  const sideNews = dedupMerge(
   256→    [d.featured, d.hot, d.rankings.clicks, d.rankings.updates, d.rankings.finished],
   257→    30,
   258→  )
   259→  const friendLinks = dedupMerge(
   260→    [d.rankings.updates, d.rankings.finished, d.featured],
   261→    12,
   262→  )
   263→
   264→  return (
   265→    <Container className="mt-1">
   266→      <div className="grid gap-5 md:grid-cols-[minmax(0,73%)_minmax(0,1fr)] md:gap-[2%]">
   267→        {/* 第一行左：热门小说推荐（2 列封面简介卡） */}
   268→        <section>
   269→          <GH2>热门小说推荐</GH2>
   270→          <div className="grid gap-x-5 gap-y-4 sm:grid-cols-2">
   271→            {hotItems.map((n) => (
   272→              <HotItem key={n.id} novel={n} navigate={navigate} />
   273→            ))}
   274→          </div>
   275→        </section>
   276→
   277→        {/* 第一行右：搜索框 + 阅读排行榜 */}
   278→        <aside>
   279→          <SearchBox navigate={navigate} />
   280→          <GH2 className="mt-4">阅读排行榜</GH2>
   281→          <div>
   282→            {clickRank.map((n) => (
   283→              <RankRow key={n.id} novel={n} navigate={navigate} />
   284→            ))}
   285→          </div>
   286→        </aside>
   287→
   288→        {/* 第二行左：最近更新（五列字段表） */}
   289→        <section>
   290→          <GH2>最近更新</GH2>
   291→          <div>
   292→            {updateRows.map((n) => (
   293→              <UpdateRow key={n.id} novel={n} navigate={navigate} />
   294→            ))}
   295→          </div>
   296→        </section>
   297→
   298→        {/* 第二行右：最新小说（两字段行榜单） */}
   299→        <aside>
   300→          <GH2>最新小说</GH2>
   301→          <div>
   302→            {sideNews.map((n) => (
   303→              <RankRow key={n.id} novel={n} navigate={navigate} />

   305→          </div>
   306→        </aside>
   307→      </div>
   308→
   309→      {/* 友情链接（源站首页底部同款单行文字链；无外站可交换，用站内书页入口填充） */}

   317→              className={linkCls}
   318→            >
   319→              {n.title}
   320→            </button>
   321→          ))}
   322→          <button onClick={() => navigate({ name: 'category' })} className={linkCls}>
   323→            全部分类
   324→          </button>
   325→        </span>
   326→      </section>
   327→    </Container>
   328→  )
   329→}

   331→/* ==================== 分类页（分类导航条 + 3 列虚线盒 + 数字分页） ==================== */
   332→
   333→export function Category({
   334→  navigate,
   335→  categoryId,
   336→  page = 1,
   337→}: ViewProps & { categoryId?: number; page?: number }) {
   338→  const cats = useCategories()
   339→  const list = useNovels({ categoryId, page, pageSize: 20 })
   340→  const catName = cats.data?.find((c) => c.id === categoryId)?.name
   341→
   342→  const navCls = (active: boolean) =>
   343→    active
   344→      ? 'grid place-items-center h-[38px] rounded-[3px] bg-[#56ccb5] text-[15px] text-white transition-colors duration-200 cursor-pointer'
   345→      : 'grid place-items-center h-[38px] cursor-pointer rounded-[3px] text-[15px] text-[#00886d] transition-colors duration-200 hover:bg-[#cdf3eb] hover:text-[#f50]'
   346→
   347→  return (
   348→    <Container className="mt-1">
   349→      {/* 分类导航条 */}
   350→      <nav className="mb-3 grid grid-cols-3 gap-1 rounded-[4px] border border-[#ccc] bg-white p-1 sm:grid-cols-4 md:grid-cols-8">
   351→        <button
   352→          onClick={() => navigate({ name: 'category', page: 1 })}
   353→          className={navCls(categoryId == null)}
   354→        >
   355→          全部
   356→        </button>
   357→        {(cats.data ?? []).map((c) => (
   358→          <button
   359→            key={c.id}
   360→            onClick={() => navigate({ name: 'category', categoryId: c.id, page: 1 })}
   361→            className={navCls(categoryId === c.id)}
   362→          >
   363→            {c.name}
   364→          </button>
   365→        ))}
   366→      </nav>
   367→
   368→      {/* 书籍列表 */}
   369→      <section className="rounded-[4px] border border-[#ccc] bg-white p-3">
   370→        <GH2 className="text-center">
   371→          {catName ?? '全部小说'}
   372→          {list.data ? <span className="ml-2 text-[13px] font-normal text-[#999]">共 {list.data.total} 本</span> : null}
   373→        </GH2>
   374→
   375→        {list.isPending ? (
   376→          <GBoxSkeleton count={6} />
   377→        ) : list.isError ? (
   378→          <GErrorBox onRetry={() => list.refetch()} />
   379→        ) : (list.data?.list.length ?? 0) === 0 ? (
   380→          <GEmptyBox text="该分类下暂无收录小说" />
   381→        ) : (
   382→          <>
   383→            <div className="grid gap-4 pt-2 sm:grid-cols-2 lg:grid-cols-3">
   384→              {list.data!.list.map((n, i) => (
   385→                <BookBoxItem
   386→                  key={n.id}
   387→                  novel={n}
   388→                  index={(list.data!.page - 1) * (list.data!.pageSize ?? 20) + i + 1}
   389→                  navigate={navigate}
   390→                />
   391→              ))}
   392→            </div>
   393→            <GPager
   394→              page={list.data!.page}
   395→              totalPages={list.data!.totalPages}
   396→              onGo={(p) => navigate({ name: 'category', categoryId, page: p })}
   397→            />
   398→          </>
   399→        )}
   400→      </section>
   401→    </Container>
   402→  )
   403→}

   793→        navigate({ name: 'toc', novelId: ch.novelId })
   794→      } else if (e.key === 'ArrowLeft' && ch.prevId) {
   795→        navigate({ name: 'chapter', chapterId: ch.prevId })
   796→      } else if (e.key === 'ArrowRight' && ch.nextId) {
   797→        navigate({ name: 'chapter', chapterId: ch.nextId })
   798→      }
   799→    }
   800→    window.addEventListener('keydown', onKey)
   801→    return () => window.removeEventListener('keydown', onKey)

   806→      <Container className="mt-1">

   824→      </Container>
   825→    )
   826→  }
   827→
   828→  const paragraphs = ch.content.split('\n').map((s) => s.trim()).filter(Boolean)
   829→  const scene = SCENES[prefs.scene] ?? SCENES.day
   830→
   831→  const navBtnCls =
   832→    'cursor-pointer rounded-[4px] border border-[#ccc] bg-white py-2 text-[15px] text-[#333] transition-colors duration-200 hover:border-[#56ccb5] hover:text-[#00886d] disabled:pointer-events-none disabled:opacity-40'

   835→    <Container className="mt-1">
   836→      <GBreadcrumb
   837→        items={[
   838→          { label: '首页', view: { name: 'home' } },
   839→          ...(novel.data
   840→            ? [
   841→                {
   842→                  label: novel.data.categoryName,
   843→                  view: {

   924→/* ==================== 书签（localStorage 按书分组持久化，与其他主题行为一致） ==================== */
   925→
   926→function loadMarks(novelId: number): number[] {
   927→  if (typeof window === 'undefined') return []
   928→  try {
   929→    return JSON.parse(window.localStorage.getItem(`ggd66-marks-${novelId}`) ?? '[]') as number[]
   930→  } catch {
   931→    return []
   932→  }
   933→}
   934→
   935→function toggleMark(novelId: number, chapterId: number): boolean {
   936→  if (typeof window === 'undefined') return false
   937→  try {
   938→    const arr = loadMarks(novelId)
   939→    const has = arr.includes(chapterId)
   940→    window.localStorage.setItem(
   941→      `ggd66-marks-${novelId}`,
   942→      JSON.stringify(has ? arr.filter((x) => x !== chapterId) : [...arr.slice(-199), chapterId]),
   943→    )
   944→    return !has
   945→  } catch {
   946→    return false
   947→  }
   948→}
