   670→function toggleShelf(id: number): boolean {
   671→  if (typeof window === 'undefined') return false
   672→  try {
   673→    const arr = loadShelf()
   674→    const has = arr.includes(id)
   675→    window.localStorage.setItem(SHELF_KEY, JSON.stringify(has ? arr.filter((x) => x !== id) : [...arr, id]))
   676→    return !has
   677→  } catch {
   678→    return false
   679→  }
