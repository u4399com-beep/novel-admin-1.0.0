   229→          </div>
   230→        </div>
   231→      </Container>
   232→    )
   233→  }
   234→
   235→  if (home.isError || !home.data) {
   236→    return (
   237→      <Container className="mt-4">
   238→        <GErrorBox onRetry={() => home.refetch()} />
   239→      </Container>
   240→    )
   241→  }
   242→  if (updates.isError) {
   243→    return (
   244→      <Container className="mt-4">
   245→        <GErrorBox onRetry={() => updates.refetch()} />
   246→      </Container>
   247→    )
   248→  }
   249→
   250→  const d = home.data
   251→  const hotItems = d.hot.slice(0, 6)
   252→  const clickRank = d.rankings.clicks.slice(0, 12)
   253→  const updateRows = (updates.data?.list ?? []).slice(0, 30)
   254→  const sideNews = dedupMerge(
   255→    [d.featured, d.hot, d.rankings.clicks, d.rankings.updates, d.rankings.finished],
   256→    30,
   257→  )
   258→  const friendLinks = dedupMerge(
   259→    [d.rankings.updates, d.rankings.finished, d.featured],
   260→    12,
   261→  )
   262→
   263→  return (
   264→    <Container className="mt-1">
   265→      <div className="grid gap-5 md:grid-cols-[minmax(0,73%)_minmax(0,1fr)] md:gap-[2%]">
   266→        {/* 第一行左：热门小说推荐（2 列封面简介卡） */}
   267→        <section>
   268→          <GH2>热门小说推荐</GH2>
   269→          <div className="grid gap-x-5 gap-y-4 sm:grid-cols-2">
   270→            {hotItems.map((n) => (
   271→              <HotItem key={n.id} novel={n} navigate={navigate} />
   272→            ))}
   273→          </div>
   274→        </section>
   275→
   276→        {/* 第一行右：搜索框 + 阅读排行榜 */}
   277→        <aside>
   278→          <SearchBox navigate={navigate} />
   279→          <GH2 className="mt-4">阅读排行榜</GH2>
   280→          <div>
   281→            {clickRank.map((n) => (
   282→              <RankRow key={n.id} novel={n} navigate={navigate} />
   283→            ))}
   284→          </div>
   285→        </aside>
   286→
   287→        {/* 第二行左：最近更新（五列字段表） */}
   288→        <section>
   289→          <GH2>最近更新</GH2>
   290→          <div>
   291→            {updateRows.map((n) => (
   292→              <UpdateRow key={n.id} novel={n} navigate={navigate} />
   293→            ))}
   294→          </div>
   295→        </section>
   296→
   297→        {/* 第二行右：最新小说（两字段行榜单） */}
   298→        <aside>
   299→          <GH2>最新小说</GH2>

   302→              <RankRow key={n.id} novel={n} navigate={navigate} />

   304→          </div>
   305→        </aside>
   306→      </div>
   307→

   322→            全部分类
   323→          </button>
   324→        </span>
   325→      </section>
   326→    </Container>
   327→  )
   328→}

   330→/* ==================== 分类页（分类导航条 + 3 列虚线盒 + 数字分页） ==================== */

   332→export function Category({
   333→  navigate,
   334→  categoryId,
   335→  page = 1,
   336→}: ViewProps & { categoryId?: number; page?: number }) {
   337→  const cats = useCategories()
   338→  const list = useNovels({ categoryId, page, pageSize: 20 })
   339→  const catName = cats.data?.find((c) => c.id === categoryId)?.name
   340→
   341→  const navCls = (active: boolean) =>
   342→    active
   343→      ? 'grid place-items-center h-[38px] rounded-[3px] bg-[#56ccb5] text-[15px] text-white transition-colors duration-200 cursor-pointer'
   344→      : 'grid place-items-center h-[38px] cursor-pointer rounded-[3px] text-[15px] text-[#00886d] transition-colors duration-200 hover:bg-[#cdf3eb] hover:text-[#f50]'
   345→
   346→  return (
   347→    <Container className="mt-1">
   348→      {/* 分类导航条 */}
   349→      <nav className="mb-3 grid grid-cols-3 gap-1 rounded-[4px] border border-[#ccc] bg-white p-1 sm:grid-cols-4 md:grid-cols-8">
   350→        <button
   351→          onClick={() => navigate({ name: 'category', page: 1 })}
   352→          className={navCls(categoryId == null)}
   353→        >
   354→          全部
   355→        </button>
   356→        {(cats.data ?? []).map((c) => (
   357→          <button
   358→            key={c.id}
   359→            onClick={() => navigate({ name: 'category', categoryId: c.id, page: 1 })}
   360→            className={navCls(categoryId === c.id)}
   361→          >
   362→            {c.name}
   363→          </button>
   364→        ))}
   365→      </nav>
   366→
   367→      {/* 书籍列表 */}
   368→      <section className="rounded-[4px] border border-[#ccc] bg-white p-3">
   369→        <GH2 className="text-center">
   370→          {catName ?? '全部小说'}
   371→          {list.data ? <span className="ml-2 text-[13px] font-normal text-[#999]">共 {list.data.total} 本</span> : null}
   372→        </GH2>
   373→
   374→        {list.isPending ? (
   375→          <GBoxSkeleton count={6} />
   376→        ) : list.isError ? (
   377→          <GErrorBox onRetry={() => list.refetch()} />
   378→        ) : (list.data?.list.length ?? 0) === 0 ? (
   379→          <GEmptyBox text="该分类下暂无收录小说" />
   380→        ) : (
   381→          <>
   382→            <div className="grid gap-4 pt-2 sm:grid-cols-2 lg:grid-cols-3">
   383→              {list.data!.list.map((n, i) => (
   384→                <BookBoxItem
   385→                  key={n.id}
   386→                  novel={n}
   387→                  index={(list.data!.page - 1) * (list.data!.pageSize ?? 20) + i + 1}
   388→                  navigate={navigate}
   389→                />
   390→              ))}
   391→            </div>
   392→            <GPager
   393→              page={list.data!.page}
   394→              totalPages={list.data!.totalPages}
   395→              onGo={(p) => navigate({ name: 'category', categoryId, page: p })}
   396→            />
   397→          </>
   398→        )}
   399→      </section>
   400→    </Container>
   401→  )
   402→}

   776→export function Chapter({ navigate, chapterId }: ViewProps & { chapterId: number }) {
   777→  const { data: ch, isPending, isError, refetch } = useChapter(chapterId)
   778→  const novel = useNovel(ch?.novelId)
   779→  const [prefs] = useReaderPrefs()
   780→
   781→  useEffect(() => {
   782→    window.scrollTo(0, 0)
   783→  }, [chapterId])
   784→
   785→  /* 键盘翻页：Enter 返回书目 / ← 上一章 / → 下一章（焦点在输入框/下拉/按钮上时不触发，避免误触与双重导航） */
   786→  useEffect(() => {
   787→    if (!ch) return
   788→    const onKey = (e: KeyboardEvent) => {
   789→      const t = e.target as HTMLElement | null
   790→      if (t && (t.tagName === 'INPUT' || t.tagName === 'TEXTAREA' || t.tagName === 'SELECT' || t.tagName === 'BUTTON' || t.isContentEditable)) return
   791→      if (e.key === 'Enter') {
   792→        navigate({ name: 'toc', novelId: ch.novelId })
   793→      } else if (e.key === 'ArrowLeft' && ch.prevId) {
   794→        navigate({ name: 'chapter', chapterId: ch.prevId })
   795→      } else if (e.key === 'ArrowRight' && ch.nextId) {
   796→        navigate({ name: 'chapter', chapterId: ch.nextId })
   797→      }
   798→    }
   799→    window.addEventListener('keydown', onKey)
   800→    return () => window.removeEventListener('keydown', onKey)
   801→  }, [ch, navigate])
   802→
   803→  if (isPending) {
   804→    return (
   805→      <Container className="mt-1">
   806→        <GSkel className="mb-3 h-10 w-full" />
   807→        <div className="rounded-[4px] border border-[#ccc] bg-[#FBF4EC] p-4">
   808→          <GSkel className="mx-auto h-7 w-1/2" />
   809→          <div className="mt-6 space-y-4">
   810→            {Array.from({ length: 9 }, (_, i) => (
   811→              <GSkel key={i} className="h-5 w-full" />
   812→            ))}
   813→          </div>
   814→        </div>
   815→      </Container>
   816→    )
   817→  }
   818→
   819→  if (isError || !ch) {
   820→    return (
   821→      <Container className="mt-4">
   822→        <GErrorBox onRetry={() => refetch()} />
   823→      </Container>
   824→    )
   825→  }
   826→
   827→  const paragraphs = ch.content.split('\n').map((s) => s.trim()).filter(Boolean)
   828→  const scene = SCENES[prefs.scene] ?? SCENES.day
   829→
   830→  const navBtnCls =
   831→    'cursor-pointer rounded-[4px] border border-[#ccc] bg-white py-2 text-[15px] text-[#333] transition-colors duration-200 hover:border-[#56ccb5] hover:text-[#00886d] disabled:pointer-events-none disabled:opacity-40'
   832→
   833→  return (
   834→    <Container className="mt-1">
   835→      <GBreadcrumb
   836→        items={[
   837→          { label: '首页', view: { name: 'home' } },
   838→          ...(novel.data
   839→            ? [
   840→                {
   841→                  label: novel.data.categoryName,
   842→                  view: {
   843→                    name: 'category' as const,
   844→                    categoryId: novel.data.categoryId,
   845→                    page: 1,
   846→                  },
   847→                },
   848→              ]
   849→            : []),
   850→          { label: ch.novelTitle, view: { name: 'book', novelId: ch.novelId } },
   851→          { label: ch.title },
   852→        ]}
   853→        navigate={navigate}
   854→      />
   855→
   856→      {/* 阅读卡（米黄底，阅读场景换肤） */}
   857→      <article className="rounded-[4px] border p-3 transition-colors" style={{ background: scene.paper, borderColor: scene.line }}>
   858→        <h1 className="px-2 py-3 text-center text-[20px] font-bold md:text-[26px]" style={{ color: scene.ink }}>
   859→          {ch.title}
   860→        </h1>
   861→        <div className="pb-2 text-center">
   862→          <MarkButton key={`${ch.novelId}-${ch.id}`} novelId={ch.novelId} chapterId={ch.id} />
   863→        </div>
   864→        <SettingsBar />
   865→        <div
   866→          className="break-words border-t px-3 py-2 tracking-[0.1em]"
   867→          style={{
   868→            borderColor: scene.line,
   869→            fontSize: prefs.fontSize,
   870→            lineHeight: prefs.lineHeight,
   871→            fontFamily: readerFontStack(prefs.font),
   872→            color: readerInk(prefs, scene),
   873→          }}
   874→        >

   882→        {/* 翻页导航：上一章 / 书页目录 / 下一章（移动端换行堆叠） */}
   883→        <div className="flex flex-wrap justify-center gap-2 px-2 pt-3 pb-1">

   923→/* ==================== 书签（localStorage 按书分组持久化，与其他主题行为一致） ==================== */
   924→
   925→function loadMarks(novelId: number): number[] {
   926→  if (typeof window === 'undefined') return []
   927→  try {
   928→    return JSON.parse(window.localStorage.getItem(`ggd66-marks-${novelId}`) ?? '[]') as number[]
   929→  } catch {
   930→    return []
   931→  }
   932→}
   933→
   934→function toggleMark(novelId: number, chapterId: number): boolean {
   935→  if (typeof window === 'undefined') return false
   936→  try {
   937→    const arr = loadMarks(novelId)
   938→    const has = arr.includes(chapterId)
   939→    window.localStorage.setItem(
   940→      `ggd66-marks-${novelId}`,
   941→      JSON.stringify(has ? arr.filter((x) => x !== chapterId) : [...arr.slice(-199), chapterId]),
   942→    )
   943→    return !has
   944→  } catch {
   945→    return false
   946→  }
   947→}

   952→  return (
   953→    <button
   954→      onClick={() => setMarked(toggleMark(novelId, chapterId))}
   955→      className="cursor-pointer rounded-full border border-[#ffb0b4] bg-white/70 px-4 py-[3px] text-[12px] text-[#bf2c24] transition-colors duration-200 hover:border-[#f50] hover:text-[#f50]"
