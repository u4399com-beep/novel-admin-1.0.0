   155→        {catName ?? '小说书库'}
   156→        {cats.data ? (
   157→          <span className="ml-3 text-[13px] font-normal text-[#94a3b8]">
   158→            {cats.data.find((c) => c.id === categoryId)?.novelCount ?? list.data?.total ?? ''}
   159→          </span>
   160→        ) : null}
   161→      </h1>
   162→

   167→          className={
   168→            categoryId == null
   169→              ? 'cursor-pointer rounded-full bg-[#2563eb] px-4 py-1.5 text-[13px] text-white transition-colors duration-200'
   170→              : 'cursor-pointer rounded-full bg-[#f0f4fb] px-4 py-1.5 text-[13px] text-[#1e293b] transition-colors duration-200 hover:bg-[#e8f1ff] hover:text-[#2563eb]'

   181→                ? 'cursor-pointer rounded-full bg-[#2563eb] px-4 py-1.5 text-[13px] text-white transition-colors duration-200'

   419→    </div>
   420→  )

   425→export function Toc({ navigate, novelId }: ViewProps & { novelId: number }) {

   427→  const { data: chapters, isPending, isError, refetch } = useChapters(novelId)

   429→  return (

   435→          { label: '章节目录' },
   436→        ]}
   437→        navigate={navigate}
   438→      />
   439→      {/* 最新章节（全书倒数 12 章，新→旧） */}
   440→      {chapters && chapters.length > 0 && (
   441→        <section className="rounded-[10px] border border-[#e6edf7] bg-white p-6 shadow-[0_1px_3px_rgba(37,99,235,0.06)]">
   442→          <SectionTitle
   443→            title="最新章节"

   446→              <button
   447→                onClick={() => navigate({ name: 'book', novelId })}
   448→                className="shrink-0 cursor-pointer rounded-[10px] border-[1.5px] border-[#dbe4f0] bg-white px-5 py-2 text-[13px] text-[#1e293b] transition-colors duration-200 hover:border-[#2563eb] hover:text-[#2563eb]"
   449→              >
   450→                返回书页
   451→              </button>
   452→            }
   453→          />
   454→          <ChapterPills chapters={[...chapters].slice(-12).reverse()} navigate={navigate} />
   455→        </section>
   456→      )}
   457→
   458→      <section className="rounded-[10px] border border-[#e6edf7] bg-white p-6 shadow-[0_1px_3px_rgba(37,99,235,0.06)]">
   459→        <SectionTitle
   460→          title={novel ? `${novel.title} · 章节目录` : '章节目录'}
   461→          small={chapters ? `共 ${chapters.length} 章` : undefined}
   462→          action={
   463→            <button
   464→              onClick={() => navigate({ name: 'book', novelId })}
   465→              className="shrink-0 cursor-pointer rounded-[10px] border-[1.5px] border-[#dbe4f0] bg-white px-5 py-2 text-[13px] text-[#1e293b] transition-colors duration-200 hover:border-[#2563eb] hover:text-[#2563eb]"

   467→              返回书页
   468→            </button>
   469→          }
   470→        />
   471→        {isPending ? (
   472→          <PillSkeleton count={16} />
   473→        ) : isError ? (
   474→          <ErrorBox onRetry={() => refetch()} />
   475→        ) : (chapters?.length ?? 0) === 0 ? (

   481→    </div>

   791→function SearchPanel({ query, navigate }: { query: string; navigate: Nav }) {

   793→  const [page, setPage] = useState(1)
   794→  const list = useNovels({ q: query || undefined, page, pageSize: 20 })

   797→    const kw = input.trim()

   799→  }
   800→

   802→    <div className="space-y-5">

   813→          className="cursor-pointer rounded-[10px] bg-[#2563eb] px-8 text-[14px] text-white transition-colors duration-200 hover:bg-[#1d4ed8]"
   814→        >
   815→          搜索
   816→        </button>
   817→      </div>
   818→
   819→      <SectionTitle
   820→        title="搜索结果"
   821→        small={query ? `“${query}” · ${list.data?.total ?? 0} 条` : '请输入关键词'}

   825→        <CardGridSkeleton count={6} />

   827→        <ErrorBox onRetry={() => list.refetch()} />

   829→        <EmptyBox text={`没有找到与“${query}”相关的小说，换个关键词试试`} />

   834→              <BookTextCard key={n.id} novel={n} navigate={navigate} />

   836→          </div>
   837→          <Pager
   838→            page={list.data!.page}
   839→            totalPages={list.data!.totalPages}
   840→            onGo={(p) => setPage(p)}
   841→          />
   842→        </>

   844→    </div>
   845→  )
