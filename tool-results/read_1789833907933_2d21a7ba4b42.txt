    84→/**
    85→ * 僵尸任务恢复（心跳判定）：worker 常驻在 Next.js 进程内存里（fire-and-forget promise），
    86→ * 进程重启后内存任务丢失，DB 中 status=pending/running 的记录会永久卡死。
    87→ *
    88→ * 判定依据：运行中任务每完成一章/一书都会 flush 触碰 updatedAt（心跳）；真正的僵尸
    89→ * （进程消失/模块实例丢失）updatedAt 停摆。仅回收「仍处 pending/running 且心跳停摆
    90→ * 超过 ZOMBIE_STALE_MS」的任务——不按 createdAt/进程启动时刻判定，因为 dev 下
    91→ * Turbopack 每个路由 bundle 各有一份本模块实例，任何晚加载实例的时钟基线都不同，
    92→ * 按创建时刻会把「重跑老任务」等活任务误杀（实测踩坑）。
    93→ * 心跳最长间隙 = 引擎单请求预算 60s + 落库时间，远小于 5 分钟阈值。
    94→ *
    95→ * 防重入标志同步置位（await 前置位）：dev 下同进程可能存在多份模块实例
    96→ * （Turbopack 按路由分包），异步置位会留下双重回收的竞态窗口。
    97→ */
    98→const ZOMBIE_STALE_MS = 5 * 60_000
    99→
   100→async function recoverStaleTasks(): Promise<void> {
   101→  if (gWorker.__scrapeRecovered) return
   102→  gWorker.__scrapeRecovered = true // 同步置位：跨实例/跨 HMR 防重入
   103→  try {
   104→    const stale = await db.scrapeTask.findMany({
   105→      where: {
   106→        status: { in: ['pending', 'running'] },
   107→        updatedAt: { lt: new Date(Date.now() - ZOMBIE_STALE_MS) },
   108→      },
   109→      select: { id: true, log: true },
   110→    })
   111→    let recovered = 0
   112→    for (const t of stale) {
   113→      if (running.has(t.id)) continue // 本进程内存中仍有实例在执行（共享 Set 时双保险）
   114→      const line = `[${new Date().toTimeString().slice(0, 8)}] 心跳停摆超 5 分钟，判定僵尸任务并回收`
   115→      const log = `${t.log ? `${t.log}\n` : ''}${line}`
   116→      const res = await db.scrapeTask
   117→        .updateMany({
   118→          where: { id: t.id, status: { in: ['pending', 'running'] } },
   119→          data: {
   120→            status: 'failed',
   121→            message: '任务心跳超时，自动回收',
   122→            // 与 Run 同规约：只保留最近 100 行
   123→            log: log.split('\n').slice(-MAX_LOG_LINES).join('\n'),
   124→          },
   125→        })
   126→        .catch(() => null)
   127→      if (res && res.count > 0) recovered++
   128→    }
   129→    if (recovered > 0) console.log(`[scrape-worker] 僵尸任务回收: ${recovered} 条`)
   130→  } catch {
   131→    // 回收失败不影响模块加载；标志已置位，等待下次进程重启重试（代价可接受：僵尸多驻留一轮）
   132→  }
   133→}

   136→/** 协作式取消检查：记录不存在视为取消；DB 瞬时错误不误判为取消（fail-open） */
   137→async function isCanceled(taskId: number): Promise<boolean> {
   138→  const t = await db.scrapeTask
   139→    .findUnique({ where: { id: taskId }, select: { status: true } })
   140→    .catch(() => undefined)
   141→  if (t === undefined) return false // 查询失败 ≠ 被取消，避免瞬时 DB 错误误停任务
   142→  return !t || t.status === 'canceled'
   143→}

   145→/**
   146→ * 并行池：cursor 抢占式派发，concurrency 路 worker 持续取件直至耗尽。
   147→ * fn 内部通过 stop signal 提前收敛（停止派发，在途任务跑完当前单元后自然退出）。
   148→ */
   149→async function runPool<T>(items: T[], concurrency: number, fn: (item: T, index: number) => Promise<void>): Promise<void> {
   150→  let cursor = 0
   151→  const workers = Array.from({ length: Math.max(1, Math.min(concurrency, items.length)) }, async () => {
   152→    for (;;) {
   153→      const i = cursor++
   154→      if (i >= items.length) return
   155→      await fn(items[i], i)
   156→    }
   157→  })
   158→  await Promise.all(workers)
   159→}

   161→/** 生成第 k 页候选 URL：?page=k（已有 query 则 &page=k）与 /page/k 两种变体 */
   162→function pageVariants(url: string, k: number): string[] {
   163→  const out: string[] = []
   164→  try {
   165→    const u = new URL(url)
   166→    u.searchParams.set('page', String(k))
   167→    out.push(u.toString())
   168→    const p = new URL(url)
   169→    p.pathname = `${p.pathname.replace(/\/+$/, '')}/page/${k}`
   170→    p.search = ''
   171→    if (p.toString() !== u.toString()) out.push(p.toString())
   172→  } catch {
   173→    out.push(url.includes('?') ? `${url}&page=${k}` : `${url}?page=${k}`)
   174→    out.push(`${url.replace(/\/+$/, '')}/page/${k}`)
   175→  }
   176→  return [...new Set(out)]
   177→}

   181→/**
   182→ * Phase 1：抓取书页+完整目录 → 书籍 upsert（含封面落盘/下拉词绑定）→ 章节目录骨架入库。
   183→ *
   184→ * 骨架入库语义（与 Phase 2 去重逻辑配合，支持重跑续采）：
   185→ * - 已存在且有内容（wordCount>0）→ skip 占位任务（不重抓，进度照常推进）；
   186→ * - 已存在空骨架（wordCount=0，上次任务未抓到）→ 保持骨架，Phase 2 重抓填充；
   187→ * - 不存在 → 创建骨架（content=''、wordCount=0、idx=目录位次保序）。
   188→ * 已有章节盘点只取 id/idx/title/volume/wordCount（绝不 select content 全文——
   189→ * 用 wordCount 区分内容章节与骨架，正常入库章节 wordCount>0 恒成立）。
   190→ *
   191→ * 骨架批量入库：createMany 快路径；整批失败（并发双任务同书撞 (novelId,idx) 等）时
   192→ * 逐条 create 兜底，createChapterSkeleton 撞唯一约束查回已有记录按「已存在」处理。
   193→ */
   194→async function processBookMeta(
   195→  run: Run,
   196→  bookUrl: string,
   197→  rule: LoadedRule,
   198→  opts: { trackTotal: boolean; referer?: string | null },
   199→): Promise<BookMetaOutcome> {
   200→  // 失败语义三分（P1-4）：canceled=用户取消/记录删除（停整个任务）；
   201→  // hard=DB 记录级硬失败（list 跳过继续、single 按 failed 收尾，不再借用 canceled 通道）；
   202→  // plain=普通书籍级失败（书页提取失败/空标题等，处理同 hard）
   203→  const fail = (message: string, kind: 'plain' | 'canceled' | 'hard' = 'plain'): BookMetaOutcome => ({
   204→    ok: false,
   205→    canceled: kind === 'canceled',
   206→    hardFailed: kind === 'hard',
   207→    novelId: 0,
   208→    refsTotal: 0,
   209→    skipped: 0,
   210→    jobs: [],
   211→    message,
   212→  })
   213→
   214→  run.log(`抓取书页 ${bookUrl.slice(0, 120)}…`)
   215→  // Referer 链：list 模式传「发现本书的列表页」作来路；single 模式缺省由引擎回落站内首页。
   216→  // 熔断感知：书页撞上熔断（站点临时限流封禁）时等待冷却后自动重试，而非立即把本书记失败
   217→  const page = await withCircuitRetry(hostOf(bookUrl), () => fetchBookPage(run, bookUrl, rule, opts.referer ?? undefined), {
   218→    detect: (r) => (r.ok ? null : circuitWaitMs(r.error, r.circuitRetryAfterMs)),
   219→    isAlive: async () => !(await isCanceled(run.taskId)),
   220→    onWait: (ms, cycle, max) =>
   221→      run.log(`目标主机熔断冷却中，暂停 ${Math.round(ms / 1000)}s 后自动重试书页（第 ${cycle}/${max} 轮，等待期间不向源站发请求）`),
   222→  })
   223→  if (!page.ok) {
   224→    run.log(`书页提取失败: ${page.error}`)
   225→    return fail(page.error)
   226→  }
   227→  const book = page.book
   228→  // 引擎响应未经 schema 校验（callEngine 直接 as 断言），关键字段做形态兜底，
   229→  // 防畸形载荷（引擎版本错位等）在任务中途抛 TypeError 把整任务拖成 failed
   230→  if (typeof book.title !== 'string') book.title = ''
   231→  if (!Array.isArray(book.chapters)) book.chapters = []
   232→  if (typeof book.description !== 'string') book.description = ''
   233→  if (typeof book.author !== 'string') book.author = ''
   234→  if (typeof book.category !== 'string') book.category = ''
   235→  if (typeof book.status !== 'string') book.status = ''
   236→  if (typeof book.chapterCount !== 'number') book.chapterCount = book.chapters.length
   237→  // 全字段兜底清洗（引擎侧已清洗，此处防版本错位/旧引擎，幂等成本低）：
   238→  // 实体解码（&#091；类全角分号变体）、字面 \n 还原、残缺标签剥除、简介样板句/SEO 伪简介处理
   239→  book.title = cleanTextField(book.title)
   240→  book.author = cleanTextField(book.author)
   241→  book.description = cleanDescriptionField(book.description)
   242→  book.category = cleanTextField(book.category)
   243→  book.status = cleanTextField(book.status)
   244→  run.log(
   245→    `书页提取成功：《${book.title.slice(0, 40)}》${book.author ? ` / ${book.author.slice(0, 20)}` : ''}，章节链接 ${book.chapterCount} 条`,
   246→  )
   247→
   248→  // ---- 完整目录页二次提取（bookRule.catalogLinkSelector，如 23qb 新模板书页仅含最新几章）----
   249→  let allRefs = book.chapters
   250→  const catalogSel = rule.bookRule.catalogLinkSelector
   251→  if (typeof catalogSel === 'string' && catalogSel) {
   252→    if (await isCanceled(run.taskId)) return fail('任务已取消', 'canceled')
   253→    if (book.catalogUrl) {
   254→      run.log(`发现完整目录页 ${book.catalogUrl.slice(0, 100)}，尝试整目提取…`)
   255→      const catalogUrl = book.catalogUrl // 闭包捕获需局部常量（TS 收窄不跨回调）
   256→      // 熔断感知：目录页撞熔断时等待冷却后自动重试（超限时按旧语义回退书页章节链接）
   257→      const catalog = await withCircuitRetry(
   258→        hostOf(catalogUrl),
   259→        () => fetchCatalogChapters(run, catalogUrl, rule, bookUrl),
   260→        {
   261→          detect: (r) => (r.error ? circuitWaitMs(r.error) : null),
   262→          isSuccess: (r) => r.error === null,
   263→          isAlive: async () => !(await isCanceled(run.taskId)),
   264→          onWait: (ms, cycle, max) =>
   265→            run.log(`目标主机熔断冷却中，暂停 ${Math.round(ms / 1000)}s 后自动重试目录页（第 ${cycle}/${max} 轮）`),
   266→        },
   267→      )
   268→      const catalogRefs = catalog.refs
   269→      if (catalogRefs.length > allRefs.length) {
   270→        run.log(`目录页提取到 ${catalogRefs.length} 条章节链接（书页仅 ${allRefs.length} 条），采用目录页结果`)
   271→        allRefs = catalogRefs
   272→      } else if (catalogRefs.length > 0) {
   273→        run.log(`目录页提取 ${catalogRefs.length} 条不多于书页 ${allRefs.length} 条，维持书页结果`)
   274→      }
   275→    } else {
   276→      run.log(`catalogLinkSelector "${catalogSel.slice(0, 60)}" 在书页无命中，仅用书页章节链接`)
   277→    }
   278→  }
   279→
   280→  if (await isCanceled(run.taskId)) return fail('任务已取消', 'canceled')
   281→
   282→  // ---- 乱序重排（充分考虑分卷：同号不同卷不跨卷错排，卷名随章节移动）----
   283→  // 源站目录常见「最新章节块（新→旧）+ 正文块（旧→新）」或整本倒序；直接按抓取序编 idx 会错乱。
   284→  // 引擎响应未经 schema 校验，逐条形态兑底后再重排。
   285→  allRefs = (Array.isArray(allRefs) ? allRefs : []).map((c) => ({
   286→    title: typeof c?.title === 'string' ? c.title : '',
   287→    url: typeof c?.url === 'string' ? c.url : null,
   288→    volume: typeof c?.volume === 'string' && c.volume.trim() ? c.volume.trim().slice(0, 50) : undefined,
   289→  })) as ChapterRef[]
   290→  const rr = reorderChapterRefs(allRefs)
   291→  if (rr.reordered) {
   292→    allRefs = rr.refs
   293→    run.log(`目录乱序重排：${rr.note}`)
   294→  }
   295→
   296→  // ---- 分类 ----
   297→  let categoryId: number
   298→  try {
   299→    categoryId = await ensureCategory(book.category)
   300→  } catch (e) {
   301→    const msg = e instanceof Error ? e.message : '分类处理失败'
   302→    run.log(msg)
   303→    return fail(msg)
   304→  }
   305→
   306→  // ---- 书籍 upsert（title+author 查重；DB 层 @@unique([title,author]) 兜底并发）----
   307→  const up = await upsertBook(run, book, categoryId, rule.proxy)
   308→  if (!up.ok) {
   309→    // P1-4：记录级 DB 失败（up.hard=true，如书籍在采集中被删除）按书籍级硬失败处理，
   310→    // list 模式不再停整个任务（旧实现借 canceled 通道把整任务收成 canceled，语义失真）；
   311→    // 书籍级普通失败（空标题等）同样只算本书失败
   312→    return fail(up.message, up.hard ? 'hard' : 'plain')
   313→  }
   314→  // 采集的同时取搜索引擎下拉词 → 绑定书籍 + 生成 PSEO 书籍页（fire-and-forget）
   315→  scheduleBindBookSuggestKeywords(run, up.novelId, up.title || book.title, up.createdNew)
   316→
   317→  // ---- 章节列表准备 ----
   318→  const refs = allRefs.filter(
   319→    (c): c is ChapterRef & { url: string } => !!c && typeof c.url === 'string' && !!c.url,
   320→  )
   321→  if (refs.length === 0) {
   322→    run.log('未提取到任何有效章节链接')
   323→    return { ok: true, canceled: false, hardFailed: false, novelId: up.novelId, refsTotal: 0, skipped: 0, jobs: [], message: '书籍已入库（未提取到章节链接）' }
   324→  }
   325→  // 同目录内预去重（骨架入库前完成）：按清洗后标题优先、URL 兜底去重，防同批重复入库
   326→  const seenRef = new Set<string>()
   327→  const uniqueRefs: (ChapterRef & { url: string })[] = []
   328→  for (const c of refs) {
   329→    const key = c.title.trim() ? `t:${cleanTextField(c.title)}` : `u:${c.url}`
   330→    if (seenRef.has(key)) continue
   331→    seenRef.add(key)
   332→    uniqueRefs.push(c)
   333→  }
   334→  if (uniqueRefs.length < refs.length) run.log(`目录内去重：${refs.length} → ${uniqueRefs.length} 条（无数量上限，全量采集）`)
   335→
   336→  if (await isCanceled(run.taskId)) return fail('任务已取消', 'canceled')
   337→
   338→  // ---- 已有章节盘点（不含 content 全文；wordCount 区分内容章节与骨架） ----
   339→  const existingChapters = await db.chapter
   340→    .findMany({
   341→      where: { novelId: up.novelId },
   342→      select: { id: true, idx: true, title: true, volume: true, wordCount: true },
   343→    })
   344→    .catch(() => [] as { id: number; idx: number; title: string; volume: string; wordCount: number }[])
   345→  // 去重键与入库标题共用同一清洗规约（cleanTextField 幂等）：存量脏标题与本次净标题可命中比对，
   346→  // 避免清洗规约升级后同一章以两种写法重复入库
   347→  const titleGroups = new Map<string, { id: number; idx: number; wordCount: number }[]>()
   348→  for (const c of existingChapters) {
   349→    const k = cleanTextField(c.title)
   350→    const arr = titleGroups.get(k)
   351→    if (arr) arr.push({ id: c.id, idx: c.idx, wordCount: c.wordCount })
   352→    else titleGroups.set(k, [{ id: c.id, idx: c.idx, wordCount: c.wordCount }])
   353→  }
   354→  const maxAgg = await db.chapter
   355→    .aggregate({ where: { novelId: up.novelId }, _max: { idx: true } })
   356→    .catch(() => ({ _max: { idx: null as number | null } }))
   357→  // 骨架槽位基准：现有最大 idx+1；每条按目录位次预分配（base+位次），保序且并行安全
   358→  const baseIdx = (maxAgg._max.idx ?? 0) + 1
   359→
   360→  // ---- 目录逐条分类：skip（已有内容）/ fill（已有空骨架或新骨架）/ 骨架待建 ----
   361→  const jobs: ChapterFillJob[] = []
   362→  /** 待创建骨架（jobIndex 回指 jobs 中对应的 fill 任务，创建结果落地后回填定位） */
   363→  const pendingSkeletons: { jobIndex: number; idx: number; title: string; volume: string; refTitle: string }[] = []
   364→  /** 已存在章节的卷名回填队列（title → volume；仅唯一标题且存量为空卷时写入） */
   365→  const volumeBackfill = new Map<string, string>()
   366→  let skipped = 0
   367→
   368→  for (let position = 0; position < uniqueRefs.length; position++) {
   369→    const ref = uniqueRefs[position]
   370→    const refTitle = cleanTextField(String(ref.title ?? '')).slice(0, 200)
   371→    if (refTitle && titleGroups.has(refTitle)) {
   372→      const matches = titleGroups.get(refTitle)!
   373→      if (matches.some((m) => m.wordCount > 0)) {
   374→        // 已存在且有内容：跳过不重抓（Phase 2 仅作进度占位）
   375→        skipped++
   376→        jobs.push({ kind: 'skip', novelId: up.novelId, chapterId: matches[0].id, idx: matches[0].idx, url: '', refTitle, volume: '' })
   377→        // 分卷补齐：本次源站目录携带卷名、存量章节为空卷且标题在全书中唯一时，回填卷名
   378→        const rv = ref.volume
   379→        if (rv && matches.length === 1 && !volumeBackfill.has(refTitle)) volumeBackfill.set(refTitle, rv)
   380→        continue
   381→      }
   382→      // 已存在空骨架（上次任务未抓到内容）：保持骨架不重建，Phase 2 重抓填充（不跳过）
   383→      const target = matches.reduce((a, b) => (a.idx <= b.idx ? a : b))
   384→      jobs.push({ kind: 'fill', novelId: up.novelId, chapterId: target.id, idx: target.idx, url: ref.url, refTitle, volume: ref.volume ?? '' })
   385→      continue
   386→    }
   387→    // 不存在 → 骨架占位（idx=目录位次预分配保序），创建结果落地后回填 chapterId
   388→    const idx = baseIdx + position
   389→    pendingSkeletons.push({ jobIndex: jobs.length, idx, title: refTitle || `第${idx}章`, volume: ref.volume ?? '', refTitle })
   390→    jobs.push({ kind: 'fill', novelId: up.novelId, chapterId: null, idx, url: ref.url, refTitle, volume: ref.volume ?? '' })
   391→  }
   392→
   393→  // ---- 骨架批量入库：createMany 快路径；整批失败（并发撞 (novelId,idx) 等）逐条 create 兜底 ----
   394→  if (pendingSkeletons.length > 0) {
   395→    let batchErr: unknown = null
   396→    try {
   397→      await db.chapter.createMany({
   398→        data: pendingSkeletons.map((s) => ({
   399→          novelId: up.novelId,
   400→          idx: s.idx,
   401→          title: s.title,
   402→          content: '',
   403→          wordCount: 0,
   404→          volume: s.volume,
   405→        })),
   406→      })
   407→    } catch (e) {
   408→      batchErr = e
   409→    }
   410→    if (batchErr !== null) {
   411→      // P1-3：整批撞 FK（书在采集中被删除，Novel←Chapter onDelete:Cascade）→ 立即中止本书，
   412→      // 不再逐条空转（旧行为对已删书逐章重试刷屏）
   413→      if (isRecordMissingError(batchErr)) {
   414→        run.log(`书籍已消失（采集中被删除），中止本书章节骨架入库`)
   415→        return fail('书籍已消失（采集中被删除）', 'hard')
   416→      }
   417→      // 并发撞 (novelId,idx) 等其他批量失败 → 逐条 create 兜底（createChapterSkeleton
   418→      // 撞唯一约束查回已有记录按「已存在」处理）
   419→      for (const s of pendingSkeletons) {
   420→        if (await isCanceled(run.taskId)) return fail('任务已取消', 'canceled')
   421→        const job = jobs[s.jobIndex]
   422→        const sk = await createChapterSkeleton(run, up.novelId, s.idx, { title: s.title, volume: s.volume }, s.refTitle)
   423→        if (!sk.ok) {
   424→          // P1-3：逐条也撞 FK → 书已消失，立即中止本书（不再把剩余全部骨架标记失败）
   425→          if (sk.missing) {
   426→            run.log(`书籍已消失（采集中被删除），中止本书章节骨架入库`)
   427→            return fail('书籍已消失（采集中被删除）', 'hard')
   428→          }
   429→          run.log(`骨架章节「${s.title.slice(0, 30)}」入库失败: ${sk.message.slice(0, 120)}`)
   430→          job.kind = 'create' // 无骨架可填：Phase 2 抓到正文后走 storeChapter 兜底新建
   431→          continue
   432→        }
   433→        job.idx = sk.idx
   434→        if (!sk.created) {
   435→          // 撞唯一约束查回已有记录 → 按「已存在」处理（并发双任务同书场景）
   436→          if (sk.existingHasContent) {
   437→            job.kind = 'skip'
   438→            job.url = ''
   439→            skipped++
   440→          }
   441→          job.chapterId = sk.existingId
   442→        }
   443→      }
   444→    }
   445→  }
   446→
   447→  // ---- 已存在章节的卷名回填（逐卷 updateMany，仅在唯一标题且存量空卷时入队）----
   448→  if (volumeBackfill.size > 0) {
   449→    let backfilled = 0
   450→    for (const [title, volume] of volumeBackfill) {
   451→      const res = await db.chapter
   452→        .updateMany({ where: { novelId: up.novelId, title, volume: '' }, data: { volume } })
   453→        .catch(() => null)
   454→      if (res) backfilled += res.count
   455→    }
   456→    if (backfilled > 0) run.log(`分卷回填：${backfilled} 个已存在章节补记卷名`)
   457→  }
   458→
   459→  // ---- 进度分母 ----
   460→  if (opts.trackTotal) {
   461→    // single 模式：total=骨架章节数（分母），done 随 Phase 2 逐章递增
   462→    //（flush false=任务记录已删，立即停）
   463→    if (!(await run.flush({ total: uniqueRefs.length }))) return fail('任务记录已删除', 'canceled')
   464→  }
   465→
   466→  const toFetch = jobs.length - skipped
   467→  run.log(`目录骨架就绪：${uniqueRefs.length} 章（已存在 ${skipped} 章，待抓取 ${toFetch} 章）`)
   468→  return { ok: true, canceled: false, hardFailed: false, novelId: up.novelId, refsTotal: uniqueRefs.length, skipped, jobs, message: '' }
   469→}
   470→
   471→// ==================== Phase 2：跨书批量章节内容填充 ====================
   472→
   473→/** Phase 2 单书上下文（统计就地回填，供书籍级成败判定与日志节流） */
   474→interface Phase2Book {
   475→  novelId: number
   476→  title: string
   477→  /** 章节页 Referer 来路（书页 URL，站点常见「书页→章节」导航校验） */
   478→  bookUrl: string
   479→  jobs: ChapterFillJob[]
   480→  filled: number
   481→  failed: number
   482→  /** P1-3：书在采集中被删除（P2025/FK 实证）→ 本本书剩余章节不再向源站发请求 */
   483→  vanished: boolean
   484→  /** 清洗日志节流：每本书最多记 3 条，防止日志爆炸 */
   485→  cleanLogCount: number
   486→}
   487→
   488→interface Phase2Result {
   489→  canceled: boolean
   490→  filled: number
   491→  failed: number
   492→  skipped: number
   493→  /** P1-3：采集中被删除的书籍数（finalize 消息提示用） */
   494→  vanishedBooks: number
   495→}
   496→
   497→/**
   498→ * Phase 2：把 Phase 1 收集的全部章节平铺（按书分组顺序），runPool 章节级并发
   499→ * 跨书批量抓取正文，update 填充骨架章节 content/wordCount/volume。
   500→ *
   501→ * - 抓取失败的骨架章节保留骨架不删（重跑任务时 wordCount=0 的骨架会被再次抓取填充）；
   502→ * - novel.wordCount increment 只对抓取成功的章节（fill 填充在此增量，create 走
   503→ *   storeChapter 内部增量），收尾对触及书籍 recalcNovelWordCount 全量校准；
   504→ * - P1-3：填充/入库撞 P2025/FK（书在采集中被删除）→ 实证本书已消失后立即中止本书
   505→ *   剩余章节（不再空抓源站），书籍级成败按失败计；
   506→ * - 保留全部既有机制：熔断感知重试/瞬时失败单次重试/协作式取消/flush 串行链/
   507→ *   入库前清洗/MAX_CONTENT_CHARS 截断。
   508→ * - 进度：single 模式 done 随本章递增（total=骨架章节数已在 Phase 1 写入）；
   509→ *   list 模式累加共享 chapterProgress（chaptersDone/chaptersTotal）。
   510→ */
   511→async function fillChapters(
   512→  run: Run,
   513→  books: Phase2Book[],
   514→  rule: LoadedRule,
   515→  opts: { trackTotal: boolean; chapterProgress?: { done: number; total: number }; concurrency: number },
   516→): Promise<Phase2Result> {
   517→  // 平铺（按书分组顺序）：任务单元携带所属书上下文（Referer 链/日志节流按书隔离）
   518→  const units: { job: ChapterFillJob; book: Phase2Book }[] = []
   519→  for (const b of books) for (const job of b.jobs) units.push({ job, book: b })
   520→
   521→  let filled = 0
   522→  let failed = 0
   523→  let skipped = 0
   524→  let done = 0 // single 模式主口径进度（done/total=章节）
   525→  let stopRun = false // 取消/记录删除 → 停止派发新章节，在途章节收尾后退出
   526→  const touchedNovels = new Set<number>()
   527→
   528→  /** P1-3：防误杀——update 撞 P2025 也可能是单章被单独删除，实证 novel 已不存在才判定整书消失
   529→   * （查询失败按仍存在处理，不因 DB 抖动中止本书） */
   530→  const isNovelGone = async (novelId: number): Promise<boolean> => {
   531→    const n = await db.novel.findUnique({ where: { id: novelId }, select: { id: true } }).catch(() => undefined)
   532→    return n === null
   533→  }
   534→
   535→  /** P1-3：标记本书已消失并中止剩余章节（只记一条日志防刷屏） */
   536→  const markVanished = (book: Phase2Book): void => {
   537→    if (book.vanished) return
   538→    book.vanished = true
   539→    run.log(`书籍《${book.title.slice(0, 30) || book.novelId}》已消失（采集中被删除），中止本书剩余章节抓取`)
   540→  }
   541→
   542→  const alive = async (): Promise<boolean> => {
   543→    if (stopRun) return false
   544→    return !(await isCanceled(run.taskId))
   545→  }
   546→
   547→  // flush 串行链：并行分支共享 Run 缓冲与进度字段，写盘串行化避免交错/竞态
   548→  let flushChain = Promise.resolve(true)
   549→  const scheduleFlush = (fields: TaskFlushFields): Promise<boolean> => {
   550→    flushChain = flushChain.then(async (ok) => {
   551→      if (!ok || stopRun) return false
   552→      const ok2 = await run.flush(fields)
   553→      if (!ok2) stopRun = true // 任务记录被删除 → 全池停止派发
   554→      return ok2
   555→    })
   556→    return flushChain
   557→  }
   558→
   559→  /** 每处理完一个章节后要刷出的进度字段（不含任务级成果计数） */
   560→  const progressFields = (): TaskFlushFields =>
   561→    opts.trackTotal
   562→      ? { done }
   563→      : {
   564→          chaptersDone: opts.chapterProgress?.done ?? 0,
   565→          chaptersTotal: opts.chapterProgress?.total ?? 0,
   566→        }
   567→
   568→  const processUnit = async ({ job, book }: { job: ChapterFillJob; book: Phase2Book }): Promise<void> => {
   569→    if (book.vanished) {
   570→      // P1-3：书已消失——剩余章节不再向源站发请求，按失败记账推进进度（骨架已随书级联删除）
   571→      failed++
   572→      book.failed++
   573→      done++
   574→      if (opts.chapterProgress) opts.chapterProgress.done++
   575→      return
   576→    }
   577→    if (job.kind === 'skip') {
   578→      // 已存在且有内容：不重抓，仅推进进度（与旧版「已存在，跳过」口径一致）
   579→      skipped++
   580→      done++
   581→      if (opts.chapterProgress) opts.chapterProgress.done++
   582→      run.log(`章节「${job.refTitle.slice(0, 30)}」已存在，跳过`)
   583→      await scheduleFlush(progressFields())
   584→      return
   585→    }
   586→
   587→    if (!(await alive())) {
   588→      stopRun = true
   589→      return
   590→    }
   591→    run.log(`抓取章节「${(job.refTitle || job.url).slice(0, 36)}」`)
   592→    // Referer 链：章节页带书页来路（站点常见「书页→章节」导航校验）；
   593→    // fetchChapterPaged：同章分页（…_2.html / …/2.html）自动翻页拼接，避免长章只存半页。
   594→    // 熔断感知：章节撞熔断（站点临时限流封禁）时同主机并发 worker 合流等待冷却后自动重试，
   595→    // 而非把剩余章节全部烧成失败（Task 36 复盘：ggd66 一次临时封禁曾致 232 章被永久记失败）
   596→    const circuitFetch = (): ReturnType<typeof fetchChapterPaged> =>
   597→      withCircuitRetry(hostOf(job.url), () => fetchChapterPaged(job.url, rule, book.bookUrl), {
   598→        detect: (r) => (r.ok ? null : circuitWaitMs(r.error, r.circuitRetryAfterMs)),
   599→        isAlive: alive,
   600→        onWait: (ms, cycle, max) =>
   601→          run.log(`目标主机熔断冷却中，暂停 ${Math.round(ms / 1000)}s 后自动重试章节（第 ${cycle}/${max} 轮，等待期间不向源站发请求）`),
   602→      })
   603→    let ch = await circuitFetch()
   604→    // 瞬时失败单次重试（预算耗尽/超时类）：高并发下域名限速排队可吃光引擎 55s 链预算，
   605→    // 属暂时性压力而非站点拒绝——稍候片刻让队列排空后重试一次，减少无谓的永久失败章节
   606→    if (!ch.ok && isTransientEngineError(ch.error)) {
   607→      run.log(`章节「${(job.refTitle || job.url).slice(0, 30)}」因引擎预算耗尽/超时失败，稍候重试一次`)
   608→      await sleep(3_000 + Math.random() * 5_000)
   609→      if (stopRun || !(await alive())) return
   610→      ch = await circuitFetch()
   611→    }
   612→    if (stopRun || !(await alive())) return // 取消：抓取结果直接丢弃，尽快退出
   613→    const data = ch.ok ? ch.data : null
   614→
   615→    // 入库前统一清洗（去 \r\n/行首缩进/空行/噪声行），存储契约：无空行、无行首缩进
   616→    const cleaned = cleanChapterContent(data?.content ?? '')
   617→    let content = cleaned.text
   618→    if (content.length > MAX_CONTENT_CHARS) {
   619→      content = content.slice(0, MAX_CONTENT_CHARS)
   620→      run.log(`章节正文超长（>${MAX_CONTENT_CHARS} 字符），截断保存（内存保护）`)
   621→    }
   622→
   623→    if (!data || !content.trim()) {
   624→      // 抓取失败：骨架保留不删（重跑任务续采），仅记账
   625→      failed++
   626→      book.failed++
   627→      done++
   628→      if (opts.chapterProgress) opts.chapterProgress.done++
   629→      run.log(`章节抓取失败: ${ch.ok ? '正文为空' : ch.error}`)
   630→      await scheduleFlush({ ...progressFields(), chapters: run.counters.chapters })
   631→      return
   632→    }
   633→    if (ch.warnings.length) run.logWarnings(ch.warnings)
   634→
   635→    const chTitle = (job.refTitle || cleanTextField(data.title) || `第${job.idx}章`).slice(0, 200)
   636→    if (cleaned.removedLines > 0 && book.cleanLogCount < 3) {
   637→      book.cleanLogCount++
   638→      run.log(`章节「${chTitle.slice(0, 30)}」清洗 ${cleaned.removedLines} 行噪声`)
   639→    }
   640→    const wordCount = content.replace(/\s/g, '').length
   641→
   642→    let ok = false
   643→    if (job.kind === 'fill') {
   644→      // 填充骨架/已有空章节：chapterId 优先定位，缺省按 (novelId,idx) 复合唯一定位；
   645→      // 卷名仅在本次目录携带时写入（不回退为空串，防抹掉既有卷名）
   646→      const upd = await db.chapter
   647→        .update({
   648→          where: job.chapterId
   649→            ? { id: job.chapterId }
   650→            : { novelId_idx: { novelId: job.novelId, idx: job.idx } },
   651→          data: {
   652→            title: chTitle,
   653→            content,
   654→            wordCount,
   655→            ...(job.volume ? { volume: job.volume } : {}),
   656→          },
   657→        })
   658→        .then(() => true as const)
   659→        .catch((e: unknown) => (e instanceof Error ? e : new Error('章节填充失败')))
   660→      ok = upd === true
   661→      if (!ok) {
   662→        // P1-3：撞 P2025 → 章节已不存在（书被删级联删/单章被删）；实证整书已消失才中止本书，
   663→        // 单章被删按普通失败继续（保留原有章节级语义）
   664→        if (isRecordMissingError(upd) && (await isNovelGone(job.novelId))) {
   665→          markVanished(book)
   666→        } else {
   667→          run.log(`章节填充失败（骨架可能已被删除）: ${chTitle.slice(0, 40)}`)
   668→        }
   669→      }
   670→    } else {
   671→      // 兜底新建（骨架未能入库）：storeChapter 内部处理 idx 唯一冲突顺延重试与字数增量
   672→      const stored = await storeChapter(run, job.novelId, job.idx, {
   673→        title: chTitle,
   674→        content,
   675→        wordCount,
   676→        volume: job.volume,
   677→      })
   678→      ok = stored.ok
   679→      if (!stored.ok) {
   680→        // P1-3：create 撞 FK（P2003）= novelId 引用不存在，即书已删除
   681→        if (stored.missing && (await isNovelGone(job.novelId))) {
   682→          markVanished(book)
   683→        } else {
   684→          run.log(`章节入库失败: ${stored.message.slice(0, 120)}`)
   685→        }
   686→      }
   687→    }
   688→
   689→    if (ok) {
   690→      filled++
   691→      book.filled++
   692→      run.counters.chapters++
   693→      touchedNovels.add(job.novelId)
   694→      // 实时可见：fill 成功即增量推进书籍字数（create 已在 storeChapter 内增量；
   695→      // 收尾 recalc 全量校准，兜底并发双任务同书的双计误差）
   696→      if (job.kind === 'fill') {
   697→        await db.novel
   698→          .update({ where: { id: job.novelId }, data: { wordCount: { increment: wordCount } } })
   699→          .catch(() => null)
   700→      }
   701→    } else {
   702→      failed++
   703→      book.failed++
   704→    }
   705→    done++
   706→    if (opts.chapterProgress) opts.chapterProgress.done++
   707→    await scheduleFlush({ ...progressFields(), chapters: run.counters.chapters })
   708→  }
   709→
   710→  await runPool(units, clampConcurrency(opts.concurrency), processUnit)
   711→  const flushOk = await flushChain
   712→
   713→  // 收尾校准：对有成功填充的书籍重算字数（增量累计误差/并发双计兜底；失败静默）
   714→  for (const novelId of touchedNovels) await recalcNovelWordCount(novelId)
   715→
   716→  if (stopRun || !flushOk) {
   717→    run.log('任务已取消/记录已删除，停止章节填充')
   718→    await run.flush({ ...progressFields(), chapters: run.counters.chapters }).catch(() => undefined)
   719→  } else if (failed > 0) {
   720→    run.log('抓取失败的章节已保留骨架（wordCount=0），重跑任务可自动续采')
   721→  }
   722→
   723→  return {
   724→    canceled: stopRun || !flushOk,
   725→    filled,
   726→    failed,
   727→    skipped,
   728→    vanishedBooks: books.filter((b) => b.vanished).length,
   729→  }
   730→}

   734→async function runSingle(run: Run, task: TaskRecord, rule: LoadedRule): Promise<void> {
   735→  // Phase 1：书页+目录 → 书籍入库 + 章节骨架（total=骨架章节数）
   736→  const meta = await processBookMeta(run, task.targetUrl, rule, { trackTotal: true })
   737→  if (meta.canceled) {
   738→    run.log('任务已取消')
   739→    await finalize(run, 'canceled', meta.message || '任务已取消')
   740→    return
   741→  }
   742→  await run.flush({ created: run.counters.created, updated: run.counters.updated, chapters: run.counters.chapters })
   743→  if (!meta.ok) {
   744→    await finalize(run, 'failed', meta.message || '书页提取失败')
   745→    return
   746→  }
   747→  // Phase 2：章节内容填充（done 随填充递增，保持 done/total=章节 口径）
   748→  const res = await fillChapters(
   749→    run,
   750→    [{ novelId: meta.novelId, title: '', bookUrl: task.targetUrl, jobs: meta.jobs, filled: 0, failed: 0, vanished: false, cleanLogCount: 0 }],
   751→    rule,
   752→    { trackTotal: true, concurrency: task.concurrency },
   753→  )
   754→  if (res.canceled) {
   755→    await finalize(run, 'canceled', '任务已取消')
   756→    return
   757→  }
   758→  if (res.failed === 0) {
   759→    if (res.filled > 0) {
   760→      await finalize(run, 'success', `采集完成：${res.filled} 章`)
   761→      return
   762→    }
   763→    await finalize(run, 'success', meta.jobs.length === 0 ? meta.message : '无新增章节（章节均已存在）')
   764→    return
   765→  }
   766→  if (res.filled > 0) {
   767→    await finalize(
   768→      run,
   769→      'partial',
   770→      res.vanishedBooks > 0
   771→        ? `${res.filled} 章成功 / ${res.failed} 章失败（书籍在采集中被删除，剩余章节中止）`
   772→        : `${res.filled} 章成功 / ${res.failed} 章失败`,
   773→    )
   774→    return
   775→  }
   776→  await finalize(
   777→    run,
   778→    'failed',
   779→    res.vanishedBooks > 0 ? '书籍已消失（采集中被删除），章节采集中止' : '章节采集全部失败',
   780→  )
   781→}

   783→async function runList(run: Run, task: TaskRecord, rule: LoadedRule): Promise<void> {
   784→  // ---- Phase 1a：列表页收集（翻页变体/熔断重试/Referer 链机制保持不变）----
   785→  const startPage = Math.max(1, Math.floor(task.startPage) || 1)
   786→  const endPage = startPage + Math.max(1, task.pages) - 1
   787→  run.log(`抓取列表页第 ${startPage} 页…`)
   788→  // 熔断感知：起始列表页撞熔断时等待冷却后自动重试；抓取失败（超限）与「空列表页」自此可区分
   789→  const firstPage = await withCircuitRetry(hostOf(task.targetUrl), () => fetchListPage(run, task.targetUrl, rule), {
   790→    detect: (r) => (r.error ? circuitWaitMs(r.error) : null),
   791→    isSuccess: (r) => r.error === null,
   792→    isAlive: async () => !(await isCanceled(run.taskId)),
   793→    onWait: (ms, cycle, max) =>
   794→      run.log(`目标主机熔断冷却中，暂停 ${Math.round(ms / 1000)}s 后自动重试列表页（第 ${cycle}/${max} 轮，等待期间不向源站发请求）`),
   795→  })
   796→  const first = firstPage.items
   797→  if (await isCanceled(run.taskId)) {
   798→    await finalize(run, 'canceled', '任务已取消')
   799→    return
   800→  }
   801→  if (firstPage.error) {
   802→    run.log('起始列表页持续抓取失败（熔断等待轮数耗尽或持续失败），任务终止')
   803→    await finalize(run, 'failed', `列表页抓取失败: ${firstPage.error.slice(0, 180)}`)
   804→    return
   805→  }
   806→  if (first.length === 0) {
   807→    run.log('列表页未提取到书籍条目')
   808→    await finalize(run, 'failed', '列表页未提取到书籍条目')
   809→    return
   810→  }
   811→  run.log(`第 ${startPage} 页提取 ${first.length} 条`)
   812→
   813→  const items = [...first]
   814→  // 当前生效的列表页 URL（含翻页命中页）：作为后续书页抓取的 Referer 来路
   815→  let currentListUrl = task.targetUrl
   816→  for (let k = startPage + 1; k <= endPage; k++) {
   817→    if (await isCanceled(run.taskId)) break
   818→    let got: ListItem[] | null = null
   819→    for (const v of pageVariants(task.targetUrl, k)) {
   820→      // 熔断感知：翻页变体撞熔断时等待冷却后自动重试；变体持续失败 → 尝试下一变体
   821→      const pv = await withCircuitRetry(hostOf(v), () => fetchListPage(run, v, rule), {
   822→        detect: (r) => (r.error ? circuitWaitMs(r.error) : null),
   823→        isSuccess: (r) => r.error === null,
   824→        isAlive: async () => !(await isCanceled(run.taskId)),
   825→        onWait: (ms, cycle, max) =>
   826→          run.log(`目标主机熔断冷却中，暂停 ${Math.round(ms / 1000)}s 后自动重试列表页（第 ${cycle}/${max} 轮）`),
   827→      })
   828→      if (pv.error) continue
   829→      if (pv.items.length > 0) {
   830→        got = pv.items
   831→        currentListUrl = v
   832→        run.log(`第 ${k} 页命中: ${v.slice(0, 100)}`)
   833→        break
   834→      }
   835→    }
   836→    if (!got || got.length === 0) {
   837→      run.log(`第 ${k} 页无结果，跳过`)
   838→      continue
   839→    }
   840→    run.log(`第 ${k} 页提取 ${got.length} 条`)
   841→    items.push(...got)
   842→  }
   843→
   844→  // 合并去重（按 URL，缺 URL 按标题）；不设书籍数量上限，全量采集
   845→  const seen = new Set<string>()
   846→  const merged: ListItem[] = []
   847→  for (const it of items) {
   848→    const key = it.url ?? `t:${it.title}`
   849→    if (seen.has(key)) continue
   850→    seen.add(key)
   851→    merged.push(it)
   852→  }
   853→  const total = merged.length
   854→  await run.flush({ total, done: 0 })
   855→  run.log(`去重后共 ${total} 本书待采集（并发 ${task.concurrency} 路：先书级元数据+骨架，后章级批量内容）`)
   856→
   857→  // ---- Phase 1b：书级并发抓取元数据+目录骨架（done/total=书，主口径不变）----
   858→  const books: Phase2Book[] = []
   859→  let doneBooks = 0
   860→  let failBooks = 0
   861→  let stopRun = false
   862→  // 任务级章节进度（跨书累计）：list 模式 done/total 主口径是「书」，章节进度走 chaptersDone/chaptersTotal
   863→  const chapterProgress = { done: 0, total: 0 }
   864→  // flush 串行链：并行书籍分支的进度写盘串行化
   865→  let flushChain = Promise.resolve(true)
   866→  const scheduleBookFlush = (): Promise<boolean> => {
   867→    flushChain = flushChain.then(async (ok) => {
   868→      if (!ok || stopRun) return false
   869→      const ok2 = await run.flush({
   870→        done: doneBooks,
   871→        chaptersDone: chapterProgress.done,
   872→        chaptersTotal: chapterProgress.total,
   873→        created: run.counters.created,
   874→        updated: run.counters.updated,
   875→        chapters: run.counters.chapters,
   876→      })
   877→      if (!ok2) stopRun = true
   878→      return ok2
   879→    })
   880→    return flushChain
   881→  }
   882→
   883→  const processItem = async (item: ListItem): Promise<void> => {
   884→    if (stopRun) return
   885→    if (await isCanceled(run.taskId)) {
   886→      stopRun = true
   887→      return
   888→    }
   889→    run.log(`━━ 《${(item.title || '未命名').slice(0, 30)}》`)
   890→    const meta = await processBookMeta(run, item.url as string, rule, { trackTotal: false, referer: currentListUrl })
   891→    doneBooks++
   892→    if (meta.canceled) stopRun = true // 取消传播：停止派发后续书籍
   893→    else if (meta.ok) {
   894→      books.push({
   895→        novelId: meta.novelId,
   896→        title: item.title || '',
   897→        bookUrl: item.url as string,
   898→        jobs: meta.jobs,
   899→        filled: 0,
   900→        failed: 0,
   901→        vanished: false,
   902→        cleanLogCount: 0,
   903→      })
   904→      chapterProgress.total += meta.refsTotal
   905→    } else failBooks++
   906→    await scheduleBookFlush()
   907→  }
   908→
   909→  await runPool(merged, clampConcurrency(task.concurrency), processItem)
   910→  const flushOk = await flushChain
   911→
   912→  if (stopRun || !flushOk) {
   913→    run.log('任务已取消')
   914→    await finalize(run, 'canceled', `已取消（完成 ${doneBooks}/${total} 本）`)
   915→    return
   916→  }
   917→
   918→  // ---- Phase 2：全部书籍章节平铺（按书分组顺序），章级并发批量填充 ----
   919→  let res: Phase2Result = { canceled: false, filled: 0, failed: 0, skipped: 0, vanishedBooks: 0 }
   920→  if (books.length > 0) {
   921→    const pending = books.reduce((n, b) => n + b.jobs.filter((j) => j.kind !== 'skip').length, 0)
   922→    run.log(`元数据阶段完成：${books.length} 本书目录骨架就绪，开始批量抓取章节内容（待抓 ${pending} 章，并发 ${task.concurrency} 路）`)
   923→    res = await fillChapters(run, books, rule, { trackTotal: false, chapterProgress, concurrency: task.concurrency })
   924→  }
   925→  if (res.canceled) {
   926→    await finalize(run, 'canceled', `已取消（完成 ${doneBooks}/${total} 本，已填充 ${res.filled} 章）`)
   927→    return
   928→  }
   929→
   930→  // ---- 书籍级成败判定：元数据失败 + 章节全部抓取失败（已存在跳过/无章节链接仍算成功）----
   931→  const okBooks = books.filter((b) => b.filled > 0 || b.failed === 0).length
   932→  const chapterFailBooks = books.filter((b) => b.filled === 0 && b.failed > 0).length
   933→  const failBooksFinal = failBooks + chapterFailBooks
   934→  // 消失提示后缀（P1-3：书在采集中被删除，中止本书剩余章节——与站点侧失败相区分）
   935→  const vanishedNote = res.vanishedBooks > 0 ? `（${res.vanishedBooks} 本采集中被删除，已中止）` : ''
   936→
   937→  if (okBooks === 0) {
   938→    await finalize(
   939→      run,
   940→      'failed',
   941→      res.vanishedBooks > 0 && failBooksFinal === 0 ? '书籍已消失（采集中被删除），无书籍采集成功' : '无书籍采集成功',
   942→    )
   943→    return
   944→  }
   945→  if (failBooksFinal > 0) {
   946→    await finalize(run, 'partial', `${okBooks} 本成功 / ${failBooksFinal} 本失败，入库 ${res.filled} 章${vanishedNote}`)
   947→    return
   948→  }
   949→  await finalize(
   950→    run,
   951→    'success',
   952→    res.failed > 0
   953→      ? `范围采集完成：共 ${total} 本，入库 ${res.filled} 章，失败 ${res.failed} 章${vanishedNote}`
   954→      : `范围采集完成：共 ${total} 本，入库 ${res.filled} 章${vanishedNote}`,
   955→  )
   956→}

   960→/** 终态写入：仅当仍处于 running 时写终态；已被取消/删除则只保留日志 */
   961→async function finalize(run: Run, status: string, message: string): Promise<void> {
   962→  const cur = await db.scrapeTask
   963→    .findUnique({ where: { id: run.taskId }, select: { status: true } })
   964→    .catch(() => null)
   965→  if (!cur) return
   966→  if (cur.status === 'running') {
   967→    await db.scrapeTask
   968→      .updateMany({
   969→        where: { id: run.taskId, status: 'running' },
   970→        data: { status, message: message.slice(0, 500), log: run.logText() },
   971→      })
   972→      .catch(() => {})
   973→  } else {
   974→    await db.scrapeTask
   975→      .updateMany({ where: { id: run.taskId }, data: { log: run.logText() } })
   976→      .catch(() => {})
   977→  }
   978→}

   980→async function runTask(taskId: number): Promise<void> {
   981→  const run = new Run(taskId)
   982→  try {
   983→    // pending → running 条件更新：pending 阶段已被取消/删除的任务不再启动
   984→    const started = await db.scrapeTask
   985→      .updateMany({ where: { id: taskId, status: 'pending' }, data: { status: 'running' } })
   986→      .catch(() => ({ count: 0 }))
   987→    if (started.count === 0) return
   988→    const task = await db.scrapeTask.findUnique({ where: { id: taskId } }).catch(() => null)
   989→    if (!task) return
   990→    run.ruleId = task.ruleId ?? null // 封面回填链路：入库时记入 Novel.sourceRuleId
   991→    const record: TaskRecord = {
   992→      id: task.id,
   993→      mode: task.mode,
   994→      targetUrl: task.targetUrl,
   995→      pages: task.pages,
   996→      startPage: task.startPage,
   997→      concurrency: clampConcurrency(task.concurrency),
   998→      ruleId: task.ruleId,
   999→    }
  1000→    const rule = await loadRule(record.ruleId)
  1001→    run.log(
  1002→      record.mode === 'list'
  1003→        ? `任务开始（范围采集）目标: ${record.targetUrl}，起始页: ${record.startPage}，页数: ${record.pages}，并发: ${record.concurrency}`
  1004→        : `任务开始（单本采集）目标: ${record.targetUrl}，并发: ${record.concurrency}`,
  1005→    )
  1006→    run.log(
  1007→      record.ruleId
  1008→        ? `使用规则「${rule.name ?? record.ruleId}」（charset=${rule.charset ?? 'auto'}）`
  1009→        : '未使用规则，依赖引擎内置启发式提取',
  1010→    )
  1011→    await run.flush()
  1012→    if (record.mode === 'list') await runList(run, record, rule)
  1013→    else await runSingle(run, record, rule)
  1014→  } catch (e) {
  1015→    run.log(`发生未预期错误: ${e instanceof Error ? e.message.slice(0, 200) : String(e).slice(0, 200)}`)
  1016→    await finalize(run, 'failed', '任务执行异常')
  1017→  }
  1018→}

  1020→/**
  1021→ * fire-and-forget 入口：API 路由创建任务后调用，不 await。
  1022→ * 多任务天然并行：每个任务独立 worker promise，互不阻塞；
  1023→ * 对同一目标站的压力由引擎层域名限速统一收敛。
  1024→ */
  1025→export function triggerScrapeTask(taskId: number): void {
  1026→  if (running.has(taskId)) return
  1027→  running.add(taskId)
  1028→  void runTask(taskId)
  1029→    .catch((e: unknown) => {
  1030→      console.error(`[scrape-worker] task ${taskId} 兜底异常:`, e instanceof Error ? e.message : e)
  1031→    })
  1032→    .finally(() => {
  1033→      running.delete(taskId)
  1034→    })
  1035→}
