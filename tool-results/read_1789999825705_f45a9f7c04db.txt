    44→func handleChapterCreate(w http.ResponseWriter, r *http.Request, _ map[string]string) {
    45→        v, ok := readBodyValue(r)
    46→        if !ok {
    47→                writeJSON(w, 400, map[string]string{"error": "请求体不是合法 JSON"})
    48→                return
    49→        }
    50→        body := bodyMap(v)
    51→
    52→        novelId, okN := intFieldStrict(body["novelId"])
    53→        if !okN || novelId == 0 {
    54→                writeJSON(w, 400, map[string]string{"error": "novelId 必填"})
    55→                return
    56→        }
    57→        rawTitle := strField(body["title"], 0)
    58→        if trimSpaceStr(rawTitle) == "" {
    59→                writeJSON(w, 400, map[string]string{"error": "章节标题不能为空"})
    60→                return
    61→        }
    62→        var nid int64
    63→        if err := queryOne(`SELECT "id" FROM "Novel" WHERE "id" = ?`, []any{&nid}, novelId); err != nil {
    64→                if isNoRows(err) {
    65→                        writeJSON(w, 404, map[string]string{"error": "小说不存在"})
    66→                        return
    67→                }
    68→                failJSON(w, "服务器错误", firstLineErr(err), 500)
    69→                return
    70→        }
    71→
    72→        chTitle := truncateRunes(trimSpaceStr(rawTitle), 120) // TS 内联 slice(0,120)
    73→        // 非字符串 content 按空正文处理（与 PUT 语义一致）
    74→        content := strField(body["content"], 0)
    75→        wc := wordCountJS(content)
    76→
    77→        createChapter := func(idxVal int64) (int64, error) {
    78→                return execReturningID(
    79→                        `INSERT INTO "Chapter" ("novelId","idx","title","content","wordCount","createdAt") VALUES (?,?,?,?,?,?)`,
    80→                        novelId, idxVal, chTitle, content, wc, nowMillis(),
    81→                )
    82→        }
    83→
    84→        var maxIdx sql.NullInt64
    85→        if err := queryOne(`SELECT MAX("idx") FROM "Chapter" WHERE "novelId" = ?`, []any{&maxIdx}, novelId); err != nil && !isNoRows(err) {
    86→                failJSON(w, "服务器错误", firstLineErr(err), 500)
    87→                return
    88→        }
    89→        idx := maxIdx.Int64 + 1
    90→        chapterID, err := createChapter(idx)
    91→        if err != nil {
    92→                // 并发竞态兜底：撞 [novelId, idx] 唯一约束后重读最大 idx 重试一次
    93→                if !isUniqueConflict(err) {
    94→                        failJSON(w, "服务器错误", firstLineErr(err), 500)
    95→                        return
    96→                }
    97→                var retryIdx sql.NullInt64
    98→                if err2 := queryOne(`SELECT MAX("idx") FROM "Chapter" WHERE "novelId" = ?`, []any{&retryIdx}, novelId); err2 != nil && !isNoRows(err2) {
    99→                        failJSON(w, "服务器错误", firstLineErr(err2), 500)
   100→                        return
   101→                }
   102→                idx = retryIdx.Int64 + 1
   103→                chapterID, err = createChapter(idx)
   104→                if err != nil {
   105→                        failJSON(w, "创建章节失败", firstLineErr(err), 500)
   106→                        return
   107→                }
   108→        }
   109→
   110→        // 同步小说字数与更新时间（显式 updatedAt，对齐 Prisma @updatedAt）
   111→        var sumWC sql.NullInt64
   112→        if err := queryOne(`SELECT SUM("wordCount") FROM "Chapter" WHERE "novelId" = ?`, []any{&sumWC}, novelId); err != nil && !isNoRows(err) {
   113→                failJSON(w, "服务器错误", firstLineErr(err), 500)
   114→                return
   115→        }
   116→        if _, err := exec(`UPDATE "Novel" SET "wordCount" = ?, "updatedAt" = ? WHERE "id" = ?`, sumWC.Int64, nowMillis(), novelId); err != nil {
   117→                failJSON(w, "服务器错误", firstLineErr(err), 500)
   118→                return
   119→        }
   120→        writeJSON(w, 201, map[string]any{"id": chapterID, "idx": idx})
   121→}

   125→func handleChapterDetail(w http.ResponseWriter, r *http.Request, ps map[string]string) {
   126→        cid, ok := routeIntID(w, ps["id"], "无效 ID")
   127→        if !ok {
   128→                return
   129→        }
   130→        var id, novelId, idx, wordCount int64
   131→        var title, content string
   132→        var novelTitle sql.NullString
   133→        err := queryOne(
   134→                `SELECT ch."id", ch."novelId", ch."idx", ch."title", ch."content", ch."wordCount", n."title" FROM "Chapter" ch LEFT JOIN "Novel" n ON n."id" = ch."novelId" WHERE ch."id" = ?`,
   135→                []any{&id, &novelId, &idx, &title, &content, &wordCount, &novelTitle},
   136→                cid,
   137→        )
   138→        if err != nil {
   139→                if isNoRows(err) {
   140→                        writeJSON(w, 404, map[string]string{"error": "章节不存在"})
   141→                        return
   142→                }
   143→                failJSON(w, "服务器错误", firstLineErr(err), 500)
   144→                return
   145→        }
   146→        var prevID, nextID sql.NullInt64
   147→        _ = queryOne(`SELECT "id" FROM "Chapter" WHERE "novelId" = ? AND "idx" < ? ORDER BY "idx" DESC LIMIT 1`, []any{&prevID}, novelId, idx)
   148→        _ = queryOne(`SELECT "id" FROM "Chapter" WHERE "novelId" = ? AND "idx" > ? ORDER BY "idx" ASC LIMIT 1`, []any{&nextID}, novelId, idx)
   149→
   150→        var prev, next any
   151→        if prevID.Valid {
   152→                prev = prevID.Int64
   153→        }
   154→        if nextID.Valid {
   155→                next = nextID.Int64
   156→        }
   157→        nt := ""
   158→        if novelTitle.Valid {
   159→                nt = novelTitle.String
   160→        }
   161→        writeJSON(w, 200, map[string]any{
   162→                "id":         id,
   163→                "novelId":    novelId,
   164→                "novelTitle": nt,
   165→                "idx":        idx,
   166→                "title":      title,
   167→                "content":    content,
   168→                "wordCount":  wordCount,
   169→                "prevId":     prev,
   170→                "nextId":     next,
   171→        })
   172→}

   176→func handleChapterUpdate(w http.ResponseWriter, r *http.Request, ps map[string]string) {
   177→        cid, ok := routePosIntID(w, ps["id"], "无效 ID")
   178→        if !ok {
   179→                return
   180→        }
   181→        v, ok := readBodyValue(r)
   182→        if !ok {
   183→                writeJSON(w, 400, map[string]string{"error": "请求体不是合法 JSON"})
   184→                return
   185→        }
   186→        body := bodyMap(v)
   187→
   188→        var novelId int64
   189→        if err := queryOne(`SELECT "novelId" FROM "Chapter" WHERE "id" = ?`, []any{&novelId}, cid); err != nil {
   190→                // TS update 抛错（含不存在）一律 404
   191→                writeJSON(w, 404, map[string]string{"error": "章节不存在或更新失败"})
   192→                return
   193→        }
   194→
   195→        sets := []string{}
   196→        args := []any{}
   197→        if s, ok2 := body["title"].(string); ok2 {
   198→                if t := trimSpaceStr(s); t != "" {
   199→                        sets = append(sets, `"title" = ?`)
   200→                        args = append(args, truncateRunes(t, 120))
   201→                }
   202→        }
   203→        if s, ok2 := body["content"].(string); ok2 {
   204→                sets = append(sets, `"content" = ?`, `"wordCount" = ?`)
   205→                args = append(args, s, wordCountJS(s))
   206→        }
   207→        if len(sets) > 0 {
   208→                if _, err := exec(`UPDATE "Chapter" SET `+strings.Join(sets, ", ")+` WHERE "id" = ?`, append(args, cid)...); err != nil {
   209→                        writeJSON(w, 404, map[string]string{"error": "章节不存在或更新失败"})
   210→                        return
   211→                }
   212→        }
   213→        // 内容变化同步书籍字数合计并触碰 updatedAt（对齐 TS 语义）
   214→        if err := resumNovelWordCount(novelId, true); err != nil {
   215→                failJSON(w, "服务器错误", firstLineErr(err), 500)
   216→                return
   217→        }
   218→        writeJSON(w, 200, map[string]any{"ok": true})
   219→}

   223→func handleChapterDelete(w http.ResponseWriter, r *http.Request, ps map[string]string) {
   224→        cid, ok := routePosIntID(w, ps["id"], "无效 ID")
   225→        if !ok {
   226→                return
   227→        }
   228→        var novelId int64
   229→        if err := queryOne(`SELECT "novelId" FROM "Chapter" WHERE "id" = ?`, []any{&novelId}, cid); err != nil {
   230→                writeJSON(w, 404, map[string]string{"error": "章节不存在"})
   231→                return
   232→        }
   233→        res, err := exec(`DELETE FROM "Chapter" WHERE "id" = ?`, cid)
   234→        if err != nil {
   235→                writeJSON(w, 404, map[string]string{"error": "章节不存在"})
   236→                return
   237→        }
   238→        if n, _ := res.RowsAffected(); n == 0 {
   239→                writeJSON(w, 404, map[string]string{"error": "章节不存在"})
   240→                return
   241→        }
   242→        // 与 PUT/POST 对齐：内容变化同时触碰 updatedAt，让「最近更新」排序如实反映删章
   243→        if err := resumNovelWordCount(novelId, true); err != nil {
   244→                failJSON(w, "服务器错误", firstLineErr(err), 500)
   245→                return
   246→        }
   247→        writeJSON(w, 200, map[string]any{"ok": true})
   248→}

   250→// resumNovelWordCount 重算书籍字数合计（touch=true 时同步触碰 updatedAt）
   251→func resumNovelWordCount(novelId int64, touch bool) error {
   252→        var sumWC sql.NullInt64
   253→        if err := queryOne(`SELECT SUM("wordCount") FROM "Chapter" WHERE "novelId" = ?`, []any{&sumWC}, novelId); err != nil {
   254→                return err
   255→        }
   256→        if touch {
   257→                _, err := exec(`UPDATE "Novel" SET "wordCount" = ?, "updatedAt" = ? WHERE "id" = ?`, sumWC.Int64, nowMillis(), novelId)
   258→                return err
   259→        }
   260→        _, err := exec(`UPDATE "Novel" SET "wordCount" = ? WHERE "id" = ?`, sumWC.Int64, novelId)
   261→        return err
   262→}

   271→const cleanAllBatchSize = 500
   272→
   273→var (
   274→        cleanAllMu      sync.Mutex
   275→        cleanAllRunning bool

   278→// scanChapters 遍历全部章节做清洗比对；write=true 落库（章节 content/wordCount + 书籍字数合计）
   279→func scanChapters(write bool) (checked, changed, novels int, err error) {
   280→        cursor := int64(0)
   281→        touchedNovels := map[int64]bool{}
   282→        for {
   283→                type row struct {
   284→                        id      int64
   285→                        novelId int64
   286→                        content string
   287→                }
   288→                batch := make([]row, 0, cleanAllBatchSize)
   289→                err := queryList(`SELECT "id", "novelId", "content" FROM "Chapter" WHERE "id" > ? ORDER BY "id" ASC LIMIT ?`,
   290→                        func(rows *sql.Rows) error {
   291→                                var r row
   292→                                if err := rows.Scan(&r.id, &r.novelId, &r.content); err != nil {
   293→                                        return err
   294→                                }
   295→                                batch = append(batch, r)
   296→                                return nil
   297→                        }, cursor, cleanAllBatchSize)
   298→                if err != nil {
   299→                        return checked, changed, len(touchedNovels), err
   300→                }
   301→                if len(batch) == 0 {
   302→                        break
   303→                }
   304→                for _, ch := range batch {
   305→                        cursor = ch.id
   306→                        checked++
   307→                        res := cleanChapterContent(ch.content)
   308→                        if res.Text == ch.content {
   309→                                continue // 无变化不写库
   310→                        }
   311→                        if write {
   312→                                // 更新失败（瞬时锁等）不计入 cleaned，避免虚报；下轮 dryRun 可复查
   313→                                if _, err := exec(`UPDATE "Chapter" SET "content" = ?, "wordCount" = ? WHERE "id" = ?`, res.Text, wordCountJS(res.Text), ch.id); err != nil {
   314→                                        continue
   315→                                }
   316→                        }
   317→                        changed++
   318→                        touchedNovels[ch.novelId] = true
   319→                }
   320→        }
   321→        if write && len(touchedNovels) > 0 {
   322→                // 章节 wordCount 变了，书籍级字数合计一并重算（update 触碰 @updatedAt → 显式 set）
   323→                for novelId := range touchedNovels {
   324→                        var sumWC sql.NullInt64
   325→                        if err := queryOne(`SELECT SUM("wordCount") FROM "Chapter" WHERE "novelId" = ?`, []any{&sumWC}, novelId); err != nil {
   326→                                continue
   327→                        }
   328→                        _, _ = exec(`UPDATE "Novel" SET "wordCount" = ?, "updatedAt" = ? WHERE "id" = ?`, sumWC.Int64, nowMillis(), novelId)
   329→                }
   330→        }
   331→        return checked, changed, len(touchedNovels), nil
   332→}

   345→func handleChapterCleanAllPost(w http.ResponseWriter, r *http.Request, _ map[string]string) {
   346→        cleanAllMu.Lock()

   368→// ==================== /api/chapters/audit：中文数字解析 ====================
   369→
   370→var cnDigit = map[rune]int{
   371→        '零': 0, '〇': 0, '一': 1, '壹': 1, '二': 2, '贰': 2, '两': 2, '三': 3, '叁': 3, '四': 4, '肆': 4,
   372→        '五': 5, '伍': 5, '六': 6, '陆': 6, '七': 7, '柒': 7, '八': 8, '捌': 8, '九': 9, '玖': 9,
   373→}
   374→var cnUnit = map[rune]int{'十': 10, '拾': 10, '百': 100, '佰': 100, '千': 1000, '仟': 1000}
   375→
   376→// parseChapterNumber 中文/全角数字 → 阿拉伯数字；不可解析返回 false
   377→func parseChapterNumber(raw string) (int64, bool) {
   378→        var b strings.Builder
   379→        for _, c := range raw {
   380→                if c >= 0xFF10 && c <= 0xFF19 { // 全角０-９ → ASCII
   381→                        c = c - 0xFEE0
   382→                }
   383→                b.WriteRune(c)
   384→        }
   385→        s := jsTrim(b.String())
   386→        if s == "" {
   387→                return 0, false
   388→        }
   389→        if allDigitsRe.MatchString(s) {
   390→                return parseDigitsASCII(s), true
   391→        }
   392→        var result, section int64
   393→        cur := int64(-1)
   394→        for _, ch := range s {
   395→                switch {
   396→                case ch == '万' || ch == '萬':
   397→                        section = (section + maxInt64(cur, 0)) * 10000
   398→                        result += section
   399→                        section = 0
   400→                        cur = -1
   401→                default:
   402→                        if d, ok := cnDigit[ch]; ok {
   403→                                cur = int64(d)
   404→                        } else if u, ok := cnUnit[ch]; ok {
   405→                                base := cur
   406→                                if base < 0 {
   407→                                        base = 1
   408→                                }
   409→                                section += base * int64(u)
   410→                                cur = -1
   411→                        } else {
   412→                                return 0, false
   413→                        }
   414→                }
   415→        }
   416→        return result + section + maxInt64(cur, 0), true
   417→}
   418→
   419→var allDigitsRe = regexp.MustCompile(`^\d+$`)
   420→
   421→// parseDigitsASCII 纯数字串转 int64（标题编号场景，不适用 strconv 以外的错误分支）
   422→func parseDigitsASCII(s string) int64 {
   423→        var n int64
   424→        for _, c := range s {
   425→                n = n*10 + int64(c-'0')
   426→                if n > 1<<53 {
   427→                        break // JS Number 超出安全整数后精度丢失，截断即可（章节编号不会这么大）
   428→                }
   429→        }
   430→        return n
   431→}
   432→
   433→func maxInt64(a, b int64) int64 {
   434→        if a > b {
   435→                return a
   436→        }
   437→        return b
   438→}
   439→
   440→const auditCNClass = `[0-9\x{ff10}-\x{ff19}]+|[零〇一二三四五六七八九十百千两壹贰叁肆伍陆柒捌玖拾佰仟萬万]+`
   441→
   442→var (
   443→        // 第N章/节/開頭（严格版：章回体「回」量词排除；\s 用 JS 空白全集）
   444→        chNumRe  = regexp.MustCompile(`^第` + jsSpaceClass + `*(` + auditCNClass + `)` + jsSpaceClass + `*(?:章|节|節)`)
   445→        volumeRe = regexp.MustCompile(`第` + jsSpaceClass + `*(?:` + auditCNClass + `)` + jsSpaceClass + `*(?:卷|部|篇)`)
   446→)
   447→
   448→// extractNum 提取章节编号（严格版）：标题必须以「第N章/节」开头；编号后剩余超 30 字不参与判定
   449→func extractNum(title string) (int64, bool) {
   450→        m := chNumRe.FindStringSubmatch(title)
   451→        if m == nil {
   452→                return 0, false
   453→        }
   454→        restRunes := []rune(title)
   455→        prefixRunes := len([]rune(m[0]))
   456→        if prefixRunes < len(restRunes) {
   457→                restRunes = restRunes[prefixRunes:]
   458→        } else {
   459→                restRunes = nil
   460→        }
   461→        if len(restRunes) > 30 {
   462→                return 0, false
   463→        }
   464→        return parseChapterNumber(m[1])
   465→}

   469→// auditNovel 逐书体检 → NovelAuditItem 形状（字段名与 TS 完全一致）
   470→func auditNovel(novelId int64, title string) map[string]any {
   471→        type row struct {
   472→                id, idx, wordCount int64
   473→                title              string
   474→        }
   475→        rows := make([]row, 0)
   476→        _ = queryList(`SELECT "id", "idx", "title", "wordCount" FROM "Chapter" WHERE "novelId" = ? ORDER BY "idx" ASC`,
   477→                func(rs *sql.Rows) error {
   478→                        var r row
   479→                        if err := rs.Scan(&r.id, &r.idx, &r.title, &r.wordCount); err != nil {
   480→                                return err
   481→                        }
   482→                        rows = append(rows, r)
   483→                        return nil
   484→                }, novelId)
   485→
   486→        // 重复标题分组
   487→        byTitle := map[string]int{}
   488→        for _, r := range rows {
   489→                byTitle[r.title]++
   490→        }
   491→        dupGroups, dupRows := 0, 0
   492→        for _, c := range byTitle {
   493→                if c > 1 {
   494→                        dupGroups++
   495→                        dupRows += c - 1
   496→                }
   497→        }
   498→
   499→        // idx 断档（rows 与 maxIdx 不符；idx 从 1 连续为健康）
   500→        var maxIdx int64
   501→        for _, r := range rows {
   502→                if r.idx > maxIdx {
   503→                        maxIdx = r.idx
   504→                }
   505→        }
   506→        idxGaps := maxIdx - int64(len(rows))
   507→        if idxGaps < 0 {
   508→                idxGaps = 0
   509→        }
   510→
   511→        emptyRows := 0
   512→        for _, r := range rows {
   513→                if r.wordCount == 0 {
   514→                        emptyRows++
   515→                }
   516→        }
   517→
   518→        // 编号乱序：相邻回退检测（前章编号大于后章编号）
   519→        disorderSamples := make([]string, 0)
   520→        var prevN int64
   521→        var prevTitle string
   522→        hasPrev := false
   523→        for _, r := range rows {
   524→                n, ok := extractNum(r.title)
   525→                if !ok {
   526→                        continue
   527→                }
   528→                if hasPrev && n < prevN && len(disorderSamples) < 3 {
   529→                        disorderSamples = append(disorderSamples, "「"+truncateRunes(prevTitle, 24)+"」→「"+truncateRunes(r.title, 24)+"」")
   530→                }
   531→                prevN, prevTitle, hasPrev = n, r.title, true
   532→        }
   533→
   534→        item := map[string]any{
   535→                "novelId":         novelId,
   536→                "title":           title,
   537→                "chapters":        len(rows),
   538→                "dupGroups":       dupGroups,
   539→                "dupRows":         dupRows,
   540→                "idxGaps":         idxGaps,
   541→                "emptyRows":       emptyRows,
   542→                "disordered":      len(disorderSamples) > 0,
   543→                "disorderSamples": disorderSamples,
   544→        }
   545→
   546→        // 分卷结构：以「第X卷/部/篇」标题行为卷边界
   547→        if len(rows) > 0 {
   548→                volumes := make([]map[string]any, 0)
   549→                var current map[string]any
   550→                for _, r := range rows {
   551→                        if volumeRe.MatchString(r.title) {
   552→                                current = map[string]any{"title": truncateRunes(r.title, 60), "from": r.idx, "to": r.idx, "chapters": 0}
   553→                                volumes = append(volumes, current)
   554→                        }
   555→                        if current != nil {
   556→                                current["to"] = r.idx
   557→                                current["chapters"] = current["chapters"].(int) + 1
   558→                        }
   559→                }
   560→                if len(volumes) > 0 {
   561→                        item["volumes"] = volumes
   562→                }
   563→        }
   564→        return item
   565→}

   567→// ==================== GET /api/chapters/audit ====================
   568→
   569→func handleChapterAuditGet(w http.ResponseWriter, r *http.Request, _ map[string]string) {
   570→        novelIdParam := r.URL.Query().Get("novelId")
   571→        if novelIdParam != "" {
   572→                f, ok := jsParseFloat(novelIdParam)
   573→                if !ok || !numIsInt(f) || f <= 0 {
   574→                        writeJSON(w, 400, map[string]string{"error": "novelId 非法"})
   575→                        return
   576→                }
   577→                novelId := int64(f)
   578→                var title string
   579→                if err := queryOne(`SELECT "title" FROM "Novel" WHERE "id" = ?`, []any{&title}, novelId); err != nil {
   580→                        if isNoRows(err) {
   581→                                writeJSON(w, 404, map[string]string{"error": "书籍不存在"})
   582→                                return
   583→                        }
   584→                        writeJSON(w, 500, map[string]string{"error": "目录体检失败", "detail": firstLineErr(err)})
   585→                        return
   586→                }
   587→                item := auditNovel(novelId, title)
   588→                writeJSON(w, 200, map[string]any{"mode": "single", "item": item})
   589→                return
   590→        }
   591→
   592→        // 全站概览：逐书体检
   593→        type novelRow struct {
   594→                id    int64
   595→                title string
   596→        }
   597→        novels := make([]novelRow, 0)
   598→        if err := queryList(`SELECT "id", "title" FROM "Novel" ORDER BY "id" ASC`, func(rows *sql.Rows) error {
   599→                var n novelRow
   600→                if err := rows.Scan(&n.id, &n.title); err != nil {
   601→                        return err
   602→                }
   603→                novels = append(novels, n)
   604→                return nil
   605→        }); err != nil {
   606→                writeJSON(w, 500, map[string]string{"error": "目录体检失败", "detail": firstLineErr(err)})
   607→                return
   608→        }
   609→        items := make([]map[string]any, 0, len(novels))
   610→        for _, n := range novels {
   611→                items = append(items, auditNovel(n.id, n.title))
   612→        }
   613→        type scored struct {
   614→                item  map[string]any
   615→                score int
   616→        }
   617→        problem := make([]scored, 0)
   618→        for _, it := range items {
   619→                if it["dupRows"].(int) > 0 || it["idxGaps"].(int64) > 0 || it["disordered"].(bool) {
   620→                        problem = append(problem, scored{it, it["dupRows"].(int)*10 + int(it["idxGaps"].(int64))})
   621→                }
   622→        }
   623→        // 稳定降序（对齐 JS Array.prototype.sort 稳定性）后取 100
   624→        for i := 1; i < len(problem); i++ {
   625→                for j := i; j > 0 && problem[j].score > problem[j-1].score; j-- {
   626→                        problem[j], problem[j-1] = problem[j-1], problem[j]
   627→                }
   628→        }
   629→        if len(problem) > 100 {
   630→                problem = problem[:100]
   631→        }
   632→        out := make([]map[string]any, 0, len(problem))
   633→        for _, p := range problem {
   634→                out = append(out, p.item)
   635→        }
   636→        writeJSON(w, 200, map[string]any{
   637→                "mode":    "overview",
   638→                "novels":  len(items),
   639→                "healthy": len(items) - len(out),
   640→                "problem": out,
   641→        })
   642→}

   644→// ==================== POST /api/chapters/audit（dedupe / reindex） ====================
   645→
   646→func handleChapterAuditPost(w http.ResponseWriter, r *http.Request, _ map[string]string) {
   647→        v, ok := readBodyValue(r)
   648→        if !ok {
   649→                writeJSON(w, 400, map[string]string{"error": "请求体非法 JSON"})
   650→                return
   651→        }
   652→        body := bodyMap(v)
   653→        action := strField(body["action"], 0)
   654→        novelIdF := jsNumber(body["novelId"])
   655→        if action != "dedupe" && action != "reindex" {
   656→                writeJSON(w, 400, map[string]string{"error": "action 仅支持 dedupe / reindex"})
   657→                return
   658→        }
   659→        if !numIsInt(novelIdF) || novelIdF <= 0 {
   660→                writeJSON(w, 400, map[string]string{"error": "novelId 必填（正整数）"})
   661→                return
   662→        }
   663→        novelId := int64(novelIdF)
   664→
   665→        var title string
   666→        if err := queryOne(`SELECT "title" FROM "Novel" WHERE "id" = ?`, []any{&title}, novelId); err != nil {
   667→                if isNoRows(err) {
   668→                        writeJSON(w, 404, map[string]string{"error": "书籍不存在"})
   669→                        return
   670→                }
   671→                writeJSON(w, 500, map[string]string{"error": "目录修复失败", "detail": firstLineErr(err)})
   672→                return
   673→        }
   674→
   675→        type row struct {
   676→                id, idx, wordCount int64
   677→                title              string
   678→        }
   679→        rows := make([]row, 0)
   680→        if err := queryList(`SELECT "id", "idx", "title", "wordCount" FROM "Chapter" WHERE "novelId" = ? ORDER BY "idx" ASC, "id" ASC`,
   681→                func(rs *sql.Rows) error {
   682→                        var x row
   683→                        if err := rs.Scan(&x.id, &x.idx, &x.title, &x.wordCount); err != nil {
   684→                                return err
   685→                        }
   686→                        rows = append(rows, x)
   687→                        return nil
   688→                }, novelId); err != nil {
   689→                writeJSON(w, 500, map[string]string{"error": "目录修复失败", "detail": firstLineErr(err)})
   690→                return
   691→        }
   692→        if len(rows) == 0 {
   693→                writeJSON(w, 400, map[string]string{"error": "该书无章节"})
   694→                return
   695→        }
   696→
   697→        // ---- 去重：同书同名保留 wordCount 最大（并列取 idx 最小）----
   698→        removed := 0
   699→        if action == "dedupe" {
   700→                type best struct {
   701→                        id, wordCount, idx int64
   702→                }
   703→                keep := map[string]best{}
   704→                for _, x := range rows {
   705→                        b, exists := keep[x.title]
   706→                        if !exists || x.wordCount > b.wordCount || (x.wordCount == b.wordCount && x.idx < b.idx) {
   707→                                keep[x.title] = best{id: x.id, wordCount: x.wordCount, idx: x.idx}
   708→                        }
   709→                }
   710→                if len(keep) < len(rows) {
   711→                        keepIDs := map[int64]bool{}
   712→                        for _, b := range keep {
   713→                                keepIDs[b.id] = true
   714→                        }
   715→                        dropIDs := make([]int64, 0)
   716→                        for _, x := range rows {
   717→                                if !keepIDs[x.id] {
   718→                                        dropIDs = append(dropIDs, x.id)
   719→                                }
   720→                        }
   721→                        // 分批 IN 删除（SQLite 变量上限防御）
   722→                        for i := 0; i < len(dropIDs); i += 500 {
   723→                                end := i + 500
   724→                                if end > len(dropIDs) {
   725→                                        end = len(dropIDs)
   726→                                }
   727→                                part := dropIDs[i:end]
   728→                                ph := strings.TrimSuffix(strings.Repeat("?,", len(part)), ",")
   729→                                args := make([]any, len(part))
   730→                                for j, idv := range part {
   731→                                        args[j] = idv
   732→                                }
   733→                                res, err := exec(`DELETE FROM "Chapter" WHERE "id" IN (`+ph+`)`, args...)
   734→                                if err != nil {
   735→                                        writeJSON(w, 500, map[string]string{"error": "目录修复失败", "detail": firstLineErr(err)})
   736→                                        return
   737→                                }
   738→                                n, _ := res.RowsAffected()
   739→                                removed += int(n)
   740→                        }
   741→                }
   742→        }
   743→
   744→        // ---- 重排：有编号按编号升序（稳定），无编号按原相对顺序置后；压实 idx 1..n ----
   745→        kept := make([]row, 0)
   746→        if err := queryList(`SELECT "id", "idx", "title" FROM "Chapter" WHERE "novelId" = ? ORDER BY "idx" ASC, "id" ASC`,
   747→                func(rs *sql.Rows) error {
   748→                        var x row
   749→                        if err := rs.Scan(&x.id, &x.idx, &x.title); err != nil {
   750→                                return err
   751→                        }
   752→                        kept = append(kept, x)
   753→                        return nil
   754→                }, novelId); err != nil {
   755→                writeJSON(w, 500, map[string]string{"error": "目录修复失败", "detail": firstLineErr(err)})
   756→                return
   757→        }
   758→        moved := 0
   759→        type numberedRow struct {
   760→                r row
   761→                i int
   762→                n int64
   763→        }
   764→        numbered := make([]numberedRow, 0)
   765→        unnumbered := make([]row, 0)
   766→        for i, x := range kept {
   767→                if n, ok := extractNum(x.title); ok {
   768→                        numbered = append(numbered, numberedRow{r: x, i: i, n: n})
   769→                } else {
   770→                        unnumbered = append(unnumbered, x)
   771→                }
   772→        }
   773→        // 稳定排序（n 升序，并列保持原相对顺序 i）
   774→        for i := 1; i < len(numbered); i++ {
   775→                for j := i; j > 0 && (numbered[j].n < numbered[j-1].n); j-- {
   776→                        numbered[j], numbered[j-1] = numbered[j-1], numbered[j]
   777→                }
   778→        }
   779→        ordered := make([]row, 0, len(kept))
   780→        for _, x := range numbered {
   781→                ordered = append(ordered, x.r)
   782→        }
   783→        ordered = append(ordered, unnumbered...)
   784→
   785→        // 两段式重排：先全部移出到负数暂存区（避开目标 idx 被占），再落位 1..n
   786→        for i, x := range ordered {
   787→                if x.idx != int64(i+1) {
   788→                        _, _ = exec(`UPDATE "Chapter" SET "idx" = ? WHERE "id" = ?`, -1_000_000-int64(i), x.id)
   789→                }
   790→        }
   791→        for i, x := range ordered {
   792→                target := int64(i + 1)
   793→                if x.idx != target {
   794→                        res, err := exec(`UPDATE "Chapter" SET "idx" = ? WHERE "id" = ?`, target, x.id)
   795→                        if err == nil {
   796→                                if n, _ := res.RowsAffected(); n > 0 {
   797→                                        moved++
   798→                                }
   799→                        }
   800→                }
   801→        }
   802→
   803→        // 重算字数（去重可能删行；update 触碰 @updatedAt → 显式 set）
   804→        if removed > 0 {
   805→                var sumWC sql.NullInt64
   806→                if err := queryOne(`SELECT SUM("wordCount") FROM "Chapter" WHERE "novelId" = ?`, []any{&sumWC}, novelId); err == nil {
   807→                        _, _ = exec(`UPDATE "Novel" SET "wordCount" = ?, "updatedAt" = ? WHERE "id" = ?`, sumWC.Int64, nowMillis(), novelId)
   808→                }
   809→        }
   810→
   811→        after := auditNovel(novelId, title)
   812→        writeJSON(w, 200, map[string]any{
   813→                "ok":      true,
   814→                "action":  action,
   815→                "novelId": novelId,
   816→                "removed": removed,
   817→                "moved":   moved,
   818→                "audit":   after,
   819→        })
   820→}
