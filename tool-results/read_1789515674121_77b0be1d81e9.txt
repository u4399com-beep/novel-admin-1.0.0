   201→export function Home({ navigate }: ViewProps) {
   202→  const home = useHomeData()
   203→  const updates = useNovels({ sort: 'latest', page: 1, pageSize: 30 })
   204→
   205→  if (home.isPending || updates.isPending) {
   206→    return (
   207→      <Container className="mt-2">
   208→        <div className="grid gap-4 md:grid-cols-[73%_1fr]">
   209→          <div>
   210→            <GSkel className="mb-2 h-6 w-40" />
   211→            <GCoverSkeleton count={4} />
   212→          </div>
   213→          <div>
   214→            <GSkel className="mb-2 h-10 w-full" />
   215→            <GSkel className="mb-2 h-6 w-32" />
   216→            <GRowSkeleton count={10} />
   217→          </div>
   218→        </div>
   219→      </Container>
   220→    )
   221→  }
   222→
   223→  if (home.isError || !home.data) {
   224→    return (
   225→      <Container className="mt-4">
   226→        <GErrorBox onRetry={() => home.refetch()} />
   227→      </Container>
   228→    )
   229→  }
   230→  if (updates.isError) {
   231→    return (
   232→      <Container className="mt-4">
   233→        <GErrorBox onRetry={() => updates.refetch()} />
   234→      </Container>
   235→    )
   236→  }
   237→
   238→  const d = home.data
   239→  const hotItems = d.hot.slice(0, 6)
   240→  const clickRank = d.rankings.clicks.slice(0, 12)
   241→  const updateRows = (updates.data?.list ?? []).slice(0, 30)
   242→  const sideNews = dedupMerge(
   243→    [d.featured, d.hot, d.rankings.clicks, d.rankings.updates, d.rankings.finished],
   244→    30,
   245→  )
   246→
   247→  return (
   248→    <Container className="mt-1">
   249→      <div className="grid gap-5 md:grid-cols-[73%_1fr] md:gap-[2%]">
   250→        {/* 第一行左：热门小说推荐（2 列封面简介卡） */}
   251→        <section>
   252→          <GH2>热门小说推荐</GH2>
   253→          <div className="grid gap-x-5 gap-y-4 sm:grid-cols-2">
   254→            {hotItems.map((n) => (
   255→              <HotItem key={n.id} novel={n} navigate={navigate} />
   256→            ))}
   257→          </div>
   258→        </section>
   259→
   260→        {/* 第一行右：搜索框 + 阅读排行榜 */}
   261→        <aside>
   262→          <SearchBox navigate={navigate} />
   263→          <GH2 className="mt-4">阅读排行榜</GH2>
   264→          <div>
   265→            {clickRank.map((n) => (
   266→              <RankRow key={n.id} novel={n} navigate={navigate} />
   267→            ))}
   268→          </div>
   269→        </aside>
   270→
   271→        {/* 第二行左：最近更新（五列字段表） */}
   272→        <section>
   273→          <GH2>最近更新</GH2>
   274→          <div>
   275→            {updateRows.map((n) => (
   276→              <UpdateRow key={n.id} novel={n} navigate={navigate} />
   277→            ))}
   278→          </div>
   279→        </section>
   280→
   281→        {/* 第二行右：最新小说（两字段行榜单） */}
   282→        <aside>
   283→          <GH2>最新小说</GH2>
   284→          <div>
   285→            {sideNews.map((n) => (
   286→              <RankRow key={n.id} novel={n} navigate={navigate} />
   287→            ))}
   288→          </div>
   289→        </aside>
   290→      </div>
   291→    </Container>
   292→  )
   293→}

   295→/* ==================== 分类页（分类导航条 + 3 列虚线盒 + 数字分页） ==================== */
   296→
   297→export function Category({
   298→  navigate,
   299→  categoryId,
   300→  page = 1,
   301→}: ViewProps & { categoryId?: number; page?: number }) {
   302→  const cats = useCategories()
   303→  const list = useNovels({ categoryId, page, pageSize: 20 })
   304→  const catName = cats.data?.find((c) => c.id === categoryId)?.name
   305→
   306→  const navCls = (active: boolean) =>
   307→    active
   308→      ? 'grid place-items-center h-[38px] rounded-[3px] bg-[#56ccb5] text-[15px] text-white transition-colors duration-200 cursor-pointer'
   309→      : 'grid place-items-center h-[38px] cursor-pointer rounded-[3px] text-[15px] text-[#00886d] transition-colors duration-200 hover:bg-[#cdf3eb] hover:text-[#f50]'
   310→
   311→  return (
   312→    <Container className="mt-1">
   313→      {/* 分类导航条 */}
   314→      <nav className="mb-3 grid grid-cols-3 gap-1 rounded-[4px] border border-[#ccc] bg-white p-1 sm:grid-cols-4 md:grid-cols-8">
   315→        <button
   316→          onClick={() => navigate({ name: 'category', page: 1 })}
   317→          className={navCls(categoryId == null)}
   318→        >
   319→          全部
   320→        </button>
   321→        {(cats.data ?? []).map((c) => (
   322→          <button
   323→            key={c.id}
   324→            onClick={() => navigate({ name: 'category', categoryId: c.id, page: 1 })}
   325→            className={navCls(categoryId === c.id)}
   326→          >
   327→            {c.name}
   328→          </button>
   329→        ))}
   330→      </nav>
   331→
   332→      {/* 书籍列表 */}
   333→      <section className="rounded-[4px] border border-[#ccc] bg-white p-3">
   334→        <GH2 className="text-center">
   335→          {catName ?? '全部小说'}
   336→          {list.data ? <span className="ml-2 text-[13px] font-normal text-[#999]">共 {list.data.total} 本</span> : null}
   337→        </GH2>
   338→
   339→        {list.isPending ? (
   340→          <GBoxSkeleton count={6} />
   341→        ) : list.isError ? (
   342→          <GErrorBox onRetry={() => list.refetch()} />
   343→        ) : (list.data?.list.length ?? 0) === 0 ? (
   344→          <GEmptyBox text="该分类下暂无收录小说" />
   345→        ) : (
   346→          <>
   347→            <div className="grid gap-4 pt-2 sm:grid-cols-2 lg:grid-cols-3">
   348→              {list.data!.list.map((n, i) => (
   349→                <BookBoxItem
   350→                  key={n.id}
   351→                  novel={n}
   352→                  index={(list.data!.page - 1) * (list.data!.pageSize ?? 20) + i + 1}
   353→                  navigate={navigate}
   354→                />
   355→              ))}
   356→            </div>
   357→            <GPager
   358→              page={list.data!.page}
   359→              totalPages={list.data!.totalPages}
   360→              onGo={(p) => navigate({ name: 'category', categoryId, page: p })}
   361→            />
   362→          </>
   363→        )}
   364→      </section>
   365→    </Container>
   366→  )
   367→}

   625→export function Chapter({ navigate, chapterId }: ViewProps & { chapterId: number }) {
   626→  const { data: ch, isPending, isError, refetch } = useChapter(chapterId)
   627→  const novel = useNovel(ch?.novelId)
   628→  /* 书签按章记忆：换章后自动失效（无 effect setState） */
   629→  const [mark, setMark] = useState<{ id: number; on: boolean }>({ id: 0, on: false })
   630→
   631→  useEffect(() => {
   632→    window.scrollTo(0, 0)
   633→  }, [chapterId])
   634→
   635→  /* 键盘翻页：Enter 返回书目 / ← 上一章 / → 下一章 */
   636→  useEffect(() => {
   637→    if (!ch) return
   638→    const onKey = (e: KeyboardEvent) => {
   639→      if (e.key === 'Enter') {
   640→        navigate({ name: 'toc', novelId: ch.novelId })
   641→      } else if (e.key === 'ArrowLeft' && ch.prevId) {
   642→        navigate({ name: 'chapter', chapterId: ch.prevId })
   643→      } else if (e.key === 'ArrowRight' && ch.nextId) {
   644→        navigate({ name: 'chapter', chapterId: ch.nextId })
   645→      }
   646→    }
   647→    window.addEventListener('keydown', onKey)
   648→    return () => window.removeEventListener('keydown', onKey)
   649→  }, [ch, navigate])
   650→
   651→  if (isPending) {
   652→    return (
   653→      <Container className="mt-1">
   654→        <GSkel className="mb-3 h-10 w-full" />
   655→        <div className="rounded-[4px] border border-[#ccc] bg-[#FBF4EC] p-4">
   656→          <GSkel className="mx-auto h-7 w-1/2" />
   657→          <div className="mt-6 space-y-4">
   658→            {Array.from({ length: 9 }, (_, i) => (
   659→              <GSkel key={i} className="h-5 w-full" />
   660→            ))}
   661→          </div>
   662→        </div>
   663→      </Container>
   664→    )
   665→  }
   666→
   667→  if (isError || !ch) {
   668→    return (
   669→      <Container className="mt-4">
   670→        <GErrorBox onRetry={() => refetch()} />
   671→      </Container>
   672→    )
   673→  }
   674→
   675→  const paragraphs = ch.content.split('\n').map((s) => s.trim()).filter(Boolean)
   676→
   677→  const navBtnCls =
   678→    'cursor-pointer rounded-[4px] border border-[#ccc] bg-white py-2 text-[15px] text-[#333] transition-colors duration-200 hover:border-[#56ccb5] hover:text-[#00886d] disabled:pointer-events-none disabled:opacity-40'
   679→
   680→  return (
   681→    <Container className="mt-1">
   682→      <GBreadcrumb
   683→        items={[
   684→          { label: '首页', view: { name: 'home' } },
   685→          ...(novel.data
   686→            ? [
   687→                {
   688→                  label: novel.data.categoryName,
   689→                  view: {
   690→                    name: 'category' as const,
   691→                    categoryId: novel.data.categoryId,
   692→                    page: 1,
   693→                  },
   694→                },
   695→              ]
   696→            : []),
   697→          { label: ch.novelTitle, view: { name: 'book', novelId: ch.novelId } },
   698→          { label: ch.title },
   699→        ]}
   700→        navigate={navigate}
   701→      />
   702→
   703→      {/* 阅读卡（米黄底） */}
   704→      <article className="rounded-[4px] border border-[#ccc] bg-[#FBF4EC] p-3">
   705→        <h1 className="px-2 py-3 text-center text-[20px] font-bold text-[#00886d] md:text-[26px]">
   706→          {ch.title}
   707→        </h1>
   708→        <div className="pb-2 text-center">
   709→          <button
   710→            onClick={() => setMark({ id: ch.id, on: !(mark.id === ch.id && mark.on) })}
   711→            className="cursor-pointer rounded-full border border-[#ffb0b4] bg-white/70 px-4 py-[3px] text-[12px] text-[#bf2c24] transition-colors duration-200 hover:border-[#f50] hover:text-[#f50]"
   712→          >
   713→            {mark.id === ch.id && mark.on ? '已加入书签 ✓' : '加入书签'}
   714→          </button>
   715→        </div>
   716→        <div className="border-t border-[#ccc] px-3 py-2 text-[18px] leading-[180%] tracking-[0.1em] text-[#333] min-[468px]:text-[24px]">
   717→          {paragraphs.length === 0 ? (
   718→            <p className="my-[10px] text-[#999]">本章内容为空</p>
   719→          ) : (
   720→            paragraphs.map((p, i) => <p key={i} className="my-[10px]">{p}</p>)
   721→          )}
   722→        </div>
   723→
   724→        {/* 翻页导航：上一章 / 书页目录 / 下一章（移动端换行堆叠） */}
   725→        <div className="flex flex-wrap justify-center gap-2 px-2 pt-3 pb-1">
   726→          <button
   727→            disabled={!ch.prevId}
   728→            onClick={() => ch.prevId && navigate({ name: 'chapter', chapterId: ch.prevId })}
   729→            className={`${navBtnCls} w-[46%] md:w-[30%]`}
   730→          >
   731→            上一章
   732→          </button>
   733→          <button
   734→            onClick={() => navigate({ name: 'toc', novelId: ch.novelId })}
   735→            className={`${navBtnCls} w-[46%] md:w-[30%]`}
   736→          >
   737→            书页目录
   738→          </button>
   739→          <button
   740→            disabled={!ch.nextId}
   741→            onClick={() => ch.nextId && navigate({ name: 'chapter', chapterId: ch.nextId })}
   742→            className={`${navBtnCls} w-[94%] md:w-[30%]`}
   743→          >
   744→            下一章
   745→          </button>
   746→        </div>
   747→        <p className="hidden pb-1 text-center text-[12px] text-[#999] md:block">
   748→          键盘操作：Enter 返回书目 · ← 上一章 · → 下一章
   749→        </p>
   750→      </article>
   751→
   752→      {/* 相关阅读 */}
   753→      {novel.data ? (
   754→        <RelatedBooks
   755→          categoryId={novel.data.categoryId}
   756→          excludeId={ch.novelId}
   757→          navigate={navigate}
   758→        />
   759→      ) : null}
   760→    </Container>
   761→  )
   762→}
