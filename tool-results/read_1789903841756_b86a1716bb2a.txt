    57→const SHELF_KEY = '23qb.shelf'
    58→
    59→function loadShelf(): number[] {
    60→  if (typeof window === 'undefined') return []
    61→  try {
    62→    return JSON.parse(window.localStorage.getItem(SHELF_KEY) ?? '[]') as number[]
    63→  } catch {
    64→    return []
    65→  }
    66→}
    67→
    68→function toggleShelf(id: number): boolean {
    69→  if (typeof window === 'undefined') return false
    70→  const arr = loadShelf()
    71→  const has = arr.includes(id)
    72→  const next = has ? arr.filter((x) => x !== id) : [...arr, id]
    73→  window.localStorage.setItem(SHELF_KEY, JSON.stringify(next))
    74→  return !has
    75→}

   163→export function CategoryView({
   164→  navigate,
   165→  categoryId,
   166→  page = 1,
   167→}: ViewProps & { categoryId?: number; page?: number }) {
   168→  const { data: categories } = useCategories()
   169→  const [sort, setSort] = useState<SortKey>('latest')
   170→  const [status, setStatus] = useState<'serial' | 'finished' | undefined>(undefined)
   171→  const { data, isLoading, isError, refetch } = useNovels({
   172→    categoryId,
   173→    page,
   174→    pageSize: 20,
   175→    sort,
   176→    status,
   177→  })
   178→
   179→  const catName = categories?.find((c) => c.id === categoryId)?.name ?? '全部'
   180→  const applyFilter = (fn: () => void) => {
   181→    fn()

   184→
   185→  return (
   186→    <Container className="space-y-6 py-6">
   187→      {/* 筛选区 */}
   188→      <Card className="space-y-3 p-5">

   212→        <div className="flex flex-wrap items-center gap-2">

   222→          </Pill>

   271→  const [rec, setRec] = useState(false)

   435→export function TocView({ navigate, novelId }: ViewProps & { novelId: number }) {
   436→  const { data: novel, isError, refetch } = useNovel(novelId)
   437→  const { data: chapters, isLoading, isError: chError, refetch: refetchCh } = useChapters(novelId)
   438→  const [bookmark, setBookmark] = useState<{ chapterId: number; title: string } | null>(
   439→    () => loadBookmarks()[String(novelId)] ?? null
   440→  )
   441→  const [bmNovelId, setBmNovelId] = useState(novelId)
   442→
   443→  if (bmNovelId !== novelId) {
   444→    setBmNovelId(novelId)
   445→    setBookmark(loadBookmarks()[String(novelId)] ?? null)
   446→  }
   447→
   448→  const err = isError || chError
   449→  if (err) {
   450→    return (
   451→      <Container className="pt-[70px]">
   452→        <ErrorRetry onRetry={() => { refetch(); refetchCh() }} />
   453→      </Container>
   454→    )
   455→  }
   456→
   457→  return (
   458→    <Container className="space-y-6 py-6">
   459→      <Card className="p-6 md:p-8">
   460→        <button
   461→          type="button"
   462→          onClick={() => navigate({ name: 'book', novelId })}
   463→          className="cursor-pointer text-left text-[30px] font-extrabold leading-tight text-[#282828] transition-colors hover:text-[#ff2a14] md:text-[38px]"
   464→        >
   465→          {novel?.title ?? '目录'}
   466→        </button>
   467→        <p className="mt-2 text-sm text-black/45">
   468→          {novel ? `作者：${novel.author} · 更新于 ${fmtDate(novel.updatedAt)} · 共 ${chapters?.length ?? novel.totalChapters} 章` : ''}
