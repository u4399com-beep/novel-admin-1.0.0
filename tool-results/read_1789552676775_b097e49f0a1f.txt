    23→async function api<T>(url: string, init?: RequestInit): Promise<T> {
    24→  const res = await fetch(url, { headers: { 'Content-Type': 'application/json' }, ...init })
    25→  const data = await res.json().catch(() => ({}))
    26→  if (!res.ok) throw new Error((data as { error?: string }).error ?? `请求失败(${res.status})`)
    27→  return data as T
    28→}

    87→function ThemesTab() {

    95→      await api('/api/settings', { method: 'PATCH', body: JSON.stringify({ activeTheme: id }) })

   110→        <Badge className="ml-1">{settings?.activeTheme ?? '-'}</Badge>

   114→          const active = settings?.activeTheme === t.id
   115→          return (
   116→            <div
   117→              key={t.id}
   118→              className={`rounded-lg border p-4 transition ${active ? 'border-neutral-900 ring-1 ring-neutral-900' : 'border-neutral-200 hover:border-neutral-400'}`}

   175→      if (form.id) {
   176→        await api(`/api/novels/${form.id}`, { method: 'PUT', body: JSON.stringify(form) })

   178→      } else {
   179→        await api('/api/novels', { method: 'POST', body: JSON.stringify(form) })
   180→        toast.success('已新增')
   181→      }
   182→      setForm(null)
   183→      await qc.invalidateQueries({ queryKey: ['novels'] })
   184→      await qc.invalidateQueries({ queryKey: ['novel'] })

   187→      toast.error(e instanceof Error ? e.message : '保存失败')

   197→      await qc.invalidateQueries({ queryKey: ['novel'] })
   198→      await qc.invalidateQueries({ queryKey: qk.home })

   200→      toast.error(e instanceof Error ? e.message : '删除失败')

   206→      <div className="mb-3 flex flex-wrap items-center gap-2">

   215→              <th className="hidden px-3 py-2 text-left font-medium sm:table-cell">分类</th>
   216→              <th className="px-3 py-2 text-left font-medium">章节</th>

   231→                <td className="px-3 py-2 text-xs">{n.chapterCount}</td>
   232→                <td className="hidden px-3 py-2 text-xs text-neutral-500 md:table-cell">{formatWordCount(n.wordCount)}</td>
   233→                <td className="px-3 py-2 text-right">
   234→                  <Button size="sm" variant="ghost" className="h-7 px-2" onClick={() => setChapterNovel(n)}>

   236→                  </Button>

   238→                    size="sm" variant="ghost" className="h-7 px-2"
   239→                    onClick={() => setForm({ id: n.id, title: n.title, author: n.author, description: n.description, categoryId: n.categoryId, status: n.status, isFeatured: n.isFeatured, isHot: n.isHot })}
   240→                  >
   241→                    <Pencil className="h-3.5 w-3.5" />
   242→                  </Button>

   261→      )}
   262→
   263→      {form && (
   264→        <div className="fixed inset-0 z-[60] flex items-center justify-center bg-black/40 p-4" onClick={() => setForm(null)}>
   265→          <div className="max-h-[85vh] w-full max-w-lg overflow-y-auto rounded-lg bg-white p-5 shadow-xl" onClick={(e) => e.stopPropagation()}>
   266→            <h3 className="mb-4 text-base font-semibold">{form.id ? '编辑小说' : '新增小说'}</h3>

   271→                <select
   272→                  className="h-9 w-full rounded-md border border-neutral-200 px-2 text-sm"
   273→                  value={form.categoryId}
   274→                  onChange={(e) => setForm({ ...form, categoryId: Number(e.target.value) })}
   275→                >
   276→                  <option value={0}>请选择…</option>
   277→                  {categories?.map((c) => <option key={c.id} value={c.id}>{c.name}</option>)}
   278→                </select>
   279→              </Field>
   280→              <Field label="状态">
   281→                <select
   282→                  className="h-9 w-full rounded-md border border-neutral-200 px-2 text-sm"
   283→                  value={form.status}
   284→                  onChange={(e) => setForm({ ...form, status: e.target.value as 'serial' | 'finished' })}
   285→                >
   286→                  <option value="serial">连载中</option>
   287→                  <option value="finished">已完本</option>
   288→                </select>
   289→              </Field>
   290→              <Field label="简介"><Textarea rows={4} value={form.description} onChange={(e) => setForm({ ...form, description: e.target.value })} /></Field>

   294→              </div>
   295→            </div>
   296→            <div className="mt-5 flex justify-end gap-2">
   297→              <Button variant="outline" onClick={() => setForm(null)}>取消</Button>

   306→  )
   307→}

   320→function ChaptersDialog({ novel, onClose }: { novel: NovelListItem; onClose: () => void }) {
   321→  const qc = useQueryClient()
   322→  const { data: chapters, isLoading } = useChapters(novel.id)
   323→  const [title, setTitle] = useState('')
   324→  const [content, setContent] = useState('')
   325→  const [editing, setEditing] = useState<{ id: number; title: string; content: string } | null>(null)
   326→  const [adding, setAdding] = useState(false)
   327→
   328→  const add = async () => {
   329→    if (!title.trim()) return toast.error('标题不能为空')
   330→    setAdding(true)
   331→    try {
   332→      await api('/api/chapters', { method: 'POST', body: JSON.stringify({ novelId: novel.id, title, content }) })
   333→      toast.success('章节已添加')
   334→      setTitle(''); setContent('')
   335→      await qc.invalidateQueries({ queryKey: qk.chapters(novel.id) })
   336→      await qc.invalidateQueries({ queryKey: ['novels'] })
   337→      await qc.invalidateQueries({ queryKey: ['novel'] })
   338→      await qc.invalidateQueries({ queryKey: qk.home })
   339→    } catch (e) {
   340→      toast.error(e instanceof Error ? e.message : '添加失败')
   341→    } finally {
   342→      setAdding(false)
   343→    }
   344→  }
   345→
   346→  const saveEdit = async () => {
   347→    if (!editing) return
   348→    try {
   349→      await api(`/api/chapters/${editing.id}`, { method: 'PUT', body: JSON.stringify({ title: editing.title, content: editing.content }) })
   350→      toast.success('已保存')
   351→      setEditing(null)
   352→      await qc.invalidateQueries({ queryKey: qk.chapters(novel.id) })
   353→      await qc.invalidateQueries({ queryKey: ['chapter'] })
   354→      await qc.invalidateQueries({ queryKey: ['novel'] })
   355→    } catch (e) {
   356→      toast.error(e instanceof Error ? e.message : '保存失败')
   357→    }
   358→  }
   359→
   360→  const remove = async (id: number) => {
   361→    if (!confirm('确认删除该章节？')) return
   362→    try {
   363→      await api(`/api/chapters/${id}`, { method: 'DELETE' })
   364→      toast.success('已删除')
   365→      await qc.invalidateQueries({ queryKey: qk.chapters(novel.id) })
   366→      await qc.invalidateQueries({ queryKey: ['novels'] })
   367→      await qc.invalidateQueries({ queryKey: ['novel'] })
   368→      await qc.invalidateQueries({ queryKey: ['chapter'] })
   369→    } catch (e) {
   370→      toast.error(e instanceof Error ? e.message : '删除失败')
   371→    }
   372→  }
   373→
   374→  return (
   375→    <div className="fixed inset-0 z-[60] flex items-center justify-center bg-black/40 p-4" onClick={onClose}>
   376→      <div className="flex max-h-[85vh] w-full max-w-2xl flex-col overflow-hidden rounded-lg bg-white shadow-xl" onClick={(e) => e.stopPropagation()}>
   377→        <div className="border-b px-5 py-4">
   378→          <h3 className="text-base font-semibold">章节管理 · {novel.title}</h3>
   379→          <p className="mt-0.5 text-xs text-neutral-500">共 {novel.chapterCount} 章</p>
   380→        </div>
   381→        <div className="min-h-0 flex-1 overflow-y-auto p-5">
   382→          <div className="mb-4 space-y-2 rounded-lg border bg-neutral-50 p-3">
   383→            <p className="text-xs font-medium text-neutral-600">新增章节</p>
   384→            <Input placeholder="章节标题" value={title} onChange={(e) => setTitle(e.target.value)} className="h-8" />
   385→            <Textarea placeholder="正文（可留空，稍后编辑）" rows={3} value={content} onChange={(e) => setContent(e.target.value)} />
   386→            <Button size="sm" onClick={add} disabled={adding}>{adding ? '添加中…' : <><Plus className="mr-1 h-3.5 w-3.5" />添加</>}</Button>
   387→          </div>
   388→          {isLoading && <p className="py-6 text-center text-sm text-neutral-400">加载中…</p>}
   389→          <div className="max-h-72 space-y-1 overflow-y-auto">
   390→            {chapters?.map((c) => (
   391→              <div key={c.id} className="flex items-center gap-2 rounded border px-3 py-1.5 text-sm">
   392→                <span className="w-10 shrink-0 text-xs text-neutral-400">{c.idx}</span>
   393→                <span className="min-w-0 flex-1 truncate">{c.title}</span>
   394→                <span className="shrink-0 text-xs text-neutral-400">{formatWordCount(c.wordCount)}字</span>
   395→                <Button size="sm" variant="ghost" className="h-6 px-1.5" onClick={async () => {
   396→                  try {
   397→                    const detail = await api<{ title: string; content: string }>(`/api/chapters/${c.id}`)
   398→                    setEditing({ id: c.id, title: detail.title, content: detail.content })
   399→                  } catch (e) {
   400→                    toast.error(e instanceof Error ? e.message : '加载章节失败')
   401→                  }
   402→                }}>
   403→                  <Pencil className="h-3 w-3" />
   404→                </Button>
   405→                <Button size="sm" variant="ghost" className="h-6 px-1.5 text-red-500" onClick={() => remove(c.id)}>
   406→                  <Trash2 className="h-3 w-3" />
   407→                </Button>
   408→              </div>
   409→            ))}
   410→          </div>
   411→        </div>
   412→        <div className="flex justify-end border-t px-5 py-3">
   413→          <Button variant="outline" onClick={onClose}>关闭</Button>
   414→        </div>
   415→      </div>
   416→
   417→      {editing && (
   418→        <div className="fixed inset-0 z-[70] flex items-center justify-center bg-black/50 p-4" onClick={() => setEditing(null)}>
