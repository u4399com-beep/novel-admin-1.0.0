   489→type SortKey = 'latest' | 'clicks' | 'words'
   490→type StatusKey = 'all' | 'serial' | 'finished'
   491→
   492→function CategoryView({
   493→  navigate,
   494→  categoryId,
   495→  page = 1,
   496→}: ViewProps & { categoryId?: number; page?: number }) {
   497→  const { data: cats } = useCategories()
   498→  const [sort, setSort] = useState<SortKey>('latest')
   499→  const [status, setStatus] = useState<StatusKey>('all')
   500→  const cur = page > 0 ? page : 1
   501→
   502→  const { data, isLoading, isError, refetch } = useNovels({
   503→    categoryId,
   504→    page: cur,
   505→    pageSize: 20,
   506→    sort,
   507→    status: status === 'all' ? undefined : status,
   508→  })
   509→
   510→  const catName =

   614→function loadShelf(): number[] {
   615→  if (typeof window === 'undefined') return []
   616→  try {

   623→function toggleShelf(id: number): boolean {
   624→  if (typeof window === 'undefined') return false

   626→    const arr = loadShelf()
   627→    const has = arr.includes(id)

   629→    return !has
   630→  } catch {
   631→    return false
   632→  }
