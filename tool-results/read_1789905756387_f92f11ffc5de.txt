   201→          <div
   202→            key={i}
   203→            className={cn(
   204→              'h-[64px] w-[44px] overflow-hidden rounded-[2px] shadow-md',

   240→export function CategoryView({
   241→  navigate,
   242→  categoryId,
   243→  page = 1,
   244→}: ViewProps & { categoryId?: number; page?: number }) {
   245→  const { data: categories } = useCategories()
   246→  const [sort, setSort] = useState<'latest' | 'clicks' | 'words'>('latest')
   247→  const [status, setStatus] = useState<'serial' | 'finished' | undefined>(undefined)
   248→
   249→  // 消费首页快捷按钮 / 导航注入的初始筛选意图（渲染期一次性消费，替代 effect）
   250→  const [intentKey, setIntentKey] = useState<string | null>(null)
   251→  const curIntentKey = `${String(categoryId)}:${String(page)}`
   252→  if (intentKey !== curIntentKey) {
   253→    setIntentKey(curIntentKey)
   254→    const intent = takeCategoryIntent()
   255→    if (intent) {
   256→      setSort(intent.sort ?? 'latest')
   257→      setStatus(intent.status)
   258→    }
   259→  }
   260→
   261→  const { data, isLoading, isError, refetch } = useNovels({ categoryId, page, pageSize: 20, sort, status })
   262→  const rank = useNovels({ categoryId, page: 1, pageSize: 10, sort: 'clicks' })
   263→
   264→  const catName = categories?.find((c) => c.id === categoryId)?.name ?? '全部分類'
   265→
   266→  return (
   267→    <Container className="space-y-4 py-4">
   268→      {/* 白卡一：分类 + 筛选 */}
   269→      <MyBox>
   270→        <SectionTitle>小說分類</SectionTitle>
   271→        <div className="no-scrollbar mt-3 flex gap-2 overflow-x-auto pb-1">
   272→          <BlueTag active={!categoryId} onClick={() => navigate({ name: 'category', page: 1 })}>
   273→            全部分類
   274→          </BlueTag>
   275→          {(categories ?? []).map((c) => (
   276→            <BlueTag
   277→              key={c.id}
   278→              active={categoryId === c.id}
   279→              onClick={() => navigate({ name: 'category', categoryId: c.id, page: 1 })}
   280→            >
   281→              {c.name}
   282→            </BlueTag>
   283→          ))}
   284→        </div>
   285→        <div className="mt-2.5 flex flex-wrap items-center gap-2">
   286→          <span className="text-xs text-[#888]">排序</span>
   287→          {(
   288→            [
   289→              { key: 'latest', label: '最新更新' },
   290→              { key: 'clicks', label: '點擊最多' },
   291→              { key: 'words', label: '字數最多' },
   292→            ] as const
   293→          ).map((o) => (
   294→            <button
   295→              key={o.key}
   296→              type="button"
   297→              onClick={() => {
   298→                setSort(o.key)
   299→                if (page !== 1) navigate({ name: 'category', categoryId, page: 1 })
   300→              }}
   301→              className={cn(
   302→                'cursor-pointer rounded-[3px] px-2 py-1 text-[13px] transition-colors hover:text-[#06c]',
   303→                sort === o.key ? 'bg-[#e8f4ff] font-bold text-[#1f6cb2]' : 'text-[#666]',
   304→              )}
   305→            >
   306→              {o.label}
   307→            </button>
   308→          ))}
   309→          <span className="ml-2 text-xs text-[#888]">狀態</span>
   310→          {(
   311→            [
   312→              { key: undefined, label: '全部' },
   313→              { key: 'serial', label: '連載' },
   314→              { key: 'finished', label: '完本' },
   315→            ] as const
   316→          ).map((o) => (
   317→            <button
   318→              key={o.label}
   319→              type="button"
   320→              onClick={() => {
   321→                setStatus(o.key)
   322→                if (page !== 1) navigate({ name: 'category', categoryId, page: 1 })
   323→              }}
   324→              className={cn(
   325→                'cursor-pointer rounded-[3px] px-2 py-1 text-[13px] transition-colors hover:text-[#06c]',
   326→                status === o.key ? 'bg-[#e8f4ff] font-bold text-[#1f6cb2]' : 'text-[#666]',
   327→              )}
   328→            >
   329→              {o.label}
   330→            </button>
   331→          ))}
   332→          <span className="ml-auto text-xs text-[#888]">
   333→            {data ? `${catName} · 共 ${data.total} 部` : ''}
   334→          </span>
   335→        </div>
   336→      </MyBox>
   337→
   338→      {/* 封面卡网格 */}
   339→      <MyBox>
   340→        {isError ? (
   341→          <ErrorRetry onRetry={() => refetch()} />
   342→        ) : isLoading ? (
   343→          <GridSkeleton />
   344→        ) : data && data.list.length > 0 ? (
   345→          <div className="grid grid-cols-3 gap-3 sm:grid-cols-4 md:grid-cols-5 lg:grid-cols-6">
   346→            {data.list.map((n) => (
   347→              <CoverCard key={n.id} novel={n} onClick={() => navigate({ name: 'book', novelId: n.id })} />
   348→            ))}
   349→          </div>
   350→        ) : (
   351→          <div className="flex flex-col items-center py-14 text-[#888]">
   352→            <SearchX className="h-9 w-9" />
   353→            <p className="mt-3 text-sm">該篩選條件下暫無收錄</p>

   355→        )}
   356→        {data && data.list.length > 0 && (
   357→          <Pager page={data.page} totalPages={data.totalPages} onGo={(p) => navigate({ name: 'category', categoryId, page: p })} />

   359→      </MyBox>
   360→
   361→      {/* 點擊排行列表（全宽横向小卡行） */}
   362→      <MyBox>
   363→        <SectionTitle>點擊排行</SectionTitle>

   367→              key={n.id}
   368→              onClick={() => navigate({ name: 'book', novelId: n.id })}
   369→              className="group flex cursor-pointer items-center gap-3 rounded-[3px] p-2 transition-colors hover:bg-[#f2f7fc]"

   371→              <span
   372→                className={cn(
   373→                  'w-6 shrink-0 text-center text-sm font-bold',
   374→                  i < 3 ? 'text-[#e84118]' : 'text-[#888]',

   376→              >
   377→                {i + 1}
   378→              </span>
   379→              <CoverFace
   380→                title={n.title}
   381→                cover={n.cover}
   382→                className="h-[80px] w-[60px] shrink-0 rounded-[3px]"
   383→                charClass="text-xl"
   384→              />
   385→              <div className="min-w-0 flex-1">
   386→                <p className="truncate text-[15px] text-[#333] transition-colors group-hover:text-[#06c]">{n.title}</p>
   387→                <p className="mt-0.5 truncate text-xs text-[#888]">{n.author}</p>

   389→                  {n.categoryName} · {statusLabel(n.status)} · {fmtWords(n.wordCount)}
   390→                </p>
   391→              </div>
   392→            </div>
   393→          ))}
   394→        </div>
   395→      </MyBox>
   396→    </Container>

   402→export function BookView({ navigate, novelId }: ViewProps & { novelId: number }) {
   403→  const { data: novel, isLoading, isError, refetch } = useNovel(novelId)
   404→  const sideRank = useNovels({ sort: 'clicks', pageSize: 20 })
   405→  const [shelf, setShelf] = useState(() => loadShelf().includes(novelId))
   406→  const [votes, setVotes] = useState(0)
   407→  const [tab, setTab] = useState<'toc' | 'intro' | 'review'>('toc')
   408→  const [sideTab, setSideTab] = useState<'hot' | 'finished'>('hot')
   409→  const [rating, setRating] = useState(0)
   410→
   411→  const [shelfNovelId, setShelfNovelId] = useState(novelId)
   412→
   413→  if (shelfNovelId !== novelId) {
   414→    setShelfNovelId(novelId)
   415→    setShelf(loadShelf().includes(novelId))
   416→    setVotes(0)
   417→    setTab('toc')
   418→  }
   419→
   420→  if (isError) return <Container className="py-6"><ErrorRetry onRetry={() => refetch()} /></Container>
   421→  if (isLoading || !novel) return <Container className="py-6"><BookSkeleton /></Container>
   422→
   423→  const sideList =
   424→    sideTab === 'hot'
   425→      ? (sideRank.data?.list ?? []).slice(0, 10)
   426→      : (sideRank.data?.list ?? []).filter((n) => n.status === 'finished').slice(0, 10)
   427→
   428→  return (
   429→    <Container className="py-4">
   430→      {/* 面包屑 */}
   431→      <nav className="flex flex-wrap items-center gap-1.5 py-3 text-sm">
   432→        <button type="button" onClick={() => navigate({ name: 'home' })} className="cursor-pointer text-[#1f6cb2] hover:text-[#06c]">
   433→          首頁
   434→        </button>
   435→        <span className="text-[#bbb]">/</span>
   436→        <button
   437→          type="button"
   438→          onClick={() => navigate({ name: 'category', categoryId: novel.categoryId, page: 1 })}
   439→          className="cursor-pointer text-[#1f6cb2] hover:text-[#06c]"
   440→        >
   441→          {novel.categoryName}
   442→        </button>
   443→        <span className="text-[#bbb]">/</span>
   444→        <span className="truncate text-[#666]">{novel.title}</span>
   445→      </nav>
   446→
   447→      {/* 66/32 两列 */}
   448→      <div className="flex flex-col gap-4 lg:flex-row lg:items-start lg:gap-5">
   449→        {/* 主栏 */}
   450→        <div className="min-w-0 flex-1 space-y-4 lg:max-w-[66%]">
   451→          {/* 盒一：书籍信息 */}
   452→          <MyBox>
   453→            <div className="flex flex-col gap-5 sm:flex-row">
   454→              {/* 封面 180×240 + 角章 + 底部浮层 */}
   455→              <div className="relative mx-auto w-[180px] shrink-0 sm:mx-0">
   456→                <CoverFace
   457→                  title={novel.title}
   458→                  cover={novel.cover}
   459→                  className="h-[240px] w-[180px] rounded-[3px] shadow-[0_4px_12px_rgba(0,0,0,.25)]"
   460→                  charClass="text-6xl"
   461→                />
   462→                <span className="absolute left-0 top-2 w-[69px] bg-[#1f6cb2] py-1 text-center text-xs text-white">
   463→                  {statusLabel(novel.status)}
   464→                </span>
   465→                <span className="absolute inset-x-0 bottom-0 rounded-b-[3px] bg-black/70 py-1 text-center text-xs text-white">
   466→                  推薦 {novel.clicks + votes}
