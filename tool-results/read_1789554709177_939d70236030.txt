    49→async function api<T>(url: string, init?: RequestInit): Promise<T> {
    50→  const res = await fetch(url, { headers: { 'Content-Type': 'application/json' }, ...init })
    51→  const data = (await res.json().catch(() => ({}))) as { error?: string }
    52→  if (!res.ok) throw new Error(data.error ?? `请求失败(${res.status})`)
    53→  return data as T
    54→}

    78→/** 仅保留非空字符串字段 */
    79→function cleanRule(obj: object): Record<string, string> {
    80→  const out: Record<string, string> = {}
    81→  for (const [k, v] of Object.entries(obj)) {
    82→    if (typeof v === 'string' && v.trim()) out[k] = v.trim()
    83→  }
    84→  return out
    85→}

    87→function truncate(s: string, n: number): string {
    88→  return s.length > n ? `${s.slice(0, n)}…` : s
    89→}

   442→                  variant="ghost"

   478→function NewTaskSection({
   479→  rules,
   480→  onCreated,
   481→}: {
   482→  rules: ScrapeRuleDto[]
   483→  onCreated: () => void
   484→}) {
   485→  const [mode, setMode] = useState<'single' | 'list'>('single')
   486→  const [targetUrl, setTargetUrl] = useState('')
   487→  const [ruleId, setRuleId] = useState<string>('none')
   488→  const [pages, setPages] = useState('1')
   489→  const [creating, setCreating] = useState(false)
   490→
   491→  const selectedRule = rules.find((r) => String(r.id) === ruleId)
   492→
   493→  const submit = async () => {
   494→    const url = targetUrl.trim()
   495→    if (!url) return toast.error('请输入目标 URL')
   496→    let p = 1
   497→    if (mode === 'list') {
   498→      p = Math.floor(Number(pages) || 0)
   499→      if (p < 1 || p > 20) return toast.error('列表页数需为 1-20 的整数')
   500→    }
   501→    setCreating(true)
   502→    try {
   503→      await api('/api/scrape-tasks', {
   504→        method: 'POST',
   505→        body: JSON.stringify({
   506→          mode,
   507→          targetUrl: url,
   508→          ruleId: ruleId === 'none' ? undefined : Number(ruleId),
   509→          pages: p,
   510→        }),
   511→      })
   512→      toast.success('采集任务已创建，开始执行')
   513→      setTargetUrl('')
   514→      onCreated()
   515→    } catch (e) {
   516→      toast.error(e instanceof Error ? e.message : '创建失败')
   517→    } finally {
   518→      setCreating(false)
   519→    }
   520→  }
   521→
   522→  return (
   523→    <section className="rounded-lg border p-4">
   524→      <h4 className="mb-3 flex items-center gap-1.5 text-sm font-semibold">
   525→        <ListPlus className="h-4 w-4" /> 新建采集任务
   526→      </h4>
   527→
   528→      <div className="space-y-3">
   529→        <RadioGroup
   530→          value={mode}
   531→          onValueChange={(v) => setMode(v as 'single' | 'list')}
   532→          className="flex flex-wrap gap-4"
   533→        >
   534→          <div className="flex items-center gap-2">
   535→            <RadioGroupItem value="single" id="mode-single" />
   536→            <Label htmlFor="mode-single" className="text-xs">
   537→              单本采集（书页 URL）
   538→            </Label>

   638→/** 任务日志对话框（执行中的任务每 2s 刷新并自动滚到底部） */
   639→function LogDialog({ task, onClose }: { task: TaskRow; onClose: () => void }) {
   640→  const isActive = task.status === 'pending' || task.status === 'running'
   641→  const { data, isLoading } = useQuery({
   642→    queryKey: ['scrape-task', task.id],
   643→    queryFn: () => api<{ task: TaskDetail }>(`/api/scrape-tasks/${task.id}`),
   644→    refetchInterval: isActive ? 2000 : false,
   645→  })
   646→  const detail = data?.task
   647→  const preRef = useRef<HTMLPreElement>(null)
   648→  const log = detail?.log ?? ''
   649→  // 日志更新后滚动到底部（ref 副作用，无 state 同步）
   650→  useEffect(() => {
   651→    if (preRef.current) preRef.current.scrollTop = preRef.current.scrollHeight
   652→  }, [log])
   653→
   654→  return (
   655→    <Dialog open onOpenChange={(o) => !o && onClose()}>
   656→      <DialogContent className="sm:max-w-xl">
   657→        <DialogHeader>
   658→          <DialogTitle className="flex items-center gap-2 text-sm">
   659→            任务 #{task.id} 日志 <StatusBadge status={detail?.status ?? task.status} />
   660→          </DialogTitle>
   661→          <DialogDescription className="break-all">
   662→            {truncate(detail?.targetUrl ?? task.targetUrl, 80)}
   663→            {detail?.message ? ` — ${detail.message}` : ''}
   664→          </DialogDescription>
   665→        </DialogHeader>
   666→        <pre
   667→          ref={preRef}
   668→          className="max-h-96 overflow-y-auto whitespace-pre-wrap break-all rounded-md bg-neutral-950 p-3 font-mono text-[11px] leading-relaxed text-emerald-300"
   669→        >
   670→          {isLoading ? '日志加载中…' : log || '暂无日志'}
   671→        </pre>
   672→        <DialogFooter>
   673→          <Button variant="outline" size="sm" onClick={onClose}>
   674→            关闭
   675→          </Button>
   676→        </DialogFooter>
   677→      </DialogContent>
   678→    </Dialog>
   679→  )
   680→}

   682→function TaskListSection() {
   683→  const qc = useQueryClient()
   684→  const [page, setPage] = useState(1)
   685→  const [logTask, setLogTask] = useState<TaskRow | null>(null)
   686→
   687→  const { data, isLoading } = useQuery({
   688→    queryKey: ['scrape-tasks', page],
   689→    queryFn: () =>
   690→      api<{ list: TaskRow[]; total: number }>(`/api/scrape-tasks?page=${page}&pageSize=${PAGE_SIZE}`),
   691→    refetchInterval: (query) => {
   692→      const list = query.state.data?.list ?? []
   693→      return list.some((t) => t.status === 'pending' || t.status === 'running') ? 3000 : false
   694→    },
   695→  })
   696→
   697→  const rows = data?.list ?? []
   698→  const total = data?.total ?? 0
   699→  const totalPages = Math.max(1, Math.ceil(total / PAGE_SIZE))
   700→
   701→  const refresh = () => qc.invalidateQueries({ queryKey: ['scrape-tasks'] })
   702→
   703→  const cancel = async (t: TaskRow) => {
   704→    try {
   705→      await api(`/api/scrape-tasks/${t.id}`, {
   706→        method: 'PATCH',
   707→        body: JSON.stringify({ action: 'cancel' }),
   708→      })
   709→      await refresh()
   710→      toast.success(`已发送取消指令（任务 #${t.id}）`)
   711→    } catch (e) {
   712→      toast.error(e instanceof Error ? e.message : '取消失败')
   713→    }
   714→  }
   715→
   716→  const remove = async (t: TaskRow) => {
   717→    try {
   718→      await api(`/api/scrape-tasks/${t.id}`, { method: 'DELETE' })
   719→      await refresh()
   720→      toast.success('任务已删除')
   721→    } catch (e) {
   722→      toast.error(e instanceof Error ? e.message : '删除失败')
   723→    }
   724→  }
   725→
   726→  return (
   727→    <section className="rounded-lg border p-4">
   728→      <div className="mb-3 flex items-center justify-between">
   729→        <h4 className="text-sm font-semibold">采集任务（{total}）</h4>
   730→        {totalPages > 1 && (
   731→          <div className="flex items-center gap-1.5 text-xs text-neutral-500">
   732→            <Button
   733→              size="sm"
   734→              variant="outline"
   735→              className="h-7"
   736→              disabled={page <= 1}
   737→              onClick={() => setPage((p) => Math.max(1, p - 1))}
   738→            >
   739→              上一页
   740→            </Button>
   741→            <span className="tabular-nums">
   742→              {page} / {totalPages}
   743→            </span>
   744→            <Button
   745→              size="sm"
   746→              variant="outline"
   747→              className="h-7"
   748→              disabled={page >= totalPages}
   749→              onClick={() => setPage((p) => Math.min(totalPages, p + 1))}
   750→            >
   751→              下一页
   752→            </Button>
   753→          </div>
   754→        )}
   755→      </div>
   756→
   757→      {isLoading ? (
   758→        <p className="py-8 text-center text-xs text-neutral-400">任务加载中…</p>
   759→      ) : rows.length === 0 ? (
   760→        <p className="py-8 text-center text-xs text-neutral-400">暂无采集任务，请在上方创建</p>
   761→      ) : (
   762→        <div className="max-h-[420px] overflow-y-auto">
   763→          <Table>
   764→            <TableHeader>
   765→              <TableRow className="hover:bg-transparent">
   766→                <TableHead className="h-8 w-10 text-xs">ID</TableHead>
   767→                <TableHead className="h-8 w-14 text-xs">模式</TableHead>
   768→                <TableHead className="h-8 text-xs">目标 URL</TableHead>
   769→                <TableHead className="h-8 w-20 text-xs">状态</TableHead>
   770→                <TableHead className="h-8 w-32 text-xs">进度</TableHead>
   771→                <TableHead className="hidden h-8 w-36 text-xs md:table-cell">成果</TableHead>
   772→                <TableHead className="hidden h-8 w-20 text-xs md:table-cell">时间</TableHead>
   773→                <TableHead className="h-8 w-24 text-right text-xs">操作</TableHead>
   774→              </TableRow>
   775→            </TableHeader>
   776→            <TableBody>
   777→              {rows.map((t) => {
   778→                const pct = t.total > 0 ? Math.min(100, Math.round((t.done / t.total) * 100)) : 0
   779→                const active = t.status === 'pending' || t.status === 'running'
   780→                return (
   781→                  <TableRow key={t.id}>
   782→                    <TableCell className="p-2 text-xs tabular-nums text-neutral-500">{t.id}</TableCell>
   783→                    <TableCell className="p-2">
   784→                      <ModeBadge mode={t.mode} />
   785→                    </TableCell>
   786→                    <TableCell className="max-w-0 p-2">
   787→                      <span className="block truncate text-xs" title={t.targetUrl}>
   788→                        {truncate(t.targetUrl, 40)}
   789→                      </span>
   790→                      {t.message && (
   791→                        <span className="block truncate text-[11px] text-neutral-400" title={t.message}>
   792→                          {truncate(t.message, 40)}
   793→                        </span>
   794→                      )}
   795→                    </TableCell>
   796→                    <TableCell className="p-2">
   797→                      <StatusBadge status={t.status} />
   798→                    </TableCell>
   799→                    <TableCell className="p-2">
   800→                      <div className="flex items-center gap-1.5">
   801→                        <span className="w-14 shrink-0 text-[11px] tabular-nums text-neutral-500">
   802→                          {t.done}/{t.total}
   803→                        </span>
   804→                        <Progress value={pct} className="h-1.5 w-16" aria-label={`任务 ${t.id} 进度 ${pct}%`} />
   805→                      </div>

   808→                      新建 {t.created} · 更新 {t.updated} · 章节 {t.chapters}
   809→                    </TableCell>

   811→                      {timeAgo(t.updatedAt)}
   812→                    </TableCell>
   813→                    <TableCell className="p-2 text-right">
   814→                      <div className="flex justify-end gap-0.5">
   815→                        <Button
   816→                          size="sm"
   817→                          variant="ghost"
   818→                          className="h-7 px-1.5"
   819→                          title="查看日志"
   820→                          aria-label={`查看任务 ${t.id} 日志`}
   821→                          onClick={() => setLogTask(t)}
   822→                        >
   823→                          <ScrollText className="h-3.5 w-3.5" />
   824→                        </Button>
   825→                        {active && (
   826→                          <Button
   827→                            size="sm"
   828→                            variant="ghost"
   829→                            className="h-7 px-1.5 text-amber-600 hover:text-amber-700"
   830→                            title="取消任务"
   831→                            aria-label={`取消任务 ${t.id}`}
   832→                            onClick={() => cancel(t)}
   833→                          >
   834→                            <Square className="h-3.5 w-3.5" />
   835→                          </Button>
   836→                        )}
   837→                        <Button
   838→                          size="sm"
   839→                          variant="ghost"
   840→                          className="h-7 px-1.5 text-red-500 hover:text-red-600"
   841→                          title="删除任务"
   842→                          aria-label={`删除任务 ${t.id}`}
   843→                          onClick={() => remove(t)}
   844→                        >
   845→                          <Trash2 className="h-3.5 w-3.5" />
   846→                        </Button>
   847→                      </div>
   848→                    </TableCell>
   849→                  </TableRow>
   850→                )
   851→              })}
   852→            </TableBody>
   853→          </Table>
   854→        </div>
   855→      )}
   856→
   857→      {logTask && <LogDialog task={logTask} onClose={() => setLogTask(null)} />}
   858→    </section>
   859→  )
