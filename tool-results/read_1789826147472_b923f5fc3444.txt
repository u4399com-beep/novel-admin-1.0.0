    66→/**
    67→ * 僵尸任务恢复（心跳判定）：worker 常驻在 Next.js 进程内存里（fire-and-forget promise），
    68→ * 进程重启后内存任务丢失，DB 中 status=pending/running 的记录会永久卡死。
    69→ *
    70→ * 判定依据：运行中任务每完成一章/一书都会 flush 触碰 updatedAt（心跳）；真正的僵尸
    71→ * （进程消失/模块实例丢失）updatedAt 停摆。仅回收「仍处 pending/running 且心跳停摆
    72→ * 超过 ZOMBIE_STALE_MS」的任务——不按 createdAt/进程启动时刻判定，因为 dev 下
    73→ * Turbopack 每个路由 bundle 各有一份本模块实例，任何晚加载实例的时钟基线都不同，
    74→ * 按创建时刻会把「重跑老任务」等活任务误杀（实测踩坑）。
    75→ * 心跳最长间隙 = 引擎单请求预算 60s + 落库时间，远小于 5 分钟阈值。
    76→ *
    77→ * 防重入标志同步置位（await 前置位）：dev 下同进程可能存在多份模块实例
    78→ * （Turbopack 按路由分包），异步置位会留下双重回收的竞态窗口。
    79→ */
    80→const ZOMBIE_STALE_MS = 5 * 60_000
    81→
    82→async function recoverStaleTasks(): Promise<void> {
    83→  if (gWorker.__scrapeRecovered) return
    84→  gWorker.__scrapeRecovered = true // 同步置位：跨实例/跨 HMR 防重入
    85→  try {
    86→    const stale = await db.scrapeTask.findMany({
    87→      where: {
    88→        status: { in: ['pending', 'running'] },
    89→        updatedAt: { lt: new Date(Date.now() - ZOMBIE_STALE_MS) },
    90→      },
    91→      select: { id: true, log: true },
    92→    })
    93→    let recovered = 0
    94→    for (const t of stale) {
    95→      if (running.has(t.id)) continue // 本进程内存中仍有实例在执行（共享 Set 时双保险）
    96→      const line = `[${new Date().toTimeString().slice(0, 8)}] 心跳停摆超 5 分钟，判定僵尸任务并回收`
    97→      const log = `${t.log ? `${t.log}\n` : ''}${line}`
    98→      const res = await db.scrapeTask
    99→        .updateMany({
   100→          where: { id: t.id, status: { in: ['pending', 'running'] } },
   101→          data: {
   102→            status: 'failed',
   103→            message: '任务心跳超时，自动回收',
   104→            // 与 Run 同规约：只保留最近 100 行
   105→            log: log.split('\n').slice(-MAX_LOG_LINES).join('\n'),
   106→          },
   107→        })
   108→        .catch(() => null)
   109→      if (res && res.count > 0) recovered++
   110→    }
   111→    if (recovered > 0) console.log(`[scrape-worker] 僵尸任务回收: ${recovered} 条`)
   112→  } catch {
   113→    // 回收失败不影响模块加载；标志已置位，等待下次进程重启重试（代价可接受：僵尸多驻留一轮）
   114→  }
   115→}

   118→/** 协作式取消检查：记录不存在视为取消；DB 瞬时错误不误判为取消（fail-open） */
   119→async function isCanceled(taskId: number): Promise<boolean> {
   120→  const t = await db.scrapeTask
   121→    .findUnique({ where: { id: taskId }, select: { status: true } })
   122→    .catch(() => undefined)
   123→  if (t === undefined) return false // 查询失败 ≠ 被取消，避免瞬时 DB 错误误停任务
   124→  return !t || t.status === 'canceled'
   125→}

   127→/**
   128→ * 并行池：cursor 抢占式派发，concurrency 路 worker 持续取件直至耗尽。
   129→ * fn 内部通过 stop signal 提前收敛（停止派发，在途任务跑完当前单元后自然退出）。
   130→ */
   131→async function runPool<T>(items: T[], concurrency: number, fn: (item: T, index: number) => Promise<void>): Promise<void> {
   132→  let cursor = 0
   133→  const workers = Array.from({ length: Math.max(1, Math.min(concurrency, items.length)) }, async () => {
   134→    for (;;) {
   135→      const i = cursor++
   136→      if (i >= items.length) return
   137→      await fn(items[i], i)
   138→    }
   139→  })
   140→  await Promise.all(workers)
   141→}

   143→/** 生成第 k 页候选 URL：?page=k（已有 query 则 &page=k）与 /page/k 两种变体 */
   144→function pageVariants(url: string, k: number): string[] {
   145→  const out: string[] = []
   146→  try {
   147→    const u = new URL(url)
   148→    u.searchParams.set('page', String(k))
   149→    out.push(u.toString())
   150→    const p = new URL(url)
   151→    p.pathname = `${p.pathname.replace(/\/+$/, '')}/page/${k}`
   152→    p.search = ''
   153→    if (p.toString() !== u.toString()) out.push(p.toString())
   154→  } catch {
   155→    out.push(url.includes('?') ? `${url}&page=${k}` : `${url}?page=${k}`)
   156→    out.push(`${url.replace(/\/+$/, '')}/page/${k}`)
   157→  }
   158→  return [...new Set(out)]
   159→}

   163→/**
   164→ * 抓取一个书页并入库（含并行章节抓取）。
   165→ * - 新书 created+1 / 已有书 updated+1（按 title+author 查重，结果记入 run.counters）
   166→ * - 章节 idx 按目录顺序预分配槽位（base+目录位次，保序），并行抓取、按槽位入库
   167→ * - 入库成功后异步触发下拉词绑定 → PSEO 书籍页（不阻塞章节抓取）
   168→ * - 进度写入：
   169→ *   - trackTotal=true（single 模式）：章节总数写入 task.total，done 随章节递增（done/total=章节，主口径）
   170→ *   - trackTotal=false（list 模式）：done/total 由调用方按「书」维护，本函数不碰；
   171→ *     章节进度累加进共享的 opts.chapterProgress（chaptersDone/chaptersTotal）并随写盘刷出
   172→ */
   173→async function processBook(
   174→  run: Run,
   175→  bookUrl: string,
   176→  rule: LoadedRule,
   177→  opts: { trackTotal: boolean; chapterProgress?: { done: number; total: number }; referer?: string | null; concurrency: number },
   178→): Promise<BookOutcome> {
   179→  const canceledOutcome = (message: string, chapters = 0, failedChapters = 0): BookOutcome => ({
   180→    ok: false,
   181→    canceled: true,
   182→    chapters,
   183→    failedChapters,
   184→    message,
   185→  })
   186→
   187→  run.log(`抓取书页 ${bookUrl.slice(0, 120)}…`)
   188→  // Referer 链：list 模式传「发现本书的列表页」作来路；single 模式缺省由引擎回落站内首页。
   189→  // 熔断感知：书页撞上熔断（站点临时限流封禁）时等待冷却后自动重试，而非立即把本书记失败
   190→  const page = await withCircuitRetry(hostOf(bookUrl), () => fetchBookPage(run, bookUrl, rule, opts.referer ?? undefined), {
   191→    detect: (r) => (r.ok ? null : circuitWaitMs(r.error, r.circuitRetryAfterMs)),
   192→    isAlive: async () => !(await isCanceled(run.taskId)),
   193→    onWait: (ms, cycle, max) =>
   194→      run.log(`目标主机熔断冷却中，暂停 ${Math.round(ms / 1000)}s 后自动重试书页（第 ${cycle}/${max} 轮，等待期间不向源站发请求）`),
   195→  })
   196→  if (!page.ok) {
   197→    run.log(`书页提取失败: ${page.error}`)
   198→    return { ok: false, canceled: false, chapters: 0, failedChapters: 0, message: page.error }
   199→  }
   200→  const book = page.book
   201→  // 引擎响应未经 schema 校验（callEngine 直接 as 断言），关键字段做形态兜底，
   202→  // 防畸形载荷（引擎版本错位等）在任务中途抛 TypeError 把整任务拖成 failed
   203→  if (typeof book.title !== 'string') book.title = ''
   204→  if (!Array.isArray(book.chapters)) book.chapters = []
   205→  if (typeof book.description !== 'string') book.description = ''
   206→  if (typeof book.author !== 'string') book.author = ''
   207→  if (typeof book.category !== 'string') book.category = ''
   208→  if (typeof book.status !== 'string') book.status = ''
   209→  if (typeof book.chapterCount !== 'number') book.chapterCount = book.chapters.length
   210→  // 全字段兜底清洗（引擎侧已清洗，此处防版本错位/旧引擎，幂等成本低）：
   211→  // 实体解码（&#091；类全角分号变体）、字面 \n 还原、残缺标签剥除、简介样板句/SEO 伪简介处理
   212→  book.title = cleanTextField(book.title)
   213→  book.author = cleanTextField(book.author)
   214→  book.description = cleanDescriptionField(book.description)
   215→  book.category = cleanTextField(book.category)
   216→  book.status = cleanTextField(book.status)
   217→  run.log(
   218→    `书页提取成功：《${book.title.slice(0, 40)}》${book.author ? ` / ${book.author.slice(0, 20)}` : ''}，章节链接 ${book.chapterCount} 条`,
   219→  )
   220→
   221→  // ---- 完整目录页二次提取（bookRule.catalogLinkSelector，如 23qb 新模板书页仅含最新几章）----
   222→  let allRefs = book.chapters
   223→  const catalogSel = rule.bookRule.catalogLinkSelector
   224→  if (typeof catalogSel === 'string' && catalogSel) {
   225→    if (await isCanceled(run.taskId)) return canceledOutcome('任务已取消')
   226→    if (book.catalogUrl) {
   227→      run.log(`发现完整目录页 ${book.catalogUrl.slice(0, 100)}，尝试整目提取…`)
   228→      const catalogUrl = book.catalogUrl // 闭包捕获需局部常量（TS 收窄不跨回调）
   229→      // 熔断感知：目录页撞熔断时等待冷却后自动重试（超限时按旧语义回退书页章节链接）
   230→      const catalog = await withCircuitRetry(
   231→        hostOf(catalogUrl),
   232→        () => fetchCatalogChapters(run, catalogUrl, rule, bookUrl),
   233→        {
   234→          detect: (r) => (r.error ? circuitWaitMs(r.error) : null),
   235→          isSuccess: (r) => r.error === null,
   236→          isAlive: async () => !(await isCanceled(run.taskId)),
   237→          onWait: (ms, cycle, max) =>
   238→            run.log(`目标主机熔断冷却中，暂停 ${Math.round(ms / 1000)}s 后自动重试目录页（第 ${cycle}/${max} 轮）`),
   239→        },
   240→      )
   241→      const catalogRefs = catalog.refs
   242→      if (catalogRefs.length > allRefs.length) {
   243→        run.log(`目录页提取到 ${catalogRefs.length} 条章节链接（书页仅 ${allRefs.length} 条），采用目录页结果`)
   244→        allRefs = catalogRefs
   245→      } else if (catalogRefs.length > 0) {
   246→        run.log(`目录页提取 ${catalogRefs.length} 条不多于书页 ${allRefs.length} 条，维持书页结果`)
   247→      }
   248→    } else {
   249→      run.log(`catalogLinkSelector "${catalogSel.slice(0, 60)}" 在书页无命中，仅用书页章节链接`)
   250→    }
   251→  }
   252→
   253→  if (await isCanceled(run.taskId)) return canceledOutcome('任务已取消')
   254→
   255→  // ---- 乱序重排（充分考虑分卷：同号不同卷不跨卷错排，卷名随章节移动）----
   256→  // 源站目录常见「最新章节块（新→旧）+ 正文块（旧→新）」或整本倒序；直接按抓取序编 idx 会错乱。
   257→  // 引擎响应未经 schema 校验，逐条形态兑底后再重排。
   258→  allRefs = (Array.isArray(allRefs) ? allRefs : []).map((c) => ({
   259→    title: typeof c?.title === 'string' ? c.title : '',
   260→    url: typeof c?.url === 'string' ? c.url : null,
   261→    volume: typeof c?.volume === 'string' && c.volume.trim() ? c.volume.trim().slice(0, 50) : undefined,
   262→  })) as ChapterRef[]
   263→  const rr = reorderChapterRefs(allRefs)
   264→  if (rr.reordered) {
   265→    allRefs = rr.refs
   266→    run.log(`目录乱序重排：${rr.note}`)
   267→  }
   268→
   269→  // ---- 分类 ----
   270→  let categoryId: number
   271→  try {
   272→    categoryId = await ensureCategory(book.category)
   273→  } catch (e) {
   274→    const msg = e instanceof Error ? e.message : '分类处理失败'
   275→    run.log(msg)
   276→    return { ok: false, canceled: false, chapters: 0, failedChapters: 0, message: msg }
   277→  }
   278→
   279→  // ---- 书籍 upsert（title+author 查重；DB 层 @@unique([title,author]) 兜底并发）----
   280→  const up = await upsertBook(run, book, categoryId, rule.proxy)
   281→  if (!up.ok) {
   282→    // 记录级失败（up.canceled=true：入库/更新失败、并发冲突后找不到记录）沿用取消通道停整个任务；
   283→    // 书籍级失败（up.canceled=false，如空标题）只算本书失败：single 按 failed 收尾，list 继续下一本
   284→    if (up.canceled) return canceledOutcome(up.message)
   285→    return { ok: false, canceled: false, chapters: 0, failedChapters: 0, message: up.message }
   286→  }
   287→  // 采集的同时取搜索引擎下拉词 → 绑定书籍 + 生成 PSEO 书籍页（fire-and-forget，不阻塞章节抓取）
   288→  scheduleBindBookSuggestKeywords(run, up.novelId, up.title || book.title, up.createdNew)
   289→
   290→  // ---- 章节列表准备 ----
   291→  const refs = allRefs.filter(
   292→    (c): c is ChapterRef & { url: string } => !!c && typeof c.url === 'string' && !!c.url,
   293→  )
   294→  if (refs.length === 0) {
   295→    run.log('未提取到任何有效章节链接')
   296→    return { ok: true, canceled: false, chapters: 0, failedChapters: 0, message: '书籍已入库（未提取到章节链接）' }
   297→  }
   298→  // 同目录内预去重（并行抓取前完成）：按清洗后标题优先、URL 兜底去重，防同批重复入库
   299→  const seenRef = new Set<string>()
   300→  const uniqueRefs: (ChapterRef & { url: string })[] = []
   301→  for (const c of refs) {
   302→    const key = c.title.trim() ? `t:${cleanTextField(c.title)}` : `u:${c.url}`
   303→    if (seenRef.has(key)) continue
   304→    seenRef.add(key)
   305→    uniqueRefs.push(c)
   306→  }
   307→  if (uniqueRefs.length < refs.length) run.log(`目录内去重：${refs.length} → ${uniqueRefs.length} 条（无数量上限，全量采集）`)
   308→
   309→  if (opts.trackTotal) {
   310→    // single 模式：进度主口径=本章节数，total 为分母（flush false=任务记录已删，立即停）
   311→    if (!(await run.flush({ total: uniqueRefs.length }))) return canceledOutcome('任务记录已删除')
   312→  } else if (opts.chapterProgress) {
   313→    // list 模式：章节进度独立于 done/total（书）累计
   314→    opts.chapterProgress.total += uniqueRefs.length
   315→  }
   316→
   317→  /** 每处理完一个章节后要刷出的进度字段（不含任务级成果计数） */
   318→  const progressFields = (): TaskFlushFields =>
   319→    opts.trackTotal
   320→      ? { done }
   321→      : {
   322→          chaptersDone: opts.chapterProgress?.done ?? 0,
   323→          chaptersTotal: opts.chapterProgress?.total ?? 0,
   324→        }
   325→
   326→  const existingChapters = await db.chapter
   327→    .findMany({ where: { novelId: up.novelId }, select: { title: true, volume: true } })
   328→    .catch(() => [] as { title: string; volume: string }[])
   329→  // 去重键与入库标题共用同一清洗规约（cleanTextField 幂等）：存量脏标题与本次净标题可命中比对，
   330→  // 避免清洗规约升级后同一章以两种写法重复入库
   331→  const existingTitles = new Set(existingChapters.map((c) => cleanTextField(c.title)))
   332→  // 同书内标题计数：用于「已存在章节」的卷名回填防误伤（重名章不回填）
   333→  const existingTitleCount = new Map<string, number>()
   334→  for (const c of existingChapters) {
   335→    const k = cleanTextField(c.title)
   336→    existingTitleCount.set(k, (existingTitleCount.get(k) ?? 0) + 1)
   337→  }
   338→  /** 已存在章节的卷名回填队列（title → volume；仅唯一标题且存量为空卷时写入） */
   339→  const volumeBackfill = new Map<string, string>()
   340→  const maxAgg = await db.chapter
   341→    .aggregate({ where: { novelId: up.novelId }, _max: { idx: true } })
   342→    .catch(() => ({ _max: { idx: null as number | null } }))
   343→  // 章节槽位基准：现有最大 idx+1；每条按目录位次预分配（base+位次），保序且并行安全
   344→  const baseIdx = (maxAgg._max.idx ?? 0) + 1
   345→
   346→  // ---- 并行章节抓取入库 ----
   347→  let done = 0
   348→  let chaptersStored = 0
   349→  let failedChapters = 0
   350→  let cleanLogCount = 0 // 清洗日志节流：每本书最多记 3 条，防止日志爆炸
   351→  let stopRun = false // 取消/记录删除 → 停止派发新章节，在途章节收尾后退出
   352→
   353→  const alive = async (): Promise<boolean> => {
   354→    if (stopRun) return false
   355→    return !(await isCanceled(run.taskId))
   356→  }
   357→
   358→  // flush 串行链：并行分支共享 Run 缓冲与进度字段，写盘串行化避免交错/竞态
   359→  let flushChain = Promise.resolve(true)
   360→  const scheduleFlush = (fields: TaskFlushFields): Promise<boolean> => {
   361→    flushChain = flushChain.then(async (ok) => {
   362→      if (!ok || stopRun) return false
   363→      const ok2 = await run.flush(fields)
   364→      if (!ok2) stopRun = true // 任务记录被删除 → 全池停止派发
   365→      return ok2
   366→    })
   367→    return flushChain
   368→  }
   369→
   370→  const processRef = async (ref: ChapterRef & { url: string }, position: number): Promise<void> => {
   371→    const refTitle = cleanTextField(String(ref.title ?? '')).slice(0, 200)
   372→    if (refTitle && existingTitles.has(refTitle)) {
   373→      done++
   374→      if (opts.chapterProgress) opts.chapterProgress.done++
   375→      run.log(`章节「${refTitle.slice(0, 30)}」已存在，跳过`)
   376→      // 分卷补齐：本次源站目录携带卷名、存量章节为空卷且标题在全书中唯一时，回填卷名
   377→      const rv = ref.volume
   378→      if (rv && existingTitleCount.get(refTitle) === 1 && !volumeBackfill.has(refTitle)) {
   379→        volumeBackfill.set(refTitle, rv)
   380→      }
   381→      await scheduleFlush(progressFields())
   382→      return
   383→    }
   384→
   385→    if (!(await alive())) {
   386→      stopRun = true
   387→      return
   388→    }
   389→    run.log(`抓取章节「${(refTitle || ref.url).slice(0, 36)}」`)
   390→    // Referer 链：章节页带书页来路（站点常见「书页→章节」导航校验）；
   391→    // fetchChapterPaged：同章分页（…_2.html / …/2.html）自动翻页拼接，避免长章只存半页。
   392→    // 熔断感知：章节撞熔断（站点临时限流封禁）时同主机并发 worker 合流等待冷却后自动重试，
   393→    // 而非把剩余章节全部烧成失败（Task 36 复盘：ggd66 一次临时封禁曾致 232 章被永久记失败）
   394→    const circuitFetch = (): ReturnType<typeof fetchChapterPaged> =>
   395→      withCircuitRetry(hostOf(ref.url), () => fetchChapterPaged(ref.url, rule, bookUrl), {
   396→        detect: (r) => (r.ok ? null : circuitWaitMs(r.error, r.circuitRetryAfterMs)),
   397→        isAlive: alive,
   398→        onWait: (ms, cycle, max) =>
   399→          run.log(`目标主机熔断冷却中，暂停 ${Math.round(ms / 1000)}s 后自动重试章节（第 ${cycle}/${max} 轮，等待期间不向源站发请求）`),
   400→      })
   401→    let ch = await circuitFetch()
   402→    // 瞬时失败单次重试（预算耗尽/超时类）：高并发下域名限速排队可吃光引擎 55s 链预算，
   403→    // 属暂时性压力而非站点拒绝——稍候片刻让队列排空后重试一次，减少无谓的永久失败章节
   404→    if (!ch.ok && isTransientEngineError(ch.error)) {
   405→      run.log(`章节「${(refTitle || ref.url).slice(0, 30)}」因引擎预算耗尽/超时失败，稍候重试一次`)
   406→      await sleep(3_000 + Math.random() * 5_000)
   407→      if (stopRun || !(await alive())) return
   408→      ch = await circuitFetch()
   409→    }
   410→    if (stopRun || !(await alive())) return // 取消：抓取结果直接丢弃，尽快退出
   411→    const data = ch.ok ? ch.data : null
   412→
   413→    // 入库前统一清洗（去 \r\n/行首缩进/空行/噪声行），存储契约：无空行、无行首缩进
   414→    const cleaned = cleanChapterContent(data?.content ?? '')
   415→    let content = cleaned.text
   416→    if (content.length > MAX_CONTENT_CHARS) {
   417→      content = content.slice(0, MAX_CONTENT_CHARS)
   418→      run.log(`章节正文超长（>${MAX_CONTENT_CHARS} 字符），截断保存（内存保护）`)
   419→    }
   420→
   421→    if (!data || !content.trim()) {
   422→      failedChapters++
   423→      done++
   424→      if (opts.chapterProgress) opts.chapterProgress.done++
   425→      run.log(`章节抓取失败: ${ch.ok ? '正文为空' : ch.error}`)
   426→      await scheduleFlush({ ...progressFields(), chapters: run.counters.chapters })
   427→      return
   428→    }
   429→    if (ch.warnings.length) run.logWarnings(ch.warnings)
   430→
   431→    const idx = baseIdx + position // 槽位保序：目录顺序即阅读顺序
   432→    const chTitle = (refTitle || cleanTextField(data.title) || `第${idx}章`).slice(0, 200)
   433→    if (cleaned.removedLines > 0 && cleanLogCount < 3) {
   434→      cleanLogCount++
   435→      run.log(`章节「${chTitle.slice(0, 30)}」清洗 ${cleaned.removedLines} 行噪声`)
   436→    }
   437→    const wordCount = content.replace(/\s/g, '').length
   438→    // storeChapter 内部处理 idx 唯一冲突顺延重试（并发任务写同书等场景）
   439→    const stored = await storeChapter(run, up.novelId, idx, {
   440→      title: chTitle,
   441→      content,
   442→      wordCount,
   443→      volume: ref.volume ?? '',
   444→    })
   445→    if (stored.ok) {
   446→      existingTitles.add(chTitle)
   447→      chaptersStored++
   448→      run.counters.chapters++
   449→    } else {
   450→      run.log(`章节入库失败: ${stored.message.slice(0, 120)}`)
   451→      failedChapters++
   452→    }
   453→    done++
   454→    if (opts.chapterProgress) opts.chapterProgress.done++
   455→    await scheduleFlush({ ...progressFields(), chapters: run.counters.chapters })
   456→  }
   457→
   458→  await runPool(uniqueRefs, clampConcurrency(opts.concurrency), processRef)
   459→  const flushOk = await flushChain
   460→  if (stopRun || !flushOk) {
   461→    run.log('任务已取消/记录已删除，停止章节抓取')
   462→    await recalcNovelWordCount(up.novelId) // 取消点前已入库章节，重算字数保持一致性
   463→    await run.flush({ ...progressFields(), chapters: run.counters.chapters }).catch(() => undefined)
   464→    return canceledOutcome('任务已取消', chaptersStored, failedChapters)
   465→  }
   466→
   467→  // ---- 已存在章节的卷名回填（逐卷 updateMany，仅在唯一标题且存量空卷时入队）----
   468→  if (volumeBackfill.size > 0) {
   469→    let backfilled = 0
   470→    for (const [title, volume] of volumeBackfill) {
   471→      const res = await db.chapter
   472→        .updateMany({ where: { novelId: up.novelId, title, volume: '' }, data: { volume } })
   473→        .catch(() => null)
   474→      if (res) backfilled += res.count
   475→    }
   476→    if (backfilled > 0) run.log(`分卷回填：${backfilled} 个已存在章节补记卷名`)
   477→  }
   478→
   479→  // ---- 重算书籍字数 ----
   480→  await recalcNovelWordCount(up.novelId)
   481→
   482→  run.log(`本书完成：入库 ${chaptersStored} 章，失败 ${failedChapters} 章`)
   483→  return {
   484→    ok: chaptersStored > 0 || failedChapters === 0, // 全部为"已存在跳过"也算成功
   485→    canceled: false,
   486→    chapters: chaptersStored,
   487→    failedChapters,
   488→    message:
   489→      chaptersStored > 0
   490→        ? `入库 ${chaptersStored} 章`
   491→        : failedChapters === 0
   492→          ? '无新增章节（章节均已存在）'
   493→          : '章节采集全部失败',
   494→  }
   495→}

   499→async function runSingle(run: Run, task: TaskRecord, rule: LoadedRule): Promise<void> {
   500→  const outcome = await processBook(run, task.targetUrl, rule, { trackTotal: true, concurrency: task.concurrency })
   501→  if (outcome.canceled) {
   502→    run.log('任务已取消')
   503→    await finalize(run, 'canceled', '任务已取消')
   504→    return
   505→  }
   506→  await run.flush({ created: run.counters.created, updated: run.counters.updated, chapters: run.counters.chapters })
   507→  if (!outcome.ok) {
   508→    await finalize(run, 'failed', outcome.message || '书页提取失败')
   509→    return
   510→  }
   511→  if (outcome.failedChapters === 0) {
   512→    await finalize(run, 'success', outcome.chapters > 0 ? `采集完成：${outcome.chapters} 章` : outcome.message)
   513→    return
   514→  }
   515→  if (outcome.chapters > 0) {
   516→    await finalize(run, 'partial', `${outcome.chapters} 章成功 / ${outcome.failedChapters} 章失败`)
   517→    return
   518→  }
   519→  await finalize(run, 'failed', '章节采集全部失败')
   520→}

   522→async function runList(run: Run, task: TaskRecord, rule: LoadedRule): Promise<void> {
   523→  const startPage = Math.max(1, Math.floor(task.startPage) || 1)
   524→  const endPage = startPage + Math.max(1, task.pages) - 1
   525→  run.log(`抓取列表页第 ${startPage} 页…`)
   526→  // 熔断感知：起始列表页撞熔断时等待冷却后自动重试；抓取失败（超限）与「空列表页」自此可区分
   527→  const firstPage = await withCircuitRetry(hostOf(task.targetUrl), () => fetchListPage(run, task.targetUrl, rule), {
   528→    detect: (r) => (r.error ? circuitWaitMs(r.error) : null),
   529→    isSuccess: (r) => r.error === null,
   530→    isAlive: async () => !(await isCanceled(run.taskId)),
   531→    onWait: (ms, cycle, max) =>
   532→      run.log(`目标主机熔断冷却中，暂停 ${Math.round(ms / 1000)}s 后自动重试列表页（第 ${cycle}/${max} 轮，等待期间不向源站发请求）`),
   533→  })
   534→  const first = firstPage.items
   535→  if (await isCanceled(run.taskId)) {
   536→    await finalize(run, 'canceled', '任务已取消')
   537→    return
   538→  }
   539→  if (firstPage.error) {
   540→    run.log('起始列表页持续抓取失败（熔断等待轮数耗尽或持续失败），任务终止')
   541→    await finalize(run, 'failed', `列表页抓取失败: ${firstPage.error.slice(0, 180)}`)
   542→    return
   543→  }
   544→  if (first.length === 0) {
   545→    run.log('列表页未提取到书籍条目')
   546→    await finalize(run, 'failed', '列表页未提取到书籍条目')
   547→    return
   548→  }
   549→  run.log(`第 ${startPage} 页提取 ${first.length} 条`)
   550→
   551→  const items = [...first]
   552→  // 当前生效的列表页 URL（含翻页命中页）：作为后续书页抓取的 Referer 来路
   553→  let currentListUrl = task.targetUrl
   554→  for (let k = startPage + 1; k <= endPage; k++) {
   555→    if (await isCanceled(run.taskId)) break
   556→    let got: ListItem[] | null = null
   557→    for (const v of pageVariants(task.targetUrl, k)) {
   558→      // 熔断感知：翻页变体撞熔断时等待冷却后自动重试；变体持续失败 → 尝试下一变体
   559→      const pv = await withCircuitRetry(hostOf(v), () => fetchListPage(run, v, rule), {
   560→        detect: (r) => (r.error ? circuitWaitMs(r.error) : null),
   561→        isSuccess: (r) => r.error === null,
   562→        isAlive: async () => !(await isCanceled(run.taskId)),
   563→        onWait: (ms, cycle, max) =>
   564→          run.log(`目标主机熔断冷却中，暂停 ${Math.round(ms / 1000)}s 后自动重试列表页（第 ${cycle}/${max} 轮）`),
   565→      })
   566→      if (pv.error) continue
   567→      if (pv.items.length > 0) {
   568→        got = pv.items
   569→        currentListUrl = v
   570→        run.log(`第 ${k} 页命中: ${v.slice(0, 100)}`)
   571→        break
   572→      }
   573→    }
   574→    if (!got || got.length === 0) {
   575→      run.log(`第 ${k} 页无结果，跳过`)
   576→      continue
   577→    }
   578→    run.log(`第 ${k} 页提取 ${got.length} 条`)
   579→    items.push(...got)
   580→  }
   581→
   582→  // 合并去重（按 URL，缺 URL 按标题）；不设书籍数量上限，全量采集
   583→  const seen = new Set<string>()
   584→  const merged: ListItem[] = []
   585→  for (const it of items) {
   586→    const key = it.url ?? `t:${it.title}`
   587→    if (seen.has(key)) continue
   588→    seen.add(key)
   589→    merged.push(it)
   590→  }
   591→  const total = merged.length
   592→  await run.flush({ total, done: 0 })
   593→  run.log(`去重后共 ${total} 本书待采集（并发 ${task.concurrency} 路书籍并行，书内章节再并行）`)
   594→
   595→  let doneBooks = 0
   596→  let okBooks = 0
   597→  let failBooks = 0
   598→  let stopRun = false
   599→  // 任务级章节进度（跨书累计）：list 模式 done/total 主口径是「书」，章节进度走 chaptersDone/chaptersTotal
   600→  const chapterProgress = { done: 0, total: 0 }
   601→  // flush 串行链：并行书籍分支的进度写盘串行化
   602→  let flushChain = Promise.resolve(true)
   603→  const scheduleBookFlush = (): Promise<boolean> => {
   604→    flushChain = flushChain.then(async (ok) => {
   605→      if (!ok || stopRun) return false
   606→      const ok2 = await run.flush({
   607→        done: doneBooks,
   608→        chaptersDone: chapterProgress.done,
   609→        chaptersTotal: chapterProgress.total,
   610→        created: run.counters.created,
   611→        updated: run.counters.updated,
   612→        chapters: run.counters.chapters,
   613→      })
   614→      if (!ok2) stopRun = true
   615→      return ok2
   616→    })
   617→    return flushChain
   618→  }
   619→
   620→  const processItem = async (item: ListItem): Promise<void> => {
   621→    if (stopRun) return
   622→    if (await isCanceled(run.taskId)) {
   623→      stopRun = true
   624→      return
   625→    }
   626→    run.log(`━━ 《${(item.title || '未命名').slice(0, 30)}》`)
   627→    const outcome = await processBook(run, item.url as string, rule, {
   628→      trackTotal: false,
   629→      chapterProgress,
   630→      referer: currentListUrl,
   631→      concurrency: task.concurrency,
   632→    })
   633→    doneBooks++
   634→    if (outcome.canceled) stopRun = true // 取消传播：停止派发后续书籍
   635→    else if (outcome.ok) okBooks++
   636→    else failBooks++
   637→    await scheduleBookFlush()
   638→  }
   639→
   640→  await runPool(merged, clampConcurrency(task.concurrency), processItem)
   641→  const flushOk = await flushChain
   642→
   643→  if (stopRun || !flushOk) {
   644→    run.log('任务已取消')
   645→    await finalize(run, 'canceled', `已取消（完成 ${doneBooks}/${total} 本）`)
   646→    return
   647→  }
   648→  if (okBooks === 0) {
   649→    await finalize(run, 'failed', '无书籍采集成功')
   650→    return
   651→  }
   652→  if (failBooks > 0) {
   653→    await finalize(run, 'partial', `${okBooks} 本成功 / ${failBooks} 本失败`)
   654→    return
   655→  }
   656→  await finalize(run, 'success', `范围采集完成：共 ${total} 本`)
   657→}

   661→/** 终态写入：仅当仍处于 running 时写终态；已被取消/删除则只保留日志 */
   662→async function finalize(run: Run, status: string, message: string): Promise<void> {
   663→  const cur = await db.scrapeTask
   664→    .findUnique({ where: { id: run.taskId }, select: { status: true } })
   665→    .catch(() => null)
   666→  if (!cur) return
   667→  if (cur.status === 'running') {
   668→    await db.scrapeTask
   669→      .updateMany({
   670→        where: { id: run.taskId, status: 'running' },
   671→        data: { status, message: message.slice(0, 500), log: run.logText() },
   672→      })
   673→      .catch(() => {})
   674→  } else {
   675→    await db.scrapeTask
   676→      .updateMany({ where: { id: run.taskId }, data: { log: run.logText() } })
   677→      .catch(() => {})
   678→  }
   679→}

   681→async function runTask(taskId: number): Promise<void> {
   682→  const run = new Run(taskId)
   683→  try {
   684→    // pending → running 条件更新：pending 阶段已被取消/删除的任务不再启动
   685→    const started = await db.scrapeTask
   686→      .updateMany({ where: { id: taskId, status: 'pending' }, data: { status: 'running' } })
   687→      .catch(() => ({ count: 0 }))
   688→    if (started.count === 0) return
   689→    const task = await db.scrapeTask.findUnique({ where: { id: taskId } }).catch(() => null)
   690→    if (!task) return
   691→    run.ruleId = task.ruleId ?? null // 封面回填链路：入库时记入 Novel.sourceRuleId
   692→    const record: TaskRecord = {
   693→      id: task.id,
   694→      mode: task.mode,
   695→      targetUrl: task.targetUrl,
   696→      pages: task.pages,
   697→      startPage: task.startPage,
   698→      concurrency: clampConcurrency(task.concurrency),
   699→      ruleId: task.ruleId,
   700→    }
   701→    const rule = await loadRule(record.ruleId)
   702→    run.log(
   703→      record.mode === 'list'
   704→        ? `任务开始（范围采集）目标: ${record.targetUrl}，起始页: ${record.startPage}，页数: ${record.pages}，并发: ${record.concurrency}`
   705→        : `任务开始（单本采集）目标: ${record.targetUrl}，并发: ${record.concurrency}`,
   706→    )
   707→    run.log(
   708→      record.ruleId
   709→        ? `使用规则「${rule.name ?? record.ruleId}」（charset=${rule.charset ?? 'auto'}）`
   710→        : '未使用规则，依赖引擎内置启发式提取',
   711→    )
   712→    await run.flush()
   713→    if (record.mode === 'list') await runList(run, record, rule)
   714→    else await runSingle(run, record, rule)
   715→  } catch (e) {
   716→    run.log(`发生未预期错误: ${e instanceof Error ? e.message.slice(0, 200) : String(e).slice(0, 200)}`)
   717→    await finalize(run, 'failed', '任务执行异常')
   718→  }
   719→}

   721→/**
   722→ * fire-and-forget 入口：API 路由创建任务后调用，不 await。
   723→ * 多任务天然并行：每个任务独立 worker promise，互不阻塞；
   724→ * 对同一目标站的压力由引擎层域名限速统一收敛。
