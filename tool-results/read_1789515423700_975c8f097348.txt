    23→    23→async function api<T>(url: string, init?: RequestInit): Promise<T> {
    24→    24→  const res = await fetch(url, { headers: { 'Content-Type': 'application/json' }, ...init })
    25→    25→  const data = await res.json().catch(() => ({}))
    26→    26→  if (!res.ok) throw new Error((data as { error?: string }).error ?? `请求失败(${res.status})`)
    27→    27→  return data as T
    28→    28→}
