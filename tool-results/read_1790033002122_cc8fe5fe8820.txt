   104→   104→// lastLines 保留最后 n 行（与 Run 同规约：只保留最近 MAX_LOG_LINES 行）
   105→   105→func lastLines(s string, n int) string {
   106→   106→        lines := strings.Split(s, "\n")
   107→   107→        if len(lines) > n {
   108→   108→                lines = lines[len(lines)-n:]
   109→   109→        }
   110→   110→        return strings.Join(lines, "\n")
   111→   111→}

   113→   113→// recoverStaleTasks 僵尸任务回收：进程首次启动时，把「创建于本进程启动之前」且仍处于
   114→   114→// pending/running 的任务标记为 failed——它们只可能属于已消失的旧进程。条件更新
   115→   115→// （status 仍为 pending/running）保证与 worker 终态写入竞态安全。
   116→   116→func recoverStaleTasks() {
   117→   117→        type staleRow struct {
   118→   118→                id   int
   119→   119→                logv string
   120→   120→        }
   121→   121→        var stale []staleRow
   122→   122→        err := queryList(
   123→   123→                "SELECT id, log FROM ScrapeTask WHERE status IN ('pending','running') AND createdAt < ? ORDER BY id ASC",
   124→   124→                func(rows *sql.Rows) error {
   125→   125→                        var r staleRow
   126→   126→                        if err := rows.Scan(&r.id, &r.logv); err != nil {
   127→   127→                                return err
   128→   128→                        }
   129→   129→                        stale = append(stale, r)
   130→   130→                        return nil
   131→   131→                }, gBootAt)
   132→   132→        if err != nil {
   133→   133→                return // 回收失败不阻塞启动
   134→   134→        }
   135→   135→        recovered := 0
   136→   136→        for _, t := range stale {
   137→   137→                if runningHas(t.id) { // 本进程仍在执行（防御性；启动时刻 running 恒为空）
   138→   138→                        continue
   139→   139→                }
   140→   140→                line := "[" + time.Now().Format("15:04:05") + "] 服务重启，任务中断（自动回收）"
   141→   141→                logv := t.logv
   142→   142→                if logv != "" {
   143→   143→                        logv += "\n"
   144→   144→                }
   145→   145→                logv = lastLines(logv+line, MAX_LOG_LINES)
   146→   146→                res, err := execRetry(
   147→   147→                        "UPDATE ScrapeTask SET status = 'failed', message = '服务重启，任务中断', log = ? WHERE id = ? AND status IN ('pending','running')",
   148→   148→                        logv, t.id)
   149→   149→                if err == nil && rowCountOf(res) > 0 {
   150→   150→                        recovered++
   151→   151→                }
   152→   152→        }
   153→   153→        if recovered > 0 {
   154→   154→                log.Printf("[scrape-worker] 僵尸任务回收: %d 条", recovered)
   155→   155→        }
   156→   156→}

   163→   163→        err := queryOne("SELECT status FROM ScrapeTask WHERE id = ?", []any{&status}, taskID)

   168→   168→                return false // 查询失败 ≠ 被取消，避免瞬时 DB 错误误停任务

   171→   171→}

   195→   195→// normalizeRefs 章节链接规范化：过滤无 URL、噪声按钮文案、trim、URL 去重（同 URL 不同标题视为同章）
   196→   196→func normalizeRefs(refs []ChapterRef) []refPair {

   205→   205→                if noiseTOCTitles[title] {
   206→   206→                        continue

   209→   209→                        continue
   210→   210→                }

   248→   248→type Phase1Outcome struct {

   261→   261→// phase1Skeletons 书目骨架（BOOK_CONCURRENCY 有界并发）：抓书页 → 目录二次提取 →
   262→   262→// 分类保障 → upsert 书籍 → 骨架入库，产出 Phase 2 填充计划。
   263→   263→func phase1Skeletons(run *Run, rule LoadedRule, items []ListItem, listURL string) Phase1Outcome {
   264→   264→        var mu sync.Mutex
   265→   265→        fillMap := map[int]fillPlan{}
   266→   266→        var novelIDs []int
   267→   267→        okBooks, failBooks, failStreak := 0, 0, 0
   268→   268→        firstError := ""
   269→   269→        totalRefs, skippedFilled, fillTotal := 0, 0, 0
   270→   270→        sawCatalog := 0
   271→   271→        var fatal error
   272→   272→
   273→   273→        setFirstError := func(msg string) {
   274→   274→                if firstError == "" {
   275→   275→                        firstError = msg
   276→   276→                }
   277→   277→        }
   278→   278→
   279→   279→        // 取消 + 快速终止（连续多本全败且 0 成功 → 判定站点不可达，中止防空转）
   280→   280→        shouldStop := throttledCheck(func() bool {
   281→   281→                if isCanceled(run.TaskID) {
   282→   282→                        return true
   283→   283→                }
   284→   284→                mu.Lock()
   285→   285→                defer mu.Unlock()
   286→   286→                if fatal != nil {
   287→   287→                        return true
   288→   288→                }
   289→   289→                return failStreak >= MAX_CONSECUTIVE_BOOK_FAILS && okBooks == 0
   290→   290→        })
   291→   291→
   292→   292→        runPool(items, BOOK_CONCURRENCY, func(item ListItem, _ int) {
   293→   293→                bookURL := item.URL
   294→   294→                page := fetchBookPage(run, bookURL, rule, listURL, false)
   295→   295→                if !page.OK {
   296→   296→                        mu.Lock()
   297→   297→                        setFirstError(page.Err)
   298→   298→                        failBooks++
   299→   299→                        failStreak++
   300→   300→                        mu.Unlock()
   301→   301→                        run.Log(fmt.Sprintf("书页失败(%s): %s", truncateRunes(bookURL, 90), truncateRunes(page.Err, 120)))
   302→   302→                        return
   303→   303→                }
   304→   304→                book := shapeBook(page.Book)
   305→   305→                authorPart := ""
   306→   306→                if a := truncateRunes(book.Author, 20); a != "" {
   307→   307→                        authorPart = " / " + a
   308→   308→                }
   309→   309→                run.Log(fmt.Sprintf("书页命中《%s》%s，章节链接 %d 条",
   310→   310→                        truncateRunes(book.Title, 40), authorPart, *book.ChapterCount))
   311→   311→
   312→   312→                // 完整目录页二次提取（书页仅含最新几章时）
   313→   313→                allRefs := book.Chapters
   314→   314→                catalogSel := rule.BookRule["catalogLinkSelector"]
   315→   315→                if catalogSel != "" && book.CatalogURL != "" {
   316→   316→                        catalogRefs := fetchCatalogChapters(run, book.CatalogURL, rule, bookURL, false)
   317→   317→                        if len(catalogRefs) > len(allRefs) {
   318→   318→                                mu.Lock()
   319→   319→                                sawCatalog++
   320→   320→                                mu.Unlock()
   321→   321→                                allRefs = catalogRefs
   322→   322→                        }
   323→   323→                }
   324→   324→                refs := normalizeRefs(allRefs)
   325→   325→                // refs 为空：仍入库书籍（原语义「书籍已入库（未提取到章节链接）」），不计失败
   326→   326→
   327→   327→                // 源站分类名归并失败时用书名+简介 LLM 推断（防「未分类」堆积）
   328→   328→                categoryID, cerr := ensureCategory(book.Category, book.Title, book.Description)
   329→   329→                if cerr != nil {
   330→   330→                        msg := cerr.Error()
   331→   331→                        mu.Lock()
   332→   332→                        setFirstError(msg)
   333→   333→                        failBooks++
   334→   334→                        failStreak++
   335→   335→                        mu.Unlock()
   336→   336→                        run.Log(fmt.Sprintf("《%s》%s", truncateRunes(book.Title, 24), msg))
   337→   337→                        return
   338→   338→                }
   339→   339→
   340→   340→                up := upsertBook(run, book, categoryID, rule.Proxy)
   341→   341→                if !up.OK {
   342→   342→                        mu.Lock()
   343→   343→                        setFirstError(up.Message)
   344→   344→                        failBooks++
   345→   345→                        failStreak++
   346→   346→                        mu.Unlock()
   347→   347→                        run.Log(fmt.Sprintf("《%s》入库失败: %s", truncateRunes(book.Title, 24), up.Message))
   348→   348→                        return
   349→   349→                }
   350→   350→
   351→   351→                sk, skerr := storeChapterSkeletons(run, up.NovelID, refs, MAX_CHAPTERS_PER_BOOK)
   352→   352→                if skerr != nil {
   353→   353→                        mu.Lock()
   354→   354→                        setFirstError(skerr.Error())
   355→   355→                        failBooks++
   356→   356→                        failStreak++
   357→   357→                        fatal = skerr
   358→   358→                        mu.Unlock()
   359→   359→                        run.Log(fmt.Sprintf("《%s》骨架入库异常: %s", truncateRunes(up.Title, 24), truncateRunes(skerr.Error(), 120)))
   360→   360→                        return
   361→   361→                }
   362→   362→                if sk.Capped {
   363→   363→                        run.Log(fmt.Sprintf("《%s》已达单本上限（%d 章），超出部分未采集", truncateRunes(up.Title, 24), MAX_CHAPTERS_PER_BOOK))
   364→   364→                }
   365→   365→                mu.Lock()
   366→   366→                novelIDs = append(novelIDs, up.NovelID)
   367→   367→                okBooks++
   368→   368→                failStreak = 0
   369→   369→                totalRefs += sk.Total
   370→   370→                skippedFilled += sk.SkippedFilled
   371→   371→                fillTotal += len(sk.FillRows)
   372→   372→                if len(sk.FillRows) > 0 {
   373→   373→                        fillMap[up.NovelID] = fillPlan{Referer: bookURL, Rows: sk.FillRows}
   374→   374→                }
   375→   375→                mu.Unlock()
   376→   376→                run.Log(fmt.Sprintf("《%s》骨架入库 %d 章（已有正文跳过 %d，待填充 %d）",
   377→   377→                        truncateRunes(up.Title, 24), sk.Stored, sk.SkippedFilled, len(sk.FillRows)))
   378→   378→        }, shouldStop)
   379→   379→
   380→   380→        if sawCatalog > 0 {
   381→   381→                run.Log(fmt.Sprintf("目录页二次提取生效 %d 本", sawCatalog))
   382→   382→        }
   383→   383→        mu.Lock()
   384→   384→        defer mu.Unlock()
   385→   385→        return Phase1Outcome{
   386→   386→                OKBooks: okBooks, FailBooks: failBooks, FirstError: firstError,
   387→   387→                NovelIDs: novelIDs, FillMap: fillMap,
   388→   388→                TotalRefs: totalRefs, SkippedFilled: skippedFilled, FillTotal: fillTotal,
   389→   389→                Fatal: fatal,
   390→   390→        }
   391→   391→}

   393→   393→// ==================== Phase 2：正文填充（跨书平铺并发） ====================
   394→   394→
   395→   395→// Phase2Outcome Phase 2 结果
   396→   396→type Phase2Outcome struct {
   397→   397→        Filled       int
   398→   398→        Failed       int
   399→   399→        StoppedEarly bool
   400→   400→}
   401→   401→
   402→   402→// phase2Fill 消费 Phase 1 的填充计划：逐书分批（200/批）拉取 DB 空骨架行（wordCount=0，
   403→   403→// 含历史中断遗留的同名重复行），书内有界并发抓正文并 update 填充；处理完一本书即从
   404→   404→// fillMap 释放其 rows（内存渐减）。章节抓取失败保留骨架（wordCount=0），重发任务自动续传。
   405→   405→// onProgress(doneSoFar) 由调用方节流落库进度；返回 false 表示任务记录已删除，立即停止。
   406→   406→func phase2Fill(run *Run, rule LoadedRule, fillMap map[int]fillPlan, onProgress func(doneSoFar int) bool) Phase2Outcome {
   407→   407→        var filledCtr, failedCtr atomic.Int64
   408→   408→        var warnLogged atomic.Int64
   409→   409→        stoppedEarly := false
   410→   410→        isStopped := throttledCheck(func() bool { return isCanceled(run.TaskID) })
   411→   411→
   412→   412→        // TS Map 迭代=插入序（Phase 1 并发完成序）；Go map 无序 → 按 novelId 升序确定性处理
   413→   413→        ids := make([]int, 0, len(fillMap))
   414→   414→        for id := range fillMap {
   415→   415→                ids = append(ids, id)
   416→   416→        }
   417→   417→        sort.Ints(ids)
   418→   418→
   419→   419→        for _, novelID := range ids {
   420→   420→                if stoppedEarly {
   421→   421→                        break
   422→   422→                }
   423→   423→                if isStopped() {
   424→   424→                        stoppedEarly = true
   425→   425→                        break
   426→   426→                }
   427→   427→                plan := fillMap[novelID]
   428→   428→                var bookFilled, bookFailed atomic.Int64
   429→   429→                for off := 0; off < len(plan.Rows); off += SKELETON_BATCH {
   430→   430→                        end := min(off+SKELETON_BATCH, len(plan.Rows))
   431→   431→                        slice := plan.Rows[off:end]
   432→   432→                        titles := make([]string, len(slice))
   433→   433→                        for i, r := range slice {
   434→   434→                                titles[i] = r.Title
   435→   435→                        }
   436→   436→                        // 同名行可能有多条（历史遗留重复）：一起拉出来填同一内容，后续由目录体检工具去重
   437→   437→                        type dbRow struct {
   438→   438→                                id    int
   439→   439→                                title string
   440→   440→                        }
   441→   441→                        var dbRows []dbRow
   442→   442→                        q := "SELECT id, title FROM Chapter WHERE novelId = ? AND wordCount = 0 AND title IN (" +
   443→   443→                                sqlPlaceholders(len(titles)) + ")"
   444→   444→                        args := make([]any, 0, len(titles)+1)
   445→   445→                        args = append(args, novelID)
   446→   446→                        for _, t := range titles {
   447→   447→                                args = append(args, t)
   448→   448→                        }
   449→   449→                        _ = queryList(q, func(rows *sql.Rows) error { // 失败 → 空数组（与 TS catch 一致）
   450→   450→                                var r dbRow
   451→   451→                                if err := rows.Scan(&r.id, &r.title); err != nil {
   452→   452→                                        return err
   453→   453→                                }
   454→   454→                                dbRows = append(dbRows, r)
   455→   455→                                return nil
   456→   456→                        }, args...)
   457→   457→                        if len(dbRows) == 0 {
   458→   458→                                continue
   459→   459→                        }
   460→   460→                        urlByTitle := map[string]string{}
   461→   461→                        for _, r := range slice {
   462→   462→                                urlByTitle[r.Title] = r.URL
   463→   463→                        }
   464→   464→                        runPool(dbRows, CHAPTER_CONCURRENCY, func(row dbRow, _ int) {
   465→   465→                                u, ok := urlByTitle[row.title]
   466→   466→                                if !ok {
   467→   467→                                        return
   468→   468→                                }
   469→   469→                                referer := plan.Referer
   470→   470→                                if referer == "" {
   471→   471→                                        referer = safeOrigin(u)
   472→   472→                                }
   473→   473→                                res := fetchChapterPaged(u, rule, referer)
   474→   474→                                if !res.OK {
   475→   475→                                        failedCtr.Add(1)
   476→   476→                                        bookFailed.Add(1)
   477→   477→                                        return
   478→   478→                                }
   479→   479→                                cleaned := cleanChapterContent(res.Data.Content)
   480→   480→                                content := truncateRunes(cleaned.Text, MAX_CONTENT_CHARS)
   481→   481→                                if trimSpaceStr(content) == "" {
   482→   482→                                        failedCtr.Add(1) // 源站空壳章：保留骨架，重发任务自动重试
   483→   483→                                        bookFailed.Add(1)
   484→   484→                                        return
   485→   485→                                }
   486→   486→                                if len(res.Warnings) > 0 && warnLogged.Load() < 10 {
   487→   487→                                        warnLogged.Add(1)
   488→   488→                                        run.LogWarnings(res.Warnings)
   489→   489→                                }
   490→   490→                                wordCount := countNonSpaceRunes(content)
   491→   491→                                title := trimSpaceStr(row.title)
   492→   492→                                if title == "" {
   493→   493→                                        title = trimSpaceStr(res.Data.Title)
   494→   494→                                }
   495→   495→                                if title == "" {
   496→   496→                                        title = "第" + itoa(row.id) + "章"
   497→   497→                                }
   498→   498→                                title = truncateRunes(title, chapterTitleMax)
   499→   499→                                res2, uerr := execRetry("UPDATE Chapter SET title = ?, content = ?, wordCount = ? WHERE id = ?",
   500→   500→                                        title, content, wordCount, row.id)
   501→   501→                                if uerr == nil && rowCountOf(res2) > 0 {
   502→   502→                                        filledCtr.Add(1)
   503→   503→                                        bookFilled.Add(1)
   504→   504→                                        run.IncChapters()
   505→   505→                                } else {
   506→   506→                                        failedCtr.Add(1)
   507→   507→                                        bookFailed.Add(1)
   508→   508→                                }
   509→   509→                        }, isStopped)
   510→   510→                        if !onProgress(int(filledCtr.Load())) {
   511→   511→                                stoppedEarly = true
   512→   512→                                break
   513→   513→                        }
   514→   514→                        if isStopped() {
   515→   515→                                stoppedEarly = true
   516→   516→                                break
   517→   517→                        }
   518→   518→                }
   519→   519→                run.Log(fmt.Sprintf("书籍 #%d 正文填充完成：成功 %d / 失败 %d", novelID, bookFilled.Load(), bookFailed.Load()))
   520→   520→                delete(fillMap, novelID) // 处理完即释放，长任务内存渐减
   521→   521→        }
   522→   522→        return Phase2Outcome{Filled: int(filledCtr.Load()), Failed: int(failedCtr.Load()), StoppedEarly: stoppedEarly}
   523→   523→}
   524→   524→
   525→   525→// ==================== 字数汇总 ====================
   526→   526→
   527→   527→// recalcWordCountsFor 任务收尾：对涉及的书籍统一重算字数（GROUP BY 汇总，避免逐书聚合）
   528→   528→func recalcWordCountsFor(novelIDs []int) {
   529→   529→        if len(novelIDs) == 0 {
   530→   530→                return
   531→   531→        }
   532→   532→        const chunk = 500
   533→   533→        for i := 0; i < len(novelIDs); i += chunk {
   534→   534→                part := novelIDs[i:min(i+chunk, len(novelIDs))]
   535→   535→                args := make([]any, 0, len(part))
   536→   536→                for _, id := range part {
   537→   537→                        args = append(args, id)
   538→   538→                }
   539→   539→                type sumRow struct {
   540→   540→                        novelID int
   541→   541→                        total   int64
   542→   542→                }
   543→   543→                var sums []sumRow
   544→   544→                q := "SELECT novelId, SUM(wordCount) FROM Chapter WHERE novelId IN (" + sqlPlaceholders(len(part)) + ") GROUP BY novelId"
   545→   545→                if err := queryList(q, func(rows *sql.Rows) error {
   546→   546→                        var s sumRow
   547→   547→                        if err := rows.Scan(&s.novelID, &s.total); err != nil {
   548→   548→                                return err
   549→   549→                        }
   550→   550→                        sums = append(sums, s)
   551→   551→                        return nil
   552→   552→                }, args...); err != nil {
   553→   553→                        continue
   554→   554→                }
   555→   555→                for _, s := range sums {
   556→   556→                        _, _ = execRetry("UPDATE Novel SET wordCount = ?, updatedAt = ? WHERE id = ?", int(s.total), nowMillis(), s.novelID)
   557→   557→                }
   558→   558→        }
   559→   559→}

   561→   561→// ==================== Phase 0：列表页条目收集 ====================
   562→   562→
   563→   563→// collectListItems 列表页翻页收集（模板优先、连续失败快速终止、合并去重、任务上限截断）。
   564→   564→// 返回 (条目, 最后命中的列表页 URL 作 referer)。
   565→   565→func collectListItems(run *Run, task TaskRecord, rule LoadedRule) ([]ListItem, string) {
   566→   566→        run.Log("抓取列表页第 1 页…")
   567→   567→        first := fetchListPage(run, task.TargetURL, rule, "")
   568→   568→        if len(first) == 0 {
   569→   569→                run.Log("列表页未提取到书籍条目")
   570→   570→                return nil, task.TargetURL
   571→   571→        }
   572→   572→        run.Log(fmt.Sprintf("第 1 页提取 %d 条", len(first)))
   573→   573→
   574→   574→        items := append([]ListItem(nil), first...)
   575→   575→        currentListURL := task.TargetURL
   576→   576→        consecutiveFails := 0
   577→   577→        if hasPaginationTemplate(rule.ListRule) {
   578→   578→                run.Log("分页模板生效: " + truncateRunes(rule.ListRule["paginationTemplate"], 120))
   579→   579→        }
   580→   580→
   581→   581→        for k := 2; k <= task.Pages; k++ {
   582→   582→                if isCanceled(run.TaskID) {
   583→   583→                        break
   584→   584→                }
   585→   585→                got := 0
   586→   586→                hit := false
   587→   587→                for _, v := range buildPageVariants(rule.ListRule, task.TargetURL, k) {
   588→   588→                        pageItems := fetchListPage(run, v, rule, "")
   589→   589→                        if len(pageItems) > 0 {
   590→   590→                                got = len(pageItems)
   591→   591→                                hit = true
   592→   592→                                currentListURL = v
   593→   593→                                items = append(items, pageItems...)
   594→   594→                                break
   595→   595→                        }
   596→   596→                }
   597→   597→                if !hit {
   598→   598→                        consecutiveFails++
   599→   599→                        run.Log(fmt.Sprintf("第 %d 页无结果（连续失败 %d）", k, consecutiveFails))
   600→   600→                        if consecutiveFails >= MAX_CONSECUTIVE_PAGE_FAILS {
   601→   601→                                run.Log(fmt.Sprintf("连续 %d 页翻页失败，提前终止翻页", MAX_CONSECUTIVE_PAGE_FAILS))
   602→   602→                                break
   603→   603→                        }
   604→   604→                        continue
   605→   605→                }
   606→   606→                consecutiveFails = 0
   607→   607→                run.Log(fmt.Sprintf("第 %d 页命中: %s，提取 %d 条", k, truncateRunes(currentListURL, 100), got))
   608→   608→        }
   609→   609→
   610→   610→        // 合并去重（按 URL，缺 URL 按标题）
   611→   611→        seen := map[string]bool{}
   612→   612→        merged := []ListItem{}
   613→   613→        for _, it := range items {
   614→   614→                key := it.URL
   615→   615→                if key == "" {
   616→   616→                        key = "t:" + it.Title
   617→   617→                }
   618→   618→                if seen[key] {
   619→   619→                        continue
   620→   620→                }
   621→   621→                seen[key] = true
   622→   622→                merged = append(merged, it)
   623→   623→        }
   624→   624→        if len(merged) > MAX_BOOKS_PER_TASK {
   625→   625→                merged = merged[:MAX_BOOKS_PER_TASK]
   626→   626→                run.Log(fmt.Sprintf("条目数超出单任务上限（%d），已截断", MAX_BOOKS_PER_TASK))
   627→   627→        }
   628→   628→        return merged, currentListURL
   629→   629→}
   630→   630→
   631→   631→// ==================== 两种模式 ====================
   632→   632→
   633→   633→// runList list 范围模式：Phase 0 列表 → Phase 1 骨架 → Phase 2 填充
   634→   634→func runList(run *Run, task TaskRecord, rule LoadedRule) {
   635→   635→        // ---- Phase 0：列表页 ----
   636→   636→        collected, currentListURL := collectListItems(run, task, rule)
   637→   637→        if isCanceled(run.TaskID) {
   638→   638→                finalize(run, "canceled", "任务已取消")
   639→   639→                return
   640→   640→        }
   641→   641→        if len(collected) == 0 {
   642→   642→                finalize(run, "failed", "列表页未提取到书籍条目")
   643→   643→                return
   644→   644→        }
   645→   645→        total := len(collected)
   646→   646→        zero := 0
   647→   647→        run.Flush(&TaskFlushFields{Total: &total, Done: &zero})
   648→   648→        run.Log(fmt.Sprintf("去重后共 %d 本书待采集", total))
   649→   649→
   650→   650→        // ---- Phase 1：书目骨架 ----
   651→   651→        run.Log(fmt.Sprintf("━━ 阶段 1/2 书目骨架（并发 %d）", BOOK_CONCURRENCY))
   652→   652→        p1 := phase1Skeletons(run, rule, collected, currentListURL)
   653→   653→        if p1.Fatal != nil { // TS：store 层非唯一冲突错误沿 promise 链冒泡 → catch 收尾
   654→   654→                run.Log("发生未预期错误: " + truncateRunes(p1.Fatal.Error(), 200))
   655→   655→                finalize(run, "failed", "任务执行异常")
   656→   656→                return
   657→   657→        }
   658→   658→        if isCanceled(run.TaskID) {
   659→   659→                recalcWordCountsFor(p1.NovelIDs)
   660→   660→                finalize(run, "canceled", fmt.Sprintf("已取消（书目完成 %d/%d 本）", p1.OKBooks, total))
   661→   661→                return
   662→   662→        }
   663→   663→        created, updated, _ := run.Snapshot()
   664→   664→        run.Flush(&TaskFlushFields{Done: &p1.OKBooks, ChaptersTotal: &p1.FillTotal, ChaptersDone: &zero, Created: &created, Updated: &updated})
   665→   665→
   666→   666→        if p1.OKBooks == 0 {
   667→   667→                msg := p1.FirstError
   668→   668→                if msg == "" {
   669→   669→                        msg = "无书籍采集成功"
   670→   670→                }
   671→   671→                finalize(run, "failed", msg)
   672→   672→                return
   673→   673→        }
   674→   674→
   675→   675→        // ---- Phase 2：正文填充 ----
   676→   676→        run.Log(fmt.Sprintf("━━ 阶段 2/2 正文填充（并发 %d，待填充 %d 章）", CHAPTER_CONCURRENCY, p1.FillTotal))
   677→   677→        var lastFlush int64
   678→   678→        p2 := phase2Fill(run, rule, p1.FillMap, func(doneSoFar int) bool {
   679→   679→                now := nowMillis()
   680→   680→                if now-lastFlush < int64(FLUSH_INTERVAL_MS) {
   681→   681→                        return true
   682→   682→                }
   683→   683→                lastFlush = now
   684→   684→                c2, u2, ch := run.Snapshot()
   685→   685→                return run.Flush(&TaskFlushFields{ChaptersDone: &doneSoFar, Chapters: &ch, Created: &c2, Updated: &u2})
   686→   686→        })
   687→   687→        recalcWordCountsFor(p1.NovelIDs)
   688→   688→        created, updated, chapters := run.Snapshot()
   689→   689→        run.Flush(&TaskFlushFields{ChaptersDone: &p2.Filled, Chapters: &chapters, Created: &created, Updated: &updated})
   690→   690→
   691→   691→        if p2.StoppedEarly {
   692→   692→                finalize(run, "canceled", fmt.Sprintf("已取消（正文填充 %d/%d 章）", p2.Filled, p1.FillTotal))
   693→   693→                return
   694→   694→        }
   695→   695→        if p2.Filled == 0 && p1.FillTotal == 0 {
   696→   696→                // 可续跑语义：重发已完成任务时骨架全部已有正文（FillTotal=0），Phase 2 无事可做
   697→   697→                // 应报成功而非「正文采集全部失败」（旧版误报 failed 会诱导用户无意义重跑；
   698→   698→                // FillTotal>0 且全败的失败语义保持不变）
   699→   699→                finalize(run, "success", fmt.Sprintf("书目 %d 本已全部有正文，无需续采", p1.OKBooks))
   700→   700→                return
   701→   701→        }
   702→   702→        if p2.Filled == 0 {
   703→   703→                finalize(run, "failed", fmt.Sprintf("书目 %d 本入库，但正文采集全部失败", p1.OKBooks))
   704→   704→                return
   705→   705→        }
   706→   706→        if p2.Failed > 0 {
   707→   707→                finalize(run, "partial", fmt.Sprintf("%d 本书 / 正文 %d 章成功，%d 章失败", p1.OKBooks, p2.Filled, p2.Failed))
   708→   708→                return
   709→   709→        }
   710→   710→        finalize(run, "success", fmt.Sprintf("范围采集完成：%d 本书，正文 %d 章", p1.OKBooks, p2.Filled))
   711→   711→}
   712→   712→
   713→   713→// runSingle single 单本模式：复用两阶段管线，一个条目 → 骨架 → 填充；done/total 主口径=章节
   714→   714→func runSingle(run *Run, task TaskRecord, rule LoadedRule) {
   715→   715→        items := []ListItem{{Title: "", URL: task.TargetURL}}
   716→   716→        run.Log(fmt.Sprintf("━━ 阶段 1/2 书目骨架（并发 %d）", BOOK_CONCURRENCY))
   717→   717→        p1 := phase1Skeletons(run, rule, items, "")
   718→   718→        if p1.Fatal != nil {
   719→   719→                run.Log("发生未预期错误: " + truncateRunes(p1.Fatal.Error(), 200))
   720→   720→                finalize(run, "failed", "任务执行异常")
   721→   721→                return
   722→   722→        }
   723→   723→        if p1.OKBooks == 0 {
   724→   724→                msg := p1.FirstError
   725→   725→                if msg == "" {
   726→   726→                        msg = "书页提取失败"
   727→   727→                }
   728→   728→                finalize(run, "failed", msg)
   729→   729→                return
   730→   730→        }
   731→   731→        if isCanceled(run.TaskID) {
   732→   732→                recalcWordCountsFor(p1.NovelIDs)
   733→   733→                finalize(run, "canceled", "任务已取消")
   734→   734→                return
   735→   735→        }
   736→   736→        totalChapters := p1.TotalRefs
   737→   737→        run.Flush(&TaskFlushFields{Total: &totalChapters, Done: &p1.SkippedFilled, ChaptersTotal: &totalChapters, ChaptersDone: &p1.SkippedFilled})
   738→   738→        if totalChapters == 0 {
   739→   739→                finalize(run, "success", "书籍已入库（未提取到章节链接）")
   740→   740→                return
   741→   741→        }
   742→   742→
   743→   743→        run.Log(fmt.Sprintf("━━ 阶段 2/2 正文填充（并发 %d，待填充 %d 章）", CHAPTER_CONCURRENCY, p1.FillTotal))
   744→   744→        var lastFlush int64
   745→   745→        p2 := phase2Fill(run, rule, p1.FillMap, func(doneSoFar int) bool {
   746→   746→                now := nowMillis()
   747→   747→                if now-lastFlush < int64(FLUSH_INTERVAL_MS) {
   748→   748→                        return true
   749→   749→                }
   750→   750→                lastFlush = now
   751→   751→                done := p1.SkippedFilled + doneSoFar
   752→   752→                return run.Flush(&TaskFlushFields{Done: &done, ChaptersDone: &done})
   753→   753→        })
   754→   754→
   755→   755→        recalcWordCountsFor(p1.NovelIDs)
   756→   756→        done := p1.SkippedFilled + p2.Filled
   757→   757→        _, _, chapters := run.Snapshot()
   758→   758→        run.Flush(&TaskFlushFields{Done: &done, ChaptersDone: &done, Chapters: &chapters})
   759→   759→
   760→   760→        if p2.StoppedEarly {
   761→   761→                finalize(run, "canceled", "任务已取消")
   762→   762→                return
   763→   763→        }
   764→   764→        if p2.Filled == 0 && p1.FillTotal == 0 {
   765→   765→                // 同 runList：骨架已全部有正文的可续跑重发报成功（旧版误报 failed）
   766→   766→                finalize(run, "success", fmt.Sprintf("书籍已入库且正文齐全（%d 章，续跑无待填充章节）", p1.SkippedFilled))
   767→   767→                return
   768→   768→        }
   769→   769→        if p2.Filled == 0 {
   770→   770→                finalize(run, "failed", "章节采集全部失败")
   771→   771→                return
   772→   772→        }
   773→   773→        if p2.Failed > 0 {
   774→   774→                finalize(run, "partial", fmt.Sprintf("%d 章成功 / %d 章失败", p2.Filled, p2.Failed))
   775→   775→                return
   776→   776→        }
   777→   777→        finalize(run, "success", fmt.Sprintf("采集完成：%d 章", p2.Filled))
   778→   778→}

   782→   782→// finalize 终态写入：仅当仍处于 running 时写终态；已被取消/删除则只保留日志
   783→   783→func finalize(run *Run, status, message string) {
   784→   784→        var cur string
   785→   785→        if err := queryOne("SELECT status FROM ScrapeTask WHERE id = ?", []any{&cur}, run.TaskID); err != nil {
   786→   786→                return
   787→   787→        }
   788→   788→        msg := truncateRunes(message, 500)
   789→   789→        if cur == "running" {
   790→   790→                _, _ = execRetry("UPDATE ScrapeTask SET status = ?, message = ?, log = ? WHERE id = ? AND status = 'running'",
   791→   791→                        status, msg, run.LogText(), run.TaskID)
   792→   792→        } else {
   793→   793→                _, _ = execRetry("UPDATE ScrapeTask SET log = ? WHERE id = ?", run.LogText(), run.TaskID)
   794→   794→        }
   795→   795→}

   797→   797→// runTask 任务执行主链路（pending → running 条件更新 → 分模式执行 → 终态）
   798→   798→func runTask(taskID int) {
   799→   799→        run := NewRun(taskID)
   800→   800→        defer func() {
   801→   801→                if r := recover(); r != nil {
   802→   802→                        run.Log("发生未预期错误: " + truncateRunes(fmt.Sprintf("%v", r), 200))
   803→   803→                        finalize(run, "failed", "任务执行异常")
   804→   804→                }
   805→   805→        }()
   806→   806→
   807→   807→        // pending → running 条件更新：pending 阶段已被取消/删除的任务不再启动
   808→   808→        res, err := execRetry("UPDATE ScrapeTask SET status = 'running' WHERE id = ? AND status = 'pending'", taskID)
   809→   809→        if err != nil || rowCountOf(res) == 0 {
   810→   810→                return
   811→   811→        }
   812→   812→        var mode, targetURL string
   813→   813→        var pages int
   814→   814→        var ruleID sql.NullInt64
   815→   815→        if err := queryOne("SELECT mode, targetUrl, pages, ruleId FROM ScrapeTask WHERE id = ?",
   816→   816→                []any{&mode, &targetURL, &pages, &ruleID}, taskID); err != nil {
   817→   817→                return
   818→   818→        }
   819→   819→        task := TaskRecord{ID: taskID, Mode: mode, TargetURL: targetURL, Pages: pages}
   820→   820→        if ruleID.Valid {
   821→   821→                v := int(ruleID.Int64)
   822→   822→                task.RuleID = &v
   823→   823→        }
   824→   824→        rule := loadRule(task.RuleID)
   825→   825→        if task.Mode == "list" {
   826→   826→                run.Log(fmt.Sprintf("任务开始（范围采集·两阶段并发）目标: %s，页数上限: %d", task.TargetURL, task.Pages))
   827→   827→        } else {
   828→   828→                run.Log(fmt.Sprintf("任务开始（单本采集·两阶段并发）目标: %s", task.TargetURL))
   829→   829→        }
   830→   830→        if task.RuleID != nil {
   831→   831→                name := rule.Name
   832→   832→                if name == "" {
   833→   833→                        name = itoa(*task.RuleID)
   834→   834→                }
   835→   835→                charset := rule.Charset
   836→   836→                if charset == "" {
   837→   837→                        charset = "auto"
   838→   838→                }
   839→   839→                run.Log(fmt.Sprintf("使用规则「%s」（charset=%s）", name, charset))
   840→   840→        } else {
   841→   841→                run.Log("未使用规则，依赖引擎内置启发式提取")
   842→   842→        }
   843→   843→        run.Flush(nil)
   844→   844→        if task.Mode == "list" {
   845→   845→                runList(run, task, rule)
   846→   846→        } else {
   847→   847→                runSingle(run, task, rule)
   848→   848→        }
   849→   849→}

   851→   851→// triggerScrapeTask fire-and-forget 入口：runner 轮询与任务创建 API 共用，不阻塞调用方
   852→   852→func triggerScrapeTask(taskID int) {
   853→   853→        gRunningMu.Lock()
   854→   854→        if gRunning[taskID] {
   855→   855→                gRunningMu.Unlock()
   856→   856→                return
   857→   857→        }
   858→   858→        gRunning[taskID] = true
   859→   859→        gRunningMu.Unlock()
   860→   860→        go func() {
   861→   861→                defer runningDelete(taskID)
   862→   862→                defer func() {
   863→   863→                        if r := recover(); r != nil {
   864→   864→                                log.Printf("[scrape-worker] task %d 兜底异常: %v", taskID, r)
   865→   865→                        }
   866→   866→                }()
   867→   867→                runTask(taskID)
   868→   868→        }()
   869→   869→}
