   432→/* Category：左 190 榜单侧栏 + 右 760 六列数据表 + 翻页                   */
   433→/* ==================================================================== */
   434→
   435→type SortKey = 'latest' | 'clicks' | 'words'
   436→type StatusKey = 'all' | 'serial' | 'finished'
   437→type PresetKey = 'clicks' | 'latest' | 'finished' | 'words'
   438→
   439→function CategoryView({
   440→  navigate,
   441→  categoryId,
   442→  page = 1,
   443→}: ViewProps & { categoryId?: number; page?: number }) {
   444→  const { data: cats } = useCategories()
   445→  const { data: home } = useHomeData()
   446→  const [sort, setSort] = useState<SortKey>('latest')
   447→  const [status, setStatus] = useState<StatusKey>('all')
   448→  const cur = page > 0 ? page : 1
   449→
   450→  const { data, isLoading, isError, refetch } = useNovels({
   451→    categoryId,
   452→    page: cur,
   453→    pageSize: 20,
   454→    sort,
   455→    status: status === 'all' ? undefined : status,
   456→  })
   457→
   458→  const catName =
   459→    categoryId != null ? (cats ?? []).find((c) => c.id === categoryId)?.name : undefined
   460→  const goPage = (p: number) => navigate({ name: 'category', categoryId, page: p })
   461→  const applySort = (s: SortKey) => {
   462→    setSort(s)
   463→    setStatus('all')
   464→    if (cur !== 1) goPage(1)
   465→  }
   466→  const applyStatus = (s: StatusKey) => {
   467→    setStatus(s)
   468→    if (cur !== 1) goPage(1)
   469→  }
   470→  const applyPreset = (p: PresetKey) => {
   471→    if (p === 'finished') {
   472→      setSort('latest')
   473→      setStatus('finished')
   474→    } else {
   475→      setSort(p)
   476→      setStatus('all')
   477→    }
   478→    if (cur !== 1) goPage(1)
   479→  }
   480→
   481→  return (
   482→    <div className="mx-auto w-full max-w-[960px] px-2 py-3">
   483→      <Crumbs
   484→        items={[{ label: '首页', view: { name: 'home' } }, { label: catName ?? '全部小说' }]}
   485→        navigate={navigate}
   486→      />
   487→

   621→function loadShelf(): number[] {
   622→  if (typeof window === 'undefined') return []
   623→  try {

   630→function toggleShelf(id: number): boolean {
   631→  if (typeof window === 'undefined') return false
   632→  try {
   633→    const arr = loadShelf()
   634→    const has = arr.includes(id)
   635→    window.localStorage.setItem(SHELF_KEY, JSON.stringify(has ? arr.filter((x) => x !== id) : [...arr, id]))
   636→    return !has
   637→  } catch {
   638→    return false
   639→  }
   640→}
