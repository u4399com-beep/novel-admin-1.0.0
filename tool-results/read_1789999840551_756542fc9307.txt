    46→    46→// jsParseFloat 模拟 JS Number(string)：空白串→0，非法→NaN(false)
    47→    47→func jsParseFloat(s string) (float64, bool) {
    48→    48→        t := trimSpaceStr(s)
    49→    49→        if t == "" {
    50→    50→                return 0, true
    51→    51→        }
    52→    52→        f, err := strconv.ParseFloat(t, 64)
    53→    53→        if err != nil {
    54→    54→                return math.NaN(), false
    55→    55→        }
    56→    56→        return f, true
    57→    57→}

    59→    59→// numIsInt JS Number.isInteger 数值判定
    60→    60→func numIsInt(f float64) bool {
    61→    61→        if math.IsNaN(f) || math.IsInf(f, 0) {
    62→    62→                return false
    63→    63→        }
    64→    64→        return f == math.Trunc(f)
    65→    65→}

    67→    67→// intFieldStrict 严格类型整数字段（JS Number.isInteger(v)：v 必须本身是 number）
    68→    68→func intFieldStrict(v any) (int, bool) {
    69→    69→        f, ok := v.(float64)
    70→    70→        if !ok || !numIsInt(f) {
    71→    71→                return 0, false
    72→    72→        }
    73→    73→        return int(f), true

    76→    76→// jsNumber JS Number() 对 JSON 值的宽式转换
    77→    77→func jsNumber(v any) float64 {
    78→    78→        switch x := v.(type) {
    79→    79→        case float64:
    80→    80→                return x
    81→    81→        case bool:
    82→    82→                if x {
    83→    83→                        return 1
    84→    84→                }
    85→    85→                return 0
    86→    86→        case nil:
    87→    87→                return 0
    88→    88→        case string:
    89→    89→                f, _ := jsParseFloat(x)
    90→    90→                return f
    91→    91→        case []any:
    92→    92→                if len(x) == 0 {
    93→    93→                        return 0
    94→    94→                }
    95→    95→                if len(x) == 1 {
    96→    96→                        return jsNumber(x[0])
    97→    97→                }
    98→    98→                return math.NaN()
    99→    99→        default:
   100→   100→                return math.NaN()
   101→   101→        }
   102→   102→}

   116→   116→func routeIntID(w http.ResponseWriter, raw, errMsg string) (int, bool) {

   119→   119→                writeJSON(w, 400, map[string]string{"error": errMsg})
   120→   120→                return 0, false
   121→   121→        }
   122→   122→        return int(f), true
   123→   123→}

   138→   138→// pageParamFloor 对齐 TS `Math.floor(Number(v) || def)`（Number(”)/NaN/0 → def）
   139→   139→func pageParamFloor(raw string, def int) int {
   140→   140→        if raw == "" {
   141→   141→                return def
   142→   142→        }
   143→   143→        f, ok := jsParseFloat(raw)
   144→   144→        if !ok || f == 0 {
   145→   145→                return def
   146→   146→        }
   147→   147→        if f > 2e9 {
   148→   148→                return 2_000_000_000
   149→   149→        }
   150→   150→        if f < -2e9 {
   151→   151→                return -2_000_000_000
   152→   152→        }
   153→   153→        return int(math.Floor(f))
   154→   154→}

   157→   157→func pageParamList(raw string) int {

   161→   161→        }

   165→   165→// readBodyValue 读请求体为任意 JSON 值（不写错误响应；由路由按 TS 契约还原文案）。

   167→   167→func readBodyValue(r *http.Request) (any, bool) {

   169→   169→        if err != nil {

   171→   171→        }

   173→   173→        if json.Unmarshal(b, &v) != nil {
   174→   174→                return nil, false
   175→   175→        }
   176→   176→        return v, true
   177→   177→}

   208→   208→// firstLineErr 移植 src/lib/errors.ts firstLine：错误首行 200 字截断（防泄露内部路径）
   209→   209→func firstLineErr(err error) string {
   210→   210→        if err == nil {
   211→   211→                return ""
   212→   212→        }
   213→   213→        msg := err.Error()
   214→   214→        if i := strings.IndexByte(msg, '\n'); i >= 0 {

   220→   220→// ---- JS 空白语义（\s 全集：Go \s 缺 \v 与 unicode 空白，必须显式补齐）----

   222→   222→var (
   223→   223→        jsSpaceClass    = `[\t\n\v\f\r \x{00a0}\x{1680}\x{2000}-\x{200a}\x{2028}\x{2029}\x{202f}\x{205f}\x{3000}\x{feff}]`
   224→   224→        jsSpacePlusRe   = regexp.MustCompile(`[\t\n\v\f\r \x{00a0}\x{1680}\x{2000}-\x{200a}\x{2028}\x{2029}\x{202f}\x{205f}\x{3000}\x{feff}]+`)
   225→   225→        jsTrimLeadRe    = regexp.MustCompile(`^[\t\n\v\f\r \x{00a0}\x{1680}\x{2000}-\x{200a}\x{2028}\x{2029}\x{202f}\x{205f}\x{3000}\x{feff}]+`)
   226→   226→        jsTrimTrailRe   = regexp.MustCompile(`[\t\n\v\f\r \x{00a0}\x{1680}\x{2000}-\x{200a}\x{2028}\x{2029}\x{202f}\x{205f}\x{3000}\x{feff}]+$`)
   227→   227→        jsSpaceRuneSet  = map[rune]bool{'\t': true, '\n': true, '\v': true, '\f': true, '\r': true, ' ': true, 0x00a0: true, 0x1680: true, 0x2028: true, 0x2029: true, 0x202f: true, 0x205f: true, 0x3000: true, 0xfeff: true}
   228→   228→        jsSpaceRuneFold = func() map[rune]bool {
   229→   229→                m := map[rune]bool{}
   230→   230→                for r := rune(0x2000); r <= 0x200a; r++ {
   231→   231→                        m[r] = true
   232→   232→                }
   233→   233→                for r := range jsSpaceRuneSet {
   234→   234→                        m[r] = true
   235→   235→                }
   236→   236→                return m
   237→   237→        }()
   238→   238→)

   258→   258→// splitJSSpace 模拟 s.split(/\s+/).filter(Boolean)
   259→   259→func splitJSSpace(s string) []string {
   260→   260→        parts := jsSpacePlusRe.Split(s, -1)
   261→   261→        out := make([]string, 0, len(parts))
   262→   262→        for _, p := range parts {
   263→   263→                if p != "" {
   264→   264→                        out = append(out, p)
   265→   265→                }
   266→   266→        }
   267→   267→        return out
   268→   268→}

   273→   273→// marshalCompact JSON.stringify 等价（无 HTML 转义、无缩进、无尾随换行）

   275→   275→        var buf strings.Builder

   331→   331→        q := "SELECT " + novelListCols + novelListFrom + whereSQL + orderSQL + " LIMIT ? OFFSET ?"

   341→   341→        if err != nil {
   342→   342→                return nil, err
   343→   343→        }

   349→   349→func handleNovelsList(w http.ResponseWriter, r *http.Request, _ map[string]string) {
   350→   350→        sp := r.URL.Query()
   351→   351→        q := truncateRunes(trimSpaceStr(sp.Get("q")), 100)
   352→   352→        status := sp.Get("status")
   353→   353→        sort := sp.Get("sort")
   354→   354→        if sort == "" {
   355→   355→                sort = "latest"
   356→   356→        }
   357→   357→        page := pageParamList(sp.Get("page"))
   358→   358→        pageSize := clampInt(pageParamFloor(sp.Get("pageSize"), 20), 4, 60)
   359→   359→
   360→   360→        // 非法 categoryId（abc/1.5/-3）→ 400（对齐 TS 明确报错而非静默全库查询）
   361→   361→        categoryId := 0
   362→   362→        if raw := sp.Get("categoryId"); raw != "" {
   363→   363→                f, ok := jsParseFloat(raw)
   364→   364→                if !ok || !numIsInt(f) || f < 0 {
   365→   365→                        writeJSON(w, 400, map[string]string{"error": "无效 categoryId"})
   366→   366→                        return
   367→   367→                }
   368→   368→                categoryId = int(f)
   369→   369→        }
   370→   370→
   371→   371→        conds := []string{}
   372→   372→        args := []any{}
   373→   373→        if categoryId > 0 {
   374→   374→                conds = append(conds, `n."categoryId" = ?`)
   375→   375→                args = append(args, categoryId)
   376→   376→        }
   377→   377→        if status == "serial" || status == "finished" {
   378→   378→                conds = append(conds, `n."status" = ?`)
   379→   379→                args = append(args, status)
   380→   380→        }
   381→   381→        if q != "" {
   382→   382→                conds = append(conds, `(n."title" LIKE ? OR n."author" LIKE ? OR n."description" LIKE ?)`)
   383→   383→                like := likeWrap(q)
   384→   384→                args = append(args, like, like, like)
   385→   385→        }
   386→   386→        whereSQL := ""
   387→   387→        if len(conds) > 0 {
   388→   388→                whereSQL = " WHERE " + strings.Join(conds, " AND ")
   389→   389→        }
   390→   390→        var orderSQL string
   391→   391→        switch sort {
   392→   392→        case "clicks":
   393→   393→                orderSQL = ` ORDER BY n."clicks" DESC, n."id" DESC`
   394→   394→        case "words":
   395→   395→                orderSQL = ` ORDER BY n."wordCount" DESC, n."id" DESC`
   396→   396→        case "featured":
   397→   397→                orderSQL = ` ORDER BY n."isFeatured" DESC, n."updatedAt" DESC, n."id" DESC`
   398→   398→        default:
   399→   399→                orderSQL = ` ORDER BY n."updatedAt" DESC, n."id" DESC`
   400→   400→        }
   401→   401→
   402→   402→        var total int64
   403→   403→        countQ := `SELECT COUNT(*) FROM "Novel" n` + whereSQL
   404→   404→        if err := queryOne(countQ, []any{&total}, args...); err != nil {
   405→   405→                failJSON(w, "查询失败", firstLineErr(err), 500)
   406→   406→                return
   407→   407→        }
   408→   408→        list, err := queryNovelList(whereSQL, orderSQL, args, pageSize, (page-1)*pageSize)
   409→   409→        if err != nil {
   410→   410→                failJSON(w, "查询失败", firstLineErr(err), 500)
   411→   411→                return
   412→   412→        }
   413→   413→        totalPages := (total + int64(pageSize) - 1) / int64(pageSize)
   414→   414→        if totalPages < 1 {
   415→   415→                totalPages = 1
   416→   416→        }
   417→   417→        writeJSON(w, 200, map[string]any{
   418→   418→                "list":       list,
   419→   419→                "total":      total,
   420→   420→                "page":       page,
   421→   421→                "pageSize":   pageSize,

   604→   604→func handleNovelUpdate(w http.ResponseWriter, r *http.Request, ps map[string]string) {
   605→   605→        nid, ok := routePosIntID(w, ps["id"], "无效 ID")
   606→   606→        if !ok {
   607→   607→                return
   608→   608→        }
   609→   609→        v, ok := readBodyValue(r)
   610→   610→        if !ok {
   611→   611→                writeJSON(w, 400, map[string]string{"error": "请求体不是合法 JSON"})
   612→   612→                return
   613→   613→        }
   614→   614→        body := bodyMap(v)
   615→   615→
   616→   616→        sets := []string{}
   617→   617→        args := []any{}
   618→   618→        if s, ok2 := body["title"].(string); ok2 {
   619→   619→                if t := trimSpaceStr(s); t != "" {
   620→   620→                        sets = append(sets, `"title" = ?`)
   621→   621→                        args = append(args, truncateRunes(t, novelTitleMax))
   622→   622→                }
   623→   623→        }
   624→   624→        if s, ok2 := body["author"].(string); ok2 {
   625→   625→                if t := trimSpaceStr(s); t != "" {
   626→   626→                        sets = append(sets, `"author" = ?`)
   627→   627→                        args = append(args, truncateRunes(t, novelAuthorMax))
   628→   628→                }
   629→   629→        }
   630→   630→        if s, ok2 := body["description"].(string); ok2 {
   631→   631→                sets = append(sets, `"description" = ?`)
   632→   632→                args = append(args, truncateRunes(s, novelDescriptionMax))
   633→   633→        }
   634→   634→        if s, ok2 := body["cover"].(string); ok2 {
   635→   635→                if coverTokenRe.MatchString(s) {
   636→   636→                        sets = append(sets, `"cover" = ?`)
   637→   637→                        args = append(args, s)
   638→   638→                }
   639→   639→        }
   640→   640→        if n, ok2 := intFieldStrict(body["categoryId"]); ok2 && n > 0 {
   641→   641→                sets = append(sets, `"categoryId" = ?`)
   642→   642→                args = append(args, n)
   643→   643→        }
   644→   644→        if s, ok2 := body["status"].(string); ok2 && (s == "serial" || s == "finished") {
   645→   645→                sets = append(sets, `"status" = ?`)
   646→   646→                args = append(args, s)
   647→   647→        }
   648→   648→        if b, ok2 := body["isFeatured"].(bool); ok2 {
   649→   649→                sets = append(sets, `"isFeatured" = ?`)
   650→   650→                args = append(args, b)
   651→   651→        }
   652→   652→        if b, ok2 := body["isHot"].(bool); ok2 {
   653→   653→                sets = append(sets, `"isHot" = ?`)
   654→   654→                args = append(args, b)
   655→   655→        }
   656→   656→        // Prisma @updatedAt 任何 update 都会触碰 → 显式 set now
   657→   657→        sets = append(sets, `"updatedAt" = ?`)
   658→   658→        args = append(args, nowMillis())
   659→   659→
   660→   660→        res, err := exec(`UPDATE "Novel" SET `+strings.Join(sets, ", ")+` WHERE "id" = ?`, append(args, nid)...)
   661→   661→        if err != nil {
   662→   662→                // P2025 之外的一切错误（分类外键不存在等）→ 400（对齐 TS 分类注释语义）
   663→   663→                failJSON(w, "更新失败（分类不存在？）", "", 400)
   664→   664→                return
   665→   665→        }
   666→   666→        if n, _ := res.RowsAffected(); n == 0 {
   667→   667→                writeJSON(w, 404, map[string]string{"error": "小说不存在"})
   668→   668→                return
   669→   669→        }
   670→   670→        writeJSON(w, 200, map[string]any{"ok": true})
   671→   671→}
