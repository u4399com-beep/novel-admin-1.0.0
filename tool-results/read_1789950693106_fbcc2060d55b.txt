   148→/* ==================== 分类排行榜模块卡（源站 .ranking-module） ==================== */
   149→
   150→function RankModule({
   151→  categoryId,
   152→  catName,
   153→  navigate,
   154→}: {
   155→  categoryId: number
   156→  catName: string
   157→  navigate: Nav
   158→}) {
   159→  const q = useNovels({ categoryId, sort: 'clicks', page: 1, pageSize: 10 })
   160→  const rows = q.data?.list ?? []
   161→  return (
   162→    <div className="overflow-hidden rounded-[10px] border border-[#dbe4f0] bg-white shadow-[0_1px_2px_rgba(15,23,42,0.04),0_4px_16px_rgba(37,99,235,0.06)] transition-shadow duration-300 hover:shadow-[0_8px_24px_rgba(37,99,235,0.14),0_2px_8px_rgba(15,23,42,0.06)]">
   163→      <div className="flex items-center gap-2 border-b border-[#dbe4f0] bg-[#f0f4fb] px-4 py-3 text-[18px] font-semibold text-[#1e293b]">
   164→        <span className="h-4 w-[3px] rounded-sm bg-[#2563eb]" />
   165→        {catName}小说榜
   166→      </div>
   167→      {q.isPending ? (
   168→        <div className="p-3">
   169→          {Array.from({ length: 8 }, (_, i) => (
   170→            <Skel key={i} className="mt-2 h-5" />
   171→          ))}
   172→        </div>
   173→      ) : rows.length === 0 ? (
   174→        <p className="px-4 py-6 text-center text-[14px] text-[#94a3b8]">该分类暂无上榜小说</p>
   175→      ) : (
   176→        <ol className="py-1">
   177→          {rows.map((n, i) => (
   178→            <RankRow key={n.id} novel={n} index={i} navigate={navigate} />
   179→          ))}
   180→        </ol>
   181→      )}
   182→    </div>
   183→  )
   184→}

   188→export function Category({
   189→  navigate,
   190→  categoryId,
   191→  page = 1,
   192→}: ViewProps & { categoryId?: number; page?: number }) {
   193→  const cats = useCategories()
   194→  const list = useNovels({ categoryId, page, pageSize: 20 })
   195→
   196→  const catName = cats.data?.find((c) => c.id === categoryId)?.name
   197→
   198→  return (
   199→    <div className="mx-auto max-w-[1180px] space-y-5 px-4 py-6">
   200→      <Crumb items={[{ label: '首页', view: { name: 'home' } }, { label: '小说书库' }]} navigate={navigate} />
   201→      <h1 className="border-l-4 border-[#2563eb] pl-4 text-[24px] leading-8 font-semibold text-[#1e293b]">
   202→        {catName ?? '小说书库'}
   203→        {cats.data ? (
   204→          <span className="ml-3 text-[13px] font-normal text-[#94a3b8]">
   205→            {cats.data.find((c) => c.id === categoryId)?.novelCount ?? list.data?.total ?? ''}
   206→          </span>
   207→        ) : null}
   208→      </h1>
   209→
   210→      {/* 筛选条 */}
   211→      <div className="flex flex-wrap gap-2 rounded-[10px] border border-[#e6edf7] bg-white p-4 shadow-[0_1px_3px_rgba(37,99,235,0.06)]">
   212→        <button
   213→          onClick={() => navigate({ name: 'category', page: 1 })}
   214→          className={
   215→            categoryId == null
   216→              ? 'cursor-pointer rounded-full bg-[#2563eb] px-4 py-1.5 text-[13px] text-white transition-colors duration-200'
   217→              : 'cursor-pointer rounded-full bg-[#f0f4fb] px-4 py-1.5 text-[13px] text-[#1e293b] transition-colors duration-200 hover:bg-[#e8f1ff] hover:text-[#2563eb]'
   218→          }
   219→        >
   220→          全部
   221→        </button>
   222→        {(cats.data ?? []).map((c) => (
   223→          <button
   224→            key={c.id}
   225→            onClick={() => navigate({ name: 'category', categoryId: c.id, page: 1 })}
   226→            className={
   227→              categoryId === c.id
   228→                ? 'cursor-pointer rounded-full bg-[#2563eb] px-4 py-1.5 text-[13px] text-white transition-colors duration-200'
   229→                : 'cursor-pointer rounded-full bg-[#f0f4fb] px-4 py-1.5 text-[13px] text-[#1e293b] transition-colors duration-200 hover:bg-[#e8f1ff] hover:text-[#2563eb]'
   230→            }
   231→          >
   232→            {c.name}
   233→          </button>
   234→        ))}
   235→      </div>
   236→
   237→      {list.isPending ? (
   238→        <CardGridSkeleton count={6} />
   239→      ) : list.isError ? (
   240→        <ErrorBox onRetry={() => list.refetch()} />
   241→      ) : (list.data?.list.length ?? 0) === 0 ? (
   242→        <EmptyBox text="该分类下暂无收录小说" />
   243→      ) : (
   244→        <>
   245→          <div className="grid gap-4 md:grid-cols-2 xl:grid-cols-3">
   246→            {list.data!.list.map((n) => (
   247→              <BookTextCard key={n.id} novel={n} navigate={navigate} />
   248→            ))}
   249→          </div>
   250→          <Pager
   251→            page={list.data!.page}
   252→            totalPages={list.data!.totalPages}
   253→            onGo={(p) => navigate({ name: 'category', categoryId, page: p })}
   254→          />
   255→        </>
   256→      )}
   257→    </div>
   258→  )
   259→}
