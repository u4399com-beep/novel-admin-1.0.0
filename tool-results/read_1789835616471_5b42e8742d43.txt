   147→export function Category({
   148→  navigate,
   149→  categoryId,
   150→  page = 1,
   151→}: ViewProps & { categoryId?: number; page?: number }) {
   152→  const cats = useCategories()
   153→  const list = useNovels({ categoryId, page, pageSize: 20 })
   154→
   155→  const catName = cats.data?.find((c) => c.id === categoryId)?.name
   156→
   157→  return (
   158→    <div className="mx-auto max-w-[1180px] space-y-5 px-4 py-6">
   159→      <Crumb items={[{ label: '首页', view: { name: 'home' } }, { label: '小说书库' }]} navigate={navigate} />
   160→      <h1 className="border-l-4 border-[#2563eb] pl-4 text-[24px] leading-8 font-semibold text-[#1e293b]">
   161→        {catName ?? '小说书库'}
   162→        {cats.data ? (
   163→          <span className="ml-3 text-[13px] font-normal text-[#94a3b8]">
   164→            {cats.data.find((c) => c.id === categoryId)?.novelCount ?? list.data?.total ?? ''}
   165→          </span>
   166→        ) : null}
   167→      </h1>
   168→
   169→      {/* 筛选条 */}
   170→      <div className="flex flex-wrap gap-2 rounded-[10px] border border-[#e6edf7] bg-white p-4 shadow-[0_1px_3px_rgba(37,99,235,0.06)]">
   171→        <button
   172→          onClick={() => navigate({ name: 'category', page: 1 })}
   173→          className={
   174→            categoryId == null
   175→              ? 'cursor-pointer rounded-full bg-[#2563eb] px-4 py-1.5 text-[13px] text-white transition-colors duration-200'
   176→              : 'cursor-pointer rounded-full bg-[#f0f4fb] px-4 py-1.5 text-[13px] text-[#1e293b] transition-colors duration-200 hover:bg-[#e8f1ff] hover:text-[#2563eb]'
   177→          }
   178→        >
   179→          全部
   180→        </button>
   181→        {(cats.data ?? []).map((c) => (
   182→          <button
   183→            key={c.id}
   184→            onClick={() => navigate({ name: 'category', categoryId: c.id, page: 1 })}
   185→            className={
   186→              categoryId === c.id
   187→                ? 'cursor-pointer rounded-full bg-[#2563eb] px-4 py-1.5 text-[13px] text-white transition-colors duration-200'
   188→                : 'cursor-pointer rounded-full bg-[#f0f4fb] px-4 py-1.5 text-[13px] text-[#1e293b] transition-colors duration-200 hover:bg-[#e8f1ff] hover:text-[#2563eb]'
   189→            }
   190→          >
   191→            {c.name}
   192→          </button>
   193→        ))}
   194→      </div>
   195→
   196→      {list.isPending ? (
   197→        <CardGridSkeleton count={6} />
   198→      ) : list.isError ? (
   199→        <ErrorBox onRetry={() => list.refetch()} />
   200→      ) : (list.data?.list.length ?? 0) === 0 ? (
   201→        <EmptyBox text="该分类下暂无收录小说" />
   202→      ) : (
   203→        <>
   204→          <div className="grid gap-4 md:grid-cols-2 xl:grid-cols-3">
   205→            {list.data!.list.map((n) => (
   206→              <BookTextCard key={n.id} novel={n} navigate={navigate} />
   207→            ))}
   208→          </div>
   209→          <Pager
   210→            page={list.data!.page}
   211→            totalPages={list.data!.totalPages}
   212→            onGo={(p) => navigate({ name: 'category', categoryId, page: p })}
   213→          />
   214→        </>
   215→      )}
   216→    </div>
   217→  )
   218→}

   423→export function Toc({ navigate, novelId }: ViewProps & { novelId: number }) {

   425→  const { data: chapters, isPending, isError, refetch } = useChapters(novelId)

   432→          { label: novel?.title ?? '…', view: { name: 'book', novelId } },
   433→          { label: '章节目录' },
   434→        ]}
   435→        navigate={navigate}
   436→      />
   437→      {/* 最新章节（全书倒数 12 章，新→旧） */}

   439→        <section className="rounded-[10px] border border-[#e6edf7] bg-white p-6 shadow-[0_1px_3px_rgba(37,99,235,0.06)]">
   440→          <SectionTitle
   441→            title="最新章节"

   444→              <button

   446→                className="shrink-0 cursor-pointer rounded-[10px] border-[1.5px] border-[#dbe4f0] bg-white px-5 py-2 text-[13px] text-[#1e293b] transition-colors duration-200 hover:border-[#2563eb] hover:text-[#2563eb]"
   447→              >
   448→                返回书页
   449→              </button>
   450→            }
   451→          />
   452→          <ChapterPills chapters={[...chapters].slice(-12).reverse()} navigate={navigate} />
   453→        </section>
   454→      )}
   455→
   456→      <section className="rounded-[10px] border border-[#e6edf7] bg-white p-6 shadow-[0_1px_3px_rgba(37,99,235,0.06)]">
   457→        <SectionTitle
   458→          title={novel ? `${novel.title} · 章节目录` : '章节目录'}
   459→          small={chapters ? `共 ${chapters.length} 章` : undefined}
   460→          action={
   461→            <button
   462→              onClick={() => navigate({ name: 'book', novelId })}
   463→              className="shrink-0 cursor-pointer rounded-[10px] border-[1.5px] border-[#dbe4f0] bg-white px-5 py-2 text-[13px] text-[#1e293b] transition-colors duration-200 hover:border-[#2563eb] hover:text-[#2563eb]"

   465→              返回书页

   467→          }
   468→        />

   493→    </div>
   494→  )

   754→
   755→function SearchPanel({ query, navigate }: { query: string; navigate: Nav }) {
   756→  const [input, setInput] = useState(query)
   757→  const [page, setPage] = useState(1)
   758→  const list = useNovels({ q: query || undefined, page, pageSize: 20 })
   759→
   760→  const submit = () => {
   761→    const kw = input.trim()
   762→    if (kw) navigate({ name: 'search', query: kw })
   763→  }
   764→
   765→  return (
   766→    <div className="space-y-5">
   767→      <div className="flex gap-3 rounded-[10px] border border-[#e6edf7] bg-white p-4 shadow-[0_1px_3px_rgba(37,99,235,0.06)]">
   768→        <input
   769→          value={input}
   770→          onChange={(e) => setInput(e.target.value)}
   771→          onKeyDown={(e) => e.key === 'Enter' && submit()}
   772→          placeholder="输入书名 / 作者关键词"
   773→          className="h-10 min-w-0 flex-1 rounded-[10px] border border-[#dbe4f0] bg-[#f0f4fb] px-4 text-[14px] text-[#1e293b] outline-none transition-colors duration-200 placeholder:text-[#94a3b8] focus:border-[#2563eb] focus:bg-white"
   774→        />
   775→        <button
   776→          onClick={submit}
   777→          className="cursor-pointer rounded-[10px] bg-[#2563eb] px-8 text-[14px] text-white transition-colors duration-200 hover:bg-[#1d4ed8]"
   778→        >
   779→          搜索
   780→        </button>
   781→      </div>
   782→
   783→      <SectionTitle
   784→        title="搜索结果"
   785→        small={query ? `“${query}” · ${list.data?.total ?? 0} 条` : '请输入关键词'}
   786→      />
   787→
   788→      {list.isPending ? (
   789→        <CardGridSkeleton count={6} />
   790→      ) : list.isError ? (
   791→        <ErrorBox onRetry={() => list.refetch()} />
   792→      ) : (list.data?.list.length ?? 0) === 0 ? (
   793→        <EmptyBox text={`没有找到与“${query}”相关的小说，换个关键词试试`} />
   794→      ) : (
   795→        <>
   796→          <div className="grid gap-4 md:grid-cols-2 xl:grid-cols-3">
   797→            {list.data!.list.map((n) => (
   798→              <BookTextCard key={n.id} novel={n} navigate={navigate} />
   799→            ))}
   800→          </div>
   801→          <Pager
   802→            page={list.data!.page}
   803→            totalPages={list.data!.totalPages}
   804→            onGo={(p) => setPage(p)}
   805→          />
   806→        </>
   807→      )}
   808→    </div>
   809→  )
   810→}
