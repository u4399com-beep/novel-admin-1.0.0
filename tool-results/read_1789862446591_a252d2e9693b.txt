   422→/* Category：左 190 榜单侧栏 + 右 760 六列数据表 + 翻页                   */
   423→/* ==================================================================== */
   424→
   425→type SortKey = 'latest' | 'clicks' | 'words'
   426→type StatusKey = 'all' | 'serial' | 'finished'
   427→type PresetKey = 'clicks' | 'latest' | 'finished' | 'words'
   428→
   429→function CategoryView({
   430→  navigate,
   431→  categoryId,
   432→  page = 1,
   433→}: ViewProps & { categoryId?: number; page?: number }) {
   434→  const { data: cats } = useCategories()
   435→  const { data: home } = useHomeData()
   436→  const [sort, setSort] = useState<SortKey>('latest')
   437→  const [status, setStatus] = useState<StatusKey>('all')
   438→  const cur = page > 0 ? page : 1
   439→
   440→  const { data, isLoading, isError, refetch } = useNovels({
   441→    categoryId,
   442→    page: cur,
   443→    pageSize: 20,
   444→    sort,
   445→    status: status === 'all' ? undefined : status,
   446→  })
   447→
   448→  const catName =
   449→    categoryId != null ? (cats ?? []).find((c) => c.id === categoryId)?.name : undefined
   450→  const goPage = (p: number) => navigate({ name: 'category', categoryId, page: p })
   451→  const applySort = (s: SortKey) => {
   452→    setSort(s)
   453→    setStatus('all')
   454→    if (cur !== 1) goPage(1)
   455→  }
   456→  const applyStatus = (s: StatusKey) => {
   457→    setStatus(s)
   458→    if (cur !== 1) goPage(1)
   459→  }
   460→  const applyPreset = (p: PresetKey) => {
   461→    if (p === 'finished') {
   462→      setSort('latest')
   463→      setStatus('finished')
   464→    } else {
   465→      setSort(p)
   466→      setStatus('all')
   467→    }
   468→    if (cur !== 1) goPage(1)
   469→  }
   470→
   471→  return (
   472→    <div className="mx-auto w-full max-w-[960px] px-2 py-3">
   473→      <Crumbs
   474→        items={[{ label: '首页', view: { name: 'home' } }, { label: catName ?? '全部小说' }]}
   475→        navigate={navigate}
   476→      />
   477→
   478→      <div className="mt-3 grid gap-3 md:grid-cols-[190px_minmax(0,1fr)]">
   479→        {/* 左侧栏 190：推荐榜 + 榜单直达 */}
   480→        <aside className="order-2 min-w-0 space-y-3 md:order-1">
   481→          <Block title="会员推荐榜" bodyClass="px-2 py-1">
   482→            <RankRows novels={(home?.rankings.clicks ?? []).slice(0, 10)} navigate={navigate} limit={10} />
   483→          </Block>
   484→          <Block title="榜单直达" bodyClass="p-2">
   485→            <div className="grid grid-cols-2 gap-1.5">
   486→              {(

   609→const SHELF_KEY = 'trxsw-shelf'
   610→
   611→function loadShelf(): number[] {
   612→  if (typeof window === 'undefined') return []
   613→  try {
   614→    return JSON.parse(window.localStorage.getItem(SHELF_KEY) ?? '[]') as number[]
   615→  } catch {
   616→    return []
   617→  }
   618→}
   619→
   620→function toggleShelf(id: number): boolean {
   621→  if (typeof window === 'undefined') return false
   622→  try {
   623→    const arr = loadShelf()
   624→    const has = arr.includes(id)
   625→    window.localStorage.setItem(SHELF_KEY, JSON.stringify(has ? arr.filter((x) => x !== id) : [...arr, id]))
   626→    return !has
   627→  } catch {
   628→    return false
   629→  }
   630→}

  1044→function ChapterView(props: ViewProps & { chapterId: number }) {
  1045→  /* key=chapterId：换章重挂载，重置书架/投票等本地状态 */

  1049→function ChapterInner({ navigate, chapterId }: ViewProps & { chapterId: number }) {
  1050→  const { data: ch, isLoading, isError, refetch } = useChapter(chapterId)
  1051→  const [prefs] = useReaderPrefs()
  1052→  const [shelf, setShelf] = useState(false)
  1053→  const [voted, setVoted] = useState(false)
  1054→  /* 章节加载后从 storage 同步书架态（书架按书持久化） */
  1055→  const [shelfNovelId, setShelfNovelId] = useState<number | null>(null)
  1056→  if (ch && shelfNovelId !== ch.novelId) {
  1057→    setShelfNovelId(ch.novelId)
  1058→    setShelf(loadShelf().includes(ch.novelId))
  1059→  }
  1060→
  1061→  const paras = useMemo(
  1062→    () =>
  1063→      (ch?.content ?? '')
  1064→        .split(/\n+/)
  1065→        .map((s) => s.trim())
  1066→        .filter(Boolean),
  1067→    [ch?.content],
  1068→  )
  1069→
  1070→  const scene = SCENES[prefs.scene] ?? SCENES.day
  1071→
  1072→  const goPrev = () => {
  1073→    if (ch?.prevId) navigate({ name: 'chapter', chapterId: ch.prevId })
  1074→  }
  1075→  const goNext = () => {
  1076→    if (ch?.nextId) navigate({ name: 'chapter', chapterId: ch.nextId })
  1077→  }
  1078→
  1079→  /* 键盘 ←/→ 翻章（焦点在输入框/下拉/按钮上时不触发；无依赖数组：每次渲染绑定最新闭包） */
  1080→  useEffect(() => {
  1081→    const onKey = (e: KeyboardEvent) => {
  1082→      const t = e.target as HTMLElement | null
  1083→      if (t && (t.tagName === 'INPUT' || t.tagName === 'TEXTAREA' || t.tagName === 'SELECT' || t.tagName === 'BUTTON' || t.isContentEditable)) return
  1084→      if (e.key === 'ArrowLeft') goPrev()
  1085→      else if (e.key === 'ArrowRight') goNext()
  1086→    }
  1087→    window.addEventListener('keydown', onKey)
  1088→    return () => window.removeEventListener('keydown', onKey)
  1089→  })
  1090→
  1091→  if (isLoading) {
  1092→    return (
  1093→      <div className="mx-auto w-full max-w-[960px] px-2 py-3">
  1094→        <div aria-hidden className="animate-pulse border border-[#D0E4F5] bg-[#E6F3FF] px-4 py-8 sm:px-10">
  1095→          <div className="mx-auto h-5 w-1/2 bg-[#D2E6F7]" />
  1096→          <div className="mx-auto mt-3 h-8 w-2/3 bg-white/70" />
  1097→          <div className="mt-5 space-y-3 rounded-sm bg-white px-6 py-6">
  1098→            {Array.from({ length: 12 }).map((_, i) => (
  1099→              <div key={i} className="h-4 bg-[#F0F0F0]" style={{ width: `${92 - ((i * 17) % 40)}%` }} />
  1100→            ))}
  1101→          </div>
  1102→        </div>
  1103→      </div>
  1104→    )
  1105→  }
  1106→  if (isError || !ch) {
  1107→    return (
  1108→      <div className="mx-auto max-w-[960px] px-2 py-6">
  1109→        <ErrorBox onRetry={() => refetch()} />
  1110→      </div>
  1111→    )
  1112→  }
  1113→
  1114→  return (
  1115→    <div className="mx-auto w-full max-w-[960px] px-2 py-3">
  1116→      {/* 淡蓝底阅读区（场景换肤） */}
  1117→      <div className="border px-3 py-4 transition-colors sm:px-6" style={{ background: scene.page, borderColor: scene.line }}>
  1118→        {/* 面包屑 + 操作链接 */}
  1119→        <div className="flex flex-wrap items-center justify-between gap-2 text-xs">
  1120→          <Crumbs
  1121→            className="min-w-0"
  1122→            items={[
  1123→              { label: '首页', view: { name: 'home' } },
  1124→              { label: ch.novelTitle, view: { name: 'book', novelId: ch.novelId } },
  1125→              { label: ch.title },
  1126→            ]}
  1127→            navigate={navigate}
  1128→          />
  1129→          <div className="flex shrink-0 gap-3 text-[#2F468F]">
  1130→            <button
  1131→              type="button"
  1132→              onClick={() => ch && setShelf(toggleShelf(ch.novelId))}
  1133→              className="cursor-pointer hover:text-[#FF6600]"
  1134→            >
  1135→              {shelf ? '已在书架' : '加入书架'}
  1136→            </button>
  1137→            <button
  1138→              type="button"
  1139→              onClick={() => setVoted(true)}
  1140→              className="cursor-pointer hover:text-[#FF6600]"
  1141→            >
  1142→              {voted ? '已推荐' : '推荐本书'}
  1143→            </button>
  1144→          </div>
  1145→        </div>
  1146→
  1147→        {/* 居中章节标题 */}
  1148→        <h1 className="py-3 text-center text-lg font-bold text-[#333]">{ch.title}</h1>
  1149→
  1150→        {/* 章首翻章导航 */}
  1151→        <ChapterNav ch={ch} navigate={navigate} />
  1152→
  1153→        {/* 阅读设置 */}
  1154→        <ReaderBar />
  1155→
  1156→        {/* 正文：居中约 85% 宽，缩进 2em，字号/行距/字体/背景可调 */}
  1157→        <article
  1158→          className="mt-3 border px-4 py-6 transition-colors sm:px-10"
  1159→          style={{ background: scene.paper, borderColor: scene.line }}
  1160→        >
  1161→          <div className="mx-auto w-[92%] max-w-[760px] break-words md:w-[85%]">
  1162→            {paras.length === 0 ? (
  1163→              <p className="py-6 text-center text-sm" style={{ color: scene.muted }}>本章内容为空</p>

  1180→            <p className="mt-6 text-center text-[11px]" style={{ color: scene.muted }}>

  1185→
  1186→        {/* 尾部按钮排 */}
  1187→        <div className="mt-4 flex flex-wrap items-center justify-center gap-2 text-xs">
  1188→          <button
  1189→            type="button"
  1190→            onClick={() => ch && setShelf(toggleShelf(ch.novelId))}
  1191→            className={cn(
  1192→              'cursor-pointer border px-4 py-1.5 transition-colors',
  1193→              shelf
  1194→                ? 'border-[#FF6600] bg-[#FFF3E8] text-[#E05A00]'
  1195→                : 'border-[#CBE0F2] bg-white text-[#2F468F] hover:border-[#FF6600] hover:text-[#FF6600]',
  1196→            )}
  1197→          >
  1198→            {shelf ? '已在书架 ✓' : '加入书架'}
  1199→          </button>
  1200→          <button

  1203→            className={cn(
