    23→async function api<T>(url: string, init?: RequestInit): Promise<T> {
    24→  const res = await fetch(url, { headers: { 'Content-Type': 'application/json' }, ...init })
    25→  const data = await res.json().catch(() => ({}))
    26→  if (!res.ok) throw new Error((data as { error?: string }).error ?? `请求失败(${res.status})`)
    27→  return data as T
    28→}

    87→function ThemesTab() {
    88→  const { data: settings, refetch } = useSettings()
    89→  const qc = useQueryClient()
    90→  const [busy, setBusy] = useState<string | null>(null)
    91→
    92→  const activate = async (id: string) => {
    93→    setBusy(id)
    94→    try {
    95→      await api('/api/settings', { method: 'PATCH', body: JSON.stringify({ activeTheme: id }) })
    96→      await qc.invalidateQueries({ queryKey: qk.settings })
    97→      refetch()

    99→    } catch (e) {
   100→      toast.error(e instanceof Error ? e.message : '切换失败')
   101→    } finally {
   102→      setBusy(null)
   103→    }
   104→  }
   105→
   106→  return (
   107→    <div>
   108→      <p className="mb-4 text-xs text-neutral-500">
   109→        共 {THEME_LIST.length} 套结构级主题模板（首页/分类/书页/目录/章节/搜索全套）。当前激活：
   110→        <Badge className="ml-1">{settings?.activeTheme ?? '-'}</Badge>
   111→      </p>
   112→      <div className="grid grid-cols-1 gap-3 sm:grid-cols-2">
   113→        {THEME_LIST.map((t) => {
   114→          const active = settings?.activeTheme === t.id
   115→          return (
   116→            <div
   117→              key={t.id}
   118→              className={`rounded-lg border p-4 transition ${active ? 'border-neutral-900 ring-1 ring-neutral-900' : 'border-neutral-200 hover:border-neutral-400'}`}
   119→            >
   120→              <div className="flex items-center gap-3">
   121→                <div className="flex h-10 w-10 overflow-hidden rounded-md" aria-hidden>
   122→                  <div className="h-full w-1/2" style={{ background: t.swatch[0] }} />
   123→                  <div className="h-full w-1/2" style={{ background: t.swatch[1] }} />
   124→                </div>
   125→                <div className="min-w-0 flex-1">
   126→                  <p className="truncate text-sm font-semibold">{t.name}</p>
   127→                  <p className="truncate text-xs text-neutral-500">{t.source}</p>
   128→                </div>
   129→                {active ? (
   130→                  <Badge>使用中</Badge>
   131→                ) : (
   132→                  <Button size="sm" disabled={busy === t.id} onClick={() => activate(t.id)}>
   133→                    {busy === t.id ? '切换中…' : '启用'}
   134→                  </Button>
   135→                )}
   136→              </div>
   137→              <p className="mt-2 line-clamp-2 text-xs text-neutral-500">{t.description}</p>
   138→            </div>
   139→          )
   140→        })}
   141→      </div>
   142→    </div>
   143→  )
   144→}

   161→function NovelsTab() {

   168→  const [chapterNovel, setChapterNovel] = useState<NovelListItem | null>(null)
   169→
   170→  const save = async () => {
   171→    if (!form) return
   172→    if (!form.title.trim()) return toast.error('书名不能为空')
   173→    if (!form.categoryId) return toast.error('请选择分类')
   174→    try {
   175→      if (form.id) {
   176→        await api(`/api/novels/${form.id}`, { method: 'PUT', body: JSON.stringify(form) })
   177→        toast.success('已保存')
   178→      } else {
   179→        await api('/api/novels', { method: 'POST', body: JSON.stringify(form) })
   180→        toast.success('已新增')
   181→      }
   182→      setForm(null)
   183→      await qc.invalidateQueries({ queryKey: ['novels'] })
   184→      await qc.invalidateQueries({ queryKey: qk.home })
   185→    } catch (e) {
   186→      toast.error(e instanceof Error ? e.message : '保存失败')
   187→    }
   188→  }
   189→
   190→  const remove = async (n: NovelListItem) => {
   191→    if (!confirm(`确认删除《${n.title}》及其全部章节？此操作不可恢复。`)) return
   192→    try {
   193→      await api(`/api/novels/${n.id}`, { method: 'DELETE' })
   194→      toast.success('已删除')
   195→      await qc.invalidateQueries({ queryKey: ['novels'] })
   196→      await qc.invalidateQueries({ queryKey: qk.home })
   197→    } catch (e) {
   198→      toast.error(e instanceof Error ? e.message : '删除失败')
   199→    }
   200→  }
   201→
   202→  return (
   203→    <div>
   204→      <div className="mb-3 flex flex-wrap items-center gap-2">
   205→        <Input placeholder="搜索书名/作者" value={q} onChange={(e) => { setQ(e.target.value); setPage(1) }} className="h-9 max-w-48" />
   206→        <Button size="sm" onClick={() => setForm({ ...EMPTY_FORM })}><Plus className="mr-1 h-3.5 w-3.5" />新增小说</Button>
   207→      </div>
   208→      <div className="overflow-hidden rounded-lg border">
   209→        <table className="w-full text-sm">
   210→          <thead className="bg-neutral-50 text-xs text-neutral-500">
   211→            <tr>
   212→              <th className="px-3 py-2 text-left font-medium">书名</th>
   213→              <th className="hidden px-3 py-2 text-left font-medium sm:table-cell">分类</th>
   214→              <th className="px-3 py-2 text-left font-medium">章节</th>
   215→              <th className="hidden px-3 py-2 text-left font-medium md:table-cell">字数</th>
   216→              <th className="px-3 py-2 text-right font-medium">操作</th>
   217→            </tr>
   218→          </thead>
   219→          <tbody>
   220→            {data?.list.map((n) => (
   221→              <tr key={n.id} className="border-t">
   222→                <td className="max-w-36 truncate px-3 py-2">
   223→                  {n.title}
   224→                  <span className="ml-1 text-xs text-neutral-400">{n.author}</span>
   225→                  {n.isFeatured && <Badge className="ml-1 px-1 py-0 text-[10px]">荐</Badge>}
   226→                  {n.isHot && <Badge variant="secondary" className="ml-1 px-1 py-0 text-[10px]">热</Badge>}
   227→                </td>
   228→                <td className="hidden px-3 py-2 text-xs text-neutral-500 sm:table-cell">{n.categoryName}</td>
   229→                <td className="px-3 py-2 text-xs">{n.chapterCount}</td>
   230→                <td className="hidden px-3 py-2 text-xs text-neutral-500 md:table-cell">{formatWordCount(n.wordCount)}</td>
   231→                <td className="px-3 py-2 text-right">
   232→                  <Button size="sm" variant="ghost" className="h-7 px-2" onClick={() => setChapterNovel(n)}>
   233→                    <BookOpen className="h-3.5 w-3.5" />
   234→                  </Button>
   235→                  <Button
   236→                    size="sm" variant="ghost" className="h-7 px-2"
   237→                    onClick={() => setForm({ id: n.id, title: n.title, author: n.author, description: n.description, categoryId: n.categoryId, status: n.status, isFeatured: n.isFeatured, isHot: n.isHot })}
   238→                  >
   239→                    <Pencil className="h-3.5 w-3.5" />
   240→                  </Button>
   241→                  <Button size="sm" variant="ghost" className="h-7 px-2 text-red-500 hover:text-red-600" onClick={() => remove(n)}>
   242→                    <Trash2 className="h-3.5 w-3.5" />
   243→                  </Button>
   244→                </td>
   245→              </tr>
   246→            ))}
   247→            {data?.list.length === 0 && (
   248→              <tr><td colSpan={5} className="px-3 py-8 text-center text-sm text-neutral-400">暂无数据</td></tr>
   249→            )}
   250→          </tbody>
   251→        </table>
   252→      </div>
   253→      {data && data.totalPages > 1 && (
   254→        <div className="mt-3 flex items-center justify-center gap-2 text-sm">

   257→          <Button size="sm" variant="outline" disabled={page >= data.totalPages} onClick={() => setPage(page + 1)}>下一页</Button>
   258→        </div>
   259→      )}
   260→
   261→      {form && (
   262→        <div className="fixed inset-0 z-[60] flex items-center justify-center bg-black/40 p-4" onClick={() => setForm(null)}>
   263→          <div className="max-h-[85vh] w-full max-w-lg overflow-y-auto rounded-lg bg-white p-5 shadow-xl" onClick={(e) => e.stopPropagation()}>
   264→            <h3 className="mb-4 text-base font-semibold">{form.id ? '编辑小说' : '新增小说'}</h3>
   265→            <div className="space-y-3">
   266→              <Field label="书名"><Input value={form.title} onChange={(e) => setForm({ ...form, title: e.target.value })} /></Field>
   267→              <Field label="作者"><Input value={form.author} onChange={(e) => setForm({ ...form, author: e.target.value })} /></Field>

   269→                <select
   270→                  className="h-9 w-full rounded-md border border-neutral-200 px-2 text-sm"
   271→                  value={form.categoryId}
   272→                  onChange={(e) => setForm({ ...form, categoryId: Number(e.target.value) })}
   273→                >
   274→                  <option value={0}>请选择…</option>
   275→                  {categories?.map((c) => <option key={c.id} value={c.id}>{c.name}</option>)}
   276→                </select>
   277→              </Field>
   278→              <Field label="状态">
   279→                <select
   280→                  className="h-9 w-full rounded-md border border-neutral-200 px-2 text-sm"
   281→                  value={form.status}
   282→                  onChange={(e) => setForm({ ...form, status: e.target.value as 'serial' | 'finished' })}
   283→                >
   284→                  <option value="serial">连载中</option>
   285→                  <option value="finished">已完本</option>
   286→                </select>
   287→              </Field>
   288→              <Field label="简介"><Textarea rows={4} value={form.description} onChange={(e) => setForm({ ...form, description: e.target.value })} /></Field>
   289→              <div className="flex gap-6">
   290→                <label className="flex items-center gap-2 text-sm"><Switch checked={form.isFeatured} onCheckedChange={(v) => setForm({ ...form, isFeatured: v })} />推荐</label>
   291→                <label className="flex items-center gap-2 text-sm"><Switch checked={form.isHot} onCheckedChange={(v) => setForm({ ...form, isHot: v })} />热门</label>
   292→              </div>
   293→            </div>
   294→            <div className="mt-5 flex justify-end gap-2">
   295→              <Button variant="outline" onClick={() => setForm(null)}>取消</Button>
   296→              <Button onClick={save}>保存</Button>
   297→            </div>
   298→          </div>
   299→        </div>

   301→
   302→      {chapterNovel && <ChaptersDialog novel={chapterNovel} onClose={() => setChapterNovel(null)} />}

   304→  )
   305→}

   318→function ChaptersDialog({ novel, onClose }: { novel: NovelListItem; onClose: () => void }) {
   319→  const qc = useQueryClient()
   320→  const { data: chapters, isLoading } = useChapters(novel.id)
   321→  const [title, setTitle] = useState('')
   322→  const [content, setContent] = useState('')
   323→  const [editing, setEditing] = useState<{ id: number; title: string; content: string } | null>(null)
   324→
   325→  const add = async () => {
   326→    if (!title.trim()) return toast.error('标题不能为空')
   327→    try {
   328→      await api('/api/chapters', { method: 'POST', body: JSON.stringify({ novelId: novel.id, title, content }) })
   329→      toast.success('章节已添加')
   330→      setTitle(''); setContent('')
   331→      await qc.invalidateQueries({ queryKey: qk.chapters(novel.id) })
   332→      await qc.invalidateQueries({ queryKey: ['novels'] })
   333→    } catch (e) {
   334→      toast.error(e instanceof Error ? e.message : '添加失败')
   335→    }
   336→  }
   337→
   338→  const saveEdit = async () => {
   339→    if (!editing) return
   340→    try {
   341→      await api(`/api/chapters/${editing.id}`, { method: 'PUT', body: JSON.stringify({ title: editing.title, content: editing.content }) })
   342→      toast.success('已保存')
   343→      setEditing(null)
   344→      await qc.invalidateQueries({ queryKey: qk.chapters(novel.id) })
   345→    } catch (e) {
   346→      toast.error(e instanceof Error ? e.message : '保存失败')
   347→    }
   348→  }
   349→
   350→  const remove = async (id: number) => {
   351→    if (!confirm('确认删除该章节？')) return
   352→    try {
   353→      await api(`/api/chapters/${id}`, { method: 'DELETE' })
   354→      toast.success('已删除')
   355→      await qc.invalidateQueries({ queryKey: qk.chapters(novel.id) })
   356→      await qc.invalidateQueries({ queryKey: ['novels'] })
   357→    } catch (e) {
   358→      toast.error(e instanceof Error ? e.message : '删除失败')
   359→    }
   360→  }
   361→
   362→  return (
   363→    <div className="fixed inset-0 z-[60] flex items-center justify-center bg-black/40 p-4" onClick={onClose}>
   364→      <div className="flex max-h-[85vh] w-full max-w-2xl flex-col overflow-hidden rounded-lg bg-white shadow-xl" onClick={(e) => e.stopPropagation()}>
   365→        <div className="border-b px-5 py-4">
   366→          <h3 className="text-base font-semibold">章节管理 · {novel.title}</h3>
   367→          <p className="mt-0.5 text-xs text-neutral-500">共 {novel.chapterCount} 章</p>
   368→        </div>
   369→        <div className="min-h-0 flex-1 overflow-y-auto p-5">
   370→          <div className="mb-4 space-y-2 rounded-lg border bg-neutral-50 p-3">
   371→            <p className="text-xs font-medium text-neutral-600">新增章节</p>
   372→            <Input placeholder="章节标题" value={title} onChange={(e) => setTitle(e.target.value)} className="h-8" />
   373→            <Textarea placeholder="正文（可留空，稍后编辑）" rows={3} value={content} onChange={(e) => setContent(e.target.value)} />
   374→            <Button size="sm" onClick={add}><Plus className="mr-1 h-3.5 w-3.5" />添加</Button>
   375→          </div>
   376→          {isLoading && <p className="py-6 text-center text-sm text-neutral-400">加载中…</p>}
   377→          <div className="max-h-72 space-y-1 overflow-y-auto">
   378→            {chapters?.map((c) => (
   379→              <div key={c.id} className="flex items-center gap-2 rounded border px-3 py-1.5 text-sm">
   380→                <span className="w-10 shrink-0 text-xs text-neutral-400">{c.idx}</span>
   381→                <span className="min-w-0 flex-1 truncate">{c.title}</span>
   382→                <span className="shrink-0 text-xs text-neutral-400">{formatWordCount(c.wordCount)}字</span>
   383→                <Button size="sm" variant="ghost" className="h-6 px-1.5" onClick={async () => {
   384→                  const detail = await api<{ title: string; content: string }>(`/api/chapters/${c.id}`)
   385→                  setEditing({ id: c.id, title: detail.title, content: detail.content })
   386→                }}>
   387→                  <Pencil className="h-3 w-3" />
   388→                </Button>
   389→                <Button size="sm" variant="ghost" className="h-6 px-1.5 text-red-500" onClick={() => remove(c.id)}>
   390→                  <Trash2 className="h-3 w-3" />
   391→                </Button>
   392→              </div>
   393→            ))}
   394→          </div>
   395→        </div>
   396→        <div className="flex justify-end border-t px-5 py-3">
   397→          <Button variant="outline" onClick={onClose}>关闭</Button>
   398→        </div>
   399→      </div>
   400→
   401→      {editing && (
   402→        <div className="fixed inset-0 z-[70] flex items-center justify-center bg-black/50 p-4" onClick={() => setEditing(null)}>
   403→          <div className="flex max-h-[85vh] w-full max-w-xl flex-col rounded-lg bg-white p-5 shadow-xl" onClick={(e) => e.stopPropagation()}>
   404→            <h4 className="mb-3 text-sm font-semibold">编辑章节</h4>
   405→            <Input value={editing.title} onChange={(e) => setEditing({ ...editing, title: e.target.value })} className="mb-2" />
   406→            <Textarea rows={12} value={editing.content} onChange={(e) => setEditing({ ...editing, content: e.target.value })} className="flex-1" />
   407→            <div className="mt-3 flex justify-end gap-2">
   408→              <Button variant="outline" onClick={() => setEditing(null)}>取消</Button>
   409→              <Button onClick={saveEdit}>保存</Button>
   410→            </div>
   411→          </div>
   412→        </div>
   413→      )}
   414→    </div>
   415→  )
   416→}

   672→    if (!kw.trim()) return toast.error('请输入种子关键词')

   675→      const res = await api<{ added: number; generated: number; suggestions: { engine: string; ok: boolean; count: number; error?: string }[] }>(

   686→      setRunning(false)

   692→      <section className="rounded-lg border p-4">
