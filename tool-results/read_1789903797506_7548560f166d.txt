   212→
   213→export function Home({ navigate }: ViewProps) {
   214→  const home = useHomeData()
   215→  const updates = useNovels({ sort: 'latest', page: 1, pageSize: 30 })
   216→
   217→  if (home.isPending || updates.isPending) {
   218→    return (
   219→      <Container className="mt-2">
   220→        <div className="grid gap-4 md:grid-cols-[minmax(0,73%)_minmax(0,1fr)]">
   221→          <div>
   222→            <GSkel className="mb-2 h-6 w-40" />
   223→            <GCoverSkeleton count={4} />
   224→          </div>
   225→          <div>
   226→            <GSkel className="mb-2 h-10 w-full" />
   227→            <GSkel className="mb-2 h-6 w-32" />
   228→            <GRowSkeleton count={10} />
   229→          </div>
   230→        </div>
   231→      </Container>
   232→    )
   233→  }
   234→
   235→  if (home.isError || !home.data) {
   236→    return (
   237→      <Container className="mt-4">
   238→        <GErrorBox onRetry={() => home.refetch()} />
   239→      </Container>
   240→    )
   241→  }
   242→  if (updates.isError) {
   243→    return (
   244→      <Container className="mt-4">
   245→        <GErrorBox onRetry={() => updates.refetch()} />
   246→      </Container>
   247→    )
   248→  }
   249→
   250→  const d = home.data
   251→  const hotItems = d.hot.slice(0, 6)
   252→  const clickRank = d.rankings.clicks.slice(0, 12)
   253→  const updateRows = (updates.data?.list ?? []).slice(0, 30)
   254→  const sideNews = dedupMerge(
   255→    [d.featured, d.hot, d.rankings.clicks, d.rankings.updates, d.rankings.finished],
   256→    30,
   257→  )
   258→
   259→  return (
   260→    <Container className="mt-1">
   261→      <div className="grid gap-5 md:grid-cols-[minmax(0,73%)_minmax(0,1fr)] md:gap-[2%]">
   262→        {/* 第一行左：热门小说推荐（2 列封面简介卡） */}
   263→        <section>
   264→          <GH2>热门小说推荐</GH2>
   265→          <div className="grid gap-x-5 gap-y-4 sm:grid-cols-2">
   266→            {hotItems.map((n) => (
   267→              <HotItem key={n.id} novel={n} navigate={navigate} />
   268→            ))}
   269→          </div>
   270→        </section>
   271→
   272→        {/* 第一行右：搜索框 + 阅读排行榜 */}
   273→        <aside>
   274→          <SearchBox navigate={navigate} />
   275→          <GH2 className="mt-4">阅读排行榜</GH2>
   276→          <div>
   277→            {clickRank.map((n) => (
   278→              <RankRow key={n.id} novel={n} navigate={navigate} />
   279→            ))}
   280→          </div>
   281→        </aside>
   282→
   283→        {/* 第二行左：最近更新（五列字段表） */}
   284→        <section>
   285→          <GH2>最近更新</GH2>
   286→          <div>
   287→            {updateRows.map((n) => (
   288→              <UpdateRow key={n.id} novel={n} navigate={navigate} />
   289→            ))}
   290→          </div>
   291→        </section>
   292→
   293→        {/* 第二行右：最新小说（两字段行榜单） */}
   294→        <aside>
   295→          <GH2>最新小说</GH2>
   296→          <div>
   297→            {sideNews.map((n) => (
   298→              <RankRow key={n.id} novel={n} navigate={navigate} />
   299→            ))}
   300→          </div>
   301→        </aside>
   302→      </div>
   303→    </Container>
   304→  )
   305→}

   307→/* ==================== 分类页（分类导航条 + 3 列虚线盒 + 数字分页） ==================== */

   309→export function Category({
   310→  navigate,
   311→  categoryId,
   312→  page = 1,
   313→}: ViewProps & { categoryId?: number; page?: number }) {
   314→  const cats = useCategories()
   315→  const list = useNovels({ categoryId, page, pageSize: 20 })
   316→  const catName = cats.data?.find((c) => c.id === categoryId)?.name
   317→
   318→  const navCls = (active: boolean) =>
   319→    active
   320→      ? 'grid place-items-center h-[38px] rounded-[3px] bg-[#56ccb5] text-[15px] text-white transition-colors duration-200 cursor-pointer'
   321→      : 'grid place-items-center h-[38px] cursor-pointer rounded-[3px] text-[15px] text-[#00886d] transition-colors duration-200 hover:bg-[#cdf3eb] hover:text-[#f50]'
   322→
   323→  return (
   324→    <Container className="mt-1">
   325→      {/* 分类导航条 */}
   326→      <nav className="mb-3 grid grid-cols-3 gap-1 rounded-[4px] border border-[#ccc] bg-white p-1 sm:grid-cols-4 md:grid-cols-8">
   327→        <button
   328→          onClick={() => navigate({ name: 'category', page: 1 })}
   329→          className={navCls(categoryId == null)}
   330→        >
   331→          全部
   332→        </button>
   333→        {(cats.data ?? []).map((c) => (
   334→          <button
   335→            key={c.id}
   336→            onClick={() => navigate({ name: 'category', categoryId: c.id, page: 1 })}
   337→            className={navCls(categoryId === c.id)}
   338→          >
   339→            {c.name}
   340→          </button>
   341→        ))}
   342→      </nav>
   343→
   344→      {/* 书籍列表 */}
   345→      <section className="rounded-[4px] border border-[#ccc] bg-white p-3">
   346→        <GH2 className="text-center">
   347→          {catName ?? '全部小说'}
   348→          {list.data ? <span className="ml-2 text-[13px] font-normal text-[#999]">共 {list.data.total} 本</span> : null}
   349→        </GH2>
   350→
   351→        {list.isPending ? (
   352→          <GBoxSkeleton count={6} />
   353→        ) : list.isError ? (
   354→          <GErrorBox onRetry={() => list.refetch()} />
   355→        ) : (list.data?.list.length ?? 0) === 0 ? (
   356→          <GEmptyBox text="该分类下暂无收录小说" />
   357→        ) : (
   358→          <>
   359→            <div className="grid gap-4 pt-2 sm:grid-cols-2 lg:grid-cols-3">
   360→              {list.data!.list.map((n, i) => (
   361→                <BookBoxItem
   362→                  key={n.id}
   363→                  novel={n}
   364→                  index={(list.data!.page - 1) * (list.data!.pageSize ?? 20) + i + 1}
   365→                  navigate={navigate}
   366→                />
   367→              ))}
   368→            </div>
   369→            <GPager
   370→              page={list.data!.page}
   371→              totalPages={list.data!.totalPages}
   372→              onGo={(p) => navigate({ name: 'category', categoryId, page: p })}
   373→            />
   374→          </>
   375→        )}
   376→      </section>
   377→    </Container>
   378→  )
   379→}

   883→function loadMarks(novelId: number): number[] {
   884→  if (typeof window === 'undefined') return []
   885→  try {
   886→    return JSON.parse(window.localStorage.getItem(`ggd66-marks-${novelId}`) ?? '[]') as number[]
   887→  } catch {
   888→    return []
   889→  }

   892→function toggleMark(novelId: number, chapterId: number): boolean {
   893→  if (typeof window === 'undefined') return false
   894→  try {
   895→    const arr = loadMarks(novelId)
   896→    const has = arr.includes(chapterId)
   897→    window.localStorage.setItem(
   898→      `ggd66-marks-${novelId}`,
   899→      JSON.stringify(has ? arr.filter((x) => x !== chapterId) : [...arr.slice(-199), chapterId]),
   900→    )
   901→    return !has
   902→  } catch {
   903→    return false
   904→  }
   905→}
