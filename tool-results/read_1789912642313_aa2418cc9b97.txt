    57→const SHELF_KEY = '23qb.shelf'
    58→
    59→function loadShelf(): number[] {
    60→  if (typeof window === 'undefined') return []
    61→  try {
    62→    return JSON.parse(window.localStorage.getItem(SHELF_KEY) ?? '[]') as number[]
    63→  } catch {
    64→    return []
    65→  }
    66→}
    67→
    68→function toggleShelf(id: number): boolean {
    69→  if (typeof window === 'undefined') return false
    70→  const arr = loadShelf()
    71→  const has = arr.includes(id)
    72→  const next = has ? arr.filter((x) => x !== id) : [...arr, id]
    73→  window.localStorage.setItem(SHELF_KEY, JSON.stringify(next))
    74→  return !has
    75→}

   167→export function CategoryView({
   168→  navigate,
   169→  categoryId,
   170→  page = 1,
   171→}: ViewProps & { categoryId?: number; page?: number }) {
   172→  const { data: categories } = useCategories()
   173→  const [sort, setSort] = useState<SortKey>('latest')
   174→  const [status, setStatus] = useState<'serial' | 'finished' | undefined>(undefined)
   175→  const { data, isLoading, isError, refetch } = useNovels({
   176→    categoryId,
   177→    page,
   178→    pageSize: 20,
   179→    sort,
   180→    status,
   181→  })
   182→
   183→  const catName = categories?.find((c) => c.id === categoryId)?.name ?? '全部'
   184→  const applyFilter = (fn: () => void) => {
   185→    fn()
   186→    if (page !== 1) navigate({ name: 'category', categoryId, page: 1 })
   187→  }
   188→
   189→  return (
   190→    <Container className="space-y-6 py-6">
   191→      {/* 筛选区 */}
   192→      <Card className="space-y-3 p-5">
   193→        <div className="flex flex-wrap items-center gap-2">
   194→          <span className="mr-1 text-sm font-bold text-black/45">分类</span>

   197→          </Pill>

   199→            <Pill
   200→              key={c.id}
   201→              active={categoryId === c.id}
   202→              onClick={() => applyFilter(() => navigate({ name: 'category', categoryId: c.id, page: 1 }))}

   205→            </Pill>
   206→          ))}

   210→          {SORT_OPTIONS.map((o) => (

   213→            </Pill>
   214→          ))}
   215→        </div>
   216→        <div className="flex flex-wrap items-center gap-2">
   217→          <span className="mr-1 text-sm font-bold text-black/45">进度</span>

   222→            连载
   223→          </Pill>

   225→            完本
   226→          </Pill>

   237→      <Card className="p-5 md:p-7">

   239→          <ErrorRetry onRetry={() => refetch()} />

   241→          <GridSkeleton />

   243→          <div className="grid grid-cols-2 gap-x-5 gap-y-8 sm:grid-cols-4 md:grid-cols-5 lg:grid-cols-6 xl:grid-cols-7 2xl:grid-cols-8">

   259→          <Pager page={data.page} totalPages={data.totalPages} onGo={(p) => navigate({ name: 'category', categoryId, page: p })} />

   275→  const [rec, setRec] = useState(false)

   439→export function TocView({ navigate, novelId }: ViewProps & { novelId: number }) {
   440→  const { data: novel, isError, refetch } = useNovel(novelId)
   441→  const { data: chapters, isLoading, isError: chError, refetch: refetchCh } = useChapters(novelId)
   442→  const [bookmark, setBookmark] = useState<{ chapterId: number; title: string } | null>(
   443→    () => loadBookmarks()[String(novelId)] ?? null
   444→  )
   445→  const [bmNovelId, setBmNovelId] = useState(novelId)
   446→
   447→  if (bmNovelId !== novelId) {
   448→    setBmNovelId(novelId)
   449→    setBookmark(loadBookmarks()[String(novelId)] ?? null)
   450→  }
   451→
   452→  const err = isError || chError
   453→  if (err) {
   454→    return (
   455→      <Container className="pt-[70px]">
   456→        <ErrorRetry onRetry={() => { refetch(); refetchCh() }} />
   457→      </Container>
   458→    )
   459→  }
   460→
   461→  return (
   462→    <Container className="space-y-6 py-6">
   463→      <Card className="p-6 md:p-8">
   464→        <button
