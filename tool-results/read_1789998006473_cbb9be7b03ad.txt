    43→func handleChapterCreate(w http.ResponseWriter, r *http.Request, _ map[string]string) {
    44→        v, ok := readBodyValue(r)
    45→        if !ok {
    46→                writeJSON(w, 400, map[string]string{"error": "请求体不是合法 JSON"})
    47→                return
    48→        }
    49→        body := bodyMap(v)
    50→
    51→        novelId, okN := intFieldStrict(body["novelId"])
    52→        if !okN || novelId == 0 {
    53→                writeJSON(w, 400, map[string]string{"error": "novelId 必填"})
    54→                return
    55→        }
    56→        rawTitle := strField(body["title"], 0)
    57→        if trimSpaceStr(rawTitle) == "" {
    58→                writeJSON(w, 400, map[string]string{"error": "章节标题不能为空"})
    59→                return
    60→        }
    61→        var nid int64
    62→        if err := queryOne(`SELECT "id" FROM "Novel" WHERE "id" = ?`, []any{&nid}, novelId); err != nil {
    63→                if isNoRows(err) {
    64→                        writeJSON(w, 404, map[string]string{"error": "小说不存在"})
    65→                        return
    66→                }
    67→                failJSON(w, "服务器错误", firstLineErr(err), 500)
    68→                return
    69→        }
    70→
    71→        chTitle := truncateRunes(trimSpaceStr(rawTitle), 120) // TS 内联 slice(0,120)
    72→        // 非字符串 content 按空正文处理（与 PUT 语义一致）
    73→        content := strField(body["content"], 0)
    74→        wc := wordCountJS(content)
    75→
    76→        createChapter := func(idxVal int64) (int64, error) {
    77→                return execReturningID(
    78→                        `INSERT INTO "Chapter" ("novelId","idx","title","content","wordCount","createdAt") VALUES (?,?,?,?,?,?)`,
    79→                        novelId, idxVal, chTitle, content, wc, nowMillis(),
    80→                )
    81→        }
    82→
    83→        var maxIdx sql.NullInt64
    84→        if err := queryOne(`SELECT MAX("idx") FROM "Chapter" WHERE "novelId" = ?`, []any{&maxIdx}, novelId); err != nil && !isNoRows(err) {
    85→                failJSON(w, "服务器错误", firstLineErr(err), 500)
    86→                return
    87→        }
    88→        idx := maxIdx.Int64 + 1
    89→        chapterID, err := createChapter(idx)
    90→        if err != nil {
    91→                // 并发竞态兜底：撞 [novelId, idx] 唯一约束后重读最大 idx 重试一次
    92→                if !isUniqueConflict(err) {
    93→                        failJSON(w, "服务器错误", firstLineErr(err), 500)
    94→                        return
    95→                }
    96→                var retryIdx sql.NullInt64
    97→                if err2 := queryOne(`SELECT MAX("idx") FROM "Chapter" WHERE "novelId" = ?`, []any{&retryIdx}, novelId); err2 != nil && !isNoRows(err2) {
    98→                        failJSON(w, "服务器错误", firstLineErr(err2), 500)
    99→                        return
   100→                }
   101→                idx = retryIdx.Int64 + 1
   102→                chapterID, err = createChapter(idx)
   103→                if err != nil {
   104→                        failJSON(w, "创建章节失败", firstLineErr(err), 500)
   105→                        return
   106→                }
   107→        }
   108→
   109→        // 同步小说字数与更新时间（显式 updatedAt，对齐 Prisma @updatedAt）
   110→        var sumWC sql.NullInt64
   111→        if err := queryOne(`SELECT SUM("wordCount") FROM "Chapter" WHERE "novelId" = ?`, []any{&sumWC}, novelId); err != nil && !isNoRows(err) {
   112→                failJSON(w, "服务器错误", firstLineErr(err), 500)
   113→                return
   114→        }
   115→        if _, err := exec(`UPDATE "Novel" SET "wordCount" = ?, "updatedAt" = ? WHERE "id" = ?`, sumWC.Int64, nowMillis(), novelId); err != nil {
   116→                failJSON(w, "服务器错误", firstLineErr(err), 500)
   117→                return
   118→        }
   119→        writeJSON(w, 201, map[string]any{"id": chapterID, "idx": idx})
   120→}

   124→func handleChapterDetail(w http.ResponseWriter, r *http.Request, ps map[string]string) {
   125→        cid, ok := routeIntID(w, ps["id"], "无效 ID")
   126→        if !ok {
   127→                return
   128→        }
   129→        var id, novelId, idx, wordCount int64
   130→        var title, content string
   131→        var novelTitle sql.NullString
   132→        err := queryOne(
   133→                `SELECT ch."id", ch."novelId", ch."idx", ch."title", ch."content", ch."wordCount", n."title" FROM "Chapter" ch LEFT JOIN "Novel" n ON n."id" = ch."novelId" WHERE ch."id" = ?`,
   134→                []any{&id, &novelId, &idx, &title, &content, &wordCount, &novelTitle},
   135→                cid,
   136→        )
   137→        if err != nil {
   138→                if isNoRows(err) {
   139→                        writeJSON(w, 404, map[string]string{"error": "章节不存在"})
   140→                        return
   141→                }
   142→                failJSON(w, "服务器错误", firstLineErr(err), 500)
   143→                return
   144→        }
   145→        var prevID, nextID sql.NullInt64
   146→        _ = queryOne(`SELECT "id" FROM "Chapter" WHERE "novelId" = ? AND "idx" < ? ORDER BY "idx" DESC LIMIT 1`, []any{&prevID}, novelId, idx)
   147→        _ = queryOne(`SELECT "id" FROM "Chapter" WHERE "novelId" = ? AND "idx" > ? ORDER BY "idx" ASC LIMIT 1`, []any{&nextID}, novelId, idx)
   148→
   149→        var prev, next any
   150→        if prevID.Valid {
   151→                prev = prevID.Int64
   152→        }
   153→        if nextID.Valid {
   154→                next = nextID.Int64
   155→        }
   156→        nt := ""
   157→        if novelTitle.Valid {
   158→                nt = novelTitle.String
   159→        }
   160→        writeJSON(w, 200, map[string]any{
   161→                "id":         id,
   162→                "novelId":    novelId,
   163→                "novelTitle": nt,
   164→                "idx":        idx,
   165→                "title":      title,
   166→                "content":    content,
   167→                "wordCount":  wordCount,
   168→                "prevId":     prev,
   169→                "nextId":     next,
   170→        })
   171→}

   175→func handleChapterUpdate(w http.ResponseWriter, r *http.Request, ps map[string]string) {
   176→        cid, ok := routePosIntID(w, ps["id"], "无效 ID")
   177→        if !ok {
   178→                return
   179→        }
   180→        v, ok := readBodyValue(r)
   181→        if !ok {
   182→                writeJSON(w, 400, map[string]string{"error": "请求体不是合法 JSON"})
   183→                return
   184→        }
   185→        body := bodyMap(v)
   186→
   187→        var novelId int64
   188→        if err := queryOne(`SELECT "novelId" FROM "Chapter" WHERE "id" = ?`, []any{&novelId}, cid); err != nil {
   189→                // TS update 抛错（含不存在）一律 404
   190→                writeJSON(w, 404, map[string]string{"error": "章节不存在或更新失败"})
   191→                return
   192→        }
   193→
   194→        sets := []string{}
   195→        args := []any{}
   196→        if s, ok2 := body["title"].(string); ok2 {
   197→                if t := trimSpaceStr(s); t != "" {
   198→                        sets = append(sets, `"title" = ?`)
   199→                        args = append(args, truncateRunes(t, 120))
   200→                }
   201→        }
   202→        if s, ok2 := body["content"].(string); ok2 {
   203→                sets = append(sets, `"content" = ?`, `"wordCount" = ?`)
   204→                args = append(args, s, wordCountJS(s))
   205→        }
   206→        if len(sets) > 0 {
   207→                if _, err := exec(`UPDATE "Chapter" SET `+strings.Join(sets, ", ")+` WHERE "id" = ?`, append(args, cid)...); err != nil {
   208→                        writeJSON(w, 404, map[string]string{"error": "章节不存在或更新失败"})
   209→                        return
   210→                }
   211→        }
   212→        // 内容变化同步书籍字数合计并触碰 updatedAt（对齐 TS 语义）
   213→        if err := resumNovelWordCount(novelId, true); err != nil {
   214→                failJSON(w, "服务器错误", firstLineErr(err), 500)
   215→                return
   216→        }
   217→        writeJSON(w, 200, map[string]any{"ok": true})
   218→}

   222→func handleChapterDelete(w http.ResponseWriter, r *http.Request, ps map[string]string) {
   223→        cid, ok := routePosIntID(w, ps["id"], "无效 ID")
   224→        if !ok {
   225→                return
   226→        }
   227→        var novelId int64
   228→        if err := queryOne(`SELECT "novelId" FROM "Chapter" WHERE "id" = ?`, []any{&novelId}, cid); err != nil {
   229→                writeJSON(w, 404, map[string]string{"error": "章节不存在"})
   230→                return
   231→        }
   232→        res, err := exec(`DELETE FROM "Chapter" WHERE "id" = ?`, cid)
   233→        if err != nil {
   234→                writeJSON(w, 404, map[string]string{"error": "章节不存在"})
   235→                return
   236→        }
   237→        if n, _ := res.RowsAffected(); n == 0 {
   238→                writeJSON(w, 404, map[string]string{"error": "章节不存在"})
   239→                return
   240→        }
   241→        // 与 PUT/POST 对齐：内容变化同时触碰 updatedAt，让「最近更新」排序如实反映删章
   242→        if err := resumNovelWordCount(novelId, true); err != nil {
   243→                failJSON(w, "服务器错误", firstLineErr(err), 500)
   244→                return
   245→        }
   246→        writeJSON(w, 200, map[string]any{"ok": true})
   247→}

   249→// resumNovelWordCount 重算书籍字数合计（touch=true 时同步触碰 updatedAt）
   250→func resumNovelWordCount(novelId int64, touch bool) error {
   251→        var sumWC sql.NullInt64
   252→        if err := queryOne(`SELECT SUM("wordCount") FROM "Chapter" WHERE "novelId" = ?`, []any{&sumWC}, novelId); err != nil {
   253→                return err
   254→        }
   255→        if touch {
   256→                _, err := exec(`UPDATE "Novel" SET "wordCount" = ?, "updatedAt" = ? WHERE "id" = ?`, sumWC.Int64, nowMillis(), novelId)
   257→                return err
   258→        }
   259→        _, err := exec(`UPDATE "Novel" SET "wordCount" = ? WHERE "id" = ?`, sumWC.Int64, novelId)
   260→        return err
   261→}

   263→// ==================== 正文噪声清洗（移植 src/lib/content-clean.ts） ====================
   264→
   265→const (
   266→        noiseShortLineMax = 30
   267→        noiseTailLineMax  = 80
   268→)
   269→
   270→var (
   271→        noiseLeadingIndentRe = regexp.MustCompile(`^[ \t\x{3000}\x{00a0}\x{feff}\x{200b}\x{200c}\x{200d}]+`)
   272→        noiseHasTextRe       = regexp.MustCompile(`[\p{L}\p{N}]`)
   273→        // RE2 无 (?![a-z])：改写为消费式 (?:[^a-z]|$)，配合 (?i) 与 TS 布尔判定等价
   274→        noiseURLLineRe    = regexp.MustCompile(`(?i)www\.|https?://|\.(?:com|net|cc|org|info|xyz|top|vip)(?:[^a-z]|$)|^[a-z0-9-]+(?:\.[a-z0-9-]+)+$`)
   275→        noiseSitePromoRe  = regexp.MustCompile(`笔趣阁|顶点小说|飞卢|起点中文|纵横中文|天才一?秒?记住|一秒记住|本章未完|点击下一页|继续阅读请|最新章节|手机阅读|无弹窗|全本小说|请记住本书|首发域名|记得收藏|请收藏本站|收藏网址|求收藏|求推荐票|求月票|投推荐票|加入书签|书迷交流|站内搜索|快速找到你想要的|TXT电子书|电子书下载|全本TXT|TXT全集|TXT下载|最快更新|第一时间更新|本站网址|备用域名|备用网址|看书神器|阅读神器|免费阅读网|小说网址|提供无错|精校版|无错版`)
   276→        noiseNavExactRe   = regexp.MustCompile(`^(?:上一章|上一页|下一章|下一页|上一頁|下一頁|目录|章节目录|章节列表|返回|返回目录|返回书页|返回列表|返回首页|返回书架|回到书架|返回顶部|回顶部|去底部|首页|书页|书签|加入书签|加入收藏|加入书架|放到书架|收藏本站|收藏本书|收藏|推荐票|点击进入|点击收藏|第一页|末页|搜索|搜索全站|正文|封面|书评|打卡|签到|赞|踩|分享)$`)
   277→        noisePageNumberRe = regexp.MustCompile(`^[0-9]{1,4}$`)
   278→        noiseTailHintRe   = regexp.MustCompile(`本章未完|未完待续|本章完|请点击下一[页章頁]继续阅读|转载请注明(?:来源|出处)|章节错误.{0,6}点此举报|手机用户请(?:浏览|阅读|访问)|关注公众号|微信公众号|(?:天才|一秒)记住本站最新网址|章节内容(?:错误|缺失)|看不到(?:结尾|结局)`)
   279→        noiseJSResidueRe  = regexp.MustCompile(`(?i)javascript:|function\s*\(|document\.|window\.|\{.*\}`)
   280→        crlfRe            = regexp.MustCompile(`\r\n?`)
   281→)

   283→// isNoiseLine 判断（已规范化）行是否噪声；空串 false（空行由调用方丢弃）
   284→func isNoiseLine(line string) bool {
   285→        t := jsTrim(line)
   286→        if t == "" {
   287→                return false
   288→        }
   289→        if !noiseHasTextRe.MatchString(t) {
   290→                return true
   291→        }
   292→        n := len([]rune(t))
   293→        if n > noiseShortLineMax {
   294→                if n > noiseTailLineMax {
   295→                        return false
   296→                }
   297→                return noiseTailHintRe.MatchString(t)
   298→        }
   299→        if noiseURLLineRe.MatchString(t) {
   300→                return true
   301→        }
   302→        if noiseSitePromoRe.MatchString(t) {
   303→                return true
   304→        }
   305→        if noiseNavExactRe.MatchString(t) {
   306→                return true
   307→        }
   308→        if noisePageNumberRe.MatchString(t) {
   309→                return true
   310→        }
   311→        if noiseTailHintRe.MatchString(t) {
   312→                return true
   313→        }
   314→        if noiseJSResidueRe.MatchString(t) {
   315→                return true
   316→        }
   317→        return false
   318→}

   320→type cleanResult struct {
   321→        text         string
   322→        removedLines int
   323→}
   324→
   325→// cleanChapterContent 移植 src/lib/content-clean.ts cleanChapterContent：
   326→// 归一化换行/去行首缩进/行内空白折叠（JS \s 语义）/去空行/滤噪声行，行间单个 \n
   327→func cleanChapterContent(raw string) cleanResult {
   328→        if raw == "" {
   329→                return cleanResult{text: "", removedLines: 0}
   330→        }
   331→        normalized := crlfRe.ReplaceAllString(raw, "\n")
   332→        kept := make([]string, 0)
   333→        removed := 0
   334→        for _, rawLine := range strings.Split(normalized, "\n") {
   335→                line := jsTrim(jsSpacePlusRe.ReplaceAllString(noiseLeadingIndentRe.ReplaceAllString(rawLine, ""), " "))
   336→                if line == "" {
   337→                        continue
   338→                }
   339→                if isNoiseLine(line) {
   340→                        removed++
   341→                        continue
   342→                }
   343→                kept = append(kept, line)
   344→        }
   345→        return cleanResult{text: strings.Join(kept, "\n"), removedLines: removed}
   346→}

   348→// ==================== /api/chapters/clean-all ====================
   349→
   350→const cleanAllBatchSize = 500
   351→
   352→var (
   353→        cleanAllMu      sync.Mutex
   354→        cleanAllRunning bool
   355→)
   356→
   357→// scanChapters 遍历全部章节做清洗比对；write=true 落库（章节 content/wordCount + 书籍字数合计）
   358→func scanChapters(write bool) (checked, changed, novels int, err error) {
   359→        cursor := int64(0)
   360→        touchedNovels := map[int64]bool{}
   361→        for {
   362→                type row struct {
   363→                        id      int64
   364→                        novelId int64
   365→                        content string
   366→                }
   367→                batch := make([]row, 0, cleanAllBatchSize)
   368→                err := queryList(`SELECT "id", "novelId", "content" FROM "Chapter" WHERE "id" > ? ORDER BY "id" ASC LIMIT ?`,
   369→                        func(rows *sql.Rows) error {
   370→                                var r row
   371→                                if err := rows.Scan(&r.id, &r.novelId, &r.content); err != nil {
   372→                                        return err
   373→                                }
   374→                                batch = append(batch, r)
   375→                                return nil
   376→                        }, cursor, cleanAllBatchSize)
   377→                if err != nil {
   378→                        return checked, changed, len(touchedNovels), err
   379→                }
   380→                if len(batch) == 0 {
   381→                        break
   382→                }
   383→                for _, ch := range batch {
   384→                        cursor = ch.id
   385→                        checked++
   386→                        res := cleanChapterContent(ch.content)
   387→                        if res.text == ch.content {
   388→                                continue // 无变化不写库
   389→                        }
   390→                        if write {
   391→                                // 更新失败（瞬时锁等）不计入 cleaned，避免虚报；下轮 dryRun 可复查
   392→                                if _, err := exec(`UPDATE "Chapter" SET "content" = ?, "wordCount" = ? WHERE "id" = ?`, res.text, wordCountJS(res.text), ch.id); err != nil {
   393→                                        continue
   394→                                }
   395→                        }
   396→                        changed++
   397→                        touchedNovels[ch.novelId] = true
   398→                }
   399→        }
   400→        if write && len(touchedNovels) > 0 {
   401→                // 章节 wordCount 变了，书籍级字数合计一并重算（update 触碰 @updatedAt → 显式 set）
   402→                for novelId := range touchedNovels {
   403→                        var sumWC sql.NullInt64
   404→                        if err := queryOne(`SELECT SUM("wordCount") FROM "Chapter" WHERE "novelId" = ?`, []any{&sumWC}, novelId); err != nil {
   405→                                continue
   406→                        }
   407→                        _, _ = exec(`UPDATE "Novel" SET "wordCount" = ?, "updatedAt" = ? WHERE "id" = ?`, sumWC.Int64, nowMillis(), novelId)
   408→                }
   409→        }
   410→        return checked, changed, len(touchedNovels), nil
   411→}

   413→func handleChapterCleanAllGet(w http.ResponseWriter, r *http.Request, _ map[string]string) {
   414→        dryRun := r.URL.Query().Get("dryRun") == "1"
   415→        // GET 恒为只读预览：write=false，绝不写库
   416→        checked, changed, _, err := scanChapters(false)
   417→        if err != nil {
   418→                writeJSON(w, 500, map[string]string{"error": "存量章节预览失败", "detail": firstLineErr(err)})
   419→                return
   420→        }
   421→        writeJSON(w, 200, map[string]any{"checked": checked, "toClean": changed, "dryRun": dryRun})
   422→}

   431→        cleanAllRunning = true
   432→        cleanAllMu.Unlock()
   433→        defer func() {
   434→                cleanAllMu.Lock()
   435→                cleanAllRunning = false
   436→                cleanAllMu.Unlock()
   437→        }()
   438→
   439→        checked, changed, novels, err := scanChapters(true)
   440→        if err != nil {

   443→        }

   447→// ==================== /api/chapters/audit：中文数字解析 ====================
   448→
   449→var cnDigit = map[rune]int{
   450→        '零': 0, '〇': 0, '一': 1, '壹': 1, '二': 2, '贰': 2, '两': 2, '三': 3, '叁': 3, '四': 4, '肆': 4,
   451→        '五': 5, '伍': 5, '六': 6, '陆': 6, '七': 7, '柒': 7, '八': 8, '捌': 8, '九': 9, '玖': 9,
   452→}
   453→var cnUnit = map[rune]int{'十': 10, '拾': 10, '百': 100, '佰': 100, '千': 1000, '仟': 1000}
   454→
   455→// parseChapterNumber 中文/全角数字 → 阿拉伯数字；不可解析返回 false
   456→func parseChapterNumber(raw string) (int64, bool) {
   457→        var b strings.Builder
   458→        for _, c := range raw {
   459→                if c >= 0xFF10 && c <= 0xFF19 { // 全角０-９ → ASCII
   460→                        c = c - 0xFEE0
   461→                }
   462→                b.WriteRune(c)
   463→        }
   464→        s := jsTrim(b.String())
   465→        if s == "" {
   466→                return 0, false
   467→        }
   468→        if allDigitsRe.MatchString(s) {
   469→                return parseDigitsASCII(s), true
   470→        }
   471→        var result, section int64
   472→        cur := int64(-1)
   473→        for _, ch := range s {
   474→                switch {
   475→                case ch == '万' || ch == '萬':
   476→                        section = (section + maxInt64(cur, 0)) * 10000
   477→                        result += section
   478→                        section = 0
   479→                        cur = -1
   480→                default:
   481→                        if d, ok := cnDigit[ch]; ok {
   482→                                cur = int64(d)
   483→                        } else if u, ok := cnUnit[ch]; ok {
   484→                                base := cur
   485→                                if base < 0 {
   486→                                        base = 1
   487→                                }
   488→                                section += base * int64(u)
   489→                                cur = -1
   490→                        } else {
   491→                                return 0, false
   492→                        }
   493→                }
   494→        }
   495→        return result + section + maxInt64(cur, 0), true
   496→}
   497→
   498→var allDigitsRe = regexp.MustCompile(`^\d+$`)
   499→
   500→// parseDigitsASCII 纯数字串转 int64（标题编号场景，不适用 strconv 以外的错误分支）
   501→func parseDigitsASCII(s string) int64 {
   502→        var n int64
   503→        for _, c := range s {
   504→                n = n*10 + int64(c-'0')
   505→                if n > 1<<53 {
   506→                        break // JS Number 超出安全整数后精度丢失，截断即可（章节编号不会这么大）
   507→                }
   508→        }
   509→        return n
   510→}
   511→
   512→func maxInt64(a, b int64) int64 {
   513→        if a > b {
   514→                return a
   515→        }
   516→        return b
   517→}
   518→
   519→const auditCNClass = `[0-9\x{ff10}-\x{ff19}]+|[零〇一二三四五六七八九十百千两壹贰叁肆伍陆柒捌玖拾佰仟萬万]+`
   520→
   521→var (
   522→        // 第N章/节/開頭（严格版：章回体「回」量词排除；\s 用 JS 空白全集）
   523→        chNumRe    = regexp.MustCompile(`^第` + jsSpaceClass + `*(` + auditCNClass + `)` + jsSpaceClass + `*(?:章|节|節)`)
   524→        volumeRe   = regexp.MustCompile(`第` + jsSpaceClass + `*(?:` + auditCNClass + `)` + jsSpaceClass + `*(?:卷|部|篇)`)
   525→)
   526→
   527→// extractNum 提取章节编号（严格版）：标题必须以「第N章/节」开头；编号后剩余超 30 字不参与判定
   528→func extractNum(title string) (int64, bool) {
   529→        m := chNumRe.FindStringSubmatch(title)
   530→        if m == nil {
   531→                return 0, false
   532→        }
   533→        restRunes := []rune(title)
   534→        prefixRunes := len([]rune(m[0]))
   535→        if prefixRunes < len(restRunes) {
   536→                restRunes = restRunes[prefixRunes:]
   537→        } else {
   538→                restRunes = nil
   539→        }
   540→        if len(restRunes) > 30 {
   541→                return 0, false
   542→        }
   543→        return parseChapterNumber(m[1])
   544→}

   548→// auditNovel 逐书体检 → NovelAuditItem 形状（字段名与 TS 完全一致）
   549→func auditNovel(novelId int64, title string) map[string]any {
   550→        type row struct {
   551→                id, idx, wordCount int64
   552→                title              string
   553→        }
   554→        rows := make([]row, 0)
   555→        _ = queryList(`SELECT "id", "idx", "title", "wordCount" FROM "Chapter" WHERE "novelId" = ? ORDER BY "idx" ASC`,
   556→                func(rs *sql.Rows) error {
   557→                        var r row
   558→                        if err := rs.Scan(&r.id, &r.idx, &r.title, &r.wordCount); err != nil {
   559→                                return err
   560→                        }
   561→                        rows = append(rows, r)
   562→                        return nil
   563→                }, novelId)
   564→
   565→        // 重复标题分组
   566→        byTitle := map[string]int{}
   567→        for _, r := range rows {
   568→                byTitle[r.title]++
   569→        }
   570→        dupGroups, dupRows := 0, 0
   571→        for _, c := range byTitle {
   572→                if c > 1 {
   573→                        dupGroups++
   574→                        dupRows += c - 1
   575→                }
   576→        }
   577→
   578→        // idx 断档（rows 与 maxIdx 不符；idx 从 1 连续为健康）
   579→        var maxIdx int64
   580→        for _, r := range rows {
   581→                if r.idx > maxIdx {
   582→                        maxIdx = r.idx
   583→                }
   584→        }
   585→        idxGaps := maxIdx - int64(len(rows))
   586→        if idxGaps < 0 {
   587→                idxGaps = 0
   588→        }
   589→
   590→        emptyRows := 0
   591→        for _, r := range rows {
   592→                if r.wordCount == 0 {
   593→                        emptyRows++
   594→                }
   595→        }
   596→
   597→        // 编号乱序：相邻回退检测（前章编号大于后章编号）
   598→        disorderSamples := make([]string, 0)
   599→        var prevN int64
   600→        var prevTitle string
   601→        hasPrev := false
   602→        for _, r := range rows {
   603→                n, ok := extractNum(r.title)
   604→                if !ok {
   605→                        continue
   606→                }
   607→                if hasPrev && n < prevN && len(disorderSamples) < 3 {
   608→                        disorderSamples = append(disorderSamples, "「"+truncateRunes(prevTitle, 24)+"」→「"+truncateRunes(r.title, 24)+"」")
   609→                }
   610→                prevN, prevTitle, hasPrev = n, r.title, true
   611→        }
   612→
   613→        item := map[string]any{
   614→                "novelId":         novelId,
   615→                "title":           title,
   616→                "chapters":        len(rows),
   617→                "dupGroups":       dupGroups,
   618→                "dupRows":         dupRows,
   619→                "idxGaps":         idxGaps,
   620→                "emptyRows":       emptyRows,
   621→                "disordered":      len(disorderSamples) > 0,
   622→                "disorderSamples": disorderSamples,
   623→        }
   624→
   625→        // 分卷结构：以「第X卷/部/篇」标题行为卷边界
   626→        if len(rows) > 0 {
   627→                volumes := make([]map[string]any, 0)
   628→                var current map[string]any
   629→                for _, r := range rows {
   630→                        if volumeRe.MatchString(r.title) {
   631→                                current = map[string]any{"title": truncateRunes(r.title, 60), "from": r.idx, "to": r.idx, "chapters": 0}
   632→                                volumes = append(volumes, current)
   633→                        }
   634→                        if current != nil {
   635→                                current["to"] = r.idx
   636→                                current["chapters"] = current["chapters"].(int) + 1
   637→                        }
   638→                }
   639→                if len(volumes) > 0 {
   640→                        item["volumes"] = volumes
   641→                }
   642→        }
   643→        return item
   644→}

   648→func handleChapterAuditGet(w http.ResponseWriter, r *http.Request, _ map[string]string) {
   649→        novelIdParam := r.URL.Query().Get("novelId")
   650→        if novelIdParam != "" {
   651→                f, ok := jsParseFloat(novelIdParam)
   652→                if !ok || !numIsInt(f) || f <= 0 {
   653→                        writeJSON(w, 400, map[string]string{"error": "novelId 非法"})
   654→                        return
   655→                }
   656→                novelId := int64(f)
   657→                var title string
   658→                if err := queryOne(`SELECT "title" FROM "Novel" WHERE "id" = ?`, []any{&title}, novelId); err != nil {
   659→                        if isNoRows(err) {
   660→                                writeJSON(w, 404, map[string]string{"error": "书籍不存在"})
   661→                                return
   662→                        }
   663→                        writeJSON(w, 500, map[string]string{"error": "目录体检失败", "detail": firstLineErr(err)})
   664→                        return
   665→                }
   666→                item := auditNovel(novelId, title)
   667→                writeJSON(w, 200, map[string]any{"mode": "single", "item": item})
   668→                return
   669→        }
   670→
   671→        // 全站概览：逐书体检
   672→        type novelRow struct {
   673→                id    int64
   674→                title string
   675→        }
   676→        novels := make([]novelRow, 0)
   677→        if err := queryList(`SELECT "id", "title" FROM "Novel" ORDER BY "id" ASC`, func(rows *sql.Rows) error {
   678→                var n novelRow
   679→                if err := rows.Scan(&n.id, &n.title); err != nil {
   680→                        return err
   681→                }
   682→                novels = append(novels, n)
   683→                return nil
   684→        }); err != nil {
   685→                writeJSON(w, 500, map[string]string{"error": "目录体检失败", "detail": firstLineErr(err)})
   686→                return
   687→        }
   688→        items := make([]map[string]any, 0, len(novels))
   689→        for _, n := range novels {
   690→                items = append(items, auditNovel(n.id, n.title))
   691→        }
   692→        type scored struct {
   693→                item  map[string]any
   694→                score int
   695→        }
   696→        problem := make([]scored, 0)
   697→        for _, it := range items {
   698→                if it["dupRows"].(int) > 0 || it["idxGaps"].(int64) > 0 || it["disordered"].(bool) {
   699→                        problem = append(problem, scored{it, it["dupRows"].(int)*10 + int(it["idxGaps"].(int64))})
   700→                }
   701→        }
   702→        // 稳定降序（对齐 JS Array.prototype.sort 稳定性）后取 100
   703→        for i := 1; i < len(problem); i++ {
   704→                for j := i; j > 0 && problem[j].score > problem[j-1].score; j-- {
   705→                        problem[j], problem[j-1] = problem[j-1], problem[j]
   706→                }
   707→        }
   708→        if len(problem) > 100 {
   709→                problem = problem[:100]
   710→        }
   711→        out := make([]map[string]any, 0, len(problem))
   712→        for _, p := range problem {
   713→                out = append(out, p.item)
   714→        }
   715→        writeJSON(w, 200, map[string]any{
   716→                "mode":    "overview",
   717→                "novels":  len(items),
   718→                "healthy": len(items) - len(out),
   719→                "problem": out,
   720→        })
   721→}

   723→// ==================== POST /api/chapters/audit（dedupe / reindex） ====================
   724→
   725→func handleChapterAuditPost(w http.ResponseWriter, r *http.Request, _ map[string]string) {
   726→        v, ok := readBodyValue(r)
   727→        if !ok {
   728→                writeJSON(w, 400, map[string]string{"error": "请求体非法 JSON"})
   729→                return
   730→        }
   731→        body := bodyMap(v)
   732→        action := strField(body["action"], 0)
   733→        novelIdF := jsNumber(body["novelId"])
   734→        if action != "dedupe" && action != "reindex" {
   735→                writeJSON(w, 400, map[string]string{"error": "action 仅支持 dedupe / reindex"})
   736→                return
   737→        }
   738→        if !numIsInt(novelIdF) || novelIdF <= 0 {
   739→                writeJSON(w, 400, map[string]string{"error": "novelId 必填（正整数）"})
   740→                return
   741→        }
   742→        novelId := int64(novelIdF)
   743→
   744→        var title string
   745→        if err := queryOne(`SELECT "title" FROM "Novel" WHERE "id" = ?`, []any{&title}, novelId); err != nil {
   746→                if isNoRows(err) {
   747→                        writeJSON(w, 404, map[string]string{"error": "书籍不存在"})
   748→                        return
   749→                }
   750→                writeJSON(w, 500, map[string]string{"error": "目录修复失败", "detail": firstLineErr(err)})
   751→                return
   752→        }
   753→
   754→        type row struct {
   755→                id, idx, wordCount int64
   756→                title              string
   757→        }
   758→        rows := make([]row, 0)
   759→        if err := queryList(`SELECT "id", "idx", "title", "wordCount" FROM "Chapter" WHERE "novelId" = ? ORDER BY "idx" ASC, "id" ASC`,
   760→                func(rs *sql.Rows) error {
   761→                        var x row
   762→                        if err := rs.Scan(&x.id, &x.idx, &x.title, &x.wordCount); err != nil {
   763→                                return err
   764→                        }
   765→                        rows = append(rows, x)
   766→                        return nil
   767→                }, novelId); err != nil {
   768→                writeJSON(w, 500, map[string]string{"error": "目录修复失败", "detail": firstLineErr(err)})
   769→                return
   770→        }
   771→        if len(rows) == 0 {
   772→                writeJSON(w, 400, map[string]string{"error": "该书无章节"})
   773→                return
   774→        }
   775→
   776→        // ---- 去重：同书同名保留 wordCount 最大（并列取 idx 最小）----
   777→        removed := 0
   778→        if action == "dedupe" {
   779→                type best struct {
   780→                        id, wordCount, idx int64
   781→                }
   782→                keep := map[string]best{}
   783→                for _, x := range rows {
   784→                        b, exists := keep[x.title]
   785→                        if !exists || x.wordCount > b.wordCount || (x.wordCount == b.wordCount && x.idx < b.idx) {
   786→                                keep[x.title] = best{id: x.id, wordCount: x.wordCount, idx: x.idx}
   787→                        }
   788→                }
   789→                if len(keep) < len(rows) {
   790→                        keepIDs := map[int64]bool{}
   791→                        for _, b := range keep {
   792→                                keepIDs[b.id] = true
   793→                        }
   794→                        dropIDs := make([]int64, 0)
   795→                        for _, x := range rows {
   796→                                if !keepIDs[x.id] {
   797→                                        dropIDs = append(dropIDs, x.id)
   798→                                }
   799→                        }
   800→                        // 分批 IN 删除（SQLite 变量上限防御）
   801→                        for i := 0; i < len(dropIDs); i += 500 {
   802→                                end := i + 500
   803→                                if end > len(dropIDs) {
   804→                                        end = len(dropIDs)
   805→                                }
   806→                                part := dropIDs[i:end]
   807→                                ph := strings.TrimSuffix(strings.Repeat("?,", len(part)), ",")
   808→                                args := make([]any, len(part))
   809→                                for j, idv := range part {
   810→                                        args[j] = idv
   811→                                }
   812→                                res, err := exec(`DELETE FROM "Chapter" WHERE "id" IN (`+ph+`)`, args...)
   813→                                if err != nil {
   814→                                        writeJSON(w, 500, map[string]string{"error": "目录修复失败", "detail": firstLineErr(err)})
   815→                                        return
   816→                                }
   817→                                n, _ := res.RowsAffected()
   818→                                removed += int(n)
   819→                        }
   820→                }
   821→        }
   822→
   823→        // ---- 重排：有编号按编号升序（稳定），无编号按原相对顺序置后；压实 idx 1..n ----
   824→        kept := make([]row, 0)
   825→        if err := queryList(`SELECT "id", "idx", "title" FROM "Chapter" WHERE "novelId" = ? ORDER BY "idx" ASC, "id" ASC`,
   826→                func(rs *sql.Rows) error {
   827→                        var x row
   828→                        if err := rs.Scan(&x.id, &x.idx, &x.title); err != nil {
   829→                                return err
   830→                        }
   831→                        kept = append(kept, x)
   832→                        return nil
   833→                }, novelId); err != nil {
   834→                writeJSON(w, 500, map[string]string{"error": "目录修复失败", "detail": firstLineErr(err)})
   835→                return
   836→        }
   837→        moved := 0
   838→        type numberedRow struct {
   839→                r row
   840→                i int
   841→                n int64
   842→        }
   843→        numbered := make([]numberedRow, 0)
   844→        unnumbered := make([]row, 0)
   845→        for i, x := range kept {
   846→                if n, ok := extractNum(x.title); ok {
   847→                        numbered = append(numbered, numberedRow{r: x, i: i, n: n})
   848→                } else {
   849→                        unnumbered = append(unnumbered, x)
   850→                }
   851→        }
   852→        // 稳定排序（n 升序，并列保持原相对顺序 i）
   853→        for i := 1; i < len(numbered); i++ {
   854→                for j := i; j > 0 && (numbered[j].n < numbered[j-1].n); j-- {
   855→                        numbered[j], numbered[j-1] = numbered[j-1], numbered[j]
   856→                }
   857→        }
   858→        ordered := make([]row, 0, len(kept))
   859→        for _, x := range numbered {
   860→                ordered = append(ordered, x.r)
   861→        }
   862→        ordered = append(ordered, unnumbered...)
   863→
   864→        // 两段式重排：先全部移出到负数暂存区（避开目标 idx 被占），再落位 1..n
   865→        for i, x := range ordered {
   866→                if x.idx != int64(i+1) {
   867→                        _, _ = exec(`UPDATE "Chapter" SET "idx" = ? WHERE "id" = ?`, -1_000_000-int64(i), x.id)
   868→                }
   869→        }
   870→        for i, x := range ordered {
   871→                target := int64(i + 1)
   872→                if x.idx != target {
   873→                        res, err := exec(`UPDATE "Chapter" SET "idx" = ? WHERE "id" = ?`, target, x.id)
   874→                        if err == nil {
   875→                                if n, _ := res.RowsAffected(); n > 0 {
   876→                                        moved++
   877→                                }
   878→                        }
   879→                }
   880→        }
   881→
   882→        // 重算字数（去重可能删行；update 触碰 @updatedAt → 显式 set）
   883→        if removed > 0 {
   884→                var sumWC sql.NullInt64
   885→                if err := queryOne(`SELECT SUM("wordCount") FROM "Chapter" WHERE "novelId" = ?`, []any{&sumWC}, novelId); err == nil {
   886→                        _, _ = exec(`UPDATE "Novel" SET "wordCount" = ?, "updatedAt" = ? WHERE "id" = ?`, sumWC.Int64, nowMillis(), novelId)
   887→                }
   888→        }
   889→
   890→        after := auditNovel(novelId, title)
   891→        writeJSON(w, 200, map[string]any{
   892→                "ok":      true,
   893→                "action":  action,
   894→                "novelId": novelId,
   895→                "removed": removed,
   896→                "moved":   moved,
   897→                "audit":   after,
   898→        })
   899→}
