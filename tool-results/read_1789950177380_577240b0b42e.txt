   447→/* Category：左 190 榜单侧栏 + 右 760 六列数据表 + 翻页                   */
   448→/* ==================================================================== */
   449→
   450→type SortKey = 'latest' | 'clicks' | 'words'
   451→type StatusKey = 'all' | 'serial' | 'finished'
   452→type PresetKey = 'clicks' | 'latest' | 'finished' | 'words'
   453→
   454→function CategoryView({
   455→  navigate,
   456→  categoryId,
   457→  page = 1,
   458→}: ViewProps & { categoryId?: number; page?: number }) {
   459→  const { data: cats } = useCategories()
   460→  const { data: home } = useHomeData()
   461→  const [sort, setSort] = useState<SortKey>('latest')
   462→  const [status, setStatus] = useState<StatusKey>('all')
   463→  const cur = page > 0 ? page : 1
   464→
   465→  const { data, isLoading, isError, refetch } = useNovels({
   466→    categoryId,
   467→    page: cur,
   468→    pageSize: 20,
   469→    sort,
   470→    status: status === 'all' ? undefined : status,
   471→  })
   472→
   473→  const catName =
   474→    categoryId != null ? (cats ?? []).find((c) => c.id === categoryId)?.name : undefined
   475→  const goPage = (p: number) => navigate({ name: 'category', categoryId, page: p })
   476→  const applySort = (s: SortKey) => {
   477→    setSort(s)
   478→    setStatus('all')
   479→    if (cur !== 1) goPage(1)
   480→  }
   481→  const applyStatus = (s: StatusKey) => {
   482→    setStatus(s)
   483→    if (cur !== 1) goPage(1)
   484→  }
   485→  const applyPreset = (p: PresetKey) => {
   486→    if (p === 'finished') {
   487→      setSort('latest')
   488→      setStatus('finished')
   489→    } else {

   634→const SHELF_KEY = 'trxsw-shelf'
   635→
   636→function loadShelf(): number[] {
   637→  if (typeof window === 'undefined') return []
   638→  try {
   639→    return JSON.parse(window.localStorage.getItem(SHELF_KEY) ?? '[]') as number[]
   640→  } catch {
   641→    return []
   642→  }
   643→}
   644→
   645→function toggleShelf(id: number): boolean {
   646→  if (typeof window === 'undefined') return false
   647→  try {
   648→    const arr = loadShelf()
   649→    const has = arr.includes(id)
   650→    window.localStorage.setItem(SHELF_KEY, JSON.stringify(has ? arr.filter((x) => x !== id) : [...arr, id]))
   651→    return !has
   652→  } catch {
   653→    return false
   654→  }
   655→}

  1257→function SearchInner({ navigate, query, field }: ViewProps & { query: string; field: SearchField }) {
