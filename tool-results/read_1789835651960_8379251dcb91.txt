    97→    97→ * （Turbopack 按路由分包），异步置位会留下双重回收的竞态窗口。
    98→    98→ */
    99→    99→const ZOMBIE_STALE_MS = 5 * 60_000
   100→   100→
   101→   101→async function recoverStaleTasks(): Promise<void> {
   102→   102→  if (gWorker.__scrapeRecovered) return
   103→   103→  gWorker.__scrapeRecovered = true // 同步置位：跨实例/跨 HMR 防重入
   104→   104→  try {
   105→   105→    const stale = await db.scrapeTask.findMany({
   106→   106→      where: {
   107→   107→        status: { in: ['pending', 'running'] },
   108→   108→        updatedAt: { lt: new Date(Date.now() - ZOMBIE_STALE_MS) },
   109→   109→      },
   110→   110→      select: { id: true, log: true },
   111→   111→    })
   112→   112→    let recovered = 0
   113→   113→    for (const t of stale) {
   114→   114→      if (running.has(t.id)) continue // 本进程内存中仍有实例在执行（共享 Set 时双保险）
   115→   115→      const line = `[${new Date().toTimeString().slice(0, 8)}] 心跳停摆超 5 分钟，判定僵尸任务并回收`
   116→   116→      const log = `${t.log ? `${t.log}\n` : ''}${line}`
   117→   117→      const res = await db.scrapeTask
   118→   118→        .updateMany({
   119→   119→          where: { id: t.id, status: { in: ['pending', 'running'] } },
   120→   120→          data: {
   121→   121→            status: 'failed',
   122→   122→            message: '任务心跳超时，自动回收',
   123→   123→            // 与 Run 同规约：只保留最近 100 行
   124→   124→            log: log.split('\n').slice(-MAX_LOG_LINES).join('\n'),
   125→   125→          },
   126→   126→        })
   127→   127→        .catch(() => null)
   128→   128→      if (res && res.count > 0) recovered++
   129→   129→    }
   130→   130→    if (recovered > 0) console.log(`[scrape-worker] 僵尸任务回收: ${recovered} 条`)
   131→   131→  } catch {
   132→   132→    // 回收失败不影响模块加载；标志已置位，等待下次进程重启重试（代价可接受：僵尸多驻留一轮）
   133→   133→  }
   134→   134→}

   137→   137→/** 协作式取消检查：记录不存在视为取消；DB 瞬时错误不误判为取消（fail-open） */
   138→   138→async function isCanceled(taskId: number): Promise<boolean> {
   139→   139→  const t = await db.scrapeTask
   140→   140→    .findUnique({ where: { id: taskId }, select: { status: true } })
   141→   141→    .catch(() => undefined)
   142→   142→  if (t === undefined) return false // 查询失败 ≠ 被取消，避免瞬时 DB 错误误停任务
   143→   143→  return !t || t.status === 'canceled'
   144→   144→}

   146→   146→/**
   147→   147→ * 并行池：cursor 抢占式派发，concurrency 路 worker 持续取件直至耗尽。
   148→   148→ * fn 内部通过 stop signal 提前收敛（停止派发，在途任务跑完当前单元后自然退出）。
   149→   149→ */
   150→   150→async function runPool<T>(items: T[], concurrency: number, fn: (item: T, index: number) => Promise<void>): Promise<void> {
   151→   151→  let cursor = 0
   152→   152→  const workers = Array.from({ length: Math.max(1, Math.min(concurrency, items.length)) }, async () => {
   153→   153→    for (;;) {
   154→   154→      const i = cursor++
   155→   155→      if (i >= items.length) return
   156→   156→      await fn(items[i], i)
   157→   157→    }
   158→   158→  })
   159→   159→  await Promise.all(workers)
   160→   160→}

   162→   162→// ==================== Phase 1：书籍元数据 + 章节目录骨架（single 与 list 共用） ====================

   164→   164→/**
   165→   165→ * Phase 1：抓取书页+完整目录 → 书籍 upsert（含封面落盘/下拉词绑定）→ 章节目录骨架入库。
   166→   166→ *

   169→   169→ * - 已存在空骨架（wordCount=0，上次任务未抓到）→ 保持骨架，Phase 2 重抓填充；
   170→   170→ * - 不存在 → 创建骨架（content=''、wordCount=0、idx=目录位次保序）。
   171→   171→ * 已有章节盘点只取 id/idx/title/volume/wordCount（绝不 select content 全文——

   173→   173→ *

   175→   175→ * 逐条 create 兜底，createChapterSkeleton 撞唯一约束查回已有记录按「已存在」处理。

   177→   177→async function processBookMeta(
   178→   178→  run: Run,
   179→   179→  bookUrl: string,
   180→   180→  rule: LoadedRule,
   181→   181→  opts: { trackTotal: boolean; referer?: string | null },
   182→   182→): Promise<BookMetaOutcome> {
   183→   183→  // 失败语义三分（P1-4）：canceled=用户取消/记录删除（停整个任务）；
   184→   184→  // hard=DB 记录级硬失败（list 跳过继续、single 按 failed 收尾，不再借用 canceled 通道）；
   185→   185→  // plain=普通书籍级失败（书页提取失败/空标题等，处理同 hard）
   186→   186→  const fail = (message: string, kind: 'plain' | 'canceled' | 'hard' = 'plain'): BookMetaOutcome => ({
   187→   187→    ok: false,
   188→   188→    canceled: kind === 'canceled',
   189→   189→    hardFailed: kind === 'hard',
   190→   190→    novelId: 0,
   191→   191→    refsTotal: 0,
   192→   192→    skipped: 0,
   193→   193→    jobs: [],
   194→   194→    message,
   195→   195→  })
   196→   196→
   197→   197→  run.log(`抓取书页 ${bookUrl.slice(0, 120)}…`)
   198→   198→  // Referer 链：list 模式传「发现本书的列表页」作来路；single 模式缺省由引擎回落站内首页。
   199→   199→  // 熔断感知：书页撞上熔断（站点临时限流封禁）时等待冷却后自动重试，而非立即把本书记失败
   200→   200→  const page = await withCircuitRetry(hostOf(bookUrl), () => fetchBookPage(run, bookUrl, rule, opts.referer ?? undefined), {
   201→   201→    detect: (r) => (r.ok ? null : circuitWaitMs(r.error, r.circuitRetryAfterMs)),
   202→   202→    isAlive: async () => !(await isCanceled(run.taskId)),
   203→   203→    onWait: (ms, cycle, max) =>
   204→   204→      run.log(`目标主机熔断冷却中，暂停 ${Math.round(ms / 1000)}s 后自动重试书页（第 ${cycle}/${max} 轮，等待期间不向源站发请求）`),
   205→   205→  })
   206→   206→  if (!page.ok) {
   207→   207→    run.log(`书页提取失败: ${page.error}`)

   212→   212→  // 防畸形载荷（引擎版本错位等）在任务中途抛 TypeError 把整任务拖成 failed
   213→   213→  if (typeof book.title !== 'string') book.title = ''
   214→   214→  if (!Array.isArray(book.chapters)) book.chapters = []
   215→   215→  if (typeof book.description !== 'string') book.description = ''
   216→   216→  if (typeof book.author !== 'string') book.author = ''
   217→   217→  if (typeof book.category !== 'string') book.category = ''
   218→   218→  if (typeof book.status !== 'string') book.status = ''
   219→   219→  if (typeof book.chapterCount !== 'number') book.chapterCount = book.chapters.length
   220→   220→  // 全字段兜底清洗（引擎侧已清洗，此处防版本错位/旧引擎，幂等成本低）：
   221→   221→  // 实体解码（&#091；类全角分号变体）、字面 \n 还原、残缺标签剥除、简介样板句/SEO 伪简介处理
   222→   222→  book.title = cleanTextField(book.title)
   223→   223→  book.author = cleanTextField(book.author)
   224→   224→  book.description = cleanDescriptionField(book.description)
   225→   225→  book.category = cleanTextField(book.category)
   226→   226→  book.status = cleanTextField(book.status)
   227→   227→  run.log(
   228→   228→    `书页提取成功：《${book.title.slice(0, 40)}》${book.author ? ` / ${book.author.slice(0, 20)}` : ''}，章节链接 ${book.chapterCount} 条`,
   229→   229→  )
   230→   230→
   231→   231→  // ---- 完整目录页二次提取（bookRule.catalogLinkSelector，如 23qb 新模板书页仅含最新几章）----
   232→   232→  let allRefs = book.chapters
   233→   233→  const catalogSel = rule.bookRule.catalogLinkSelector
   234→   234→  if (typeof catalogSel === 'string' && catalogSel) {
   235→   235→    if (await isCanceled(run.taskId)) return fail('任务已取消', 'canceled')
   236→   236→    if (book.catalogUrl) {
   237→   237→      run.log(`发现完整目录页 ${book.catalogUrl.slice(0, 100)}，尝试整目提取…`)
   238→   238→      const catalogUrl = book.catalogUrl // 闭包捕获需局部常量（TS 收窄不跨回调）
   239→   239→      // 熔断感知：目录页撞熔断时等待冷却后自动重试（超限时按旧语义回退书页章节链接）
   240→   240→      const catalog = await withCircuitRetry(
   241→   241→        hostOf(catalogUrl),
   242→   242→        () => fetchCatalogChapters(run, catalogUrl, rule, bookUrl),
   243→   243→        {
   244→   244→          detect: (r) => (r.error ? circuitWaitMs(r.error) : null),
   245→   245→          isSuccess: (r) => r.error === null,
   246→   246→          isAlive: async () => !(await isCanceled(run.taskId)),
   247→   247→          onWait: (ms, cycle, max) =>
   248→   248→            run.log(`目标主机熔断冷却中，暂停 ${Math.round(ms / 1000)}s 后自动重试目录页（第 ${cycle}/${max} 轮）`),
   249→   249→        },
   250→   250→      )
   251→   251→      const catalogRefs = catalog.refs
   252→   252→      if (catalogRefs.length > allRefs.length) {
   253→   253→        run.log(`目录页提取到 ${catalogRefs.length} 条章节链接（书页仅 ${allRefs.length} 条），采用目录页结果`)

   255→   255→      } else if (catalogRefs.length > 0) {
   256→   256→        run.log(`目录页提取 ${catalogRefs.length} 条不多于书页 ${allRefs.length} 条，维持书页结果`)
   257→   257→      }
   258→   258→    } else {
   259→   259→      run.log(`catalogLinkSelector "${catalogSel.slice(0, 60)}" 在书页无命中，仅用书页章节链接`)
   260→   260→    }
   261→   261→  }
   262→   262→
   263→   263→  if (await isCanceled(run.taskId)) return fail('任务已取消', 'canceled')
   264→   264→
   265→   265→  // ---- 乱序重排（充分考虑分卷：同号不同卷不跨卷错排，卷名随章节移动）----
   266→   266→  // 源站目录常见「最新章节块（新→旧）+ 正文块（旧→新）」或整本倒序；直接按抓取序编 idx 会错乱。
   267→   267→  // 引擎响应未经 schema 校验，逐条形态兑底后再重排。
   268→   268→  allRefs = (Array.isArray(allRefs) ? allRefs : []).map((c) => ({
   269→   269→    title: typeof c?.title === 'string' ? c.title : '',
   270→   270→    url: typeof c?.url === 'string' ? c.url : null,
   271→   271→    volume: typeof c?.volume === 'string' && c.volume.trim() ? c.volume.trim().slice(0, 50) : undefined,
   272→   272→  })) as ChapterRef[]
   273→   273→  const rr = reorderChapterRefs(allRefs)
   274→   274→  if (rr.reordered) {
   275→   275→    allRefs = rr.refs
   276→   276→    run.log(`目录乱序重排：${rr.note}`)
   277→   277→  }
   278→   278→
   279→   279→  // ---- 分类 ----
   280→   280→  let categoryId: number
   281→   281→  try {
   282→   282→    categoryId = await ensureCategory(book.category)
   283→   283→  } catch (e) {
   284→   284→    const msg = e instanceof Error ? e.message : '分类处理失败'
   285→   285→    run.log(msg)
   286→   286→    return fail(msg)
   287→   287→  }
   288→   288→
   289→   289→  // ---- 书籍 upsert（title+author 查重；DB 层 @@unique([title,author]) 兜底并发）----
   290→   290→  const up = await upsertBook(run, book, categoryId, rule.proxy)
   291→   291→  if (!up.ok) {
   292→   292→    // P1-4：记录级 DB 失败（up.hard=true，如书籍在采集中被删除）按书籍级硬失败处理，
   293→   293→    // list 模式不再停整个任务（旧实现借 canceled 通道把整任务收成 canceled，语义失真）；
   294→   294→    // 书籍级普通失败（空标题等）同样只算本书失败
   295→   295→    return fail(up.message, up.hard ? 'hard' : 'plain')
   296→   296→  }
   297→   297→  // 采集的同时取搜索引擎下拉词 → 绑定书籍 + 生成 PSEO 书籍页（fire-and-forget）
   298→   298→  scheduleBindBookSuggestKeywords(run, up.novelId, up.title || book.title, up.createdNew)
   299→   299→
   300→   300→  // ---- 章节列表准备 ----
   301→   301→  const refs = allRefs.filter(
   302→   302→    (c): c is ChapterRef & { url: string } => !!c && typeof c.url === 'string' && !!c.url,
   303→   303→  )
   304→   304→  if (refs.length === 0) {
   305→   305→    run.log('未提取到任何有效章节链接')
   306→   306→    return { ok: true, canceled: false, hardFailed: false, novelId: up.novelId, refsTotal: 0, skipped: 0, jobs: [], message: '书籍已入库（未提取到章节链接）' }
   307→   307→  }
   308→   308→  // 同目录内预去重（骨架入库前完成）：按清洗后标题优先、URL 兜底去重，防同批重复入库
   309→   309→  const seenRef = new Set<string>()
   310→   310→  const uniqueRefs: (ChapterRef & { url: string })[] = []
   311→   311→  for (const c of refs) {
   312→   312→    const key = c.title.trim() ? `t:${cleanTextField(c.title)}` : `u:${c.url}`
   313→   313→    if (seenRef.has(key)) continue
   314→   314→    seenRef.add(key)
   315→   315→    uniqueRefs.push(c)
   316→   316→  }
   317→   317→  if (uniqueRefs.length < refs.length) run.log(`目录内去重：${refs.length} → ${uniqueRefs.length} 条（无数量上限，全量采集）`)
   318→   318→
   319→   319→  if (await isCanceled(run.taskId)) return fail('任务已取消', 'canceled')
   320→   320→
   321→   321→  // ---- 已有章节盘点（不含 content 全文；wordCount 区分内容章节与骨架） ----
   322→   322→  const existingChapters = await db.chapter
   323→   323→    .findMany({
   324→   324→      where: { novelId: up.novelId },
   325→   325→      select: { id: true, idx: true, title: true, volume: true, wordCount: true },
   326→   326→    })
   327→   327→    .catch(() => [] as { id: number; idx: number; title: string; volume: string; wordCount: number }[])
   328→   328→  // 去重键与入库标题共用同一清洗规约（cleanTextField 幂等）：存量脏标题与本次净标题可命中比对，
   329→   329→  // 避免清洗规约升级后同一章以两种写法重复入库
   330→   330→  const titleGroups = new Map<string, { id: number; idx: number; wordCount: number }[]>()
   331→   331→  for (const c of existingChapters) {
   332→   332→    const k = cleanTextField(c.title)
   333→   333→    const arr = titleGroups.get(k)
   334→   334→    if (arr) arr.push({ id: c.id, idx: c.idx, wordCount: c.wordCount })
   335→   335→    else titleGroups.set(k, [{ id: c.id, idx: c.idx, wordCount: c.wordCount }])
   336→   336→  }
   337→   337→  const maxAgg = await db.chapter
   338→   338→    .aggregate({ where: { novelId: up.novelId }, _max: { idx: true } })
   339→   339→    .catch(() => ({ _max: { idx: null as number | null } }))
   340→   340→  // 骨架槽位基准：现有最大 idx+1；每条按目录位次预分配（base+位次），保序且并行安全
   341→   341→  const baseIdx = (maxAgg._max.idx ?? 0) + 1
   342→   342→
   343→   343→  // ---- 目录逐条分类：skip（已有内容）/ fill（已有空骨架或新骨架）/ 骨架待建 ----
   344→   344→  const jobs: ChapterFillJob[] = []
   345→   345→  /** 待创建骨架（jobIndex 回指 jobs 中对应的 fill 任务，创建结果落地后回填定位） */
   346→   346→  const pendingSkeletons: { jobIndex: number; idx: number; title: string; volume: string; refTitle: string }[] = []
   347→   347→  /** 已存在章节的卷名回填队列（title → volume；仅唯一标题且存量为空卷时写入） */
   348→   348→  const volumeBackfill = new Map<string, string>()
   349→   349→  let skipped = 0
   350→   350→
   351→   351→  for (let position = 0; position < uniqueRefs.length; position++) {
   352→   352→    const ref = uniqueRefs[position]
   353→   353→    const refTitle = cleanTextField(String(ref.title ?? '')).slice(0, 200)
   354→   354→    if (refTitle && titleGroups.has(refTitle)) {
   355→   355→      const matches = titleGroups.get(refTitle)!
   356→   356→      if (matches.some((m) => m.wordCount > 0)) {
   357→   357→        // 已存在且有内容：跳过不重抓（Phase 2 仅作进度占位）
   358→   358→        skipped++
   359→   359→        jobs.push({ kind: 'skip', novelId: up.novelId, chapterId: matches[0].id, idx: matches[0].idx, url: '', refTitle, volume: '' })
   360→   360→        // 分卷补齐：本次源站目录携带卷名、存量章节为空卷且标题在全书中唯一时，回填卷名
   361→   361→        const rv = ref.volume
   362→   362→        if (rv && matches.length === 1 && !volumeBackfill.has(refTitle)) volumeBackfill.set(refTitle, rv)
   363→   363→        continue
   364→   364→      }
   365→   365→      // 已存在空骨架（上次任务未抓到内容）：保持骨架不重建，Phase 2 重抓填充（不跳过）
   366→   366→      const target = matches.reduce((a, b) => (a.idx <= b.idx ? a : b))
   367→   367→      jobs.push({ kind: 'fill', novelId: up.novelId, chapterId: target.id, idx: target.idx, url: ref.url, refTitle, volume: ref.volume ?? '' })
   368→   368→      continue
   369→   369→    }
   370→   370→    // 不存在 → 骨架占位（idx=目录位次预分配保序），创建结果落地后回填 chapterId
   371→   371→    const idx = baseIdx + position
   372→   372→    pendingSkeletons.push({ jobIndex: jobs.length, idx, title: refTitle || `第${idx}章`, volume: ref.volume ?? '', refTitle })
   373→   373→    jobs.push({ kind: 'fill', novelId: up.novelId, chapterId: null, idx, url: ref.url, refTitle, volume: ref.volume ?? '' })
   374→   374→  }
   375→   375→
   376→   376→  // ---- 骨架批量入库：createMany 快路径；整批失败（并发撞 (novelId,idx) 等）逐条 create 兜底 ----
   377→   377→  if (pendingSkeletons.length > 0) {
   378→   378→    let batchErr: unknown = null
   379→   379→    try {
   380→   380→      await db.chapter.createMany({
   381→   381→        data: pendingSkeletons.map((s) => ({
   382→   382→          novelId: up.novelId,
   383→   383→          idx: s.idx,
   384→   384→          title: s.title,
   385→   385→          content: '',
   386→   386→          wordCount: 0,
   387→   387→          volume: s.volume,
   388→   388→        })),
   389→   389→      })
   390→   390→    } catch (e) {
   391→   391→      batchErr = e
   392→   392→    }
   393→   393→    if (batchErr !== null) {
   394→   394→      // P1-3：整批撞 FK（书在采集中被删除，Novel←Chapter onDelete:Cascade）→ 立即中止本书，
   395→   395→      // 不再逐条空转（旧行为对已删书逐章重试刷屏）
   396→   396→      if (isRecordMissingError(batchErr)) {
   397→   397→        run.log(`书籍已消失（采集中被删除），中止本书章节骨架入库`)
   398→   398→        return fail('书籍已消失（采集中被删除）', 'hard')
   399→   399→      }
   400→   400→      // 并发撞 (novelId,idx) 等其他批量失败 → 逐条 create 兜底（createChapterSkeleton
   401→   401→      // 撞唯一约束查回已有记录按「已存在」处理）
   402→   402→      for (const s of pendingSkeletons) {
   403→   403→        if (await isCanceled(run.taskId)) return fail('任务已取消', 'canceled')
   404→   404→        const job = jobs[s.jobIndex]
   405→   405→        const sk = await createChapterSkeleton(run, up.novelId, s.idx, { title: s.title, volume: s.volume }, s.refTitle)
   406→   406→        if (!sk.ok) {
   407→   407→          // P1-3：逐条也撞 FK → 书已消失，立即中止本书（不再把剩余全部骨架标记失败）
   408→   408→          if (sk.missing) {
   409→   409→            run.log(`书籍已消失（采集中被删除），中止本书章节骨架入库`)
   410→   410→            return fail('书籍已消失（采集中被删除）', 'hard')
   411→   411→          }
   412→   412→          run.log(`骨架章节「${s.title.slice(0, 30)}」入库失败: ${sk.message.slice(0, 120)}`)
   413→   413→          job.kind = 'create' // 无骨架可填：Phase 2 抓到正文后走 storeChapter 兜底新建
   414→   414→          continue
   415→   415→        }
   416→   416→        job.idx = sk.idx
   417→   417→        if (!sk.created) {
   418→   418→          // 撞唯一约束查回已有记录 → 按「已存在」处理（并发双任务同书场景）
   419→   419→          if (sk.existingHasContent) {
   420→   420→            job.kind = 'skip'
   421→   421→            job.url = ''
   422→   422→            skipped++
   423→   423→          }
   424→   424→          job.chapterId = sk.existingId
   425→   425→        }
   426→   426→      }
   427→   427→    }
   428→   428→  }
   429→   429→
   430→   430→  // ---- 已存在章节的卷名回填（逐卷 updateMany，仅在唯一标题且存量空卷时入队）----
   431→   431→  if (volumeBackfill.size > 0) {
   432→   432→    let backfilled = 0
   433→   433→    for (const [title, volume] of volumeBackfill) {
   434→   434→      const res = await db.chapter
   435→   435→        .updateMany({ where: { novelId: up.novelId, title, volume: '' }, data: { volume } })
   436→   436→        .catch(() => null)
   437→   437→      if (res) backfilled += res.count
   438→   438→    }
   439→   439→    if (backfilled > 0) run.log(`分卷回填：${backfilled} 个已存在章节补记卷名`)
   440→   440→  }
   441→   441→
   442→   442→  // ---- 进度分母 ----
   443→   443→  if (opts.trackTotal) {
   444→   444→    // single 模式：total=骨架章节数（分母），done 随 Phase 2 逐章递增
   445→   445→    //（flush false=任务记录已删，立即停）
   446→   446→    if (!(await run.flush({ total: uniqueRefs.length }))) return fail('任务记录已删除', 'canceled')
   447→   447→  }
   448→   448→
   449→   449→  const toFetch = jobs.length - skipped
   450→   450→  run.log(`目录骨架就绪：${uniqueRefs.length} 章（已存在 ${skipped} 章，待抓取 ${toFetch} 章）`)
   451→   451→  return { ok: true, canceled: false, hardFailed: false, novelId: up.novelId, refsTotal: uniqueRefs.length, skipped, jobs, message: '' }
   452→   452→}
   453→   453→
   454→   454→// ==================== Phase 2：跨书批量章节内容填充 ====================
   455→   455→
   456→   456→/** Phase 2 单书上下文（统计就地回填，供书籍级成败判定与日志节流） */
   457→   457→interface Phase2Book {
   458→   458→  novelId: number
   459→   459→  title: string
   460→   460→  /** 章节页 Referer 来路（书页 URL，站点常见「书页→章节」导航校验） */
   461→   461→  bookUrl: string
   462→   462→  jobs: ChapterFillJob[]
   463→   463→  filled: number
   464→   464→  failed: number
   465→   465→  /** P1-3：书在采集中被删除（P2025/FK 实证）→ 本本书剩余章节不再向源站发请求 */
   466→   466→  vanished: boolean
   467→   467→  /** 清洗日志节流：每本书最多记 3 条，防止日志爆炸 */
   468→   468→  cleanLogCount: number
   469→   469→}
   470→   470→
   471→   471→interface Phase2Result {
   472→   472→  canceled: boolean
   473→   473→  filled: number
   474→   474→  failed: number
   475→   475→  skipped: number
   476→   476→  /** P1-3：采集中被删除的书籍数（finalize 消息提示用） */
   477→   477→  vanishedBooks: number
   478→   478→}
   479→   479→
   480→   480→/**
   481→   481→ * Phase 2：把 Phase 1 收集的全部章节平铺（按书分组顺序），runPool 章节级并发
   482→   482→ * 跨书批量抓取正文，update 填充骨架章节 content/wordCount/volume。
   483→   483→ *
   484→   484→ * - 抓取失败的骨架章节保留骨架不删（重跑任务时 wordCount=0 的骨架会被再次抓取填充）；
   485→   485→ * - novel.wordCount increment 只对抓取成功的章节（fill 填充在此增量，create 走
   486→   486→ *   storeChapter 内部增量），收尾对触及书籍 recalcNovelWordCount 全量校准；
   487→   487→ * - P1-3：填充/入库撞 P2025/FK（书在采集中被删除）→ 实证本书已消失后立即中止本书
   488→   488→ *   剩余章节（不再空抓源站），书籍级成败按失败计；
   489→   489→ * - 保留全部既有机制：熔断感知重试/瞬时失败单次重试/协作式取消/flush 串行链/
   490→   490→ *   入库前清洗/MAX_CONTENT_CHARS 截断。
   491→   491→ * - 进度：single 模式 done 随本章递增（total=骨架章节数已在 Phase 1 写入）；
   492→   492→ *   list 模式累加共享 chapterProgress（chaptersDone/chaptersTotal）。
   493→   493→ */
   494→   494→async function fillChapters(
   495→   495→  run: Run,
   496→   496→  books: Phase2Book[],
   497→   497→  rule: LoadedRule,
   498→   498→  opts: { trackTotal: boolean; chapterProgress?: { done: number; total: number }; concurrency: number },
   499→   499→): Promise<Phase2Result> {
   500→   500→  // 平铺（按书分组顺序）：任务单元携带所属书上下文（Referer 链/日志节流按书隔离）
   501→   501→  const units: { job: ChapterFillJob; book: Phase2Book }[] = []
   502→   502→  for (const b of books) for (const job of b.jobs) units.push({ job, book: b })
   503→   503→
   504→   504→  let filled = 0
   505→   505→  let failed = 0
   506→   506→  let skipped = 0
   507→   507→  let done = 0 // single 模式主口径进度（done/total=章节）
   508→   508→  let stopRun = false // 取消/记录删除 → 停止派发新章节，在途章节收尾后退出
   509→   509→  const touchedNovels = new Set<number>()
   510→   510→
   511→   511→  /** P1-3：防误杀——update 撞 P2025 也可能是单章被单独删除，实证 novel 已不存在才判定整书消失
   512→   512→   * （查询失败按仍存在处理，不因 DB 抖动中止本书） */
   513→   513→  const isNovelGone = async (novelId: number): Promise<boolean> => {
   514→   514→    const n = await db.novel.findUnique({ where: { id: novelId }, select: { id: true } }).catch(() => undefined)
   515→   515→    return n === null
   516→   516→  }
   517→   517→
   518→   518→  /** P1-3：标记本书已消失并中止剩余章节（只记一条日志防刷屏） */
   519→   519→  const markVanished = (book: Phase2Book): void => {
   520→   520→    if (book.vanished) return
   521→   521→    book.vanished = true
   522→   522→    run.log(`书籍《${book.title.slice(0, 30) || book.novelId}》已消失（采集中被删除），中止本书剩余章节抓取`)
   523→   523→  }
   524→   524→
   525→   525→  const alive = async (): Promise<boolean> => {
   526→   526→    if (stopRun) return false
   527→   527→    return !(await isCanceled(run.taskId))
   528→   528→  }
   529→   529→
   530→   530→  // flush 串行链：并行分支共享 Run 缓冲与进度字段，写盘串行化避免交错/竞态
   531→   531→  let flushChain = Promise.resolve(true)
   532→   532→  const scheduleFlush = (fields: TaskFlushFields): Promise<boolean> => {
   533→   533→    flushChain = flushChain.then(async (ok) => {
   534→   534→      if (!ok || stopRun) return false
   535→   535→      const ok2 = await run.flush(fields)
   536→   536→      if (!ok2) stopRun = true // 任务记录被删除 → 全池停止派发
   537→   537→      return ok2
   538→   538→    })
   539→   539→    return flushChain
   540→   540→  }
   541→   541→
   542→   542→  /** 每处理完一个章节后要刷出的进度字段（不含任务级成果计数） */
   543→   543→  const progressFields = (): TaskFlushFields =>
   544→   544→    opts.trackTotal
   545→   545→      ? { done }
   546→   546→      : {
   547→   547→          chaptersDone: opts.chapterProgress?.done ?? 0,
   548→   548→          chaptersTotal: opts.chapterProgress?.total ?? 0,
   549→   549→        }
   550→   550→
   551→   551→  const processUnit = async ({ job, book }: { job: ChapterFillJob; book: Phase2Book }): Promise<void> => {
   552→   552→    if (book.vanished) {
   553→   553→      // P1-3：书已消失——剩余章节不再向源站发请求，按失败记账推进进度（骨架已随书级联删除）
   554→   554→      failed++
   555→   555→      book.failed++
   556→   556→      done++
   557→   557→      if (opts.chapterProgress) opts.chapterProgress.done++
   558→   558→      return
   559→   559→    }
   560→   560→    if (job.kind === 'skip') {
   561→   561→      // 已存在且有内容：不重抓，仅推进进度（与旧版「已存在，跳过」口径一致）
   562→   562→      skipped++
   563→   563→      done++
   564→   564→      if (opts.chapterProgress) opts.chapterProgress.done++
   565→   565→      run.log(`章节「${job.refTitle.slice(0, 30)}」已存在，跳过`)
   566→   566→      await scheduleFlush(progressFields())
   567→   567→      return
   568→   568→    }
   569→   569→
   570→   570→    if (!(await alive())) {
   571→   571→      stopRun = true
   572→   572→      return
   573→   573→    }
   574→   574→    run.log(`抓取章节「${(job.refTitle || job.url).slice(0, 36)}」`)
   575→   575→    // Referer 链：章节页带书页来路（站点常见「书页→章节」导航校验）；
   576→   576→    // fetchChapterPaged：同章分页（…_2.html / …/2.html）自动翻页拼接，避免长章只存半页。
   577→   577→    // 熔断感知：章节撞熔断（站点临时限流封禁）时同主机并发 worker 合流等待冷却后自动重试，
   578→   578→    // 而非把剩余章节全部烧成失败（Task 36 复盘：ggd66 一次临时封禁曾致 232 章被永久记失败）
   579→   579→    const circuitFetch = (): ReturnType<typeof fetchChapterPaged> =>
   580→   580→      withCircuitRetry(hostOf(job.url), () => fetchChapterPaged(job.url, rule, book.bookUrl), {
   581→   581→        detect: (r) => (r.ok ? null : circuitWaitMs(r.error, r.circuitRetryAfterMs)),
   582→   582→        isAlive: alive,
   583→   583→        onWait: (ms, cycle, max) =>

   717→   717→async function runSingle(run: Run, task: TaskRecord, rule: LoadedRule): Promise<void> {
   718→   718→  // Phase 1：书页+目录 → 书籍入库 + 章节骨架（total=骨架章节数）

   726→   726→  if (!meta.ok) {
   727→   727→    await finalize(run, 'failed', meta.message || '书页提取失败')

   734→   734→    rule,
   735→   735→    { trackTotal: true, concurrency: task.concurrency },
   736→   736→  )
   737→   737→  if (res.canceled) {
   738→   738→    await finalize(run, 'canceled', '任务已取消')
   739→   739→    return
   740→   740→  }
   741→   741→  if (res.failed === 0) {
   742→   742→    if (res.filled > 0) {
   743→   743→      await finalize(run, 'success', `采集完成：${res.filled} 章`)

   749→   749→  if (res.filled > 0) {

   752→   752→      'partial',
   753→   753→      res.vanishedBooks > 0
   754→   754→        ? `${res.filled} 章成功 / ${res.failed} 章失败（书籍在采集中被删除，剩余章节中止）`
   755→   755→        : `${res.filled} 章成功 / ${res.failed} 章失败`,
   756→   756→    )
   757→   757→    return
   758→   758→  }
   759→   759→  await finalize(
   760→   760→    run,
   761→   761→    'failed',
   762→   762→    res.vanishedBooks > 0 ? '书籍已消失（采集中被删除），章节采集中止' : '章节采集全部失败',
   763→   763→  )

   924→   924→      'failed',
   925→   925→      res.vanishedBooks > 0 && failBooksFinal === 0 ? '书籍已消失（采集中被删除），无书籍采集成功' : '无书籍采集成功',
   926→   926→    )
   927→   927→    return
   928→   928→  }
   929→   929→  if (failBooksFinal > 0) {
   930→   930→    await finalize(run, 'partial', `${okBooks} 本成功 / ${failBooksFinal} 本失败，入库 ${res.filled} 章${vanishedNote}`)
   931→   931→    return
   932→   932→  }
   933→   933→  await finalize(
   934→   934→    run,
   935→   935→    'success',
   936→   936→    res.failed > 0
   937→   937→      ? `范围采集完成：共 ${total} 本，入库 ${res.filled} 章，失败 ${res.failed} 章${vanishedNote}`
   938→   938→      : `范围采集完成：共 ${total} 本，入库 ${res.filled} 章${vanishedNote}`,
   939→   939→  )
   940→   940→}

   944→   944→/** 终态写入：仅当仍处于 running 时写终态；已被取消/删除则只保留日志 */
   945→   945→async function finalize(run: Run, status: string, message: string): Promise<void> {
   946→   946→  const cur = await db.scrapeTask
   947→   947→    .findUnique({ where: { id: run.taskId }, select: { status: true } })
   948→   948→    .catch(() => null)
   949→   949→  if (!cur) return
   950→   950→  if (cur.status === 'running') {
   951→   951→    await db.scrapeTask
   952→   952→      .updateMany({
   953→   953→        where: { id: run.taskId, status: 'running' },
   954→   954→        data: { status, message: message.slice(0, 500), log: run.logText() },
   955→   955→      })
   956→   956→      .catch(() => {})
   957→   957→  } else {
   958→   958→    await db.scrapeTask
   959→   959→      .updateMany({ where: { id: run.taskId }, data: { log: run.logText() } })
   960→   960→      .catch(() => {})
   961→   961→  }
   962→   962→}

   964→   964→async function runTask(taskId: number): Promise<void> {
   965→   965→  const run = new Run(taskId)
   966→   966→  try {
   967→   967→    // pending → running 条件更新：pending 阶段已被取消/删除的任务不再启动
   968→   968→    const started = await db.scrapeTask
   969→   969→      .updateMany({ where: { id: taskId, status: 'pending' }, data: { status: 'running' } })
   970→   970→      .catch(() => ({ count: 0 }))
   971→   971→    if (started.count === 0) return
   972→   972→    const task = await db.scrapeTask.findUnique({ where: { id: taskId } }).catch(() => null)
   973→   973→    if (!task) return
   974→   974→    run.ruleId = task.ruleId ?? null // 封面回填链路：入库时记入 Novel.sourceRuleId
   975→   975→    const record: TaskRecord = {
   976→   976→      id: task.id,
   977→   977→      mode: task.mode,
   978→   978→      targetUrl: task.targetUrl,
   979→   979→      pages: task.pages,
   980→   980→      startPage: task.startPage,
   981→   981→      concurrency: clampConcurrency(task.concurrency),
   982→   982→      ruleId: task.ruleId,
   983→   983→    }
   984→   984→    const rule = await loadRule(record.ruleId)
   985→   985→    run.log(
   986→   986→      record.mode === 'list'
   987→   987→        ? `任务开始（范围采集）目标: ${record.targetUrl}，起始页: ${record.startPage}，页数: ${record.pages}，并发: ${record.concurrency}`
   988→   988→        : `任务开始（单本采集）目标: ${record.targetUrl}，并发: ${record.concurrency}`,
   989→   989→    )
   990→   990→    run.log(
   991→   991→      record.ruleId
   992→   992→        ? `使用规则「${rule.name ?? record.ruleId}」（charset=${rule.charset ?? 'auto'}）`
   993→   993→        : '未使用规则，依赖引擎内置启发式提取',
   994→   994→    )
   995→   995→    await run.flush()
   996→   996→    if (record.mode === 'list') await runList(run, record, rule)
   997→   997→    else await runSingle(run, record, rule)
   998→   998→  } catch (e) {
   999→   999→    run.log(`发生未预期错误: ${e instanceof Error ? e.message.slice(0, 200) : String(e).slice(0, 200)}`)
  1000→  1000→    await finalize(run, 'failed', '任务执行异常')
  1001→  1001→  }
  1002→  1002→}

  1004→  1004→/**
  1005→  1005→ * fire-and-forget 入口：API 路由创建任务后调用，不 await。
  1006→  1006→ * 多任务天然并行：每个任务独立 worker promise，互不阻塞；
  1007→  1007→ * 对同一目标站的压力由引擎层域名限速统一收敛。
  1008→  1008→ */
  1009→  1009→export function triggerScrapeTask(taskId: number): void {
  1010→  1010→  if (running.has(taskId)) return
  1011→  1011→  running.add(taskId)
  1012→  1012→  void runTask(taskId)
  1013→  1013→    .catch((e: unknown) => {
  1014→  1014→      console.error(`[scrape-worker] task ${taskId} 兜底异常:`, e instanceof Error ? e.message : e)
  1015→  1015→    })
  1016→  1016→    .finally(() => {
  1017→  1017→      running.delete(taskId)
  1018→  1018→    })
  1019→  1019→}
