   424→/* Category：左 190 榜单侧栏 + 右 760 六列数据表 + 翻页                   */
   425→/* ==================================================================== */
   426→
   427→type SortKey = 'latest' | 'clicks' | 'words'
   428→type StatusKey = 'all' | 'serial' | 'finished'
   429→type PresetKey = 'clicks' | 'latest' | 'finished' | 'words'
   430→
   431→function CategoryView({
   432→  navigate,
   433→  categoryId,
   434→  page = 1,
   435→}: ViewProps & { categoryId?: number; page?: number }) {
   436→  const { data: cats } = useCategories()
   437→  const { data: home } = useHomeData()
   438→  const [sort, setSort] = useState<SortKey>('latest')
   439→  const [status, setStatus] = useState<StatusKey>('all')
   440→  const cur = page > 0 ? page : 1
   441→
   442→  const { data, isLoading, isError, refetch } = useNovels({
   443→    categoryId,
   444→    page: cur,
   445→    pageSize: 20,
   446→    sort,
   447→    status: status === 'all' ? undefined : status,
   448→  })
   449→
   450→  const catName =
   451→    categoryId != null ? (cats ?? []).find((c) => c.id === categoryId)?.name : undefined
   452→  const goPage = (p: number) => navigate({ name: 'category', categoryId, page: p })
   453→  const applySort = (s: SortKey) => {
   454→    setSort(s)
   455→    setStatus('all')
   456→    if (cur !== 1) goPage(1)
   457→  }
   458→  const applyStatus = (s: StatusKey) => {
   459→    setStatus(s)
   460→    if (cur !== 1) goPage(1)
   461→  }
   462→  const applyPreset = (p: PresetKey) => {
   463→    if (p === 'finished') {
   464→      setSort('latest')
   465→      setStatus('finished')
   466→    } else {
   467→      setSort(p)
   468→      setStatus('all')
   469→    }
   470→    if (cur !== 1) goPage(1)
   471→  }
   472→
   473→  return (
   474→    <div className="mx-auto w-full max-w-[960px] px-2 py-3">
   475→      <Crumbs
   476→        items={[{ label: '首页', view: { name: 'home' } }, { label: catName ?? '全部小说' }]}
   477→        navigate={navigate}
   478→      />
   479→
   480→      <div className="mt-3 grid gap-3 md:grid-cols-[190px_minmax(0,1fr)]">
   481→        {/* 左侧栏 190：推荐榜 + 榜单直达 */}
   482→        <aside className="order-2 min-w-0 space-y-3 md:order-1">
   483→          <Block title="会员推荐榜" bodyClass="px-2 py-1">
   484→            <RankRows novels={(home?.rankings.clicks ?? []).slice(0, 10)} navigate={navigate} limit={10} />
   485→          </Block>

   613→function loadShelf(): number[] {
   614→  if (typeof window === 'undefined') return []
   615→  try {

   622→function toggleShelf(id: number): boolean {
   623→  if (typeof window === 'undefined') return false
   624→  try {
   625→    const arr = loadShelf()
   626→    const has = arr.includes(id)
   627→    window.localStorage.setItem(SHELF_KEY, JSON.stringify(has ? arr.filter((x) => x !== id) : [...arr, id]))
   628→    return !has
   629→  } catch {
   630→    return false
   631→  }
   632→}
