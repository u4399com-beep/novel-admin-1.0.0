    97→    97→ */

   100→   100→async function recoverStaleTasks(): Promise<void> {
   101→   101→  if (gWorker.__scrapeRecovered) return
   102→   102→  gWorker.__scrapeRecovered = true // 同步置位：跨实例/跨 HMR 防重入
   103→   103→  try {
   104→   104→    const stale = await db.scrapeTask.findMany({
   105→   105→      where: {
   106→   106→        status: { in: ['pending', 'running'] },
   107→   107→        updatedAt: { lt: new Date(Date.now() - ZOMBIE_STALE_MS) },
   108→   108→      },
   109→   109→      select: { id: true, log: true },
   110→   110→    })
   111→   111→    let recovered = 0
   112→   112→    for (const t of stale) {
   113→   113→      if (running.has(t.id)) continue // 本进程内存中仍有实例在执行（共享 Set 时双保险）
   114→   114→      const line = `[${new Date().toTimeString().slice(0, 8)}] 心跳停摆超 5 分钟，判定僵尸任务并回收`
   115→   115→      const log = `${t.log ? `${t.log}\n` : ''}${line}`
   116→   116→      const res = await db.scrapeTask
   117→   117→        .updateMany({
   118→   118→          where: { id: t.id, status: { in: ['pending', 'running'] } },
   119→   119→          data: {
   120→   120→            status: 'failed',
   121→   121→            message: '任务心跳超时，自动回收',
   122→   122→            // 与 Run 同规约：只保留最近 100 行
   123→   123→            log: log.split('\n').slice(-MAX_LOG_LINES).join('\n'),
   124→   124→          },
   125→   125→        })
   126→   126→        .catch(() => null)
   127→   127→      if (res && res.count > 0) recovered++
   128→   128→    }
   129→   129→    if (recovered > 0) console.log(`[scrape-worker] 僵尸任务回收: ${recovered} 条`)
   130→   130→  } catch {
   131→   131→    // 回收失败不影响模块加载；标志已置位，等待下次进程重启重试（代价可接受：僵尸多驻留一轮）
   132→   132→  }
   133→   133→}

   136→   136→/** 协作式取消检查：记录不存在视为取消；DB 瞬时错误不误判为取消（fail-open） */
   137→   137→async function isCanceled(taskId: number): Promise<boolean> {
   138→   138→  const t = await db.scrapeTask
   139→   139→    .findUnique({ where: { id: taskId }, select: { status: true } })
   140→   140→    .catch(() => undefined)
   141→   141→  if (t === undefined) return false // 查询失败 ≠ 被取消，避免瞬时 DB 错误误停任务
   142→   142→  return !t || t.status === 'canceled'
   143→   143→}

   145→   145→/**
   146→   146→ * 并行池：cursor 抢占式派发，concurrency 路 worker 持续取件直至耗尽。
   147→   147→ * fn 内部通过 stop signal 提前收敛（停止派发，在途任务跑完当前单元后自然退出）。
   148→   148→ */
   149→   149→async function runPool<T>(items: T[], concurrency: number, fn: (item: T, index: number) => Promise<void>): Promise<void> {
   150→   150→  let cursor = 0
   151→   151→  const workers = Array.from({ length: Math.max(1, Math.min(concurrency, items.length)) }, async () => {
   152→   152→    for (;;) {
   153→   153→      const i = cursor++
   154→   154→      if (i >= items.length) return
   155→   155→      await fn(items[i], i)
   156→   156→    }
   157→   157→  })
   158→   158→  await Promise.all(workers)
   159→   159→}

   162→   162→function pageVariants(url: string, k: number): string[] {
   163→   163→  const out: string[] = []
   164→   164→  try {
   165→   165→    const u = new URL(url)
   166→   166→    u.searchParams.set('page', String(k))
   167→   167→    out.push(u.toString())
   168→   168→    const p = new URL(url)
   169→   169→    p.pathname = `${p.pathname.replace(/\/+$/, '')}/page/${k}`
   170→   170→    p.search = ''
   171→   171→    if (p.toString() !== u.toString()) out.push(p.toString())
   172→   172→  } catch {
   173→   173→    out.push(url.includes('?') ? `${url}&page=${k}` : `${url}?page=${k}`)
   174→   174→    out.push(`${url.replace(/\/+$/, '')}/page/${k}`)
   175→   175→  }
   176→   176→  return [...new Set(out)]
   177→   177→}

   195→   195→  run: Run,

   205→   205→    canceled: kind === 'canceled',

   218→   218→    detect: (r) => (r.ok ? null : circuitWaitMs(r.error, r.circuitRetryAfterMs)),

   222→   222→  })
   223→   223→  if (!page.ok) {
   224→   224→    run.log(`书页提取失败: ${page.error}`)

   226→   226→  }

   229→   229→  // 防畸形载荷（引擎版本错位等）在任务中途抛 TypeError 把整任务拖成 failed
   230→   230→  if (typeof book.title !== 'string') book.title = ''
   231→   231→  if (!Array.isArray(book.chapters)) book.chapters = []
   232→   232→  if (typeof book.description !== 'string') book.description = ''
   233→   233→  if (typeof book.author !== 'string') book.author = ''

   238→   238→  // 实体解码（&#091；类全角分号变体）、字面 \n 还原、残缺标签剥除、简介样板句/SEO 伪简介处理
   239→   239→  book.title = cleanTextField(book.title)
   240→   240→  book.author = cleanTextField(book.author)
   241→   241→  book.description = cleanDescriptionField(book.description)
   242→   242→  book.category = cleanTextField(book.category)
   243→   243→  book.status = cleanTextField(book.status)
   244→   244→  run.log(
   245→   245→    `书页提取成功：《${book.title.slice(0, 40)}》${book.author ? ` / ${book.author.slice(0, 20)}` : ''}，章节链接 ${book.chapterCount} 条`,
   246→   246→  )
   247→   247→
   248→   248→  // ---- 完整目录页二次提取（bookRule.catalogLinkSelector，如 23qb 新模板书页仅含最新几章）----
   249→   249→  let allRefs = book.chapters
   250→   250→  const catalogSel = rule.bookRule.catalogLinkSelector
   251→   251→  if (typeof catalogSel === 'string' && catalogSel) {
   252→   252→    if (await isCanceled(run.taskId)) return fail('任务已取消', 'canceled')
   253→   253→    if (book.catalogUrl) {
   254→   254→      run.log(`发现完整目录页 ${book.catalogUrl.slice(0, 100)}，尝试整目提取…`)

   258→   258→        hostOf(catalogUrl),

   266→   266→        },

   268→   268→      const catalogRefs = catalog.refs

   274→   274→      }
   275→   275→    } else {

   277→   277→    }
   278→   278→  }
   279→   279→
   280→   280→  if (await isCanceled(run.taskId)) return fail('任务已取消', 'canceled')
   281→   281→
   282→   282→  // ---- 乱序重排（充分考虑分卷：同号不同卷不跨卷错排，卷名随章节移动）----
   283→   283→  // 源站目录常见「最新章节块（新→旧）+ 正文块（旧→新）」或整本倒序；直接按抓取序编 idx 会错乱。
   284→   284→  // 引擎响应未经 schema 校验，逐条形态兑底后再重排。
   285→   285→  allRefs = (Array.isArray(allRefs) ? allRefs : []).map((c) => ({
   286→   286→    title: typeof c?.title === 'string' ? c.title : '',
   287→   287→    url: typeof c?.url === 'string' ? c.url : null,
   288→   288→    volume: typeof c?.volume === 'string' && c.volume.trim() ? c.volume.trim().slice(0, 50) : undefined,
   289→   289→  })) as ChapterRef[]
   290→   290→  const rr = reorderChapterRefs(allRefs)
   291→   291→  if (rr.reordered) {
   292→   292→    allRefs = rr.refs
   293→   293→    run.log(`目录乱序重排：${rr.note}`)
   294→   294→  }
   295→   295→
   296→   296→  // ---- 分类 ----
   297→   297→  let categoryId: number
   298→   298→  try {
   299→   299→    categoryId = await ensureCategory(book.category)
   300→   300→  } catch (e) {
   301→   301→    const msg = e instanceof Error ? e.message : '分类处理失败'
   302→   302→    run.log(msg)
   303→   303→    return fail(msg)
   304→   304→  }
   305→   305→
   306→   306→  // ---- 书籍 upsert（title+author 查重；DB 层 @@unique([title,author]) 兜底并发）----
   307→   307→  const up = await upsertBook(run, book, categoryId, rule.proxy)
   308→   308→  if (!up.ok) {
   309→   309→    // P1-4：记录级 DB 失败（up.hard=true，如书籍在采集中被删除）按书籍级硬失败处理，
   310→   310→    // list 模式不再停整个任务（旧实现借 canceled 通道把整任务收成 canceled，语义失真）；
   311→   311→    // 书籍级普通失败（空标题等）同样只算本书失败
   312→   312→    return fail(up.message, up.hard ? 'hard' : 'plain')
   313→   313→  }
   314→   314→  // 采集的同时取搜索引擎下拉词 → 绑定书籍 + 生成 PSEO 书籍页（fire-and-forget）
   315→   315→  scheduleBindBookSuggestKeywords(run, up.novelId, up.title || book.title, up.createdNew)
   316→   316→
   317→   317→  // ---- 章节列表准备 ----
   318→   318→  const refs = allRefs.filter(
   319→   319→    (c): c is ChapterRef & { url: string } => !!c && typeof c.url === 'string' && !!c.url,
   320→   320→  )
   321→   321→  if (refs.length === 0) {
   322→   322→    run.log('未提取到任何有效章节链接')
   323→   323→    return { ok: true, canceled: false, hardFailed: false, novelId: up.novelId, refsTotal: 0, skipped: 0, jobs: [], message: '书籍已入库（未提取到章节链接）' }
   324→   324→  }
   325→   325→  // 同目录内预去重（骨架入库前完成）：按清洗后标题优先、URL 兜底去重，防同批重复入库
   326→   326→  const seenRef = new Set<string>()
   327→   327→  const uniqueRefs: (ChapterRef & { url: string })[] = []
   328→   328→  for (const c of refs) {
   329→   329→    const key = c.title.trim() ? `t:${cleanTextField(c.title)}` : `u:${c.url}`
   330→   330→    if (seenRef.has(key)) continue
   331→   331→    seenRef.add(key)
   332→   332→    uniqueRefs.push(c)
   333→   333→  }
   334→   334→  if (uniqueRefs.length < refs.length) run.log(`目录内去重：${refs.length} → ${uniqueRefs.length} 条（无数量上限，全量采集）`)
   335→   335→
   336→   336→  if (await isCanceled(run.taskId)) return fail('任务已取消', 'canceled')
   337→   337→
   338→   338→  // ---- 已有章节盘点（不含 content 全文；wordCount 区分内容章节与骨架） ----
   339→   339→  const existingChapters = await db.chapter
   340→   340→    .findMany({
   341→   341→      where: { novelId: up.novelId },
   342→   342→      select: { id: true, idx: true, title: true, volume: true, wordCount: true },
   343→   343→    })
   344→   344→    .catch(() => [] as { id: number; idx: number; title: string; volume: string; wordCount: number }[])
   345→   345→  // 去重键与入库标题共用同一清洗规约（cleanTextField 幂等）：存量脏标题与本次净标题可命中比对，
   346→   346→  // 避免清洗规约升级后同一章以两种写法重复入库
   347→   347→  const titleGroups = new Map<string, { id: number; idx: number; wordCount: number }[]>()
   348→   348→  for (const c of existingChapters) {
   349→   349→    const k = cleanTextField(c.title)
   350→   350→    const arr = titleGroups.get(k)
   351→   351→    if (arr) arr.push({ id: c.id, idx: c.idx, wordCount: c.wordCount })
   352→   352→    else titleGroups.set(k, [{ id: c.id, idx: c.idx, wordCount: c.wordCount }])
   353→   353→  }
   354→   354→  const maxAgg = await db.chapter
   355→   355→    .aggregate({ where: { novelId: up.novelId }, _max: { idx: true } })
   356→   356→    .catch(() => ({ _max: { idx: null as number | null } }))
   357→   357→  // 骨架槽位基准：现有最大 idx+1；每条按目录位次预分配（base+位次），保序且并行安全
   358→   358→  const baseIdx = (maxAgg._max.idx ?? 0) + 1
   359→   359→
   360→   360→  // ---- 目录逐条分类：skip（已有内容）/ fill（已有空骨架或新骨架）/ 骨架待建 ----
   361→   361→  const jobs: ChapterFillJob[] = []
   362→   362→  /** 待创建骨架（jobIndex 回指 jobs 中对应的 fill 任务，创建结果落地后回填定位） */
   363→   363→  const pendingSkeletons: { jobIndex: number; idx: number; title: string; volume: string; refTitle: string }[] = []
   364→   364→  /** 已存在章节的卷名回填队列（title → volume；仅唯一标题且存量为空卷时写入） */
   365→   365→  const volumeBackfill = new Map<string, string>()
   366→   366→  let skipped = 0
   367→   367→
   368→   368→  for (let position = 0; position < uniqueRefs.length; position++) {
   369→   369→    const ref = uniqueRefs[position]
   370→   370→    const refTitle = cleanTextField(String(ref.title ?? '')).slice(0, 200)
   371→   371→    if (refTitle && titleGroups.has(refTitle)) {
   372→   372→      const matches = titleGroups.get(refTitle)!
   373→   373→      if (matches.some((m) => m.wordCount > 0)) {
   374→   374→        // 已存在且有内容：跳过不重抓（Phase 2 仅作进度占位）
   375→   375→        skipped++
   376→   376→        jobs.push({ kind: 'skip', novelId: up.novelId, chapterId: matches[0].id, idx: matches[0].idx, url: '', refTitle, volume: '' })
   377→   377→        // 分卷补齐：本次源站目录携带卷名、存量章节为空卷且标题在全书中唯一时，回填卷名
   378→   378→        const rv = ref.volume
   379→   379→        if (rv && matches.length === 1 && !volumeBackfill.has(refTitle)) volumeBackfill.set(refTitle, rv)
   380→   380→        continue
   381→   381→      }
   382→   382→      // 已存在空骨架（上次任务未抓到内容）：保持骨架不重建，Phase 2 重抓填充（不跳过）
   383→   383→      const target = matches.reduce((a, b) => (a.idx <= b.idx ? a : b))
   384→   384→      jobs.push({ kind: 'fill', novelId: up.novelId, chapterId: target.id, idx: target.idx, url: ref.url, refTitle, volume: ref.volume ?? '' })
   385→   385→      continue
   386→   386→    }
   387→   387→    // 不存在 → 骨架占位（idx=目录位次预分配保序），创建结果落地后回填 chapterId
   388→   388→    const idx = baseIdx + position
   389→   389→    pendingSkeletons.push({ jobIndex: jobs.length, idx, title: refTitle || `第${idx}章`, volume: ref.volume ?? '', refTitle })
   390→   390→    jobs.push({ kind: 'fill', novelId: up.novelId, chapterId: null, idx, url: ref.url, refTitle, volume: ref.volume ?? '' })
   391→   391→  }
   392→   392→
   393→   393→  // ---- 骨架批量入库：createMany 快路径；整批失败（并发撞 (novelId,idx) 等）逐条 create 兜底 ----
   394→   394→  if (pendingSkeletons.length > 0) {
   395→   395→    let batchErr: unknown = null
   396→   396→    try {
   397→   397→      await db.chapter.createMany({
   398→   398→        data: pendingSkeletons.map((s) => ({
   399→   399→          novelId: up.novelId,
   400→   400→          idx: s.idx,
   401→   401→          title: s.title,
   402→   402→          content: '',
   403→   403→          wordCount: 0,
   404→   404→          volume: s.volume,
   405→   405→        })),
   406→   406→      })
   407→   407→    } catch (e) {
   408→   408→      batchErr = e
   409→   409→    }
   410→   410→    if (batchErr !== null) {
   411→   411→      // P1-3：整批撞 FK（书在采集中被删除，Novel←Chapter onDelete:Cascade）→ 立即中止本书，
   412→   412→      // 不再逐条空转（旧行为对已删书逐章重试刷屏）
   413→   413→      if (isRecordMissingError(batchErr)) {
   414→   414→        run.log(`书籍已消失（采集中被删除），中止本书章节骨架入库`)
   415→   415→        return fail('书籍已消失（采集中被删除）', 'hard')
   416→   416→      }
   417→   417→      // 并发撞 (novelId,idx) 等其他批量失败 → 逐条 create 兜底（createChapterSkeleton
   418→   418→      // 撞唯一约束查回已有记录按「已存在」处理）
   419→   419→      for (const s of pendingSkeletons) {
   420→   420→        if (await isCanceled(run.taskId)) return fail('任务已取消', 'canceled')
   421→   421→        const job = jobs[s.jobIndex]
   422→   422→        const sk = await createChapterSkeleton(run, up.novelId, s.idx, { title: s.title, volume: s.volume }, s.refTitle)
   423→   423→        if (!sk.ok) {
   424→   424→          // P1-3：逐条也撞 FK → 书已消失，立即中止本书（不再把剩余全部骨架标记失败）
   425→   425→          if (sk.missing) {
   426→   426→            run.log(`书籍已消失（采集中被删除），中止本书章节骨架入库`)
   427→   427→            return fail('书籍已消失（采集中被删除）', 'hard')
   428→   428→          }
   429→   429→          run.log(`骨架章节「${s.title.slice(0, 30)}」入库失败: ${sk.message.slice(0, 120)}`)
   430→   430→          job.kind = 'create' // 无骨架可填：Phase 2 抓到正文后走 storeChapter 兜底新建
   431→   431→          continue
   432→   432→        }
   433→   433→        job.idx = sk.idx
   434→   434→        if (!sk.created) {
   435→   435→          // 撞唯一约束查回已有记录 → 按「已存在」处理（并发双任务同书场景）
   436→   436→          if (sk.existingHasContent) {
   437→   437→            job.kind = 'skip'
   438→   438→            job.url = ''
   439→   439→            skipped++
   440→   440→          }
   441→   441→          job.chapterId = sk.existingId
   442→   442→        }
   443→   443→      }
   444→   444→    }
   445→   445→  }
   446→   446→
   447→   447→  // ---- 已存在章节的卷名回填（逐卷 updateMany，仅在唯一标题且存量空卷时入队）----
   448→   448→  if (volumeBackfill.size > 0) {
   449→   449→    let backfilled = 0
   450→   450→    for (const [title, volume] of volumeBackfill) {
   451→   451→      const res = await db.chapter
   452→   452→        .updateMany({ where: { novelId: up.novelId, title, volume: '' }, data: { volume } })
   453→   453→        .catch(() => null)
   454→   454→      if (res) backfilled += res.count
   455→   455→    }
   456→   456→    if (backfilled > 0) run.log(`分卷回填：${backfilled} 个已存在章节补记卷名`)
   457→   457→  }
   458→   458→
   459→   459→  // ---- 进度分母 ----
   460→   460→  if (opts.trackTotal) {
   461→   461→    // single 模式：total=骨架章节数（分母），done 随 Phase 2 逐章递增
   462→   462→    //（flush false=任务记录已删，立即停）
   463→   463→    if (!(await run.flush({ total: uniqueRefs.length }))) return fail('任务记录已删除', 'canceled')
   464→   464→  }
   465→   465→
   466→   466→  const toFetch = jobs.length - skipped
   467→   467→  run.log(`目录骨架就绪：${uniqueRefs.length} 章（已存在 ${skipped} 章，待抓取 ${toFetch} 章）`)
   468→   468→  return { ok: true, canceled: false, hardFailed: false, novelId: up.novelId, refsTotal: uniqueRefs.length, skipped, jobs, message: '' }
   469→   469→}
   470→   470→
   471→   471→// ==================== Phase 2：跨书批量章节内容填充 ====================
   472→   472→
   473→   473→/** Phase 2 单书上下文（统计就地回填，供书籍级成败判定与日志节流） */
   474→   474→interface Phase2Book {
   475→   475→  novelId: number
   476→   476→  title: string
   477→   477→  /** 章节页 Referer 来路（书页 URL，站点常见「书页→章节」导航校验） */
   478→   478→  bookUrl: string
   479→   479→  jobs: ChapterFillJob[]
   480→   480→  filled: number
   481→   481→  failed: number
   482→   482→  /** P1-3：书在采集中被删除（P2025/FK 实证）→ 本本书剩余章节不再向源站发请求 */
   483→   483→  vanished: boolean
   484→   484→  /** 清洗日志节流：每本书最多记 3 条，防止日志爆炸 */
   485→   485→  cleanLogCount: number
   486→   486→}
   487→   487→
   488→   488→interface Phase2Result {
   489→   489→  canceled: boolean
   490→   490→  filled: number
   491→   491→  failed: number
   492→   492→  skipped: number
   493→   493→  /** P1-3：采集中被删除的书籍数（finalize 消息提示用） */
   494→   494→  vanishedBooks: number
   495→   495→}
   496→   496→
   497→   497→/**
   498→   498→ * Phase 2：把 Phase 1 收集的全部章节平铺（按书分组顺序），runPool 章节级并发
   499→   499→ * 跨书批量抓取正文，update 填充骨架章节 content/wordCount/volume。
   500→   500→ *
   501→   501→ * - 抓取失败的骨架章节保留骨架不删（重跑任务时 wordCount=0 的骨架会被再次抓取填充）；
   502→   502→ * - novel.wordCount increment 只对抓取成功的章节（fill 填充在此增量，create 走
   503→   503→ *   storeChapter 内部增量），收尾对触及书籍 recalcNovelWordCount 全量校准；
   504→   504→ * - P1-3：填充/入库撞 P2025/FK（书在采集中被删除）→ 实证本书已消失后立即中止本书
   505→   505→ *   剩余章节（不再空抓源站），书籍级成败按失败计；
   506→   506→ * - 保留全部既有机制：熔断感知重试/瞬时失败单次重试/协作式取消/flush 串行链/
   507→   507→ *   入库前清洗/MAX_CONTENT_CHARS 截断。
   508→   508→ * - 进度：single 模式 done 随本章递增（total=骨架章节数已在 Phase 1 写入）；
   509→   509→ *   list 模式累加共享 chapterProgress（chaptersDone/chaptersTotal）。
   510→   510→ */
   511→   511→async function fillChapters(
   512→   512→  run: Run,
   513→   513→  books: Phase2Book[],
   514→   514→  rule: LoadedRule,
   515→   515→  opts: { trackTotal: boolean; chapterProgress?: { done: number; total: number }; concurrency: number },
   516→   516→): Promise<Phase2Result> {
   517→   517→  // 平铺（按书分组顺序）：任务单元携带所属书上下文（Referer 链/日志节流按书隔离）
   518→   518→  const units: { job: ChapterFillJob; book: Phase2Book }[] = []
   519→   519→  for (const b of books) for (const job of b.jobs) units.push({ job, book: b })
   520→   520→
   521→   521→  let filled = 0
   522→   522→  let failed = 0
   523→   523→  let skipped = 0
   524→   524→  let done = 0 // single 模式主口径进度（done/total=章节）
   525→   525→  let stopRun = false // 取消/记录删除 → 停止派发新章节，在途章节收尾后退出
   526→   526→  const touchedNovels = new Set<number>()
   527→   527→
   528→   528→  /** P1-3：防误杀——update 撞 P2025 也可能是单章被单独删除，实证 novel 已不存在才判定整书消失
   529→   529→   * （查询失败按仍存在处理，不因 DB 抖动中止本书） */
   530→   530→  const isNovelGone = async (novelId: number): Promise<boolean> => {
   531→   531→    const n = await db.novel.findUnique({ where: { id: novelId }, select: { id: true } }).catch(() => undefined)
   532→   532→    return n === null
   533→   533→  }
   534→   534→
   535→   535→  /** P1-3：标记本书已消失并中止剩余章节（只记一条日志防刷屏） */
   536→   536→  const markVanished = (book: Phase2Book): void => {
   537→   537→    if (book.vanished) return
   538→   538→    book.vanished = true
   539→   539→    run.log(`书籍《${book.title.slice(0, 30) || book.novelId}》已消失（采集中被删除），中止本书剩余章节抓取`)
   540→   540→  }
   541→   541→
   542→   542→  const alive = async (): Promise<boolean> => {
   543→   543→    if (stopRun) return false
   544→   544→    return !(await isCanceled(run.taskId))
   545→   545→  }
   546→   546→
   547→   547→  // flush 串行链：并行分支共享 Run 缓冲与进度字段，写盘串行化避免交错/竞态
   548→   548→  let flushChain = Promise.resolve(true)
   549→   549→  const scheduleFlush = (fields: TaskFlushFields): Promise<boolean> => {
   550→   550→    flushChain = flushChain.then(async (ok) => {
   551→   551→      if (!ok || stopRun) return false
   552→   552→      const ok2 = await run.flush(fields)
   553→   553→      if (!ok2) stopRun = true // 任务记录被删除 → 全池停止派发
   554→   554→      return ok2
   555→   555→    })
   556→   556→    return flushChain
   557→   557→  }
   558→   558→
   559→   559→  /** 每处理完一个章节后要刷出的进度字段（不含任务级成果计数） */
   560→   560→  const progressFields = (): TaskFlushFields =>
   561→   561→    opts.trackTotal
   562→   562→      ? { done }
   563→   563→      : {
   564→   564→          chaptersDone: opts.chapterProgress?.done ?? 0,
   565→   565→          chaptersTotal: opts.chapterProgress?.total ?? 0,
   566→   566→        }
   567→   567→
   568→   568→  const processUnit = async ({ job, book }: { job: ChapterFillJob; book: Phase2Book }): Promise<void> => {
   569→   569→    if (book.vanished) {
   570→   570→      // P1-3：书已消失——剩余章节不再向源站发请求，按失败记账推进进度（骨架已随书级联删除）
   571→   571→      failed++
   572→   572→      book.failed++
   573→   573→      done++
   574→   574→      if (opts.chapterProgress) opts.chapterProgress.done++
   575→   575→      return
   576→   576→    }
   577→   577→    if (job.kind === 'skip') {
   578→   578→      // 已存在且有内容：不重抓，仅推进进度（与旧版「已存在，跳过」口径一致）
   579→   579→      skipped++
   580→   580→      done++
   581→   581→      if (opts.chapterProgress) opts.chapterProgress.done++
   582→   582→      run.log(`章节「${job.refTitle.slice(0, 30)}」已存在，跳过`)
   583→   583→      await scheduleFlush(progressFields())
   584→   584→      return
   585→   585→    }
   586→   586→
   587→   587→    if (!(await alive())) {
   588→   588→      stopRun = true
   589→   589→      return
   590→   590→    }
   591→   591→    run.log(`抓取章节「${(job.refTitle || job.url).slice(0, 36)}」`)
   592→   592→    // Referer 链：章节页带书页来路（站点常见「书页→章节」导航校验）；

   710→   710→  await runPool(units, clampConcurrency(opts.concurrency), processUnit)
   711→   711→  const flushOk = await flushChain
   712→   712→
   713→   713→  // 收尾校准：对有成功填充的书籍重算字数（增量累计误差/并发双计兜底；失败静默）

   715→   715→
   716→   716→  if (stopRun || !flushOk) {
   717→   717→    run.log('任务已取消/记录已删除，停止章节填充')
   718→   718→    await run.flush({ ...progressFields(), chapters: run.counters.chapters }).catch(() => undefined)
   719→   719→  } else if (failed > 0) {
   720→   720→    run.log('抓取失败的章节已保留骨架（wordCount=0），重跑任务可自动续采')
   721→   721→  }
   722→   722→
   723→   723→  return {
   724→   724→    canceled: stopRun || !flushOk,
   725→   725→    filled,
   726→   726→    failed,
   727→   727→    skipped,
   728→   728→    vanishedBooks: books.filter((b) => b.vanished).length,
   729→   729→  }
   730→   730→}

   734→   734→async function runSingle(run: Run, task: TaskRecord, rule: LoadedRule): Promise<void> {
   735→   735→  // Phase 1：书页+目录 → 书籍入库 + 章节骨架（total=骨架章节数）
   736→   736→  const meta = await processBookMeta(run, task.targetUrl, rule, { trackTotal: true })
   737→   737→  if (meta.canceled) {
   738→   738→    run.log('任务已取消')
   739→   739→    await finalize(run, 'canceled', meta.message || '任务已取消')
   740→   740→    return
   741→   741→  }
   742→   742→  await run.flush({ created: run.counters.created, updated: run.counters.updated, chapters: run.counters.chapters })
   743→   743→  if (!meta.ok) {
   744→   744→    await finalize(run, 'failed', meta.message || '书页提取失败')
   745→   745→    return
   746→   746→  }
   747→   747→  // Phase 2：章节内容填充（done 随填充递增，保持 done/total=章节 口径）
   748→   748→  const res = await fillChapters(
   749→   749→    run,
   750→   750→    [{ novelId: meta.novelId, title: '', bookUrl: task.targetUrl, jobs: meta.jobs, filled: 0, failed: 0, vanished: false, cleanLogCount: 0 }],
   751→   751→    rule,
   752→   752→    { trackTotal: true, concurrency: task.concurrency },
   753→   753→  )
   754→   754→  if (res.canceled) {
   755→   755→    await finalize(run, 'canceled', '任务已取消')
   756→   756→    return
   757→   757→  }
   758→   758→  if (res.failed === 0) {
   759→   759→    if (res.filled > 0) {
   760→   760→      await finalize(run, 'success', `采集完成：${res.filled} 章`)
   761→   761→      return
   762→   762→    }
   763→   763→    await finalize(run, 'success', meta.jobs.length === 0 ? meta.message : '无新增章节（章节均已存在）')
   764→   764→    return
   765→   765→  }
   766→   766→  if (res.filled > 0) {
   767→   767→    await finalize(
   768→   768→      run,
   769→   769→      'partial',
   770→   770→      res.vanishedBooks > 0
   771→   771→        ? `${res.filled} 章成功 / ${res.failed} 章失败（书籍在采集中被删除，剩余章节中止）`
   772→   772→        : `${res.filled} 章成功 / ${res.failed} 章失败`,
   773→   773→    )
   774→   774→    return
   775→   775→  }
   776→   776→  await finalize(
   777→   777→    run,
   778→   778→    'failed',
   779→   779→    res.vanishedBooks > 0 ? '书籍已消失（采集中被删除），章节采集中止' : '章节采集全部失败',
   780→   780→  )
   781→   781→}

   928→   928→  }

   930→   930→  // ---- 书籍级成败判定：元数据失败 + 章节全部抓取失败（已存在跳过/无章节链接仍算成功）----
   931→   931→  const okBooks = books.filter((b) => b.filled > 0 || b.failed === 0).length
   932→   932→  const chapterFailBooks = books.filter((b) => b.filled === 0 && b.failed > 0).length
   933→   933→  const failBooksFinal = failBooks + chapterFailBooks
   934→   934→  // 消失提示后缀（P1-3：书在采集中被删除，中止本书剩余章节——与站点侧失败相区分）
   935→   935→  const vanishedNote = res.vanishedBooks > 0 ? `（${res.vanishedBooks} 本采集中被删除，已中止）` : ''
   936→   936→
   937→   937→  if (okBooks === 0) {
   938→   938→    await finalize(
   939→   939→      run,
   940→   940→      'failed',
   941→   941→      res.vanishedBooks > 0 && failBooksFinal === 0 ? '书籍已消失（采集中被删除），无书籍采集成功' : '无书籍采集成功',
   942→   942→    )
   943→   943→    return
   944→   944→  }
   945→   945→  if (failBooksFinal > 0) {
   946→   946→    await finalize(run, 'partial', `${okBooks} 本成功 / ${failBooksFinal} 本失败，入库 ${res.filled} 章${vanishedNote}`)
   947→   947→    return
   948→   948→  }
   949→   949→  await finalize(
   950→   950→    run,
   951→   951→    'success',
   952→   952→    res.failed > 0
   953→   953→      ? `范围采集完成：共 ${total} 本，入库 ${res.filled} 章，失败 ${res.failed} 章${vanishedNote}`
   954→   954→      : `范围采集完成：共 ${total} 本，入库 ${res.filled} 章${vanishedNote}`,
   955→   955→  )
   956→   956→}

   960→   960→/** 终态写入：仅当仍处于 running 时写终态；已被取消/删除则只保留日志 */
   961→   961→async function finalize(run: Run, status: string, message: string): Promise<void> {
   962→   962→  const cur = await db.scrapeTask
   963→   963→    .findUnique({ where: { id: run.taskId }, select: { status: true } })
   964→   964→    .catch(() => null)
   965→   965→  if (!cur) return
   966→   966→  if (cur.status === 'running') {
   967→   967→    await db.scrapeTask
   968→   968→      .updateMany({
   969→   969→        where: { id: run.taskId, status: 'running' },
   970→   970→        data: { status, message: message.slice(0, 500), log: run.logText() },
   971→   971→      })
   972→   972→      .catch(() => {})
   973→   973→  } else {
   974→   974→    await db.scrapeTask
   975→   975→      .updateMany({ where: { id: run.taskId }, data: { log: run.logText() } })
   976→   976→      .catch(() => {})
   977→   977→  }
   978→   978→}

   980→   980→async function runTask(taskId: number): Promise<void> {
   981→   981→  const run = new Run(taskId)
   982→   982→  try {
   983→   983→    // pending → running 条件更新：pending 阶段已被取消/删除的任务不再启动
   984→   984→    const started = await db.scrapeTask
   985→   985→      .updateMany({ where: { id: taskId, status: 'pending' }, data: { status: 'running' } })
   986→   986→      .catch(() => ({ count: 0 }))
   987→   987→    if (started.count === 0) return
   988→   988→    const task = await db.scrapeTask.findUnique({ where: { id: taskId } }).catch(() => null)
   989→   989→    if (!task) return
   990→   990→    run.ruleId = task.ruleId ?? null // 封面回填链路：入库时记入 Novel.sourceRuleId
   991→   991→    const record: TaskRecord = {
   992→   992→      id: task.id,
   993→   993→      mode: task.mode,
   994→   994→      targetUrl: task.targetUrl,
   995→   995→      pages: task.pages,
   996→   996→      startPage: task.startPage,
   997→   997→      concurrency: clampConcurrency(task.concurrency),
   998→   998→      ruleId: task.ruleId,
   999→   999→    }
  1000→  1000→    const rule = await loadRule(record.ruleId)
  1001→  1001→    run.log(
  1002→  1002→      record.mode === 'list'
  1003→  1003→        ? `任务开始（范围采集）目标: ${record.targetUrl}，起始页: ${record.startPage}，页数: ${record.pages}，并发: ${record.concurrency}`
  1004→  1004→        : `任务开始（单本采集）目标: ${record.targetUrl}，并发: ${record.concurrency}`,
  1005→  1005→    )
  1006→  1006→    run.log(
  1007→  1007→      record.ruleId
  1008→  1008→        ? `使用规则「${rule.name ?? record.ruleId}」（charset=${rule.charset ?? 'auto'}）`
  1009→  1009→        : '未使用规则，依赖引擎内置启发式提取',
  1010→  1010→    )
  1011→  1011→    await run.flush()
  1012→  1012→    if (record.mode === 'list') await runList(run, record, rule)
  1013→  1013→    else await runSingle(run, record, rule)
  1014→  1014→  } catch (e) {
  1015→  1015→    run.log(`发生未预期错误: ${e instanceof Error ? e.message.slice(0, 200) : String(e).slice(0, 200)}`)
  1016→  1016→    await finalize(run, 'failed', '任务执行异常')
  1017→  1017→  }
  1018→  1018→}

  1020→  1020→/**
  1021→  1021→ * fire-and-forget 入口：API 路由创建任务后调用，不 await。
  1022→  1022→ * 多任务天然并行：每个任务独立 worker promise，互不阻塞；
  1023→  1023→ * 对同一目标站的压力由引擎层域名限速统一收敛。
  1024→  1024→ */
  1025→  1025→export function triggerScrapeTask(taskId: number): void {
  1026→  1026→  if (running.has(taskId)) return
  1027→  1027→  running.add(taskId)
  1028→  1028→  void runTask(taskId)
  1029→  1029→    .catch((e: unknown) => {
  1030→  1030→      console.error(`[scrape-worker] task ${taskId} 兜底异常:`, e instanceof Error ? e.message : e)
  1031→  1031→    })
  1032→  1032→    .finally(() => {
  1033→  1033→      running.delete(taskId)
  1034→  1034→    })
  1035→  1035→}
