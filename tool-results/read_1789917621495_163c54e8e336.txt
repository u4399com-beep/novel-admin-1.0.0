   145→export function Category({
   146→  navigate,
   147→  categoryId,
   148→  page = 1,
   149→}: ViewProps & { categoryId?: number; page?: number }) {
   150→  const cats = useCategories()
   151→  const list = useNovels({ categoryId, page, pageSize: 20 })
   152→
   153→  const catName = cats.data?.find((c) => c.id === categoryId)?.name
   154→
   155→  return (
   156→    <div className="mx-auto max-w-[1180px] space-y-5 px-4 py-6">
   157→      <Crumb items={[{ label: '首页', view: { name: 'home' } }, { label: '小说书库' }]} navigate={navigate} />
   158→      <h1 className="border-l-4 border-[#2563eb] pl-4 text-[24px] leading-8 font-semibold text-[#1e293b]">
   159→        {catName ?? '小说书库'}
   160→        {cats.data ? (
   161→          <span className="ml-3 text-[13px] font-normal text-[#94a3b8]">
   162→            {cats.data.find((c) => c.id === categoryId)?.novelCount ?? list.data?.total ?? ''}
   163→          </span>
   164→        ) : null}
   165→      </h1>
   166→
   167→      {/* 筛选条 */}
   168→      <div className="flex flex-wrap gap-2 rounded-[10px] border border-[#e6edf7] bg-white p-4 shadow-[0_1px_3px_rgba(37,99,235,0.06)]">
   169→        <button
   170→          onClick={() => navigate({ name: 'category', page: 1 })}
   171→          className={
   172→            categoryId == null
   173→              ? 'cursor-pointer rounded-full bg-[#2563eb] px-4 py-1.5 text-[13px] text-white transition-colors duration-200'
   174→              : 'cursor-pointer rounded-full bg-[#f0f4fb] px-4 py-1.5 text-[13px] text-[#1e293b] transition-colors duration-200 hover:bg-[#e8f1ff] hover:text-[#2563eb]'
   175→          }
   176→        >
   177→          全部
   178→        </button>
   179→        {(cats.data ?? []).map((c) => (
   180→          <button
   181→            key={c.id}
   182→            onClick={() => navigate({ name: 'category', categoryId: c.id, page: 1 })}
   183→            className={
   184→              categoryId === c.id
   185→                ? 'cursor-pointer rounded-full bg-[#2563eb] px-4 py-1.5 text-[13px] text-white transition-colors duration-200'
   186→                : 'cursor-pointer rounded-full bg-[#f0f4fb] px-4 py-1.5 text-[13px] text-[#1e293b] transition-colors duration-200 hover:bg-[#e8f1ff] hover:text-[#2563eb]'
   187→            }
   188→          >
   189→            {c.name}
   190→          </button>
   191→        ))}
   192→      </div>
   193→
   194→      {list.isPending ? (
   195→        <CardGridSkeleton count={6} />
   196→      ) : list.isError ? (
   197→        <ErrorBox onRetry={() => list.refetch()} />
   198→      ) : (list.data?.list.length ?? 0) === 0 ? (
   199→        <EmptyBox text="该分类下暂无收录小说" />
   200→      ) : (
   201→        <>
   202→          <div className="grid gap-4 md:grid-cols-2 xl:grid-cols-3">
   203→            {list.data!.list.map((n) => (
   204→              <BookTextCard key={n.id} novel={n} navigate={navigate} />
   205→            ))}
   206→          </div>
   207→          <Pager
   208→            page={list.data!.page}
   209→            totalPages={list.data!.totalPages}
   210→            onGo={(p) => navigate({ name: 'category', categoryId, page: p })}
   211→          />
   212→        </>
   213→      )}
   214→    </div>
   215→  )
   216→}

   409→export function Toc({ navigate, novelId }: ViewProps & { novelId: number }) {

   411→  const { data: chapters, isPending, isError, refetch } = useChapters(novelId)

   413→  return (

   418→          { label: novel?.title ?? '…', view: { name: 'book', novelId } },
   419→          { label: '章节目录' },
   420→        ]}
   421→        navigate={navigate}
   422→      />
   423→      {/* 最新章节（全书倒数 12 章，新→旧） */}
   424→      {chapters && chapters.length > 0 && (
   425→        <section className="rounded-[10px] border border-[#e6edf7] bg-white p-6 shadow-[0_1px_3px_rgba(37,99,235,0.06)]">
   426→          <SectionTitle
   427→            title="最新章节"
   428→            small="最近更新 12 章 · 新→旧"
   429→            action={
   430→              <button
   431→                onClick={() => navigate({ name: 'book', novelId })}
   432→                className="shrink-0 cursor-pointer rounded-[10px] border-[1.5px] border-[#dbe4f0] bg-white px-5 py-2 text-[13px] text-[#1e293b] transition-colors duration-200 hover:border-[#2563eb] hover:text-[#2563eb]"
   433→              >
   434→                返回书页
   435→              </button>
   436→            }
   437→          />
   438→          <ChapterPills chapters={[...chapters].slice(-12).reverse()} navigate={navigate} />
   439→        </section>
   440→      )}
   441→
   442→      <section className="rounded-[10px] border border-[#e6edf7] bg-white p-6 shadow-[0_1px_3px_rgba(37,99,235,0.06)]">
   443→        <SectionTitle
   444→          title={novel ? `${novel.title} · 章节目录` : '章节目录'}
   445→          small={chapters ? `共 ${chapters.length} 章` : undefined}
   446→          action={
   447→            <button
   448→              onClick={() => navigate({ name: 'book', novelId })}
   449→              className="shrink-0 cursor-pointer rounded-[10px] border-[1.5px] border-[#dbe4f0] bg-white px-5 py-2 text-[13px] text-[#1e293b] transition-colors duration-200 hover:border-[#2563eb] hover:text-[#2563eb]"
   450→            >
   451→              返回书页
   452→            </button>
   453→          }
   454→        />
   455→        {isPending ? (
   456→          <PillSkeleton count={16} />
   457→        ) : isError ? (
   458→          <ErrorBox onRetry={() => refetch()} />
   459→        ) : (chapters?.length ?? 0) === 0 ? (
   460→          <EmptyBox text="暂无章节" />
   461→        ) : (
   462→          <ChapterPills chapters={chapters!} navigate={navigate} />
   463→        )}
   464→      </section>
   465→    </div>
   466→  )
   467→}

   469→/* ==================== 章节正文页（900px 阅读容器） ==================== */
   470→
   471→function AuthorWorks({
   472→  author,
   473→  excludeId,
   474→  navigate,
   475→}: {
   476→  author: string
   477→  excludeId: number
   478→  navigate: Nav
   479→}) {
   480→  const q = useNovels({ q: author, pageSize: 6 })
   481→  const rows: NovelListItem[] = (q.data?.list ?? []).filter((n) => n.id !== excludeId).slice(0, 3)
   482→  if (rows.length === 0) return null

   775→function SearchPanel({ query, navigate }: { query: string; navigate: Nav }) {
   776→  const [input, setInput] = useState(query)
   777→  const [page, setPage] = useState(1)
   778→  const list = useNovels({ q: query || undefined, page, pageSize: 20 })
   779→
   780→  const submit = () => {
   781→    const kw = input.trim()
   782→    if (kw) navigate({ name: 'search', query: kw })
   783→  }
   784→
   785→  return (
   786→    <div className="space-y-5">
   787→      <div className="flex gap-3 rounded-[10px] border border-[#e6edf7] bg-white p-4 shadow-[0_1px_3px_rgba(37,99,235,0.06)]">
   788→        <input
   789→          value={input}
   790→          onChange={(e) => setInput(e.target.value)}
   791→          onKeyDown={(e) => e.key === 'Enter' && submit()}
   792→          placeholder="输入书名 / 作者关键词"
   793→          className="h-10 min-w-0 flex-1 rounded-[10px] border border-[#dbe4f0] bg-[#f0f4fb] px-4 text-[14px] text-[#1e293b] outline-none transition-colors duration-200 placeholder:text-[#94a3b8] focus:border-[#2563eb] focus:bg-white"
   794→        />
   795→        <button
   796→          onClick={submit}
   797→          className="cursor-pointer rounded-[10px] bg-[#2563eb] px-8 text-[14px] text-white transition-colors duration-200 hover:bg-[#1d4ed8]"
   798→        >
   799→          搜索
   800→        </button>
   801→      </div>
   802→
   803→      <SectionTitle
   804→        title="搜索结果"
   805→        small={query ? `“${query}” · ${list.data?.total ?? 0} 条` : '请输入关键词'}
   806→      />
   807→
   808→      {list.isPending ? (
   809→        <CardGridSkeleton count={6} />
   810→      ) : list.isError ? (
   811→        <ErrorBox onRetry={() => list.refetch()} />
   812→      ) : (list.data?.list.length ?? 0) === 0 ? (
   813→        <EmptyBox text={`没有找到与“${query}”相关的小说，换个关键词试试`} />

   815→        <>
   816→          <div className="grid gap-4 md:grid-cols-2 xl:grid-cols-3">

   818→              <BookTextCard key={n.id} novel={n} navigate={navigate} />
   819→            ))}
   820→          </div>
   821→          <Pager
   822→            page={list.data!.page}
   823→            totalPages={list.data!.totalPages}
   824→            onGo={(p) => setPage(p)}
   825→          />
   826→        </>
   827→      )}
   828→    </div>
   829→  )
