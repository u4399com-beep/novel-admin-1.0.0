    46→// jsParseFloat 模拟 JS Number(string)：空白串→0，非法→NaN(false)
    47→func jsParseFloat(s string) (float64, bool) {
    48→        t := trimSpaceStr(s)
    49→        if t == "" {
    50→                return 0, true
    51→        }
    52→        f, err := strconv.ParseFloat(t, 64)
    53→        if err != nil {
    54→                return math.NaN(), false
    55→        }
    56→        return f, true
    57→}

    59→// numIsInt JS Number.isInteger 数值判定
    60→func numIsInt(f float64) bool {
    61→        if math.IsNaN(f) || math.IsInf(f, 0) {
    62→                return false
    63→        }
    64→        return f == math.Trunc(f)
    65→}

    67→// intFieldStrict 严格类型整数字段（JS Number.isInteger(v)：v 必须本身是 number）
    68→func intFieldStrict(v any) (int, bool) {
    69→        f, ok := v.(float64)
    70→        if !ok || !numIsInt(f) {
    71→                return 0, false
    72→        }
    73→        return int(f), true
    74→}

    76→// jsNumber JS Number() 对 JSON 值的宽式转换
    77→func jsNumber(v any) float64 {
    78→        switch x := v.(type) {
    79→        case float64:
    80→                return x
    81→        case bool:
    82→                if x {
    83→                        return 1
    84→                }
    85→                return 0
    86→        case nil:
    87→                return 0
    88→        case string:
    89→                f, _ := jsParseFloat(x)
    90→                return f
    91→        case []any:
    92→                if len(x) == 0 {
    93→                        return 0
    94→                }
    95→                if len(x) == 1 {
    96→                        return jsNumber(x[0])
    97→                }
    98→                return math.NaN()
    99→        default:
   100→                return math.NaN()
   101→        }
   102→}

   104→// parsePositiveInt 移植 src/lib/scrape/api-utils.ts parsePositiveInt：
   105→// Number(raw) 后必须为正整数，否则失败
   106→func parsePositiveInt(v any) (int64, bool) {
   107→        f := jsNumber(v)
   108→        if !numIsInt(f) || f <= 0 {
   109→                return 0, false
   110→        }
   111→        return int64(f), true

   114→// routeIntID 路径段 id 解析（Number(id) 须为整数，可负/零——与 TS Number.isInteger 对齐）。
   115→// 非法时写 400 {error: errMsg}。
   116→func routeIntID(w http.ResponseWriter, raw, errMsg string) (int, bool) {
   117→        f, ok := jsParseFloat(raw)
   118→        if !ok || !numIsInt(f) || f < math.MinInt32 || f > math.MaxInt32 {
   119→                writeJSON(w, 400, map[string]string{"error": errMsg})
   120→                return 0, false
   121→        }
   122→        return int(f), true
   123→}

   125→// routePosIntID 路径段 id 解析（整数且 > 0，对齐 TS `!isInteger || <= 0` 校验）
   126→func routePosIntID(w http.ResponseWriter, raw, errMsg string) (int, bool) {
   127→        n, ok := routeIntID(w, raw, errMsg)
   128→        if !ok {
   129→                return 0, false
   130→        }
   131→        if n <= 0 {
   132→                writeJSON(w, 400, map[string]string{"error": errMsg})
   133→                return 0, false
   134→        }
   135→        return n, true
   136→}

   138→// pageParamFloor 对齐 TS `Math.floor(Number(v) || def)`（Number('')/NaN/0 → def）
   139→func pageParamFloor(raw string, def int) int {
   140→        if raw == "" {
   141→                return def
   142→        }
   143→        f, ok := jsParseFloat(raw)
   144→        if !ok || f == 0 {
   145→                return def
   146→        }
   147→        if f > 2e9 {
   148→                return 2_000_000_000
   149→        }
   150→        if f < -2e9 {
   151→                return -2_000_000_000
   152→        }
   153→        return int(math.Floor(f))
   154→}

   156→// pageParamList 对齐 TS `Math.max(1, Math.floor(Number(v) || 1))`
   157→func pageParamList(raw string) int {
   158→        p := pageParamFloor(raw, 1)
   159→        if p < 1 {
   160→                return 1
   161→        }
   162→        return p
   163→}

   165→// readBodyValue 读请求体为任意 JSON 值（不写错误响应；由路由按 TS 契约还原文案）。
   166→// TS 的 req.json() 无大小上限，这里 32MB 兜底防滥用。
   167→func readBodyValue(r *http.Request) (any, bool) {
   168→        b, err := readAllLimited(r.Body, 32<<20)
   169→        if err != nil {
   170→                return nil, false
   171→        }
   172→        var v any
   173→        if json.Unmarshal(b, &v) != nil {
   174→                return nil, false
   175→        }
   176→        return v, true
   177→}

   185→// jsStringify 模拟 JS String(v) 的字符串化（用于 scrape-tasks PUT mode 等宽式转换）
   186→func jsStringify(v any) string {
   187→        switch x := v.(type) {
   188→        case string:
   189→                return x
   190→        case nil:
   191→                return "null"
   192→        case bool:
   193→                if x {
   194→                        return "true"
   195→                }
   196→                return "false"
   197→        case float64:
   198→                return fmtF(x)
   199→        default:
   200→                b, err := json.Marshal(x)
   201→                if err != nil {
   202→                        return "[object Object]"
   203→                }
   204→                return string(b)
   205→        }
   206→}

   208→// firstLineErr 移植 src/lib/errors.ts firstLine：错误首行 200 字截断（防泄露内部路径）
   209→func firstLineErr(err error) string {
   210→        if err == nil {
   211→                return ""
   212→        }
   213→        msg := err.Error()
   214→        if i := strings.IndexByte(msg, '\n'); i >= 0 {
   215→                msg = msg[:i]
   216→        }
   217→        return truncateRunes(msg, 200)
   218→}

   220→// ---- JS 空白语义（\s 全集：Go \s 缺 \v 与 unicode 空白，必须显式补齐）----
   221→
   222→var (
   223→        jsSpaceClass    = `[\t\n\v\f\r \x{00a0}\x{1680}\x{2000}-\x{200a}\x{2028}\x{2029}\x{202f}\x{205f}\x{3000}\x{feff}]`
   224→        jsSpacePlusRe   = regexp.MustCompile(`[\t\n\v\f\r \x{00a0}\x{1680}\x{2000}-\x{200a}\x{2028}\x{2029}\x{202f}\x{205f}\x{3000}\x{feff}]+`)
   225→        jsTrimLeadRe    = regexp.MustCompile(`^[\t\n\v\f\r \x{00a0}\x{1680}\x{2000}-\x{200a}\x{2028}\x{2029}\x{202f}\x{205f}\x{3000}\x{feff}]+`)
   226→        jsTrimTrailRe   = regexp.MustCompile(`[\t\n\v\f\r \x{00a0}\x{1680}\x{2000}-\x{200a}\x{2028}\x{2029}\x{202f}\x{205f}\x{3000}\x{feff}]+$`)
   227→        jsSpaceRuneSet  = map[rune]bool{'\t': true, '\n': true, '\v': true, '\f': true, '\r': true, ' ': true, 0x00a0: true, 0x1680: true, 0x2028: true, 0x2029: true, 0x202f: true, 0x205f: true, 0x3000: true, 0xfeff: true}
   228→        jsSpaceRuneFold = func() map[rune]bool {
   229→                m := map[rune]bool{}
   230→                for r := rune(0x2000); r <= 0x200a; r++ {
   231→                        m[r] = true
   232→                }
   233→                for r := range jsSpaceRuneSet {
   234→                        m[r] = true
   235→                }
   236→                return m
   237→        }()
   238→)

   240→func isJSSpace(r rune) bool { return jsSpaceRuneFold[r] }

   242→// wordCountJS 模拟 content.replace(/\s/g, '').length（rune 计数，见头注差异 1）
   243→func wordCountJS(s string) int {
   244→        n := 0
   245→        for _, r := range s {
   246→                if !isJSSpace(r) {
   247→                        n++
   248→                }
   249→        }
   250→        return n
   251→}

   253→// jsTrim JS String.prototype.trim 精确语义
   254→func jsTrim(s string) string {
   255→        return jsTrimTrailRe.ReplaceAllString(jsTrimLeadRe.ReplaceAllString(s, ""), "")
   256→}

   258→// splitJSSpace 模拟 s.split(/\s+/).filter(Boolean)
   259→func splitJSSpace(s string) []string {
   260→        parts := jsSpacePlusRe.Split(s, -1)
   261→        out := make([]string, 0, len(parts))
   262→        for _, p := range parts {
   263→                if p != "" {
   264→                        out = append(out, p)
   265→                }
   266→        }
   267→        return out
   268→}

   270→// likeWrap Prisma contains 语义：'%kw%'（不转义通配符，与 Prisma 一致）

   273→// marshalCompact JSON.stringify 等价（无 HTML 转义、无缩进、无尾随换行）
   274→func marshalCompact(v any) string {
   275→        var buf strings.Builder
   276→        enc := json.NewEncoder(&buf)
   277→        enc.SetEscapeHTML(false)
   278→        if err := enc.Encode(v); err != nil {
   279→                return "{}"
   280→        }
   281→        return strings.TrimRight(buf.String(), "\n")
   282→}

   329→// queryNovelList 执行列表查询并映射（whereSQL 含 WHERE 前缀或空串）
   330→func queryNovelList(whereSQL, orderSQL string, args []any, limit, offset int) ([]map[string]any, error) {
   331→        q := "SELECT " + novelListCols + novelListFrom + whereSQL + orderSQL + " LIMIT ? OFFSET ?"
   332→        all := make([]map[string]any, 0)
   333→        err := queryList(q, func(rows *sql.Rows) error {
   334→                item, err := scanNovelListItem(rows)
   335→                if err != nil {
   336→                        return err
   337→                }
   338→                all = append(all, item)
   339→                return nil
   340→        }, append(args, limit, offset)...)
   341→        if err != nil {
   342→                return nil, err
   343→        }
   344→        return all, nil
   345→}

   347→// ==================== GET /api/novels ====================

   349→func handleNovelsList(w http.ResponseWriter, r *http.Request, _ map[string]string) {
   350→        sp := r.URL.Query()
   351→        q := truncateRunes(trimSpaceStr(sp.Get("q")), 100)
   352→        status := sp.Get("status")
   353→        sort := sp.Get("sort")
   354→        if sort == "" {
   355→                sort = "latest"
   356→        }
   357→        page := pageParamList(sp.Get("page"))
   358→        pageSize := clampInt(pageParamFloor(sp.Get("pageSize"), 20), 4, 60)
   359→
   360→        // 非法 categoryId（abc/1.5/-3）→ 400（对齐 TS 明确报错而非静默全库查询）
   361→        categoryId := 0
   362→        if raw := sp.Get("categoryId"); raw != "" {
   363→                f, ok := jsParseFloat(raw)
   364→                if !ok || !numIsInt(f) || f < 0 {
   365→                        writeJSON(w, 400, map[string]string{"error": "无效 categoryId"})
   366→                        return
   367→                }
   368→                categoryId = int(f)
   369→        }
   370→
   371→        conds := []string{}
   372→        args := []any{}
   373→        if categoryId > 0 {
   374→                conds = append(conds, `n."categoryId" = ?`)
   375→                args = append(args, categoryId)
   376→        }
   377→        if status == "serial" || status == "finished" {
   378→                conds = append(conds, `n."status" = ?`)
   379→                args = append(args, status)
   380→        }
   381→        if q != "" {
   382→                conds = append(conds, `(n."title" LIKE ? OR n."author" LIKE ? OR n."description" LIKE ?)`)
   383→                like := likeWrap(q)
   384→                args = append(args, like, like, like)
   385→        }
   386→        whereSQL := ""
   387→        if len(conds) > 0 {
   388→                whereSQL = " WHERE " + strings.Join(conds, " AND ")
   389→        }
   390→        var orderSQL string
   391→        switch sort {
   392→        case "clicks":
   393→                orderSQL = ` ORDER BY n."clicks" DESC, n."id" DESC`
   394→        case "words":
   395→                orderSQL = ` ORDER BY n."wordCount" DESC, n."id" DESC`
   396→        case "featured":
   397→                orderSQL = ` ORDER BY n."isFeatured" DESC, n."updatedAt" DESC, n."id" DESC`
   398→        default:
   399→                orderSQL = ` ORDER BY n."updatedAt" DESC, n."id" DESC`
   400→        }
   401→
   402→        var total int64
   403→        countQ := `SELECT COUNT(*) FROM "Novel" n` + whereSQL
   404→        if err := queryOne(countQ, []any{&total}, args...); err != nil {
   405→                failJSON(w, "查询失败", firstLineErr(err), 500)
   406→                return
   407→        }
   408→        list, err := queryNovelList(whereSQL, orderSQL, args, pageSize, (page-1)*pageSize)
   409→        if err != nil {
   410→                failJSON(w, "查询失败", firstLineErr(err), 500)
   411→                return
   412→        }
   413→        totalPages := (total + int64(pageSize) - 1) / int64(pageSize)
   414→        if totalPages < 1 {
   415→                totalPages = 1
   416→        }
   417→        writeJSON(w, 200, map[string]any{
   418→                "list":       list,
   419→                "total":      total,
   420→                "page":       page,
   421→                "pageSize":   pageSize,
   422→                "totalPages": totalPages,
   423→        })
   424→}

   430→func handleNovelsCreate(w http.ResponseWriter, r *http.Request, _ map[string]string) {
   431→        v, ok := readBodyValue(r)
   432→        body := bodyMap(v)
   433→        if !ok {
   434→                writeJSON(w, 400, map[string]string{"error": "请求体不是合法 JSON"})
   435→                return
   436→        }
   437→        title := strField(body["title"], 0)
   438→        if trimSpaceStr(title) == "" {
   439→                writeJSON(w, 400, map[string]string{"error": "书名不能为空"})
   440→                return
   441→        }
   442→        categoryId, okCat := intFieldStrict(body["categoryId"])
   443→        if !okCat || categoryId == 0 {
   444→                writeJSON(w, 400, map[string]string{"error": "请选择分类"})
   445→                return
   446→        }
   447→        var catID int64
   448→        if err := queryOne(`SELECT "id" FROM "Category" WHERE "id" = ?`, []any{&catID}, categoryId); err != nil {
   449→                if isNoRows(err) {
   450→                        writeJSON(w, 400, map[string]string{"error": "分类不存在"})
   451→                        return
   452→                }
   453→                failJSON(w, "服务器错误", firstLineErr(err), 500)
   454→                return
   455→        }
   456→
   457→        // cover：仅接受 g1-g12 字符串，否则随机（对齐 covers.includes 严格相等）
   458→        cover, _ := body["cover"].(string)
   459→        validCover := false
   460→        for _, c := range coverTokens {
   461→                if c == cover {
   462→                        validCover = true
   463→                        break
   464→                }
   465→        }
   466→        if !validCover {
   467→                cover = coverTokens[rand.Intn(len(coverTokens))]
   468→        }
   469→        author := trimSpaceStr(strField(body["author"], 0))
   470→        if author == "" {
   471→                author = "佚名"
   472→        }
   473→        description := truncateRunes(strField(body["description"], 0), novelDescriptionMax)
   474→        st := "serial"
   475→        if s, _ := body["status"].(string); s == "finished" {
   476→                st = "finished"
   477→        }
   478→        now := nowMillis()
   479→        id, err := execReturningID(
   480→                `INSERT INTO "Novel" ("title","author","description","cover","categoryId","status","isFeatured","isHot","wordCount","clicks","createdAt","updatedAt") VALUES (?,?,?,?,?,?,?,?,0,0,?,?)`,
   481→                truncateRunes(trimSpaceStr(title), novelTitleMax),
   482→                truncateRunes(author, novelAuthorMax),
   483→                description,
   484→                cover,
   485→                categoryId,
   486→                st,
   487→                boolField(body["isFeatured"]),
   488→                boolField(body["isHot"]),
   489→                now, now,
   490→        )
   491→        if err != nil {
   492→                if isUniqueConflict(err) {
   493→                        writeJSON(w, 409, map[string]string{"error": "同名同作者的书已存在"})
   494→                        return
   495→                }
   496→                failJSON(w, "服务器错误", firstLineErr(err), 500)
   497→                return
   498→        }
   499→        writeJSON(w, 201, map[string]any{"id": id})
   500→}

   504→func handleNovelDetail(w http.ResponseWriter, r *http.Request, ps map[string]string) {
   505→        nid, ok := routeIntID(w, ps["id"], "无效 ID")
   506→        if !ok {

   508→        }
   509→        var id, categoryID, wordCount, clicks, chapterCount, isFeatured, isHot, updatedAt int64
   510→        var title, author, description, cover, status string
   511→        var catName sql.NullString
   512→        err := queryOne(
   513→                `SELECT n."id", n."title", n."author", n."description", n."cover", n."categoryId", c."name", n."status", n."isFeatured", n."isHot", n."wordCount", n."clicks", n."updatedAt", (SELECT COUNT(*) FROM "Chapter" ch WHERE ch."novelId" = n."id") FROM "Novel" n LEFT JOIN "Category" c ON c."id" = n."categoryId" WHERE n."id" = ?`,
   514→                []any{&id, &title, &author, &description, &cover, &categoryID, &catName, &status, &isFeatured, &isHot, &wordCount, &clicks, &updatedAt, &chapterCount},
   515→                nid,
   516→        )
   517→        if err != nil {
   518→                if isNoRows(err) {
   519→                        writeJSON(w, 404, map[string]string{"error": "小说不存在"})

   521→                }
   522→                failJSON(w, "服务器错误", firstLineErr(err), 500)
   523→                return
   524→        }

   526→        // 浏览计数 fire-and-forget（Prisma update 会自动触碰 @updatedAt，故显式 set）
   527→        _ = exec(`UPDATE "Novel" SET "clicks" = "clicks" + 1, "updatedAt" = ? WHERE "id" = ?`, nowMillis(), nid)
   528→
   529→        var lastID sql.NullInt64
   530→        var lastTitle sql.NullString
   531→        _ = queryOne(`SELECT "id", "title" FROM "Chapter" WHERE "novelId" = ? ORDER BY "idx" DESC LIMIT 1`, []any{&lastID, &lastTitle}, nid)
   532→
   533→        type chRow struct {

   535→                title              string
   536→        }

   538→        firstID := int64(0)
   539→        hasFirst := false
   540→        err = queryList(`SELECT "id", "idx", "title", "wordCount" FROM "Chapter" WHERE "novelId" = ? ORDER BY "idx" ASC LIMIT 12`, func(rows *sql.Rows) error {
   541→                var cid, cidx, cwc int64
   542→                var ctitle string
   543→                if err := rows.Scan(&cid, &cidx, &ctitle, &cwc); err != nil {
   544→                        return err
   545→                }
   546→                if !hasFirst {
   547→                        firstID, hasFirst = cid, true
   548→                }
   549→                chapters = append(chapters, map[string]any{"id": cid, "idx": cidx, "title": ctitle, "wordCount": cwc})

   553→                failJSON(w, "服务器错误", firstLineErr(err), 500)
   554→                return
   555→        }
   556→
   557→        categoryName := "未分类"
   558→        if catName.Valid && catName.String != "" {
   559→                categoryName = catName.String
   560→        }
   561→        st := "serial"
   562→        if status == "finished" {
   563→                st = "finished"
   564→        }

   566→        var lastChapterID any
   567→        if lastTitle.Valid {
   568→                lastChapterTitle = lastTitle.String

   571→                lastChapterID = lastID.Int64

   576→        }
   577→        writeJSON(w, 200, map[string]any{
   578→                "id":               id,
   579→                "title":            title,
   580→                "author":           author,
   581→                "description":      description,
   582→                "cover":            cover,
   583→                "categoryId":       categoryID,
   584→                "categoryName":     categoryName,
   585→                "status":           st,

   588→                "wordCount":        wordCount,

   590→                "chapterCount":     chapterCount,

   594→                "firstChapterId":   firstChapterID,
   595→                "lastChapterId":    lastChapterID,

   597→        })
   598→}

   602→var coverTokenRe = regexp.MustCompile(`^g\d+$`)
   603→
   604→func handleNovelUpdate(w http.ResponseWriter, r *http.Request, ps map[string]string) {
   605→        nid, ok := routePosIntID(w, ps["id"], "无效 ID")
   606→        if !ok {
   607→                return
   608→        }
   609→        v, ok := readBodyValue(r)
   610→        if !ok {
   611→                writeJSON(w, 400, map[string]string{"error": "请求体不是合法 JSON"})
   612→                return
   613→        }
   614→        body := bodyMap(v)
   615→
   616→        sets := []string{}
   617→        args := []any{}
   618→        if s, ok2 := body["title"].(string); ok2 {
   619→                if t := trimSpaceStr(s); t != "" {
   620→                        sets = append(sets, `"title" = ?`)
   621→                        args = append(args, truncateRunes(t, novelTitleMax))
   622→                }
   623→        }
   624→        if s, ok2 := body["author"].(string); ok2 {
   625→                if t := trimSpaceStr(s); t != "" {
   626→                        sets = append(sets, `"author" = ?`)
   627→                        args = append(args, truncateRunes(t, novelAuthorMax))
   628→                }
   629→        }
   630→        if s, ok2 := body["description"].(string); ok2 {
   631→                sets = append(sets, `"description" = ?`)
   632→                args = append(args, truncateRunes(s, novelDescriptionMax))
   633→        }
   634→        if s, ok2 := body["cover"].(string); ok2 {
   635→                if coverTokenRe.MatchString(s) {
   636→                        sets = append(sets, `"cover" = ?`)
   637→                        args = append(args, s)
   638→                }
   639→        }
   640→        if n, ok2 := intFieldStrict(body["categoryId"]); ok2 && n > 0 {
   641→                sets = append(sets, `"categoryId" = ?`)
   642→                args = append(args, n)
   643→        }
   644→        if s, ok2 := body["status"].(string); ok2 && (s == "serial" || s == "finished") {
   645→                sets = append(sets, `"status" = ?`)
   646→                args = append(args, s)
   647→        }
   648→        if b, ok2 := body["isFeatured"].(bool); ok2 {
   649→                sets = append(sets, `"isFeatured" = ?`)
   650→                args = append(args, b)
   651→        }
   652→        if b, ok2 := body["isHot"].(bool); ok2 {
   653→                sets = append(sets, `"isHot" = ?`)
   654→                args = append(args, b)
   655→        }
   656→        // Prisma @updatedAt 任何 update 都会触碰 → 显式 set now
   657→        sets = append(sets, `"updatedAt" = ?`)
   658→        args = append(args, nowMillis())
   659→
   660→        res, err := exec(`UPDATE "Novel" SET `+strings.Join(sets, ", ")+` WHERE "id" = ?`, append(args, nid)...)
   661→        if err != nil {
   662→                // P2025 之外的一切错误（分类外键不存在等）→ 400（对齐 TS 分类注释语义）
   663→                failJSON(w, "更新失败（分类不存在？）", "", 400)
   664→                return
   665→        }
   666→        if n, _ := res.RowsAffected(); n == 0 {
   667→                writeJSON(w, 404, map[string]string{"error": "小说不存在"})
   668→                return
   669→        }
   670→        writeJSON(w, 200, map[string]any{"ok": true})
   671→}
