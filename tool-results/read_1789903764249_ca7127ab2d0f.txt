   145→export function Category({
   146→  navigate,
   147→  categoryId,
   148→  page = 1,
   149→}: ViewProps & { categoryId?: number; page?: number }) {
   150→  const cats = useCategories()
   151→  const list = useNovels({ categoryId, page, pageSize: 20 })
   152→
   153→  const catName = cats.data?.find((c) => c.id === categoryId)?.name
   154→
   155→  return (
   156→    <div className="mx-auto max-w-[1180px] space-y-5 px-4 py-6">
   157→      <Crumb items={[{ label: '首页', view: { name: 'home' } }, { label: '小说书库' }]} navigate={navigate} />
   158→      <h1 className="border-l-4 border-[#2563eb] pl-4 text-[24px] leading-8 font-semibold text-[#1e293b]">
   159→        {catName ?? '小说书库'}
   160→        {cats.data ? (
   161→          <span className="ml-3 text-[13px] font-normal text-[#94a3b8]">
   162→            {cats.data.find((c) => c.id === categoryId)?.novelCount ?? list.data?.total ?? ''}
   163→          </span>
   164→        ) : null}
   165→      </h1>
   166→
   167→      {/* 筛选条 */}
   168→      <div className="flex flex-wrap gap-2 rounded-[10px] border border-[#e6edf7] bg-white p-4 shadow-[0_1px_3px_rgba(37,99,235,0.06)]">
   169→        <button
   170→          onClick={() => navigate({ name: 'category', page: 1 })}
   171→          className={
   172→            categoryId == null
   173→              ? 'cursor-pointer rounded-full bg-[#2563eb] px-4 py-1.5 text-[13px] text-white transition-colors duration-200'
   174→              : 'cursor-pointer rounded-full bg-[#f0f4fb] px-4 py-1.5 text-[13px] text-[#1e293b] transition-colors duration-200 hover:bg-[#e8f1ff] hover:text-[#2563eb]'
   175→          }
   176→        >
   177→          全部
   178→        </button>
   179→        {(cats.data ?? []).map((c) => (
   180→          <button
   181→            key={c.id}
   182→            onClick={() => navigate({ name: 'category', categoryId: c.id, page: 1 })}
   183→            className={
   184→              categoryId === c.id
   185→                ? 'cursor-pointer rounded-full bg-[#2563eb] px-4 py-1.5 text-[13px] text-white transition-colors duration-200'
   186→                : 'cursor-pointer rounded-full bg-[#f0f4fb] px-4 py-1.5 text-[13px] text-[#1e293b] transition-colors duration-200 hover:bg-[#e8f1ff] hover:text-[#2563eb]'
   187→            }
   188→          >
   189→            {c.name}
   190→          </button>
   191→        ))}
   192→      </div>
   193→
   194→      {list.isPending ? (
   195→        <CardGridSkeleton count={6} />
   196→      ) : list.isError ? (
   197→        <ErrorBox onRetry={() => list.refetch()} />
   198→      ) : (list.data?.list.length ?? 0) === 0 ? (
   199→        <EmptyBox text="该分类下暂无收录小说" />
   200→      ) : (
   201→        <>
   202→          <div className="grid gap-4 md:grid-cols-2 xl:grid-cols-3">
   203→            {list.data!.list.map((n) => (
   204→              <BookTextCard key={n.id} novel={n} navigate={navigate} />
   205→            ))}
   206→          </div>
   207→          <Pager
   208→            page={list.data!.page}
   209→            totalPages={list.data!.totalPages}
   210→            onGo={(p) => navigate({ name: 'category', categoryId, page: p })}
   211→          />
   212→        </>
   213→      )}
   214→    </div>
   215→  )
   216→}

   428→                className="shrink-0 cursor-pointer rounded-[10px] border-[1.5px] border-[#dbe4f0] bg-white px-5 py-2 text-[13px] text-[#1e293b] transition-colors duration-200 hover:border-[#2563eb] hover:text-[#2563eb]"
   429→              >
   430→                返回书页
   431→              </button>
   432→            }
   433→          />
   434→          <ChapterPills chapters={[...chapters].slice(-12).reverse()} navigate={navigate} />
   435→        </section>
   436→      )}
   437→
   438→      <section className="rounded-[10px] border border-[#e6edf7] bg-white p-6 shadow-[0_1px_3px_rgba(37,99,235,0.06)]">
   439→        <SectionTitle
   440→          title={novel ? `${novel.title} · 章节目录` : '章节目录'}
   441→          small={chapters ? `共 ${chapters.length} 章` : undefined}
   442→          action={
   443→            <button
   444→              onClick={() => navigate({ name: 'book', novelId })}
   445→              className="shrink-0 cursor-pointer rounded-[10px] border-[1.5px] border-[#dbe4f0] bg-white px-5 py-2 text-[13px] text-[#1e293b] transition-colors duration-200 hover:border-[#2563eb] hover:text-[#2563eb]"
   446→            >
   447→              返回书页
   448→            </button>
   449→          }
   450→        />
   451→        {isPending ? (
   452→          <PillSkeleton count={16} />
   453→        ) : isError ? (
   454→          <ErrorBox onRetry={() => refetch()} />
   455→        ) : (chapters?.length ?? 0) === 0 ? (
   456→          <EmptyBox text="暂无章节" />
   457→        ) : (
   458→          <ChapterPills chapters={chapters!} navigate={navigate} />
   459→        )}
   460→      </section>
   461→    </div>
   462→  )
   463→}

   465→/* ==================== 章节正文页（900px 阅读容器） ==================== */
   466→
   467→function AuthorWorks({
   468→  author,
   469→  excludeId,
   470→  navigate,
   471→}: {
   472→  author: string
   473→  excludeId: number
   474→  navigate: Nav
   475→}) {
   476→  const q = useNovels({ q: author, pageSize: 6 })
   477→  const rows: NovelListItem[] = (q.data?.list ?? []).filter((n) => n.id !== excludeId).slice(0, 3)
   478→  if (rows.length === 0) return null
   479→  return (
   480→    <section>
   481→      <SectionTitle title={`「${author}」的其他作品`} />
   482→      <div className="grid gap-4 md:grid-cols-2 xl:grid-cols-3">
   483→        {rows.map((n) => (
   484→          <BookTextCard key={n.id} novel={n} navigate={navigate} />
   485→        ))}
   486→      </div>

   713→
   714→function SearchPanel({ query, navigate }: { query: string; navigate: Nav }) {
   715→  const [input, setInput] = useState(query)
   716→  const [page, setPage] = useState(1)
   717→  const list = useNovels({ q: query || undefined, page, pageSize: 20 })
   718→
   719→  const submit = () => {
   720→    const kw = input.trim()
   721→    if (kw) navigate({ name: 'search', query: kw })
   722→  }
   723→
   724→  return (
   725→    <div className="space-y-5">
   726→      <div className="flex gap-3 rounded-[10px] border border-[#e6edf7] bg-white p-4 shadow-[0_1px_3px_rgba(37,99,235,0.06)]">
   727→        <input
   728→          value={input}
   729→          onChange={(e) => setInput(e.target.value)}
   730→          onKeyDown={(e) => e.key === 'Enter' && submit()}
   731→          placeholder="输入书名 / 作者关键词"
   732→          className="h-10 min-w-0 flex-1 rounded-[10px] border border-[#dbe4f0] bg-[#f0f4fb] px-4 text-[14px] text-[#1e293b] outline-none transition-colors duration-200 placeholder:text-[#94a3b8] focus:border-[#2563eb] focus:bg-white"
   733→        />
   734→        <button
   735→          onClick={submit}
   736→          className="cursor-pointer rounded-[10px] bg-[#2563eb] px-8 text-[14px] text-white transition-colors duration-200 hover:bg-[#1d4ed8]"
   737→        >
   738→          搜索
   739→        </button>
   740→      </div>
   741→
   742→      <SectionTitle
   743→        title="搜索结果"
   744→        small={query ? `“${query}” · ${list.data?.total ?? 0} 条` : '请输入关键词'}
   745→      />
   746→
   747→      {list.isPending ? (
   748→        <CardGridSkeleton count={6} />
   749→      ) : list.isError ? (
   750→        <ErrorBox onRetry={() => list.refetch()} />
   751→      ) : (list.data?.list.length ?? 0) === 0 ? (
   752→        <EmptyBox text={`没有找到与“${query}”相关的小说，换个关键词试试`} />
   753→      ) : (
   754→        <>
   755→          <div className="grid gap-4 md:grid-cols-2 xl:grid-cols-3">
   756→            {list.data!.list.map((n) => (
   757→              <BookTextCard key={n.id} novel={n} navigate={navigate} />
   758→            ))}
   759→          </div>
   760→          <Pager
   761→            page={list.data!.page}
   762→            totalPages={list.data!.totalPages}
   763→            onGo={(p) => setPage(p)}
   764→          />
   765→        </>
   766→      )}
   767→    </div>
   768→  )
   769→}
