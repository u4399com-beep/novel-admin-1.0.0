    66→ * 模块级防并发：同一任务 id 同时只允许一个 worker 实例。
    67→ * 挂在 globalThis 上：dev HMR 重载本模块时新旧实例共享同一 Set，

    71→  __scrapeRunning?: Set<number>
    72→  __scrapeRecovered?: boolean

    76→/**
    77→ * 僵尸任务恢复（心跳判定）：worker 常驻在 Next.js 进程内存里（fire-and-forget promise），
    78→ * 进程重启后内存任务丢失，DB 中 status=pending/running 的记录会永久卡死。
    79→ *
    80→ * 判定依据：运行中任务每完成一章/一书都会 flush 触碰 updatedAt（心跳）；真正的僵尸
    81→ * （进程消失/模块实例丢失）updatedAt 停摆。仅回收「仍处 pending/running 且心跳停摆
    82→ * 超过 ZOMBIE_STALE_MS」的任务——不按 createdAt/进程启动时刻判定，因为 dev 下
    83→ * Turbopack 每个路由 bundle 各有一份本模块实例，任何晚加载实例的时钟基线都不同，
    84→ * 按创建时刻会把「重跑老任务」等活任务误杀（实测踩坑）。
    85→ * 心跳最长间隙 = 引擎单请求预算 60s + 落库时间，远小于 5 分钟阈值。
    86→ *
    87→ * 防重入标志同步置位（await 前置位）：dev 下同进程可能存在多份模块实例
    88→ * （Turbopack 按路由分包），异步置位会留下双重回收的竞态窗口。
    89→ */
    90→const ZOMBIE_STALE_MS = 5 * 60_000
    91→
    92→async function recoverStaleTasks(): Promise<void> {
    93→  if (gWorker.__scrapeRecovered) return
    94→  gWorker.__scrapeRecovered = true // 同步置位：跨实例/跨 HMR 防重入
    95→  try {
    96→    const stale = await db.scrapeTask.findMany({
    97→      where: {
    98→        status: { in: ['pending', 'running'] },
    99→        updatedAt: { lt: new Date(Date.now() - ZOMBIE_STALE_MS) },
   100→      },
   101→      select: { id: true, log: true },
   102→    })
   103→    let recovered = 0
   104→    for (const t of stale) {
   105→      if (running.has(t.id)) continue // 本进程内存中仍有实例在执行（共享 Set 时双保险）
   106→      const line = `[${new Date().toTimeString().slice(0, 8)}] 心跳停摆超 5 分钟，判定僵尸任务并回收`
   107→      const log = `${t.log ? `${t.log}\n` : ''}${line}`
   108→      const res = await db.scrapeTask
   109→        .updateMany({
   110→          where: { id: t.id, status: { in: ['pending', 'running'] } },
   111→          data: {
   112→            status: 'failed',
   113→            message: '任务心跳超时，自动回收',
   114→            // 与 Run 同规约：只保留最近 100 行
   115→            log: log.split('\n').slice(-MAX_LOG_LINES).join('\n'),
   116→          },
   117→        })
   118→        .catch(() => null)
   119→      if (res && res.count > 0) recovered++
   120→    }
   121→    if (recovered > 0) console.log(`[scrape-worker] 僵尸任务回收: ${recovered} 条`)
   122→  } catch {
   123→    // 回收失败不影响模块加载；标志已置位，等待下次进程重启重试（代价可接受：僵尸多驻留一轮）
   124→  }
   125→}

   128→/** 协作式取消检查：记录不存在视为取消；DB 瞬时错误不误判为取消（fail-open） */
   129→async function isCanceled(taskId: number): Promise<boolean> {
   130→  const t = await db.scrapeTask
   131→    .findUnique({ where: { id: taskId }, select: { status: true } })
   132→    .catch(() => undefined)
   133→  if (t === undefined) return false // 查询失败 ≠ 被取消，避免瞬时 DB 错误误停任务
   134→  return !t || t.status === 'canceled'
   135→}

   137→/**
   138→ * 并行池：cursor 抢占式派发，concurrency 路 worker 持续取件直至耗尽。
   139→ * fn 内部通过 stop signal 提前收敛（停止派发，在途任务跑完当前单元后自然退出）。
   140→ */
   141→async function runPool<T>(items: T[], concurrency: number, fn: (item: T, index: number) => Promise<void>): Promise<void> {
   142→  let cursor = 0
   143→  const workers = Array.from({ length: Math.max(1, Math.min(concurrency, items.length)) }, async () => {
   144→    for (;;) {
   145→      const i = cursor++
   146→      if (i >= items.length) return
   147→      await fn(items[i], i)
   148→    }
   149→  })
   150→  await Promise.all(workers)
   151→}

   153→/** 生成第 k 页候选 URL：?page=k（已有 query 则 &page=k）与 /page/k 两种变体 */
   154→function pageVariants(url: string, k: number): string[] {
   155→  const out: string[] = []
   156→  try {
   157→    const u = new URL(url)
   158→    u.searchParams.set('page', String(k))
   159→    out.push(u.toString())
   160→    const p = new URL(url)
   161→    p.pathname = `${p.pathname.replace(/\/+$/, '')}/page/${k}`
   162→    p.search = ''
   163→    if (p.toString() !== u.toString()) out.push(p.toString())
   164→  } catch {
   165→    out.push(url.includes('?') ? `${url}&page=${k}` : `${url}?page=${k}`)
   166→    out.push(`${url.replace(/\/+$/, '')}/page/${k}`)
   167→  }
   168→  return [...new Set(out)]
   169→}

   173→/**
   174→ * Phase 1：抓取书页+完整目录 → 书籍 upsert（含封面落盘/下拉词绑定）→ 章节目录骨架入库。
   175→ *

   177→ * - 已存在且有内容（wordCount>0）→ skip 占位任务（不重抓，进度照常推进）；
   178→ * - 已存在空骨架（wordCount=0，上次任务未抓到）→ 保持骨架，Phase 2 重抓填充；
   179→ * - 不存在 → 创建骨架（content=''、wordCount=0、idx=目录位次保序）。
   180→ * 已有章节盘点只取 id/idx/title/volume/wordCount（绝不 select content 全文——
   181→ * 用 wordCount 区分内容章节与骨架，正常入库章节 wordCount>0 恒成立）。
   182→ *
   183→ * 骨架批量入库：createMany 快路径；整批失败（并发双任务同书撞 (novelId,idx) 等）时
   184→ * 逐条 create 兜底，createChapterSkeleton 撞唯一约束查回已有记录按「已存在」处理。
   185→ */
   186→async function processBookMeta(
   187→  run: Run,
   188→  bookUrl: string,
   189→  rule: LoadedRule,
   190→  opts: { trackTotal: boolean; referer?: string | null },
   191→): Promise<BookMetaOutcome> {
   192→  const fail = (canceled: boolean, message: string): BookMetaOutcome => ({
   193→    ok: false,
   194→    canceled,
   195→    novelId: 0,
   196→    refsTotal: 0,
   197→    skipped: 0,
   198→    jobs: [],
   199→    message,
   200→  })
   201→
   202→  run.log(`抓取书页 ${bookUrl.slice(0, 120)}…`)
   203→  // Referer 链：list 模式传「发现本书的列表页」作来路；single 模式缺省由引擎回落站内首页。
   204→  // 熔断感知：书页撞上熔断（站点临时限流封禁）时等待冷却后自动重试，而非立即把本书记失败
   205→  const page = await withCircuitRetry(hostOf(bookUrl), () => fetchBookPage(run, bookUrl, rule, opts.referer ?? undefined), {
   206→    detect: (r) => (r.ok ? null : circuitWaitMs(r.error, r.circuitRetryAfterMs)),
   207→    isAlive: async () => !(await isCanceled(run.taskId)),
   208→    onWait: (ms, cycle, max) =>
   209→      run.log(`目标主机熔断冷却中，暂停 ${Math.round(ms / 1000)}s 后自动重试书页（第 ${cycle}/${max} 轮，等待期间不向源站发请求）`),
   210→  })
   211→  if (!page.ok) {
   212→    run.log(`书页提取失败: ${page.error}`)
   213→    return fail(false, page.error)
   214→  }
   215→  const book = page.book
   216→  // 引擎响应未经 schema 校验（callEngine 直接 as 断言），关键字段做形态兜底，
   217→  // 防畸形载荷（引擎版本错位等）在任务中途抛 TypeError 把整任务拖成 failed
   218→  if (typeof book.title !== 'string') book.title = ''
   219→  if (!Array.isArray(book.chapters)) book.chapters = []
   220→  if (typeof book.description !== 'string') book.description = ''
   221→  if (typeof book.author !== 'string') book.author = ''
   222→  if (typeof book.category !== 'string') book.category = ''
   223→  if (typeof book.status !== 'string') book.status = ''
   224→  if (typeof book.chapterCount !== 'number') book.chapterCount = book.chapters.length
   225→  // 全字段兜底清洗（引擎侧已清洗，此处防版本错位/旧引擎，幂等成本低）：
   226→  // 实体解码（&#091；类全角分号变体）、字面 \n 还原、残缺标签剥除、简介样板句/SEO 伪简介处理
   227→  book.title = cleanTextField(book.title)
   228→  book.author = cleanTextField(book.author)
   229→  book.description = cleanDescriptionField(book.description)
   230→  book.category = cleanTextField(book.category)
   231→  book.status = cleanTextField(book.status)
   232→  run.log(
   233→    `书页提取成功：《${book.title.slice(0, 40)}》${book.author ? ` / ${book.author.slice(0, 20)}` : ''}，章节链接 ${book.chapterCount} 条`,
   234→  )
   235→
   236→  // ---- 完整目录页二次提取（bookRule.catalogLinkSelector，如 23qb 新模板书页仅含最新几章）----
   237→  let allRefs = book.chapters
   238→  const catalogSel = rule.bookRule.catalogLinkSelector
   239→  if (typeof catalogSel === 'string' && catalogSel) {
   240→    if (await isCanceled(run.taskId)) return fail(true, '任务已取消')
   241→    if (book.catalogUrl) {
   242→      run.log(`发现完整目录页 ${book.catalogUrl.slice(0, 100)}，尝试整目提取…`)
   243→      const catalogUrl = book.catalogUrl // 闭包捕获需局部常量（TS 收窄不跨回调）
   244→      // 熔断感知：目录页撞熔断时等待冷却后自动重试（超限时按旧语义回退书页章节链接）
   245→      const catalog = await withCircuitRetry(
   246→        hostOf(catalogUrl),
   247→        () => fetchCatalogChapters(run, catalogUrl, rule, bookUrl),
   248→        {
   249→          detect: (r) => (r.error ? circuitWaitMs(r.error) : null),
   250→          isSuccess: (r) => r.error === null,
   251→          isAlive: async () => !(await isCanceled(run.taskId)),
   252→          onWait: (ms, cycle, max) =>
   253→            run.log(`目标主机熔断冷却中，暂停 ${Math.round(ms / 1000)}s 后自动重试目录页（第 ${cycle}/${max} 轮）`),
   254→        },
   255→      )
   256→      const catalogRefs = catalog.refs
   257→      if (catalogRefs.length > allRefs.length) {
   258→        run.log(`目录页提取到 ${catalogRefs.length} 条章节链接（书页仅 ${allRefs.length} 条），采用目录页结果`)
   259→        allRefs = catalogRefs
   260→      } else if (catalogRefs.length > 0) {
   261→        run.log(`目录页提取 ${catalogRefs.length} 条不多于书页 ${allRefs.length} 条，维持书页结果`)
   262→      }
   263→    } else {
   264→      run.log(`catalogLinkSelector "${catalogSel.slice(0, 60)}" 在书页无命中，仅用书页章节链接`)
   265→    }
   266→  }
   267→
   268→  if (await isCanceled(run.taskId)) return fail(true, '任务已取消')
   269→
   270→  // ---- 乱序重排（充分考虑分卷：同号不同卷不跨卷错排，卷名随章节移动）----
   271→  // 源站目录常见「最新章节块（新→旧）+ 正文块（旧→新）」或整本倒序；直接按抓取序编 idx 会错乱。
   272→  // 引擎响应未经 schema 校验，逐条形态兑底后再重排。
   273→  allRefs = (Array.isArray(allRefs) ? allRefs : []).map((c) => ({
   274→    title: typeof c?.title === 'string' ? c.title : '',
   275→    url: typeof c?.url === 'string' ? c.url : null,
   276→    volume: typeof c?.volume === 'string' && c.volume.trim() ? c.volume.trim().slice(0, 50) : undefined,
   277→  })) as ChapterRef[]
   278→  const rr = reorderChapterRefs(allRefs)
   279→  if (rr.reordered) {
   280→    allRefs = rr.refs
   281→    run.log(`目录乱序重排：${rr.note}`)
   282→  }
   283→
   284→  // ---- 分类 ----
   285→  let categoryId: number
   286→  try {
   287→    categoryId = await ensureCategory(book.category)
   288→  } catch (e) {
   289→    const msg = e instanceof Error ? e.message : '分类处理失败'
   290→    run.log(msg)
   291→    return fail(false, msg)
   292→  }
   293→
   294→  // ---- 书籍 upsert（title+author 查重；DB 层 @@unique([title,author]) 兜底并发）----
   295→  const up = await upsertBook(run, book, categoryId, rule.proxy)
   296→  if (!up.ok) {
   297→    // 记录级失败（up.canceled=true：入库/更新失败、并发冲突后找不到记录）沿用取消通道停整个任务；
   298→    // 书籍级失败（up.canceled=false，如空标题）只算本书失败：single 按 failed 收尾，list 继续下一本
   299→    return fail(up.canceled, up.message)
   300→  }
   301→  // 采集的同时取搜索引擎下拉词 → 绑定书籍 + 生成 PSEO 书籍页（fire-and-forget）
   302→  scheduleBindBookSuggestKeywords(run, up.novelId, up.title || book.title, up.createdNew)
   303→
   304→  // ---- 章节列表准备 ----
   305→  const refs = allRefs.filter(
   306→    (c): c is ChapterRef & { url: string } => !!c && typeof c.url === 'string' && !!c.url,
   307→  )
   308→  if (refs.length === 0) {
   309→    run.log('未提取到任何有效章节链接')
   310→    return { ok: true, canceled: false, novelId: up.novelId, refsTotal: 0, skipped: 0, jobs: [], message: '书籍已入库（未提取到章节链接）' }
   311→  }
   312→  // 同目录内预去重（骨架入库前完成）：按清洗后标题优先、URL 兜底去重，防同批重复入库
   313→  const seenRef = new Set<string>()
   314→  const uniqueRefs: (ChapterRef & { url: string })[] = []
   315→  for (const c of refs) {
   316→    const key = c.title.trim() ? `t:${cleanTextField(c.title)}` : `u:${c.url}`
   317→    if (seenRef.has(key)) continue
   318→    seenRef.add(key)
   319→    uniqueRefs.push(c)
   320→  }
   321→  if (uniqueRefs.length < refs.length) run.log(`目录内去重：${refs.length} → ${uniqueRefs.length} 条（无数量上限，全量采集）`)
   322→
   323→  if (await isCanceled(run.taskId)) return fail(true, '任务已取消')
   324→
   325→  // ---- 已有章节盘点（不含 content 全文；wordCount 区分内容章节与骨架） ----
   326→  const existingChapters = await db.chapter
   327→    .findMany({
   328→      where: { novelId: up.novelId },
   329→      select: { id: true, idx: true, title: true, volume: true, wordCount: true },
   330→    })
   331→    .catch(() => [] as { id: number; idx: number; title: string; volume: string; wordCount: number }[])
   332→  // 去重键与入库标题共用同一清洗规约（cleanTextField 幂等）：存量脏标题与本次净标题可命中比对，
   333→  // 避免清洗规约升级后同一章以两种写法重复入库
   334→  const titleGroups = new Map<string, { id: number; idx: number; wordCount: number }[]>()
   335→  for (const c of existingChapters) {
   336→    const k = cleanTextField(c.title)
   337→    const arr = titleGroups.get(k)
   338→    if (arr) arr.push({ id: c.id, idx: c.idx, wordCount: c.wordCount })
   339→    else titleGroups.set(k, [{ id: c.id, idx: c.idx, wordCount: c.wordCount }])
   340→  }
   341→  const maxAgg = await db.chapter
   342→    .aggregate({ where: { novelId: up.novelId }, _max: { idx: true } })
   343→    .catch(() => ({ _max: { idx: null as number | null } }))
   344→  // 骨架槽位基准：现有最大 idx+1；每条按目录位次预分配（base+位次），保序且并行安全
   345→  const baseIdx = (maxAgg._max.idx ?? 0) + 1
   346→
   347→  // ---- 目录逐条分类：skip（已有内容）/ fill（已有空骨架或新骨架）/ 骨架待建 ----
   348→  const jobs: ChapterFillJob[] = []
   349→  /** 待创建骨架（jobIndex 回指 jobs 中对应的 fill 任务，创建结果落地后回填定位） */
   350→  const pendingSkeletons: { jobIndex: number; idx: number; title: string; volume: string; refTitle: string }[] = []
   351→  /** 已存在章节的卷名回填队列（title → volume；仅唯一标题且存量为空卷时写入） */
   352→  const volumeBackfill = new Map<string, string>()
   353→  let skipped = 0
   354→
   355→  for (let position = 0; position < uniqueRefs.length; position++) {
   356→    const ref = uniqueRefs[position]
   357→    const refTitle = cleanTextField(String(ref.title ?? '')).slice(0, 200)
   358→    if (refTitle && titleGroups.has(refTitle)) {
   359→      const matches = titleGroups.get(refTitle)!
   360→      if (matches.some((m) => m.wordCount > 0)) {
   361→        // 已存在且有内容：跳过不重抓（Phase 2 仅作进度占位）
   362→        skipped++
   363→        jobs.push({ kind: 'skip', novelId: up.novelId, chapterId: matches[0].id, idx: matches[0].idx, url: '', refTitle, volume: '' })
   364→        // 分卷补齐：本次源站目录携带卷名、存量章节为空卷且标题在全书中唯一时，回填卷名
   365→        const rv = ref.volume
   366→        if (rv && matches.length === 1 && !volumeBackfill.has(refTitle)) volumeBackfill.set(refTitle, rv)
   367→        continue
   368→      }
   369→      // 已存在空骨架（上次任务未抓到内容）：保持骨架不重建，Phase 2 重抓填充（不跳过）
   370→      const target = matches.reduce((a, b) => (a.idx <= b.idx ? a : b))
   371→      jobs.push({ kind: 'fill', novelId: up.novelId, chapterId: target.id, idx: target.idx, url: ref.url, refTitle, volume: ref.volume ?? '' })
   372→      continue
   373→    }
   374→    // 不存在 → 骨架占位（idx=目录位次预分配保序），创建结果落地后回填 chapterId
   375→    const idx = baseIdx + position
   376→    pendingSkeletons.push({ jobIndex: jobs.length, idx, title: refTitle || `第${idx}章`, volume: ref.volume ?? '', refTitle })
   377→    jobs.push({ kind: 'fill', novelId: up.novelId, chapterId: null, idx, url: ref.url, refTitle, volume: ref.volume ?? '' })
   378→  }
   379→
   380→  // ---- 骨架批量入库：createMany 快路径；整批失败（并发撞 (novelId,idx) 等）逐条 create 兜底 ----
   381→  if (pendingSkeletons.length > 0) {
   382→    let batchOk = true
   383→    try {
   384→      await db.chapter.createMany({
   385→        data: pendingSkeletons.map((s) => ({
   386→          novelId: up.novelId,
   387→          idx: s.idx,
   388→          title: s.title,
   389→          content: '',
   390→          wordCount: 0,
   391→          volume: s.volume,
   392→        })),
   393→      })
   394→    } catch {
   395→      batchOk = false
   396→    }
   397→    if (!batchOk) {
   398→      for (const s of pendingSkeletons) {
   399→        const job = jobs[s.jobIndex]
   400→        const sk = await createChapterSkeleton(run, up.novelId, s.idx, { title: s.title, volume: s.volume }, s.refTitle)
   401→        if (!sk.ok) {
   402→          run.log(`骨架章节「${s.title.slice(0, 30)}」入库失败: ${sk.message.slice(0, 120)}`)
   403→          job.kind = 'create' // 无骨架可填：Phase 2 抓到正文后走 storeChapter 兜底新建
   404→          continue
   405→        }
   406→        job.idx = sk.idx
   407→        if (!sk.created) {
   408→          // 撞唯一约束查回已有记录 → 按「已存在」处理（并发双任务同书场景）
   409→          if (sk.existingHasContent) {
   410→            job.kind = 'skip'
   411→            job.url = ''
   412→            skipped++
   413→          }
   414→          job.chapterId = sk.existingId
   415→        }
   416→      }
   417→    }
   418→  }
   419→
   420→  // ---- 已存在章节的卷名回填（逐卷 updateMany，仅在唯一标题且存量空卷时入队）----
   421→  if (volumeBackfill.size > 0) {
   422→    let backfilled = 0
   423→    for (const [title, volume] of volumeBackfill) {
   424→      const res = await db.chapter
   425→        .updateMany({ where: { novelId: up.novelId, title, volume: '' }, data: { volume } })
   426→        .catch(() => null)
   427→      if (res) backfilled += res.count
   428→    }
   429→    if (backfilled > 0) run.log(`分卷回填：${backfilled} 个已存在章节补记卷名`)
   430→  }
   431→
   432→  // ---- 进度分母 ----
   433→  if (opts.trackTotal) {
   434→    // single 模式：total=骨架章节数（分母），done 随 Phase 2 逐章递增
   435→    //（flush false=任务记录已删，立即停）
   436→    if (!(await run.flush({ total: uniqueRefs.length }))) return fail(true, '任务记录已删除')
   437→  }
   438→
   439→  const toFetch = jobs.length - skipped
   440→  run.log(`目录骨架就绪：${uniqueRefs.length} 章（已存在 ${skipped} 章，待抓取 ${toFetch} 章）`)
   441→  return { ok: true, canceled: false, novelId: up.novelId, refsTotal: uniqueRefs.length, skipped, jobs, message: '' }
   442→}
   443→
   444→// ==================== Phase 2：跨书批量章节内容填充 ====================
   445→
   446→/** Phase 2 单书上下文（统计就地回填，供书籍级成败判定与日志节流） */
   447→interface Phase2Book {
   448→  novelId: number
   449→  title: string
   450→  /** 章节页 Referer 来路（书页 URL，站点常见「书页→章节」导航校验） */
   451→  bookUrl: string
   452→  jobs: ChapterFillJob[]
   453→  filled: number
   454→  failed: number
   455→  /** 清洗日志节流：每本书最多记 3 条，防止日志爆炸 */
   456→  cleanLogCount: number
   457→}
   458→
   459→interface Phase2Result {
   460→  canceled: boolean
   461→  filled: number
   462→  failed: number
   463→  skipped: number
   464→}
   465→
   466→/**
   467→ * Phase 2：把 Phase 1 收集的全部章节平铺（按书分组顺序），runPool 章节级并发
   468→ * 跨书批量抓取正文，update 填充骨架章节 content/wordCount/volume。
   469→ *
   470→ * - 抓取失败的骨架章节保留骨架不删（重跑任务时 wordCount=0 的骨架会被再次抓取填充）；
   471→ * - novel.wordCount increment 只对抓取成功的章节（fill 填充在此增量，create 走
   472→ *   storeChapter 内部增量），收尾对触及书籍 recalcNovelWordCount 全量校准；
   473→ * - 保留全部既有机制：熔断感知重试/瞬时失败单次重试/协作式取消/flush 串行链/
   474→ *   入库前清洗/MAX_CONTENT_CHARS 截断。
   475→ * - 进度：single 模式 done 随本章递增（total=骨架章节数已在 Phase 1 写入）；
   476→ *   list 模式累加共享 chapterProgress（chaptersDone/chaptersTotal）。
   477→ */
   478→async function fillChapters(
   479→  run: Run,
   480→  books: Phase2Book[],
   481→  rule: LoadedRule,
   482→  opts: { trackTotal: boolean; chapterProgress?: { done: number; total: number }; concurrency: number },
   483→): Promise<Phase2Result> {
   484→  // 平铺（按书分组顺序）：任务单元携带所属书上下文（Referer 链/日志节流按书隔离）
   485→  const units: { job: ChapterFillJob; book: Phase2Book }[] = []
   486→  for (const b of books) for (const job of b.jobs) units.push({ job, book: b })
   487→
   488→  let filled = 0
   489→  let failed = 0
   490→  let skipped = 0
   491→  let done = 0 // single 模式主口径进度（done/total=章节）
   492→  let stopRun = false // 取消/记录删除 → 停止派发新章节，在途章节收尾后退出
   493→  const touchedNovels = new Set<number>()
   494→
   495→  const alive = async (): Promise<boolean> => {
   496→    if (stopRun) return false
   497→    return !(await isCanceled(run.taskId))
   498→  }
   499→
   500→  // flush 串行链：并行分支共享 Run 缓冲与进度字段，写盘串行化避免交错/竞态
   501→  let flushChain = Promise.resolve(true)
   502→  const scheduleFlush = (fields: TaskFlushFields): Promise<boolean> => {
   503→    flushChain = flushChain.then(async (ok) => {
   504→      if (!ok || stopRun) return false
   505→      const ok2 = await run.flush(fields)
   506→      if (!ok2) stopRun = true // 任务记录被删除 → 全池停止派发
   507→      return ok2
   508→    })
   509→    return flushChain
   510→  }
   511→
   512→  /** 每处理完一个章节后要刷出的进度字段（不含任务级成果计数） */
   513→  const progressFields = (): TaskFlushFields =>
   514→    opts.trackTotal
   515→      ? { done }
   516→      : {
   517→          chaptersDone: opts.chapterProgress?.done ?? 0,
   518→          chaptersTotal: opts.chapterProgress?.total ?? 0,
   519→        }
   520→
   521→  const processUnit = async ({ job, book }: { job: ChapterFillJob; book: Phase2Book }): Promise<void> => {
   522→    if (job.kind === 'skip') {
   523→      // 已存在且有内容：不重抓，仅推进进度（与旧版「已存在，跳过」口径一致）
   524→      skipped++
   525→      done++
   526→      if (opts.chapterProgress) opts.chapterProgress.done++
   527→      run.log(`章节「${job.refTitle.slice(0, 30)}」已存在，跳过`)
   528→      await scheduleFlush(progressFields())
   529→      return
   530→    }
   531→
   532→    if (!(await alive())) {
   533→      stopRun = true
   534→      return
   535→    }
   536→    run.log(`抓取章节「${(job.refTitle || job.url).slice(0, 36)}」`)
   537→    // Referer 链：章节页带书页来路（站点常见「书页→章节」导航校验）；
   538→    // fetchChapterPaged：同章分页（…_2.html / …/2.html）自动翻页拼接，避免长章只存半页。
   539→    // 熔断感知：章节撞熔断（站点临时限流封禁）时同主机并发 worker 合流等待冷却后自动重试，
   540→    // 而非把剩余章节全部烧成失败（Task 36 复盘：ggd66 一次临时封禁曾致 232 章被永久记失败）
   541→    const circuitFetch = (): ReturnType<typeof fetchChapterPaged> =>
   542→      withCircuitRetry(hostOf(job.url), () => fetchChapterPaged(job.url, rule, book.bookUrl), {
   543→        detect: (r) => (r.ok ? null : circuitWaitMs(r.error, r.circuitRetryAfterMs)),
   544→        isAlive: alive,
   545→        onWait: (ms, cycle, max) =>
   546→          run.log(`目标主机熔断冷却中，暂停 ${Math.round(ms / 1000)}s 后自动重试章节（第 ${cycle}/${max} 轮，等待期间不向源站发请求）`),
   547→      })
   548→    let ch = await circuitFetch()
   549→    // 瞬时失败单次重试（预算耗尽/超时类）：高并发下域名限速排队可吃光引擎 55s 链预算，
   550→    // 属暂时性压力而非站点拒绝——稍候片刻让队列排空后重试一次，减少无谓的永久失败章节
   551→    if (!ch.ok && isTransientEngineError(ch.error)) {
   552→      run.log(`章节「${(job.refTitle || job.url).slice(0, 30)}」因引擎预算耗尽/超时失败，稍候重试一次`)
   553→      await sleep(3_000 + Math.random() * 5_000)
   554→      if (stopRun || !(await alive())) return
   555→      ch = await circuitFetch()
   556→    }
   557→    if (stopRun || !(await alive())) return // 取消：抓取结果直接丢弃，尽快退出
   558→    const data = ch.ok ? ch.data : null
   559→
   560→    // 入库前统一清洗（去 \r\n/行首缩进/空行/噪声行），存储契约：无空行、无行首缩进
   561→    const cleaned = cleanChapterContent(data?.content ?? '')
   562→    let content = cleaned.text
   563→    if (content.length > MAX_CONTENT_CHARS) {
   564→      content = content.slice(0, MAX_CONTENT_CHARS)
   565→      run.log(`章节正文超长（>${MAX_CONTENT_CHARS} 字符），截断保存（内存保护）`)
   566→    }
   567→
   568→    if (!data || !content.trim()) {
   569→      // 抓取失败：骨架保留不删（重跑任务续采），仅记账
   570→      failed++
   571→      book.failed++
   572→      done++
   573→      if (opts.chapterProgress) opts.chapterProgress.done++
   574→      run.log(`章节抓取失败: ${ch.ok ? '正文为空' : ch.error}`)
   575→      await scheduleFlush({ ...progressFields(), chapters: run.counters.chapters })
   576→      return
   577→    }
   578→    if (ch.warnings.length) run.logWarnings(ch.warnings)
   579→
   580→    const chTitle = (job.refTitle || cleanTextField(data.title) || `第${job.idx}章`).slice(0, 200)
   581→    if (cleaned.removedLines > 0 && book.cleanLogCount < 3) {
   582→      book.cleanLogCount++
   583→      run.log(`章节「${chTitle.slice(0, 30)}」清洗 ${cleaned.removedLines} 行噪声`)
   584→    }
   585→    const wordCount = content.replace(/\s/g, '').length
   586→
   587→    let ok = false
   588→    if (job.kind === 'fill') {
   589→      // 填充骨架/已有空章节：chapterId 优先定位，缺省按 (novelId,idx) 复合唯一定位；
   590→      // 卷名仅在本次目录携带时写入（不回退为空串，防抹掉既有卷名）
   591→      ok = await db.chapter
   592→        .update({
   593→          where: job.chapterId
   594→            ? { id: job.chapterId }
   595→            : { novelId_idx: { novelId: job.novelId, idx: job.idx } },
   596→          data: {
   597→            title: chTitle,
   598→            content,
   599→            wordCount,
   600→            ...(job.volume ? { volume: job.volume } : {}),
   601→          },
   602→        })
   603→        .then(() => true)
   604→        .catch(() => false)
   605→      if (!ok) run.log(`章节填充失败（骨架可能已被删除）: ${chTitle.slice(0, 40)}`)
   606→    } else {
   607→      // 兜底新建（骨架未能入库）：storeChapter 内部处理 idx 唯一冲突顺延重试与字数增量
   608→      const stored = await storeChapter(run, job.novelId, job.idx, {
   609→        title: chTitle,
   610→        content,
   611→        wordCount,
   612→        volume: job.volume,
   613→      })
   614→      ok = stored.ok
   615→      if (!stored.ok) run.log(`章节入库失败: ${stored.message.slice(0, 120)}`)
   616→    }
   617→
   618→    if (ok) {
   619→      filled++
   620→      book.filled++
   621→      run.counters.chapters++
   622→      touchedNovels.add(job.novelId)
   623→      // 实时可见：fill 成功即增量推进书籍字数（create 已在 storeChapter 内增量；
   624→      // 收尾 recalc 全量校准，兜底并发双任务同书的双计误差）
   625→      if (job.kind === 'fill') {
   626→        await db.novel
   627→          .update({ where: { id: job.novelId }, data: { wordCount: { increment: wordCount } } })
   628→          .catch(() => null)
   629→      }
   630→    } else {
   631→      failed++
   632→      book.failed++
   633→    }
   634→    done++
   635→    if (opts.chapterProgress) opts.chapterProgress.done++
   636→    await scheduleFlush({ ...progressFields(), chapters: run.counters.chapters })
   637→  }
   638→
   639→  await runPool(units, clampConcurrency(opts.concurrency), processUnit)
   640→  const flushOk = await flushChain
   641→
   642→  // 收尾校准：对有成功填充的书籍重算字数（增量累计误差/并发双计兜底；失败静默）
   643→  for (const novelId of touchedNovels) await recalcNovelWordCount(novelId)
   644→
   645→  if (stopRun || !flushOk) {
   646→    run.log('任务已取消/记录已删除，停止章节填充')
   647→    await run.flush({ ...progressFields(), chapters: run.counters.chapters }).catch(() => undefined)
   648→  } else if (failed > 0) {
   649→    run.log('抓取失败的章节已保留骨架（wordCount=0），重跑任务可自动续采')
   650→  }
   651→
   652→  return { canceled: stopRun || !flushOk, filled, failed, skipped }
   653→}

   655→// ==================== 两种模式（两阶段编排） ====================
   656→
   657→async function runSingle(run: Run, task: TaskRecord, rule: LoadedRule): Promise<void> {
   658→  // Phase 1：书页+目录 → 书籍入库 + 章节骨架（total=骨架章节数）
   659→  const meta = await processBookMeta(run, task.targetUrl, rule, { trackTotal: true })
   660→  if (meta.canceled) {
   661→    run.log('任务已取消')
   662→    await finalize(run, 'canceled', meta.message || '任务已取消')
   663→    return
   664→  }
   665→  await run.flush({ created: run.counters.created, updated: run.counters.updated, chapters: run.counters.chapters })
   666→  if (!meta.ok) {
   667→    await finalize(run, 'failed', meta.message || '书页提取失败')
   668→    return
   669→  }
   670→  // Phase 2：章节内容填充（done 随填充递增，保持 done/total=章节 口径）
   671→  const res = await fillChapters(
   672→    run,
   673→    [{ novelId: meta.novelId, title: '', bookUrl: task.targetUrl, jobs: meta.jobs, filled: 0, failed: 0, cleanLogCount: 0 }],
   674→    rule,
   675→    { trackTotal: true, concurrency: task.concurrency },
   676→  )
   677→  if (res.canceled) {
   678→    await finalize(run, 'canceled', '任务已取消')
   679→    return
   680→  }
   681→  if (res.failed === 0) {
   682→    if (res.filled > 0) {
   683→      await finalize(run, 'success', `采集完成：${res.filled} 章`)
   684→      return
   685→    }
   686→    await finalize(run, 'success', meta.jobs.length === 0 ? meta.message : '无新增章节（章节均已存在）')
   687→    return
   688→  }
   689→  if (res.filled > 0) {
   690→    await finalize(run, 'partial', `${res.filled} 章成功 / ${res.failed} 章失败`)
   691→    return
   692→  }
   693→  await finalize(run, 'failed', '章节采集全部失败')
   694→}

   696→async function runList(run: Run, task: TaskRecord, rule: LoadedRule): Promise<void> {
   697→  // ---- Phase 1a：列表页收集（翻页变体/熔断重试/Referer 链机制保持不变）----
   698→  const startPage = Math.max(1, Math.floor(task.startPage) || 1)
   699→  const endPage = startPage + Math.max(1, task.pages) - 1
   700→  run.log(`抓取列表页第 ${startPage} 页…`)
   701→  // 熔断感知：起始列表页撞熔断时等待冷却后自动重试；抓取失败（超限）与「空列表页」自此可区分
   702→  const firstPage = await withCircuitRetry(hostOf(task.targetUrl), () => fetchListPage(run, task.targetUrl, rule), {
   703→    detect: (r) => (r.error ? circuitWaitMs(r.error) : null),
   704→    isSuccess: (r) => r.error === null,
   705→    isAlive: async () => !(await isCanceled(run.taskId)),
   706→    onWait: (ms, cycle, max) =>
   707→      run.log(`目标主机熔断冷却中，暂停 ${Math.round(ms / 1000)}s 后自动重试列表页（第 ${cycle}/${max} 轮，等待期间不向源站发请求）`),
   708→  })
   709→  const first = firstPage.items
   710→  if (await isCanceled(run.taskId)) {
   711→    await finalize(run, 'canceled', '任务已取消')
   712→    return
   713→  }
   714→  if (firstPage.error) {
   715→    run.log('起始列表页持续抓取失败（熔断等待轮数耗尽或持续失败），任务终止')
   716→    await finalize(run, 'failed', `列表页抓取失败: ${firstPage.error.slice(0, 180)}`)
   717→    return
   718→  }
   719→  if (first.length === 0) {
   720→    run.log('列表页未提取到书籍条目')
   721→    await finalize(run, 'failed', '列表页未提取到书籍条目')
   722→    return
   723→  }
   724→  run.log(`第 ${startPage} 页提取 ${first.length} 条`)
   725→
   726→  const items = [...first]
   727→  // 当前生效的列表页 URL（含翻页命中页）：作为后续书页抓取的 Referer 来路
   728→  let currentListUrl = task.targetUrl
   729→  for (let k = startPage + 1; k <= endPage; k++) {
   730→    if (await isCanceled(run.taskId)) break
   731→    let got: ListItem[] | null = null
   732→    for (const v of pageVariants(task.targetUrl, k)) {
   733→      // 熔断感知：翻页变体撞熔断时等待冷却后自动重试；变体持续失败 → 尝试下一变体
   734→      const pv = await withCircuitRetry(hostOf(v), () => fetchListPage(run, v, rule), {
   735→        detect: (r) => (r.error ? circuitWaitMs(r.error) : null),
   736→        isSuccess: (r) => r.error === null,
   737→        isAlive: async () => !(await isCanceled(run.taskId)),
   738→        onWait: (ms, cycle, max) =>
   739→          run.log(`目标主机熔断冷却中，暂停 ${Math.round(ms / 1000)}s 后自动重试列表页（第 ${cycle}/${max} 轮）`),
   740→      })
   741→      if (pv.error) continue
   742→      if (pv.items.length > 0) {
   743→        got = pv.items
   744→        currentListUrl = v
   745→        run.log(`第 ${k} 页命中: ${v.slice(0, 100)}`)
   746→        break
   747→      }
   748→    }
   749→    if (!got || got.length === 0) {
   750→      run.log(`第 ${k} 页无结果，跳过`)
   751→      continue
   752→    }
   753→    run.log(`第 ${k} 页提取 ${got.length} 条`)
   754→    items.push(...got)
   755→  }
   756→
   757→  // 合并去重（按 URL，缺 URL 按标题）；不设书籍数量上限，全量采集
   758→  const seen = new Set<string>()
   759→  const merged: ListItem[] = []
   760→  for (const it of items) {
   761→    const key = it.url ?? `t:${it.title}`
   762→    if (seen.has(key)) continue
   763→    seen.add(key)
   764→    merged.push(it)
   765→  }
   766→  const total = merged.length
   767→  await run.flush({ total, done: 0 })
   768→  run.log(`去重后共 ${total} 本书待采集（并发 ${task.concurrency} 路：先书级元数据+骨架，后章级批量内容）`)
   769→
   770→  // ---- Phase 1b：书级并发抓取元数据+目录骨架（done/total=书，主口径不变）----
   771→  const books: Phase2Book[] = []
   772→  let doneBooks = 0
   773→  let failBooks = 0
   774→  let stopRun = false
   775→  // 任务级章节进度（跨书累计）：list 模式 done/total 主口径是「书」，章节进度走 chaptersDone/chaptersTotal
   776→  const chapterProgress = { done: 0, total: 0 }
   777→  // flush 串行链：并行书籍分支的进度写盘串行化
   778→  let flushChain = Promise.resolve(true)
   779→  const scheduleBookFlush = (): Promise<boolean> => {
   780→    flushChain = flushChain.then(async (ok) => {
   781→      if (!ok || stopRun) return false
   782→      const ok2 = await run.flush({
   783→        done: doneBooks,
   784→        chaptersDone: chapterProgress.done,
   785→        chaptersTotal: chapterProgress.total,
   786→        created: run.counters.created,
   787→        updated: run.counters.updated,
   788→        chapters: run.counters.chapters,
   789→      })
   790→      if (!ok2) stopRun = true
   791→      return ok2
   792→    })
   793→    return flushChain
   794→  }
   795→
   796→  const processItem = async (item: ListItem): Promise<void> => {
   797→    if (stopRun) return
   798→    if (await isCanceled(run.taskId)) {
   799→      stopRun = true
   800→      return
   801→    }
   802→    run.log(`━━ 《${(item.title || '未命名').slice(0, 30)}》`)
   803→    const meta = await processBookMeta(run, item.url as string, rule, { trackTotal: false, referer: currentListUrl })
   804→    doneBooks++
   805→    if (meta.canceled) stopRun = true // 取消传播：停止派发后续书籍
   806→    else if (meta.ok) {
   807→      books.push({
   808→        novelId: meta.novelId,
   809→        title: item.title || '',
   810→        bookUrl: item.url as string,
   811→        jobs: meta.jobs,
   812→        filled: 0,
   813→        failed: 0,
   814→        cleanLogCount: 0,
   815→      })
   816→      chapterProgress.total += meta.refsTotal
   817→    } else failBooks++
   818→    await scheduleBookFlush()
   819→  }
   820→
   821→  await runPool(merged, clampConcurrency(task.concurrency), processItem)
   822→  const flushOk = await flushChain
   823→
   824→  if (stopRun || !flushOk) {
   825→    run.log('任务已取消')
   826→    await finalize(run, 'canceled', `已取消（完成 ${doneBooks}/${total} 本）`)
   827→    return
   828→  }
   829→
   830→  // ---- Phase 2：全部书籍章节平铺（按书分组顺序），章级并发批量填充 ----
   831→  let res: Phase2Result = { canceled: false, filled: 0, failed: 0, skipped: 0 }
   832→  if (books.length > 0) {
   833→    const pending = books.reduce((n, b) => n + b.jobs.filter((j) => j.kind !== 'skip').length, 0)
   834→    run.log(`元数据阶段完成：${books.length} 本书目录骨架就绪，开始批量抓取章节内容（待抓 ${pending} 章，并发 ${task.concurrency} 路）`)
   835→    res = await fillChapters(run, books, rule, { trackTotal: false, chapterProgress, concurrency: task.concurrency })
   836→  }
   837→  if (res.canceled) {
   838→    await finalize(run, 'canceled', `已取消（完成 ${doneBooks}/${total} 本，已填充 ${res.filled} 章）`)
   839→    return
   840→  }
   841→
   842→  // ---- 书籍级成败判定：元数据失败 + 章节全部抓取失败（已存在跳过/无章节链接仍算成功）----
   843→  const okBooks = books.filter((b) => b.filled > 0 || b.failed === 0).length
   844→  const chapterFailBooks = books.filter((b) => b.filled === 0 && b.failed > 0).length
   845→  const failBooksFinal = failBooks + chapterFailBooks
   846→
   847→  if (okBooks === 0) {
   848→    await finalize(run, 'failed', '无书籍采集成功')
   849→    return
   850→  }
   851→  if (failBooksFinal > 0) {
   852→    await finalize(run, 'partial', `${okBooks} 本成功 / ${failBooksFinal} 本失败，入库 ${res.filled} 章`)
   853→    return
   854→  }
   855→  await finalize(
   856→    run,
   857→    'success',
   858→    res.failed > 0
   859→      ? `范围采集完成：共 ${total} 本，入库 ${res.filled} 章，失败 ${res.failed} 章`
   860→      : `范围采集完成：共 ${total} 本，入库 ${res.filled} 章`,
   861→  )
   862→}

   866→/** 终态写入：仅当仍处于 running 时写终态；已被取消/删除则只保留日志 */
   867→async function finalize(run: Run, status: string, message: string): Promise<void> {
   868→  const cur = await db.scrapeTask
   869→    .findUnique({ where: { id: run.taskId }, select: { status: true } })
   870→    .catch(() => null)
   871→  if (!cur) return
   872→  if (cur.status === 'running') {
   873→    await db.scrapeTask
   874→      .updateMany({
   875→        where: { id: run.taskId, status: 'running' },
   876→        data: { status, message: message.slice(0, 500), log: run.logText() },
   877→      })
   878→      .catch(() => {})
   879→  } else {
   880→    await db.scrapeTask
   881→      .updateMany({ where: { id: run.taskId }, data: { log: run.logText() } })
   882→      .catch(() => {})
   883→  }
   884→}

   886→async function runTask(taskId: number): Promise<void> {
   887→  const run = new Run(taskId)
   888→  try {
   889→    // pending → running 条件更新：pending 阶段已被取消/删除的任务不再启动
   890→    const started = await db.scrapeTask
   891→      .updateMany({ where: { id: taskId, status: 'pending' }, data: { status: 'running' } })
   892→      .catch(() => ({ count: 0 }))
   893→    if (started.count === 0) return
   894→    const task = await db.scrapeTask.findUnique({ where: { id: taskId } }).catch(() => null)
   895→    if (!task) return
   896→    run.ruleId = task.ruleId ?? null // 封面回填链路：入库时记入 Novel.sourceRuleId
   897→    const record: TaskRecord = {
   898→      id: task.id,
   899→      mode: task.mode,
   900→      targetUrl: task.targetUrl,
   901→      pages: task.pages,
   902→      startPage: task.startPage,
   903→      concurrency: clampConcurrency(task.concurrency),
   904→      ruleId: task.ruleId,
   905→    }
   906→    const rule = await loadRule(record.ruleId)
   907→    run.log(
   908→      record.mode === 'list'
   909→        ? `任务开始（范围采集）目标: ${record.targetUrl}，起始页: ${record.startPage}，页数: ${record.pages}，并发: ${record.concurrency}`
   910→        : `任务开始（单本采集）目标: ${record.targetUrl}，并发: ${record.concurrency}`,
   911→    )
   912→    run.log(
   913→      record.ruleId
   914→        ? `使用规则「${rule.name ?? record.ruleId}」（charset=${rule.charset ?? 'auto'}）`
   915→        : '未使用规则，依赖引擎内置启发式提取',
   916→    )
   917→    await run.flush()
   918→    if (record.mode === 'list') await runList(run, record, rule)
   919→    else await runSingle(run, record, rule)
   920→  } catch (e) {
   921→    run.log(`发生未预期错误: ${e instanceof Error ? e.message.slice(0, 200) : String(e).slice(0, 200)}`)
   922→    await finalize(run, 'failed', '任务执行异常')
   923→  }
   924→}

   926→/**
   927→ * fire-and-forget 入口：API 路由创建任务后调用，不 await。
   928→ * 多任务天然并行：每个任务独立 worker promise，互不阻塞；
   929→ * 对同一目标站的压力由引擎层域名限速统一收敛。
   930→ */
   931→export function triggerScrapeTask(taskId: number): void {
   932→  if (running.has(taskId)) return
   933→  running.add(taskId)
   934→  void runTask(taskId)
   935→    .catch((e: unknown) => {
   936→      console.error(`[scrape-worker] task ${taskId} 兜底异常:`, e instanceof Error ? e.message : e)
   937→    })
   938→    .finally(() => {
   939→      running.delete(taskId)
   940→    })
   941→}
