   147→   147→export function Category({
   148→   148→  navigate,

   150→   150→  page = 1,
   151→   151→}: ViewProps & { categoryId?: number; page?: number }) {

   164→   164→            {cats.data.find((c) => c.id === categoryId)?.novelCount ?? list.data?.total ?? ''}
   165→   165→          </span>
   166→   166→        ) : null}

   168→   168→
   169→   169→      {/* 筛选条 */}

   174→   174→            categoryId == null
   175→   175→              ? 'cursor-pointer rounded-full bg-[#2563eb] px-4 py-1.5 text-[13px] text-white transition-colors duration-200'

   210→   210→            page={list.data!.page}
   211→   211→            totalPages={list.data!.totalPages}
   212→   212→            onGo={(p) => navigate({ name: 'category', categoryId, page: p })}
   213→   213→          />
   214→   214→        </>
   215→   215→      )}
   216→   216→    </div>
