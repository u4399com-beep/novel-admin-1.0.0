   425→/* Category：面包屑 + 排序筛选 + 六段式列表 + 分页                        */
   426→/* ==================================================================== */
   427→
   428→type SortKey = 'latest' | 'clicks' | 'words'
   429→type StatusKey = 'all' | 'serial' | 'finished'
   430→
   431→function CategoryView({
   432→  navigate,
   433→  categoryId,
   434→  page = 1,
   435→}: ViewProps & { categoryId?: number; page?: number }) {
   436→  const { data: cats } = useCategories()
   437→  const [sort, setSort] = useState<SortKey>('latest')
   438→  const [status, setStatus] = useState<StatusKey>('all')
   439→  const cur = page > 0 ? page : 1
   440→
   441→  const { data, isLoading, isError, refetch } = useNovels({
   442→    categoryId,
   443→    page: cur,
   444→    pageSize: 20,
   445→    sort,
   446→    status: status === 'all' ? undefined : status,
   447→  })
   448→
   449→  const catName =
   450→    categoryId != null ? (cats ?? []).find((c) => c.id === categoryId)?.name : undefined
   451→  const goPage = (p: number) => navigate({ name: 'category', categoryId, page: p })
   452→  const changeSort = (s: SortKey) => {
   453→    setSort(s)
   454→    if (cur !== 1) goPage(1)
   455→  }
   456→  const changeStatus = (s: StatusKey) => {
   457→    setStatus(s)
   458→    if (cur !== 1) goPage(1)
   459→  }
   460→
   461→  return (
   462→    <div className="mx-auto w-full max-w-[980px] px-2 py-3">
   463→      <Crumbs
   464→        items={[
   465→          { label: '首页', view: { name: 'home' } },
   466→          { label: catName ?? '全部小说' },
   467→        ]}
   468→        navigate={navigate}
   469→      />
   470→      <Block
   471→        className="mt-3"
   472→        title={`${catName ?? '全部小说'}列表`}
   473→        extra={
   474→          <span>
   475→            共 {data?.total ?? '—'} 部 · 第 {cur}/{data?.totalPages ?? 1} 页
   476→          </span>
   477→        }
   478→        bodyClass="px-2 py-1"
   479→      >
   480→        {/* 排序 / 状态筛选行 */}
   481→        <div className="flex flex-wrap items-center gap-x-3 gap-y-1 border-b border-dotted border-[#D5E6F3] px-1 pb-2 text-xs">
   482→          <span className="text-[#999]">排序</span>
   483→          {(
   484→            [
   485→              ['latest', '最近更新'],
   486→              ['clicks', '点击最多'],
   487→              ['words', '字数最多'],
   488→            ] as const
   489→          ).map(([k, label]) => (
   490→            <button
   491→              key={k}
   492→              type="button"
   493→              onClick={() => changeSort(k)}
   494→              className={cn(
   495→                'cursor-pointer',
   496→                sort === k ? 'font-bold text-[#CC0000]' : 'text-[#3366BB] hover:text-[#FF6600]',
   497→              )}
   498→            >
   499→              {label}
   500→            </button>
   501→          ))}
   502→          <span className="ml-2 text-[#999]">状态</span>
   503→          {(
   504→            [
   505→              ['all', '全部'],
   506→              ['serial', '连载'],
   507→              ['finished', '完本'],
   508→            ] as const
   509→          ).map(([k, label]) => (
   510→            <button
   511→              key={k}
   512→              type="button"
   513→              onClick={() => changeStatus(k)}

  1020→  const [page, setPage] = useState(1)
  1021→

  1036→          onKeyDown={(e) => {
  1037→            if (e.key === 'Enter') submit()
  1038→          }}
  1039→          placeholder="输入书名 / 作者关键词"
  1040→          className="h-8 min-w-0 flex-1 border border-[#BFD8EA] px-2 text-sm outline-none focus:border-[#3B76A8]"
